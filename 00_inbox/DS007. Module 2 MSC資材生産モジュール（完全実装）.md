// lib/agrix/module2/engines/humus.engine.ts

import { HumusInput, HumusOutput } from '../types/humus.types';

/**
 * MBT腐植質生産エンジン
 * ハイパーサイクル状態から最適な腐植質生産を計画・品質評価する
 */
export function runHumusEngine(input: HumusInput): HumusOutput {
  const { hypercycle, soilData, productionConfig } = input;
  const { inputBiomass_tPerHa, retentionDays, turningFrequency, moistureTarget } = productionConfig;

  // ----- 1. 生産効率計算 -----
  // ハイパーサイクル状態が生産効率に与える影響
  const cycleEffect = hypercycle.cycleEfficiency * 0.6 + hypercycle.humificationRate * 0.4;
  
  // 投入バイオマス量の影響（適正範囲: 3-8 t/ha）
  const biomassEffect = Math.min(1, inputBiomass_tPerHa / 5.0);
  
  // 腐熟期間の影響（最適: 60-90日）
  const retentionEffect = Math.min(1, retentionDays / 75);
  
  // 切り返し頻度（最適: 1-2回/週）
  const turningEffect = Math.min(1, turningFrequency / 1.5);
  
  // 水分管理（最適: 50-60%）
  const moistureEffect = 1 - Math.abs(moistureTarget - 55) / 20;
  
  // 総合生産効率
  const productionEfficiency = Math.min(1, 
    0.30 * cycleEffect +
    0.25 * biomassEffect +
    0.20 * retentionEffect +
    0.15 * turningEffect +
    0.10 * Math.max(0, moistureEffect)
  );

  // ----- 2. 腐植質生産量 -----
  // 理論最大変換率: バイオマス投入量 × 0.6（炭素換算）
  const maxHumus = inputBiomass_tPerHa * 0.6;
  const outputHumus_tPerHa = maxHumus * productionEfficiency;
  const yieldRate = (outputHumus_tPerHa / inputBiomass_tPerHa) * 100;

  // ----- 3. 品質指標計算 -----
  // フミン酸比率（高品質指標）: サイクル効率と腐熟期間に依存
  const humicAcidRatio = Math.min(1, 
    0.3 + 0.4 * hypercycle.humificationRate +
    0.3 * Math.min(1, retentionDays / 90)
  );
  
  // フルボ酸比率（即効性指標）: 分解菌活性に依存
  const fulvicAcidRatio = Math.min(1, 0.4 + 0.4 * hypercycle.resourceConversionRate);
  
  // CEC（保肥力）: 腐植質の質に依存
  const cationExchangeCapacity = 20 + 60 * humicAcidRatio; // meq/100g
  
  // 保水力: 腐植質含有量に依存
  const waterHoldingCapacity = 50 + 150 * (outputHumus_tPerHa / 10); // %増加
  
  // 安定性指数: 腐熟度とフミン酸比率から
  const stabilityIndex = Math.min(1, 
    0.3 * (retentionDays / 90) +
    0.4 * humicAcidRatio +
    0.3 * hypercycle.stableCarbonFraction
  );

  // ----- 4. 成熟度判定 -----
  let maturityLevel: HumusOutput['quality']['maturityLevel'] = 'IMMATURE';
  const maturityScore = stabilityIndex * 0.6 + (retentionDays / 90) * 0.4;
  if (maturityScore >= 0.85) maturityLevel = 'OVER_MATURE';
  else if (maturityScore >= 0.70) maturityLevel = 'MATURE';
  else if (maturityScore >= 0.50) maturityLevel = 'SEMI_MATURE';
  else maturityLevel = 'IMMATURE';

  // ----- 5. 土壌インパクト -----
  const socIncrease = outputHumus_tPerHa * 0.3; // SOC増加（t/ha）
  const carbonSequestration = socIncrease * 3.67; // CO2換算（tCO2/ha）
  const aggregateStability = Math.min(1, 0.3 + 0.5 * humicAcidRatio + 0.2 * (outputHumus_tPerHa / 5));
  const nutrientRetention = Math.min(1, 0.2 + 0.5 * cationExchangeCapacity / 100 + 0.3 * stabilityIndex);

  // ----- 6. 品質グレード判定 -----
  const qualityScore = (humicAcidRatio * 0.3 + stabilityIndex * 0.3 + productionEfficiency * 0.2 + aggregateStability * 0.2) * 100;
  let qualityGrade: 'A' | 'B' | 'C' | 'D' = 'C';
  if (qualityScore >= 80) qualityGrade = 'A';
  else if (qualityScore >= 65) qualityGrade = 'B';
  else if (qualityScore >= 45) qualityGrade = 'C';
  else qualityGrade = 'D';

  // ----- 7. 診断メッセージ -----
  const statusMessage = generateHumusStatus(maturityLevel, qualityGrade, outputHumus_tPerHa);
  const recommendations = generateHumusRecommendations(maturityLevel, qualityGrade, productionConfig);

  return {
    production: {
      outputHumus_tPerHa: Number(outputHumus_tPerHa.toFixed(2)),
      productionEfficiency: Number(productionEfficiency.toFixed(3)),
      productionCycleDays: retentionDays,
      yieldRate: Number(yieldRate.toFixed(1)),
    },
    quality: {
      humicAcidRatio: Number(humicAcidRatio.toFixed(3)),
      fulvicAcidRatio: Number(fulvicAcidRatio.toFixed(3)),
      cationExchangeCapacity: Number(cationExchangeCapacity.toFixed(1)),
      waterHoldingCapacity: Number(waterHoldingCapacity.toFixed(0)),
      stabilityIndex: Number(stabilityIndex.toFixed(3)),
      maturityLevel,
    },
    soilImpact: {
      socIncrease: Number(socIncrease.toFixed(2)),
      carbonSequestration: Number(carbonSequestration.toFixed(2)),
      aggregateStability: Number(aggregateStability.toFixed(3)),
      nutrientRetention: Number(nutrientRetention.toFixed(3)),
    },
    statusMessage,
    recommendations,
    qualityGrade,
    calculatedAt: new Date().toISOString(),
  };
}

// ヘルパー関数
function generateHumusStatus(maturity: string, grade: string, output: number): string {
  if (maturity === 'MATURE' && grade === 'A') {
    return `✅ 高品質な腐植質が生産されました（${output.toFixed(1)} t/ha）。土壌改良効果が最大化されます。`;
  }
  if (maturity === 'MATURE' || maturity === 'OVER_MATURE') {
    return `腐植質は成熟期に達しています。品質グレード ${grade}。炭素隔離効果が期待できます。`;
  }
  if (maturity === 'SEMI_MATURE') {
    return `腐植質は半熟状態です。さらに${30 - (output / 0.05)}日程度の腐熟期間を推奨します。`;
  }
  return `腐植質は未成熟です。腐熟期間の延長と切り返し頻度の増加を検討してください。`;
}

function generateHumusRecommendations(maturity: string, grade: string, config: any): string[] {
  const recs: string[] = [];
  
  if (maturity === 'IMMATURE' || maturity === 'SEMI_MATURE') {
    recs.push('腐熟期間を延長してください（推奨: 60-90日）。');
    recs.push('切り返し頻度を週1-2回に増やし、酸素供給を改善してください。');
  }
  if (grade === 'C' || grade === 'D') {
    recs.push('投入バイオマス量を増やし（4-6 t/ha）、C/N比を適正化してください。');
    recs.push('MBT55の追加接種を検討し、分解活性を向上させてください。');
  }
  if (config.moistureTarget < 45 || config.moistureTarget > 70) {
    recs.push(`目標水分（${config.moistureTarget}%）を50-60%に調整してください。`);
  }
  if (recs.length === 0) {
    recs.push('現状の生産プロセスは良好です。継続してください。');
  }
  return recs;
}

// lib/agrix/module2/engines/fertilizer.engine.ts

import { FertilizerInput, FertilizerOutput } from '../types/fertilizer.types';

/**
 * 発酵肥料生産エンジン
 * ハイパーサイクルと分解菌活性から最適な発酵肥料を生産する
 */
export function runFertilizerEngine(input: FertilizerInput): FertilizerOutput {
  const { hypercycle, soilData, decomposerActivity, productionConfig, feedStock } = input;
  const { type, inputMaterial, fermentationDays, aerationRate, temperatureTarget, phTarget } = productionConfig;

  // ----- 1. 発酵完了度計算 -----
  // 分解菌活性の影響
  const decomposerScore = decomposerActivity?.total ?? 0.5;
  
  // 発酵環境適合度
  const tempFit = 1 - Math.abs(temperatureTarget - 30) / 15;
  const phFit = 1 - Math.abs(phTarget - 6.8) / 1.5;
  const aerationFit = Math.min(1, aerationRate / 0.5);
  
  // 原料組成適合度
  const cnRatio = feedStock.carbonSource / Math.max(feedStock.nitrogenSource, 0.01);
  const cnFit = 1 - Math.abs(cnRatio - 25) / 20; // 理想C/N比: 25
  
  // 発酵完了度
  const fermentationCompletion = Math.min(1,
    0.25 * decomposerScore +
    0.20 * tempFit +
    0.15 * phFit +
    0.20 * aerationFit +
    0.20 * Math.max(0, cnFit)
  );

  // ----- 2. 生産効率 -----
  const productionEfficiency = Math.min(1,
    0.4 * fermentationCompletion +
    0.3 * (fermentationDays / 21) + // 理想発酵期間: 21日
    0.3 * hypercycle.resourceConversionRate
  );

  // ----- 3. 生産量（液肥・固形肥） -----
  const baseProduction = 1000; // L/ha（基準値）
  const liquidVolume_L = type === 'LIQUID' || type === 'BOTH' 
    ? baseProduction * productionEfficiency * (1 + 0.5 * feedStock.moistureContent)
    : 0;
  
  const solidWeight_kg = type === 'SOLID' || type === 'BOTH'
    ? 500 * productionEfficiency * (1 + 0.3 * feedStock.carbonSource / 40)
    : 0;

  // ----- 4. 栄養プロファイル -----
  // ハイパーサイクル状態から栄養価を推定
  const nutrientProfile = {
    n: Number((2.0 + 3.0 * hypercycle.resourceConversionRate).toFixed(1)),
    p: Number((1.0 + 2.0 * (1 - (cnRatio - 25) / 30)).toFixed(1)),
    k: Number((1.5 + 2.5 * feedStock.mineralContent).toFixed(1)),
    organicMatter: Number((40 + 30 * hypercycle.humificationRate).toFixed(0)),
    aminoAcids: Number((50 + 100 * decomposerScore).toFixed(0)),
    scfa: Number((0.5 + 1.5 * decomposerScore).toFixed(2)),
    microbialCount: Number((1e6 + 9e6 * decomposerScore).toExponential(0)),
  };

  // ----- 5. 施用推奨 -----
  const recommendedRate = type === 'LIQUID' 
    ? Number((20 + 30 * productionEfficiency).toFixed(0))
    : Number((100 + 200 * productionEfficiency).toFixed(0));
  
  const frequency = productionEfficiency > 0.7 ? 3 : productionEfficiency > 0.4 ? 4 : 5;
  
  const compatibleCrops = ['野菜類', '果樹類', '穀物類', '花卉類'];
  if (inputMaterial === 'CROP_RESIDUE') compatibleCrops.push('イネ科');
  if (inputMaterial === 'MANURE') compatibleCrops.push('根菜類');

  // ----- 6. カスケード効果 -----
  const nutrientUseEfficiency = Math.min(1, 0.3 + 0.5 * productionEfficiency + 0.2 * hypercycle.resourceConversionRate);
  const microbialActivation = Math.min(1, 0.2 + 0.6 * decomposerScore + 0.2 * fermentationCompletion);
  const yieldIncrease = 5 + 25 * productionEfficiency;
  const qualityImprovement = Math.min(1, 0.3 + 0.5 * (nutrientProfile.aminoAcids / 200) + 0.2 * (nutrientProfile.scfa / 2));

  // ----- 7. 品質グレード判定 -----
  const qualityScore = (
    productionEfficiency * 0.25 +
    fermentationCompletion * 0.20 +
    nutrientProfile.n / 5 * 0.20 +
    nutrientProfile.scfa / 2 * 0.20 +
    nutrientUseEfficiency * 0.15
  ) * 100;
  
  let qualityGrade: FertilizerOutput['qualityGrade'] = 'C';
  if (qualityScore >= 80) qualityGrade = 'A';
  else if (qualityScore >= 65) qualityGrade = 'B';
  else if (qualityScore >= 45) qualityGrade = 'C';
  else qualityGrade = 'D';

  // ----- 8. 診断メッセージ -----
  const statusMessage = generateFertilizerStatus(productionEfficiency, qualityGrade, type);
  const recommendations = generateFertilizerRecommendations(productionEfficiency, qualityGrade, fermentationDays, phTarget);

  return {
    production: {
      liquidVolume_L: Number(liquidVolume_L.toFixed(0)),
      solidWeight_kg: Number(solidWeight_kg.toFixed(0)),
      productionEfficiency: Number(productionEfficiency.toFixed(3)),
      fermentationCompletion: Number(fermentationCompletion.toFixed(3)),
    },
    nutrientProfile,
    application: {
      recommendedRate,
      frequency,
      timing: productionEfficiency > 0.7 ? '生育初期・中期' : '生育初期',
      compatibleCrops,
    },
    cascadeEffect: {
      nutrientUseEfficiency: Number(nutrientUseEfficiency.toFixed(3)),
      microbialActivation: Number(microbialActivation.toFixed(3)),
      yieldIncrease: Number(yieldIncrease.toFixed(1)),
      qualityImprovement: Number(qualityImprovement.toFixed(3)),
    },
    statusMessage,
    recommendations,
    qualityGrade,
    calculatedAt: new Date().toISOString(),
  };
}

function generateFertilizerStatus(efficiency: number, grade: string, type: string): string {
  if (efficiency >= 0.7 && grade === 'A') {
    return `✅ 高品質な発酵肥料（${type}）が生産されました。栄養バランスが最適化されています。`;
  }
  if (efficiency >= 0.5) {
    return `発酵肥料（${type}）は良好な品質です（グレード${grade}）。施用による収量向上が期待できます。`;
  }

// lib/agrix/module2/engines/feed.engine.ts

import { FeedInput, FeedOutput } from '../types/feed.types';

/**
 * 発酵飼料生産エンジン
 * 家畜用発酵飼料の生産・品質・メタン削減効果を評価する
 */
export function runFeedEngine(input: FeedInput): FeedOutput {
  const { hypercycle, soilData, decomposerActivity, productionConfig, feedStock } = input;
  const { livestockType, feedBase, fermentationDays, inoculationRate, moistureTarget, compactionDensity } = productionConfig;

  // ----- 1. 発酵品質スコア -----
  // MBT55接種効果
  const inoculationEffect = Math.min(1, inoculationRate / 3.0); // 3L/tを基準
  
  // ハイパーサイクル連携効果
  const cycleEffect = hypercycle.resourceConversionRate * 0.5 + hypercycle.cycleEfficiency * 0.5;
  
  // 環境適合度
  const moistureFit = 1 - Math.abs(moistureTarget - 55) / 20;
  const densityFit = Math.min(1, compactionDensity / 600); // 600kg/m3を基準
  
  // 発酵品質スコア
  const fermentationQuality = Math.min(1,
    0.30 * inoculationEffect +
    0.25 * cycleEffect +
    0.20 * (fermentationDays / 30) +
    0.15 * moistureFit +
    0.10 * densityFit
  );

  // ----- 2. 生産効率 -----
  const productionEfficiency = Math.min(1,
    0.5 * fermentationQuality +
    0.3 * (fermentationDays / 30) +
    0.2 * (decomposerActivity?.total ?? 0.5)
  );

  // ----- 3. 乾物回収率 -----
  const dryMatterRecovery = 60 + 35 * productionEfficiency; // %

  // ----- 4. 栄養価計算 -----
  // 原料組成から発酵後の栄養価を予測
  const proteinBoost = 1 + 0.2 * fermentationQuality;
  const digestibilityBoost = 1 + 0.3 * fermentationQuality;
  
  const nutritionalValue = {
    crudeProtein: Number((feedStock.crudeProtein * proteinBoost).toFixed(1)),
    metabolizableEnergy: Number((10 + 5 * fermentationQuality).toFixed(1)),
    scfaContent: Number((50 + 150 * fermentationQuality).toFixed(0)),
    digestibility: Number((60 + 30 * digestibilityBoost / 1.3).toFixed(1)),
    palatabilityScore: Number((0.4 + 0.5 * fermentationQuality).toFixed(3)),
  };

  // ----- 5. メタン削減効果 -----
  // 発酵飼料によるメタン削減（家畜種別で効果が異なる）
  let baseReduction = 0.20; // 基準削減率（20%）
  if (livestockType === 'RUMINANT') baseReduction = 0.25; // 反芻動物は効果大
  else if (livestockType === 'PIG') baseReduction = 0.15;
  else if (livestockType === 'POULTRY') baseReduction = 0.10;
  
  const qualityMultiplier = 0.5 + 0.5 * fermentationQuality;
  const entericMethaneReduction = baseReduction * qualityMultiplier * 100;
  const manureMethaneReduction = (baseReduction * 0.7) * qualityMultiplier * 100;
  const totalMethaneReduction = (entericMethaneReduction * 0.6 + manureMethaneReduction * 0.4);
  
  // CO2換算削減量（1頭あたり年間）
  const baseEmission = livestockType === 'RUMINANT' ? 2.5 : livestockType === 'PIG' ? 0.8 : 0.3;
  const co2EquivalentReduction = baseEmission * (totalMethaneReduction / 100) * 0.5; // tCO2/頭/年

  // ----- 6. 畜産インパクト -----
  const feedEfficiency = 5 + 20 * fermentationQuality;
  const weightGain = 3 + 15 * fermentationQuality;
  const mortalityReduction = 2 + 13 * fermentationQuality;
  const immuneBoost = Math.min(1, 0.2 + 0.6 * fermentationQuality);
  const reproductivePerformance = 2 + 13 * fermentationQuality;

  // ----- 7. 品質グレード判定 -----
  const qualityScore = (
    fermentationQuality * 0.30 +
    nutritionalValue.digestibility / 100 * 0.25 +
    nutritionalValue.palatabilityScore * 0.20 +
    totalMethaneReduction / 100 * 0.15 +
    productionEfficiency * 0.10
  ) * 100;
  
  let qualityGrade: FeedOutput['qualityGrade'] = 'C';
  if (qualityScore >= 80) qualityGrade = 'A';
  else if (qualityScore >= 65) qualityGrade = 'B';
  else if (qualityScore >= 45) qualityGrade = 'C';
  else qualityGrade = 'D';

  // ----- 8. 診断メッセージ -----
  const statusMessage = generateFeedStatus(qualityGrade, livestockType, totalMethaneReduction);
  const recommendations = generateFeedRecommendations(fermentationQuality, qualityGrade, livestockType);

  return {
    production: {
      output_tPerHa: Number((4 + 6 * productionEfficiency).toFixed(2)),
      productionEfficiency: Number(productionEfficiency.toFixed(3)),
      fermentationQuality: Number(fermentationQuality.toFixed(3)),
      dryMatterRecovery: Number(dryMatterRecovery.toFixed(1)),
    },
    nutritionalValue,
    methaneReduction: {
      entericMethaneReduction: Number(entericMethaneReduction.toFixed(1)),
      manureMethaneReduction: Number(manureMethaneReduction.toFixed(1)),
      totalMethaneReduction: Number(totalMethaneReduction.toFixed(1)),
      co2EquivalentReduction: Number(co2EquivalentReduction.toFixed(2)),
    },
    livestockImpact: {
      weightGain: Number(weightGain.toFixed(1)),
      mortalityReduction: Number(mortalityReduction.toFixed(1)),
      feedEfficiency: Number(feedEfficiency.toFixed(1)),
      immuneBoost: Number(immuneBoost.toFixed(3)),
      reproductivePerformance: Number(reproductivePerformance.toFixed(1)),
    },
    statusMessage,
    recommendations,
    qualityGrade,
    calculatedAt: new Date().toISOString(),
  };
}

function generateFeedStatus(grade: string, livestockType: string, methaneReduction: number): string {
  if (grade === 'A') {
    return `✅ 高品質な発酵飼料（${livestockType}用）が生産されました。メタン削減効果は${methaneReduction.toFixed(0)}%です。`;
  }
  if (grade === 'B') {
    return `発酵飼料（${livestockType}用）は良好な品質です。メタン削減効果${methaneReduction.toFixed(0)}%が期待できます。`;
  }
  return `発酵飼料（${livestockType}用）の品質が不十分です。発酵条件の見直しが必要です。`;
}

function generateFeedRecommendations(quality: number, grade: string, livestockType: string): string[] {

    recs.push('反芻動物用は特に発酵品質が重要です。発酵期間を30日以上に延長してください。');

    recs.push('現状の発酵飼料生産は良好です。継続してください。');
  }
  return recs;

    hypercycle: input.hypercycle,

    productionConfig: input.productionConfig.humus,

      mineralContent: input.feedStock.mineralContent,
      moistureContent: input.feedStock.moistureContent,
    },
  };
  const fertilizer = runFertilizerEngine(fertilizerInput);

    hypercycle: input.hypercycle,
    soilData: input.soilData,
    decomposerActivity: input.decomposerActivity,

      crudeProtein: input.feedStock.crudeProtein,
      crudeFiber: input.feedStock.crudeFiber,
      starch: input.feedStock.starch,
      fat: input.feedStock.fat,
      ash: input.feedStock.ash,
    },
  };
  const feed = runFeedEngine(feedInput);

  // ----- 2. 統合MSCスコア計算 -----
  const mscIntegratedScore = (
    humus.quality.stabilityIndex * 0.25 +
    humus.production.productionEfficiency * 0.15 +
    fertilizer.production.productionEfficiency * 0.20 +
    fertilizer.cascadeEffect.nutrientUseEfficiency * 0.15 +
    feed.production.fermentationQuality * 0.15 +
    feed.methaneReduction.totalMethaneReduction / 100 * 0.10
  ) * 100;

  let mscStatus: Module2Output['mscStatus'] = 'FAIR';
  if (mscIntegratedScore >= 80) mscStatus = 'OPTIMAL';
  else if (mscIntegratedScore >= 65) mscStatus = 'GOOD';
  else if (mscIntegratedScore >= 45) mscStatus = 'FAIR';
  else mscStatus = 'POOR';

  // ----- 3. 資源循環効率 -----
  const resourceRecoveryRate = (
    humus.production.yieldRate / 100 * 0.4 +
    fertilizer.production.productionEfficiency * 0.3 +
    feed.production.dryMatterRecovery / 100 * 0.3
  );
  
  const wasteReductionRate = Math.min(1,
    (humus.production.productionEfficiency * 0.4 +
     fertilizer.production.fermentationCompletion * 0.3 +
     feed.production.dryMatterRecovery / 100 * 0.3)
  );
  
  const nutrientLoopEfficiency = (
    humus.soilImpact.nutrientRetention * 0.35 +
    fertilizer.cascadeEffect.nutrientUseEfficiency * 0.35 +
    feed.nutritionalValue.digestibility / 100 * 0.30
  );

  // ----- 4. Module3連携データ -----
  const totalCarbonSequestration = humus.soilImpact.carbonSequestration;
  const methaneReduction = feed.methaneReduction.co2EquivalentReduction * 0.5;
  const ghgReductionTotal = totalCarbonSequestration + methaneReduction;
  const socIncreaseTotal = humus.soilImpact.socIncrease;

    fertilizer.production.liquidVolume_L * 0.5 +
    fertilizer.production.solidWeight_kg * 0.3
  );

  // ----- 6. サマリ生成 -----

  return {
    timestamp: new Date().toISOString(),
    version: '1.0.0',

    fertilizer,

      methaneReduction: Number(methaneReduction.toFixed(2)),

    }
  }

  return {
    recommendedAction,
    priorityActions: priorityActions.length > 0 ? priorityActions : ['現状のMSC運用を継続してください。'],

import { NextRequest, NextResponse } from 'next/server';
import { runModule2Full } from '@/lib/agrix/module2';

    const body = await req.json();
    
    // 必須フィールドチェック
    const requiredFields = [
      'soilData', 'productionConfig', 'feedStock'

      module1: {
        agrixHealthScore: module1Result.agrixHealthScore,
        healthStatus: module1Result.healthStatus,
        summary: module1Result.summary,
      },
      module2: result,
    });
  } catch (error) {
    return NextResponse.json(
      { error: 'Internal server error', details: String(error) },
