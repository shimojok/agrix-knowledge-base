## 中身（そのまま貼ってOK）

```python
import numpy as np

def simulate(mbt55=True):
    # 初期値（ここは後で調整可能）
    F0 = 100   # 化学肥料
    Y0 = 5.0   # 収量（t/ha）
    C0 = 100   # コスト
    demand = 100
    
    # MBT55効果
    if mbt55:
        M = 1.0
        fertilizer_reduction = 0.65   # 65%削減
        n2o_reduction = 0.6           # 60%削減
    else:
        M = 0.3
        fertilizer_reduction = 0.0
        n2o_reduction = 0.0
    
    # モデル計算
    fertilizer = F0 * (1 - fertilizer_reduction)
    
    disease = 0.4 * np.exp(-1.2 * M)
    yield_ = Y0 * (1 - disease) + 0.5 * M
    
    cost = C0 - 40 * fertilizer_reduction
    
    price = demand / yield_
    
    co2 = 2 - (1.5 * M)
    n2o = 1.0 * (1 - n2o_reduction)
    
    return {
        "Fertilizer": round(fertilizer, 2),
        "Yield": round(yield_, 2),
        "Cost": round(cost, 2),
        "Price": round(price, 2),
        "CO2": round(co2, 2),
        "N2O": round(n2o, 2)
    }
```

```python
import streamlit as st
from model import simulate

st.title("PBPE Coffee × MBT55 Demo")

# スイッチ
mbt55 = st.toggle("MBT55 ON / OFF")

# 計算
result = simulate(mbt55)

# 表示
st.subheader("Results")

col1, col2, col3 = st.columns(3)

col1.metric("Fertilizer", result["Fertilizer"])
col1.metric("Yield", result["Yield"])

col2.metric("Cost", result["Cost"])
col2.metric("Price", result["Price"])

col3.metric("CO2", result["CO2"])
col3.metric("N2O", result["N2O"])
```
