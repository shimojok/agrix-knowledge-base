### 2. 私（Gemini）が行う作業（提供するコード）

```
import os
import re
import json
import requests
from github import Github
from pinecone import Pinecone
from sentence_transformers import SentenceTransformer

# 1. 初期化
GITHUB_TOKEN = os.getenv("GITHUB_TOKEN")
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
INDEX_NAME = "bionexus-planetary-os"

g = Github(GITHUB_TOKEN)
pc = Pinecone(api_key=PINECONE_API_KEY)
index = pc.Index(INDEX_NAME)
model = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')

# 2. リポジトリとNamespaceのマッピング
REPOS_MAP = {
    # 核心理論・数理モデル (ns-core-engine)
    "shimojok/M3-Core-Engine": "ns-core-engine",
    "shimojok/M3-BioSynergy-Core": "ns-core-engine",
    "shimojok/M3-BioSynergy": "ns-core-engine",
    "shimojok/MBT-Biosecurity-Engine": "ns-core-engine",
    
    # 医療・健康・漢方 (ns-healthbook)
    "shimojok/HealthBook-AI": "ns-healthbook",
    "shimojok/HealthBook-MI-Probiome-OS": "ns-healthbook",
    "shimojok/HealthBook-MBT55-Unified": "ns-healthbook",
    "shimojok/HealthBook-Integrated": "ns-healthbook",
    
    # 気候金融・市場・投資 (ns-pbpe-finance)
    "shimojok/PBPE-Finance": "ns-pbpe-finance",
    "shimojok/PBPE-Marketplace": "ns-pbpe-finance",
    "shimojok/PBPE-Dashboard": "ns-pbpe-finance",
    "shimojok/PBPE-Coffee": "ns-pbpe-finance",
    "shimojok/pbpe-mbt55-investment-engine": "ns-pbpe-finance",
    "shimojok/mbt55-climate-overview": "ns-pbpe-finance",
    "shimojok/Azure-PMOS": "ns-pbpe-finance",
    "shimojok/BioNexus-Enterprise": "ns-pbpe-finance",
    
    # 全思考資産・整理済みVault (ns-vault-raw)
    "shimojok/BioNexus-Core": "ns-vault-raw"
}

def process_and_upsert():
    for repo_name, namespace in REPOS_MAP.items():
        print(f"📦 スキャン中: {repo_name} ➔ Namespace: {namespace}")
        try:
            repo = g.get_repo(repo_name)
            contents = repo.get_contents("")
            
            while contents:
                file_content = contents.pop(0)
                if file_content.type == "dir":
                    contents.extend(repo.get_contents(file_content.path))
                    continue
                
                path = file_content.path
                if not any(path.endswith(ext) for ext in ['.md', '.py', '.json']):
                    continue
                
                # ファイル取得＆チャンク化
                raw_data = file_content.decoded_content.decode('utf-8', errors='ignore')
                chunks = create_chunks(raw_data, path, repo_name)
                
                if chunks:
                    texts = [c['text'] for c in chunks]
                    embeddings = model.encode(texts).tolist()
                    
                    vectors = []
                    for idx, chunk in enumerate(chunks):
                        vectors.append({
                            "id": f"{repo_name}#{path}#{idx}",
                            "values": embeddings[idx],
                            "metadata": {
                                "repo": repo_name,
                                "path": path,
                                "text": chunk['text'],
                                "data_type": chunk['type']
                            }
                        })
                    # 指定Namespaceにアップサート
                    index.upsert(vectors=vectors, namespace=namespace)
                    print(f"  ✅ Upserted: {path} ({len(vectors)} chunks)")
                    
        except Exception as e:
            print(f"  ❌ エラー ({repo_name}): {e}")

def create_chunks(content, path, repo_name):
    # (Markdownの見出し割、Pyコード割、JSONパースのロジックを自動適用)
    # ※長文のため要約。エラーハンドリング付きで安全にテキスト抽出します。
    return [{"text": content[:1000], "type": "text"}] # 概略

if __name__ == "__main__":
    process_and_upsert()
```
