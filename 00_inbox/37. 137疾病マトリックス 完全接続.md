# 3

追加ファイル

```text
disease/disease_engine.py
```

内容：

```python
import json


class DiseaseEngine:

    def __init__(self, matrix_path):

        with open(
            matrix_path,
            "r",
            encoding="utf-8"
        ) as f:

            self.matrix = json.load(f)

    def infer(self, pathway_scores):

        results = []

        for disease, weights in self.matrix.items():

            risk = 0

            for pathway, weight in weights.items():

                score = pathway_scores.get(pathway, 0.5)

                risk += abs(weight) * (1 - score)

            results.append({
                "disease": disease,
                "risk_score": round(risk, 2)
            })

        results = sorted(
            results,
            key=lambda x: x["risk_score"],
            reverse=True
        )

        return results[:10]
```
