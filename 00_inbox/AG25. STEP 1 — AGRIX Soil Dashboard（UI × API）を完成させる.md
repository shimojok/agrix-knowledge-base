## 🖥️ **AGRIX Soil Dashboard（完成版）**

### `app/agrrix/soil-dashboard/page.tsx`

tsx

```
'use client';

import { useEffect, useState } from 'react';
import SoilSummaryCard from '../components/SoilSummaryCard';
import SoilTimeSeriesChart from '../components/SoilTimeSeriesChart';

export default function AgrixSoilDashboardPage() {
  const [data, setData] = useState(null);

  useEffect(() => {
    fetch('/api/soil')
      .then((res) => res.json())
      .then(setData)
      .catch(console.error);
  }, []);

  if (!data) return <div>Loading AGRIX Soil Dashboard...</div>;

  const latest = data.snapshots[data.snapshots.length - 1];

  return (
    <main className="p-8 space-y-8">
      <h1 className="text-3xl font-bold">AGRIX Soil Dashboard</h1>
      <p className="text-gray-700">
        MBT55ハイパーサイクル × 土壌センサー × SHE™スコアの統合ビュー
      </p>

      <SoilSummaryCard
        latestShe={latest.she_score}
        latestPhenotype={latest.eco_phenotype}
      />

      <SoilTimeSeriesChart snapshots={data.snapshots} />
    </main>
  );
}
```
