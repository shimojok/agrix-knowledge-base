## 📊 MSC + カーボン統合エンジンの核（Agent-5 + Agent-6 簡易実装例）

```typescript
// lib/agrix/agents/mscOperation.ts
export function runMSCAgent(input: MSCInput): MSCOutput {
  const { hypercycle, humicAcid, fulvicAcid, biomassInput, daysElapsed } = input;

  // 稼働率 = ハイパーサイクル効率 × 腐植化速度 × 処理日数適正度
  const operationalRate = 
    0.4 * hypercycle.cycleEfficiency +
    0.3 * (humicAcid / (humicAcid + fulvicAcid)) +
    0.3 * Math.min(daysElapsed / 90, 1);

  // SOC貢献度（ベースライン2.5%から増加）
  const baseSOC = 2.5; // %
  const annualIncrease = 0.3 * hypercycle.humificationRate + 0.1 * (biomassInput / 10);
  const projectedSOC = baseSOC + annualIncrease * (daysElapsed / 365);

  return {
    mscOperationalRate: Math.min(1, operationalRate),
    socContribution: {
      currentSOC: baseSOC,
      projectedSOC: Number(projectedSOC.toFixed(2)),
      annualIncrease: Number(annualIncrease.toFixed(3)),
      contributionRate: Number((annualIncrease / (projectedSOC - baseSOC + 0.01) * 100).toFixed(1)),
    },
    resourceRecoveryRate: Math.min(1, 0.6 + 0.3 * hypercycle.resourceConversionRate),
    circularityIndex: Math.min(1, (operationalRate + hypercycle.resourceConversionRate) / 2),
    mscPhase: operationalRate > 0.8 ? 'MATURE' : operationalRate > 0.5 ? 'ACCELERATING' : 'INITIATING',
  };
}

// lib/agrix/agents/carbonAsset.ts
export function runCarbonAgent(input: CarbonInput): CarbonOutput {
  const { msc, areaHa, projectYears, scheme } = input;
  const annualTCO2PerHa = msc.socContribution.annualIncrease * 3.67; // SOC → CO2換算
  const totalTCO2 = annualTCO2PerHa * areaHa * projectYears;

  // クレジット発行ポテンシャル（認証スキーム別）
  const priceRef = scheme === 'VERRA' ? 12 : scheme === 'GOLD_STANDARD' ? 15 : 8;
  const issuableCredits = totalTCO2 * 0.9; // 10%バッファー

  return {
    carbonSequestration: {
      totalTCO2: Number(totalTCO2.toFixed(1)),
      annualTCO2PerHa: Number(annualTCO2PerHa.toFixed(2)),
      verificationStatus: 'PENDING',
    },
    creditPotential: {
      issuableCredits: Number(issuableCredits.toFixed(0)),
      estimatedRevenue: Number((issuableCredits * priceRef).toFixed(0)),
      marketPriceReference: priceRef,
      scheme,
    },
    mrvData: {
      baselineEmissions: Number((totalTCO2 * 0.7).toFixed(1)),
      actualSequestration: Number(totalTCO2.toFixed(1)),
      additionality: Number((1 - 0.2 * (projectYears / 10)).toFixed(2)),
      monitoringPeriod: `${projectYears} years`,
    },
  };
}
```
