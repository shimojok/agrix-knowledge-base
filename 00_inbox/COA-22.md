Microbial Dynamics Model — Full Version

def dN_dt(N, S, E, r, K, alpha, lam):
    """

    N : microbial biomass
    S : substrate concentration
    E : enzyme concentration

# ✅ **2_Models/enzyme_cascade.py（完全版）**

```python
"""
Enzyme Cascade Model — Full Version
Multi-step Michaelis–Menten kinetics
"""

import numpy as np

def michaelis_menten(S, Vmax, Km):
    return (Vmax * S) / (Km + S)

def cascade_step(S_list, Vmax_list, Km_list, dt=0.01):
    """
    S_list : list of substrate concentrations [S0, S1, S2, ...]
    Vmax_list, Km_list : kinetic parameters for each step
    """
    dS = np.zeros_like(S_list)

    # Step 0: input substrate decay
    v0 = michaelis_menten(S_list[0], Vmax_list[0], Km_list[0])
    dS[0] -= v0

    # Internal steps
    for i in range(1, len(S_list)):
        v_prev = michaelis_menten(S_list[i-1], Vmax_list[i-1], Km_list[i-1])
        v_curr = michaelis_menten(S_list[i], Vmax_list[i], Km_list[i])
        dS[i] += v_prev - v_curr

    return S_list + dS * dt
```

"""
Nitrogen Flux Model — Full Version
"""

def dN2_dt(E_nit, N, N2, delta, phi, eps):
    """
    E_nit : nitrogenase concentration
    N : microbial biomass
    N2 : nitrogen flux
    delta : nitrogen fixation rate
    phi : microbial nitrogen mobilization
    eps : nitrogen loss rate
    """
    return delta * E_nit + phi * N - eps * N2

def update_nitrogen(E_nit, N, N2, params, dt=0.01):
    delta, phi, eps = params
    return N2 + dN2_dt(E_nit, N, N2, delta, phi, eps) * dt

"""
Hypercycle Integration Model — Full Version
Combines microbial, enzyme, carbon, and nitrogen models
"""

import numpy as np
from microbial_dynamics import update_population
from enzyme_cascade import cascade_step, michaelis_menten
from carbon_flux import update_carbon
from nitrogen_flux import update_nitrogen

def hypercycle_step(state, params, dt=0.01):
    """
    state = {
        'N': microbial biomass,
        'S': substrate list,
        'C': carbon flux,
        'N2': nitrogen flux,
        'E_nit': nitrogenase concentration
    }
    """

    N = state['N']
    S = state['S']
    C = state['C']
    N2 = state['N2']
    E_nit = state['E_nit']

    # Unpack parameters
    micro_p, enzyme_p, carbon_p, nitrogen_p = params

    # Enzyme cascade
    S_new = cascade_step(S, enzyme_p['Vmax'], enzyme_p['Km'], dt)
    v_sum = sum(michaelis_menten(S[i], enzyme_p['Vmax'][i], enzyme_p['Km'][i]) for i in range(len(S)))

    # Microbial update
    N_new = update_population(N, S_new[0], v_sum, micro_p, dt)

    # Carbon update
    C_new = update_carbon(N_new, v_sum, C, carbon_p, dt)

    # Nitrogen update
    N2_new = update_nitrogen(E_nit, N_new, N2, nitrogen_p, dt)

    return {
        'N': N_new,
        'S': S_new,
        'C': C_new,
        'N2': N2_new,
        'E_nit': E_nit
    }
