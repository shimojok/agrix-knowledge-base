```
import os
import re
import json
import time
from github import Github
from pinecone import Pinecone
from sentence_transformers import SentenceTransformer

# ==========================================
# 1. 初期化と環境変数設定
# ==========================================
GITHUB_TOKEN = os.getenv("GITHUB_TOKEN")
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
INDEX_NAME = "bionexus-planetary-os"

if not GITHUB_TOKEN or not PINECONE_API_KEY:
    raise ValueError("❌ GITHUB_TOKEN と PINECONE_API_KEY の環境変数を設定してください。")

g = Github(GITHUB_TOKEN)
pc = Pinecone(api_key=PINECONE_API_KEY)
index = pc.Index(INDEX_NAME)

print("📥 Embeddingモデル (all-MiniLM-L6-v2) をロード中...")
model = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')

# ==========================================
# 2. 全17リポジトリ ➔ Namespace 自動マッピング
# ==========================================
REPOS_MAP = {
    # ── 核心理論・数理モデル（最優先検索ターゲット） ──
    "shimojok/M3-Core-Engine": "ns-core-engine",
    "shimojok/M3-BioSynergy-Core": "ns-core-engine",
    "shimojok/M3-BioSynergy": "ns-core-engine",
    "shimojok/MBT-Biosecurity-Engine": "ns-core-engine",

    "shimojok/HealthBook-MI-Probiome-OS": "ns-healthbook",

    "shimojok/HealthBook-Integrated": "ns-healthbook",
    
    # ── 気候金融・市場・投資・プラットフォーム ──
    "shimojok/PBPE-Finance": "ns-pbpe-finance",
    "shimojok/PBPE-Marketplace": "ns-pbpe-finance",
    "shimojok/PBPE-Dashboard": "ns-pbpe-finance",
    "shimojok/PBPE-Coffee": "ns-pbpe-finance",
    "shimojok/pbpe-mbt55-investment-engine": "ns-pbpe-finance",
    "shimojok/mbt55-climate-overview": "ns-pbpe-finance",
    "shimojok/Azure-PMOS": "ns-pbpe-finance",
    "shimojok/BioNexus-Enterprise": "ns-pbpe-finance",
    
    # ── 整理済みVault (3000弱ノートの思考資産・背景情報) ──
    "shimojok/BioNexus-Core": "ns-vault-raw"
}

# ==========================================
# 3. ファイル解析・チャンク作成ロジック
# ==========================================
def create_chunks_from_content(content: str, path: str, repo_name: str):
    """拡張子に応じてテキストを分解し、メタデータ付きチャンクを生成"""
    ext = os.path.splitext(path)[1].lower()
    base_name = os.path.basename(path)
    chunks = []
    
    # --- A. Markdown / Vaultノート (.md) ---
    if ext == ".md":
        # 見出し単位で分割
        sections = re.split(r'(^#{1,4}\s+.*$)', content, flags=re.MULTILINE)
        for i in range(0, len(sections)-1, 2):
            header = sections[i].strip().replace("#", "").strip()
            text = sections[i+1].strip() if i+1 < len(sections) else ""
            if text:
                chunks.append({
                    "id": f"{base_name}#md-{i}",
                    "text": f"[{header}]\n{text[:1500]}", # 1チャンク上限設定
                    "type": "markdown"
                })
        if not chunks and content.strip():
            chunks.append({"id": f"{base_name}#full", "text": content[:1500], "type": "markdown"})

    # --- B. Pythonコード (.py) ---
    elif ext == ".py":
        chunks.append({
            "id": f"{base_name}#code",
            "text": f"### Python Code from {repo_name}: {base_name} ###\n{content}",
            "type": "python_code"
        })

    # --- C. JSONデータ (.json) ---
    elif ext == ".json":
        chunks.append({
            "id": f"{base_name}#json",
            "text": f"### JSON Data: {base_name} ###\n{content[:2000]}",
            "type": "structured_json"
        })
        
    return chunks

# ==========================================
# 4. メイン同期実行処理
# ==========================================
def main():
    print(f"🚀 全 {len(REPOS_MAP)} リポジトリのPineconeベクトル化を開始します...\n")
    
    total_upserted = 0

    for repo_name, namespace in REPOS_MAP.items():
        print(f"📦 [{repo_name}] ➔ Namespace: [{namespace}] スキャン中...")
        try:
            repo = g.get_repo(repo_name)
            contents = repo.get_contents("")
            
            repo_vectors = []
            
            while contents:
                file_content = contents.pop(0)
                
                # ディレクトリなら再帰的に読み込み
                if file_content.type == "dir":
                    contents.extend(repo.get_contents(file_content.path))
                    continue
                
                path = file_content.path
                # 対象拡張子以外はスキップ
                if not any(path.endswith(ext) for ext in ['.md', '.py', '.json']):
                    continue
                
                try:
                    # ファイル読み込み（文字コードエラー回避）
                    raw_text = file_content.decoded_content.decode('utf-8', errors='ignore')
                    if not raw_text.strip():
                        continue
                        
                    chunks = create_chunks_from_content(raw_text, path, repo_name)
                    
                    if chunks:
                        texts = [c['text'] for c in chunks]
                        embeddings = model.encode(texts).tolist()
                        
                        for idx, chunk in enumerate(chunks):
                            vector_id = f"{repo_name.replace('/', '_')}#{path}#{idx}"
                            repo_vectors.append({
                                "id": vector_id,
                                "values": embeddings[idx],
                                "metadata": {
                                    "repo": repo_name,
                                    "file_path": path,
                                    "text": chunk['text'],
                                    "data_type": chunk['type']
                                }
                            })
                except Exception as fe:
                    print(f"  ⚠️ ファイルスキップ: {path} (エラー: {fe})")
                    continue
            
            # バッチ単位（100件ずつ）でPineconeへアップサート
            if repo_vectors:
                batch_size = 100
                for i in range(0, len(repo_vectors), batch_size):
                    batch = repo_vectors[i:i+batch_size]
                    index.upsert(vectors=batch, namespace=namespace)
                
                print(f"  ✅ アップサート完了: {len(repo_vectors)} チャンク\n")
                total_upserted += len(repo_vectors)
            else:
                print(f"  ℹ️ 対象ファイルなし\n")
                
        except Exception as re:
            print(f"  ❌ リポジトリアクセス失敗: {repo_name} (エラー: {re})\n")
            continue

    print(f"🎉 すべての同期が完了しました！ 合計 {total_upserted} チャンクがPineconeに同期されました。")

if __name__ == "__main__":
    main()
```
