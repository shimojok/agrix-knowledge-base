# 4

追加ファイル

```text
mbt55/strain_engine.py
```

内容：

```python
import json


class MBTStrainEngine:

    def __init__(self, matrix_path):

        with open(
            matrix_path,
            "r",
            encoding="utf-8"
        ) as f:

            self.matrix = json.load(f)

    def recommend(self, target_pathways):

        recommendations = []

        for item in self.matrix:

            overlap = set(
                item["target_pathways"]
            ).intersection(
                set(target_pathways)
            )

            if overlap:

                recommendations.append({
                    "cluster": item["cluster"],
                    "strain": item["strain"],
                    "matched_pathways": list(overlap),
                    "metabolites": item["major_metabolites"]
                })

        return recommendations
```
