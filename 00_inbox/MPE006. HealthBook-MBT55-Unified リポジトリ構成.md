```python
# dashboard/i18n/translator.py
class I18nManager:
    """日英統合翻訳マネージャー"""
    
    SUPPORTED_LANGUAGES = {
        "ja": "日本語",
        "en": "English"
    }
    
    def __init__(self, lang: str = "ja"):
        self.lang = lang
        self.translations = self._load_translations()
    
    def t(self, key: str) -> str:
        """翻訳キーから表示テキストを取得"""
        return self.translations.get(key, key)
    
    def _load_translations(self):
        """言語辞書を動的ロード"""
        if self.lang == "ja":
            from dashboard.i18n.ja import DICT
        else:
            from dashboard.i18n.en import DICT
        return DICT
```
