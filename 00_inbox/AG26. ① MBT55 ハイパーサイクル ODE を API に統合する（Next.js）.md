# ① **MBT55 ハイパーサイクル ODE → Next.js API Routes 統合**

あなたのドキュメント **「01_soil_dynamics.md」** に記載されている ODE モデルを Next.js API に組み込み、 **土壌センサーデータ（pH / Eh / 水分）を入力すると、MBT55ハイパーサイクルの状態量（S, X_A, M, X_B, P, X_C, C, N）を返す API** を作る。

## 📁 API Route

`app/api/mbt55/hypercycle/route.ts`

ts

```
import { NextRequest, NextResponse } from "next/server";

// --- MBT55 Hypercycle ODE Parameters ---
const params = {
  k1: 0.42,   // S → X_A
  k2: 0.31,   // X_A → M
  k3: 0.28,   // M → X_B
  k4: 0.22,   // X_B → P
  k5: 0.18,   // P → X_C
  kC: 0.12,   // Carbon fixation
  kN: 0.15,   // Nitrogen activation
};

// --- Aerobic fraction (from DA61 soil_dynamics.md) ---
function aerobicFraction(eh_mV: number, moisture_pct: number) {
  const fEh = 1 / (1 + Math.exp(-0.02 * (eh_mV - 50)));
  const fMoist = 1 - Math.max(0, (moisture_pct - 70) / 30);
  return Math.min(0.95, Math.max(0.05, fEh * fMoist));
}

// --- ODE Simulation (Euler method, 24h) ---
function simulateHypercycle(input: any) {
  let { S, X_A, M, X_B, P, X_C, C, N } = input;

  const dt = 0.5; // 30 min step
  const steps = 48; // 24h

  const aerobic = aerobicFraction(input.eh_mV, input.moisture_pct);

  for (let i = 0; i < steps; i++) {
    const dS = -params.k1 * S * aerobic;
    const dXA = params.k1 * S - params.k2 * X_A;
    const dM = params.k2 * X_A - params.k3 * M;
    const dXB = params.k3 * M - params.k4 * X_B;
    const dP = params.k4 * X_B - params.k5 * P;
    const dXC = params.k5 * P;
    const dC = params.kC * P * aerobic;
    const dN = params.kN * X_A * (1 - aerobic);

    S += dS * dt;
    X_A += dXA * dt;
    M += dM * dt;
    X_B += dXB * dt;
    P += dP * dt;
    X_C += dXC * dt;
    C += dC * dt;
    N += dN * dt;
  }

  return { S, X_A, M, X_B, P, X_C, C, N, aerobic };
}

export async function POST(req: NextRequest) {
  const body = await req.json();

  const initial = {
    S: body.S ?? 10,
    X_A: body.X_A ?? 1,
    M: body.M ?? 0.5,
    X_B: body.X_B ?? 0.2,
    P: body.P ?? 0.1,
    X_C: body.X_C ?? 0.05,
    C: body.C ?? 2,
    N: body.N ?? 1,
    eh_mV: body.eh_mV,
    moisture_pct: body.moisture_pct,
  };

  const result = simulateHypercycle(initial);

  return NextResponse.json({
    timestamp: new Date().toISOString(),
    ...result,
  });
}
```

# ② **SafetyChain™ Certificate の QR コード化（Next.js）**

SafetyChain™ Certificate は消費者の選択基準なので、 **QRコードでスマホから即アクセスできる形が必須。**

## 📁 API Route

`app/api/safetychain/certificate/route.ts`

ts

```
import { NextRequest, NextResponse } from "next/server";
import QRCode from "qrcode";

export async function POST(req: NextRequest) {
  const body = await req.json();

  const certificate = {
    lotId: body.lotId,
    vScore: body.vScore,
    functionalityIndex: body.functionalityIndex,
    metabolicLifeDays: body.metabolicLifeDays,
    polyphenolDensity: body.polyphenolDensity,
    issuedAt: new Date().toISOString(),
  };

  const qrPayload = JSON.stringify(certificate);

  const qrDataUrl = await QRCode.toDataURL(qrPayload);

  return NextResponse.json({
    certificate,
    qrCode: qrDataUrl,
  });
}
```

`app/agrrix/certificate/page.tsx`

tsx

```
'use client';

import { useState } from 'react';

export default function CertificatePage() {
  const [cert, setCert] = useState(null);

  async function generate() {
    const res = await fetch('/api/safetychain/certificate', {
      method: 'POST',
      body: JSON.stringify({
        lotId: 'LOT-20260709-001',
        vScore: 88,
        functionalityIndex: 91,
        metabolicLifeDays: 12,
        polyphenolDensity: 0.81
      })
    });

    const json = await res.json();
    setCert(json);
  }

  return (
    <main className="p-8 space-y-6">
      <h1 className="text-2xl font-bold">SafetyChain™ Certificate</h1>

      <button
        onClick={generate}
        className="px-4 py-2 bg-purple-600 text-white rounded"
      >
        証明書を生成
      </button>

      {cert && (
        <div className="space-y-4">
          <pre className="bg-gray-100 p-4 rounded">
            {JSON.stringify(cert.certificate, null, 2)}
          </pre>

          <img src={cert.qrCode} alt="SafetyChain QR Code" className="w-48" />
        </div>
      )}
    </main>
  );
}
```
