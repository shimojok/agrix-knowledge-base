# 3

追加ファイル

```text
metabolic_graph/graph_engine.py
```

内容：

```python
import json
import networkx as nx


class MetabolicGraphEngine:

    def __init__(self, graph_path):

        with open(
            graph_path,
            "r",
            encoding="utf-8"
        ) as f:

            self.edges = json.load(f)

        self.graph = nx.DiGraph()

        for edge in self.edges:

            self.graph.add_edge(
                edge["source"],
                edge["target"],
                edge_type=edge["type"]
            )

    def get_neighbors(self, node):

        if node not in self.graph:
            return []

        return list(
            self.graph.successors(node)
        )

    def get_path(self, source, target):

        try:

            return nx.shortest_path(
                self.graph,
                source,
                target
            )

        except:

            return []
```

```python
import streamlit as st
import networkx as nx
import matplotlib.pyplot as plt

G = nx.DiGraph()

edges = [
    ("Puerarin", "Daidzein"),
    ("Daidzein", "Equol"),
    ("Fiber", "Butyrate"),
    ("Glutamate", "GABA"),
    ("Collagen", "Collagen Peptides")
]

G.add_edges_from(edges)

fig, ax = plt.subplots(figsize=(10, 6))

nx.draw(
    G,
    with_labels=True,
    node_size=3000,
    font_size=9,
    ax=ax
)

st.title("Global Metabolic Graph")

st.pyplot(fig)
```
