### 5. PBPE Value 計算ページ

bash

```
mkdir -p src/pages/marketplace
touch src/pages/marketplace/value.tsx
```

tsx

```
import { useState } from "react";
import AppLayout from "../../components/layout/AppLayout";
import PBPEScoreCard from "../../components/marketplace/PBPEScoreCard";

export default function PBPEValuePage() {
  const [score, setScore] = useState<number | null>(null);

  async function calcPBPE() {
    const res = await fetch("http://localhost:8000/pbpe/value/calc", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        project_id: "UI-TEST",
        domain: "agriculture",
        country: "JP",
        time_horizon_years: 10,
        inputs: { financial: {}, social: {}, ecological: {}, governance: {} },
        weights: { financial: 0.3, social: 0.3, ecological: 0.25, governance: 0.15 }
      })
    });
    const data = await res.json();
    setScore(data.pbpe_score);
  }

  return (
    <AppLayout>
      <h2>PBPE Value Calculator</h2>
      <button onClick={calcPBPE}>Calculate PBPE</button>
      {score !== null && <PBPEScoreCard score={score} />}
    </AppLayout>
  );
}
```
