# ② 統合コード（そのまま動く版）

## ファイル：`healthbook_integrated_model.py`

```python
import numpy as np
from scipy.integrate import odeint

# -----------------------------
# ① パラメータ
# -----------------------------
params = {
    "k1": 0.3,  # lactate consumption
    "k2": 0.5,  # lactate -> butyrate
    "k3": 0.2,  # butyrate decay
    "k4": 0.1,  # acetate decay
    "alpha": 0.4,  # diversity growth
    "beta": 0.3,   # inflammation effect
    "gamma1": 0.5, # dysbiosis -> inflammation
    "gamma2": 0.6, # butyrate anti-inflammatory
    "delta1": 0.4, # inflammation -> cancer risk
    "delta2": 0.5, # butyrate suppression
    "delta3": 0.3  # diversity protection
}

# -----------------------------
# ② ODE
# -----------------------------
def hypercycle_model(y, t, p):

    L, B, A, H, M, I, C = y

    # 分解・変換
    dL = 1.0 - p["k1"] * L * M
    dB = p["k2"] * L * M - p["k3"] * B
    dA = 0.8 - p["k4"] * A

    # 電子
    dH = 0.3 * (L + A) - 0.2 * H

    # 多様性
    dM = p["alpha"] * (B + A) - p["beta"] * I

    # 炎症
    dI = p["gamma1"] * (1 - M) - p["gamma2"] * B

    # 腫瘍リスク
    dC = p["delta1"] * I - p["delta2"] * B - p["delta3"] * M

    return [dL, dB, dA, dH, dM, dI, dC]

# -----------------------------
# ③ 実行関数
# -----------------------------
def run_simulation(initial_state, days=50):

    t = np.linspace(0, days, 300)

    result = odeint(hypercycle_model, initial_state, t, args=(params,))

    return t, result

# -----------------------------
# ④ テスト実行
# -----------------------------
if __name__ == "__main__":

    initial = [1, 0.5, 0.5, 0.2, 0.6, 0.4, 0.3]

    t, res = run_simulation(initial)

    print("Simulation complete")
```
