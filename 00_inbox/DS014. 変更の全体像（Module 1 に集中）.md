// 変更後：科学モデルを呼び出すラッパー
import { exec } from 'child_process';
import { readFileSync } from 'fs';

export function runHypercycleEngine(input: HypercycleInput): HypercycleOutput {
  // 1. 入力を Data_Model.json 形式に変換
  const payload = convertToDataModel(input);

  const result = runPythonModel('Microbial_Simulation.py', payload);

  return convertFromDataModel(result);
}

function runPythonModel(modelName: string, input: any): any {
  // Pythonスクリプトを子プロセスで実行し、結果をJSONで受け取る
  const output = execSync(`python3 09_Code_Models/${modelName} --input '${JSON.stringify(input)}'`);
  return JSON.parse(output.toString());
}

import { callPythonModel } from '@/lib/bridge/python-bridge';

export async function runHypercycleEngine(input: HypercycleInput): Promise<HypercycleOutput> {
  // 1. 入力をData_Model.json準拠に変換（約10行）
  const modelInput = {

      moisture: input.moisture_pct,

    },

      decomposerActivity: input.decomposerActivity,
      // ...
    }
