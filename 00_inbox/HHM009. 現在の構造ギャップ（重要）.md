# ⑤ Python統合（最小接続実装）

```python
# --- Claudeコード（仮インポート） ---
def healthbook_model(a):
    w = np.random.uniform(0.5,1.0,len(a))
    delta = np.random.randint(0,2,(len(a),5))
    return (w * a) @ delta  # H_HB

# --- 統合モデル ---
def integrated_model(t, y, a):

    # ODE状態
    H_metabolic = y[-2:]

    # HealthBook出力
    H_HB = healthbook_model(a)

    # 統合
    H_total = 0.6 * H_metabolic + 0.4 * H_HB

    # 疾病リスク
    beta = np.array([1.2, -1.0])
    theta = 0.5

    R = 1/(1 + np.exp(-(beta @ H_total - theta)))

    return R
```
