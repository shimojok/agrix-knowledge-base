## エージェントの基本構造（base_agent.py のイメージ）
```python
class BaseAgent:
    def __init__(self, name, namespaces):
        self.name = name
        self.namespaces = namespaces
        self.pinecone_index = Pinecone(api_key="...").Index("agrix-knowledge-base")
    
    def search_knowledge(self, query, top_k=5):
        """Pineconeから関連知識を検索"""
        embedding = openai.embeddings.create(model="text-embedding-3-small", input=query)
        results = self.pinecone_index.query(
            vector=embedding.data[0].embedding,
            namespace=self.namespaces,
            top_k=top_k,
            include_metadata=True
        )
        return [m.metadata["text"] for m in results.matches]
    
    def think(self, task, context):
        """LLMによる推論（Pineconeの知識を参照）"""
        knowledge = self.search_knowledge(task)
        prompt = f"""
        あなたは{self.name}です。
        参照知識: {knowledge}
        コンテキスト: {context}
        タスク: {task}
        上記の知識に基づいて回答してください。
        """
        response = openai.chat.completions.create(model="gpt-4o", messages=[{"role":"user","content":prompt}])
        return response.choices[0].message.content
    
    def act(self, task, context):
        """タスクを実行し、結果を返す"""
        return self.think(task, context)
```

## 成果物

- Pineconeの知識を参照するため、ハルシネーションが抑制される
- 単独でも、連携しても動作する設計
