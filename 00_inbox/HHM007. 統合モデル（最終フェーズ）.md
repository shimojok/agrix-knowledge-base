# ② 数式（Integrated Global Health System Model）

## (A) 外部系：MBT55（分解・環境）

[  
\frac{dS}{dt} = - \sum_i k_i M_i^{env} S \cdot f_T(T) \cdot f_{pH}(pH)  
]

[  
\frac{dM_i^{env}}{dt} =  
M_i^{env} \left( \mu_i \frac{S}{K_{S,i} + S} + \sum_j \alpha_{ij} M_j^{env} - d_i \right)  
]

---

## (B) 内部系：腸内代謝

[  
\frac{dM_i^{gut}}{dt} =  
M_i^{gut} \left( \mu_i^{gut} \frac{N}{K_{N,i} + N} + \sum_j \alpha_{ij}^{gut} M_j^{gut} - d_i^{gut} \right)

- \kappa_i Pro_i  
    ]
    

---

## (C) 代謝カスケード

[  
\frac{dP_k}{dt} =  
\sum_i Y_{ik} M_i^{gut} + \psi_{ak} F_a - \lambda_k P_k  
]

---

## (D) フェノタイプ

[  
\frac{dH_l}{dt} =  
\sum_k w_{kl} P_k - \delta_l H_l  
]

---

## (E) 疾病リスク

[  
R_d =  
\frac{1}{1 + \exp\left(-(\sum_l \beta_{ld} H_l - \theta_d)\right)}  
]

---

## (F) GHG排出

[  
GHG =  
CO_2 + 28CH_4 + 265N_2O  
]

[  
CH_4 \propto (1-O_2)\sum_i k_i M_i^{env} S  
]

---

## (G) 経済モデル（PBPE）

[  
CF_t = Revenue_t + (\Delta GHG \cdot Price_{CO2}) - Cost_t  
]

[  
NPV = \sum \frac{CF_t}{(1+r)^t}  
]

# ④ Python実装（統合ODE）

```python
import numpy as np
from scipy.integrate import solve_ivp

# 状態ベクトル構造
# [S, M_env(3), M_gut(3), P(3), H(2)]

def model(t, y):

    S = y[0]
    M_env = np.array(y[1:4])
    M_gut = np.array(y[4:7])
    P = np.array(y[7:10])
    H = np.array(y[10:12])

    # --- 外部（MBT55） ---
    k = np.array([0.3,0.4,0.2])
    dSdt = -np.sum(k * M_env * S)

    dM_env = M_env * (0.5 * S/(1+S) - 0.1)

    # --- 内部（腸内） ---
    N = 0.1 * S + 0.05 * np.sum(P)

    dM_gut = M_gut * (0.4 * N/(1+N) - 0.1) + np.array([0.01,0,0])

    # --- 代謝 ---
    dP = 0.5 * M_gut - 0.1 * P

    # --- フェノタイプ ---
    dH = np.array([1.0, -0.5]) @ P - 0.1 * H

    return [dSdt] + list(dM_env) + list(dM_gut) + list(dP) + list(dH)

# 初期値
y0 = [50] + [1,1,1] + [0.3,0.4,0.3] + [10,10,10] + [1,1]

sol = solve_ivp(model, (0,50), y0, t_eval=np.linspace(0,50,200))

# --- GHG計算 ---
def ghg(S, M_env):
    return 0.5*np.sum(M_env*S) + 28*0.2*np.sum(M_env*S)

GHG = ghg(sol.y[0,-1], sol.y[1:4,-1])

# --- 経済 ---
carbon_price = 50
credit = GHG * carbon_price

print("GHG:", GHG)
print("Carbon Credit:", credit)
```
