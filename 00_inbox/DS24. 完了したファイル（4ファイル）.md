git clone https://github.com/Bionexus-Holdings/M3-BioSynergy.git
cd M3-BioSynergy

python -m venv venv

venv\Scripts\activate
# アクティベート（Mac/Linux）
source venv/bin/activate

pip install -r requirements.txt

# 実行
python examples/basic_simulation.py
