## 🐍 フォルダ構造から自動でメタデータを抽出するPythonスクリプト

上記の思想に基づき、Kazさんの手動作業をゼロにし、OneDrive上のフォルダ構成（`02_agrix`, `06_healthbook` など）を自動認識してPineconeへ「完全無料」で格納する修正版スクリプトです。

`scripts/sync_obsidian_to_pinecone.py` として配置してください。

Python

```
import os
import re
import glob
import json
from typing import List, Dict, Any
from pinecone import Pinecone
from sentence_transformers import SentenceTransformer

# ==========================================
# 1. 環境変数とモデルの初期化（完全無料・ローカル計算）
# ==========================================
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
PINECONE_INDEX_NAME = os.getenv("PINECONE_INDEX_NAME", "bionexus-hypercycle")
VAULT_DIR = "./"  # リポジトリのルート（BioNexus_vault直下）を想定

print("📥 埋め込みモデル（無料・ローカル実行）をロード中...")
embedding_model = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')

pc = Pinecone(api_key=PINECONE_API_KEY)
index = pc.Index(PINECONE_INDEX_NAME)

# ==========================================
# 2. フォルダ構造から守備範囲（メタデータ）を自動判別
# ==========================================
def determine_metadata_by_path(file_path: str) -> Dict[str, Any]:
    """ファイルパスからAGRIX Platformの守備範囲やモジュールを自動判別する"""
    normalized_path = file_path.replace("\\", "/")
    parts = normalized_path.split("/")
    
    # デフォルト値
    scope = "unknown"
    layer = "general_knowledge"
    folder_name = "other"

    # 親フォルダ名を探す（例: 02_agrix, 04_pbpe-finance, 06_healthbook など）
    for part in parts:
        if "01_mbt55" in part:
            scope = "MBT55_Core"
            layer = "ecosystem_infrastructure"
            folder_name = part
        elif "02_agrix" in part:
            scope = "AGRIX"
            layer = "production_infrastructure"
            folder_name = part
        elif "04_pbpe" in part:
            scope = "PBPE-Finance"
            layer = "climate_finance"
            folder_name = part
        elif "06_healthbook" in part or "07_mbt_kampo" in part or "08_MBT" in part:
            scope = "HealthBook"
            layer = "medical_metabolism"
            folder_name = part

    return {
        "target_system": scope,
        "platform_layer": layer,
        "source_folder": folder_name,
        "source_file": os.path.basename(file_path)
    }

# ==========================================
# 3. Markdownパース & チャンキング（JSON内包対応）
# ==========================================
def parse_markdown_file(file_path: str) -> List[Dict[str, Any]]:
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()

    # フロントマター（ある場合のみ、なければスキップ）を剥がす
    body = content
    if content.startswith("---"):
        match = re.match(r'^---\s*\n(.*?)\n---\s*\n', content, re.DOTALL)
        if match:
            body = content[match.end():]

    # パスからベースとなるメタデータを自動取得
    base_metadata = determine_metadata_by_path(file_path)
    processed_chunks = []
    chunk_idx = 0

    # ----------------------------------------------------
    # A. ノート内に書かれているJSONブロックの自動抽出
    # ----------------------------------------------------
    json_blocks = re.findall(r'```json\s*\n(.*?)\n```', body, re.DOTALL)
    for block in json_blocks:
        try:
            # JSONが途中で切れている場合の簡易保護
            json_data = json.loads(block.strip())
            disease_id = json_data.get("disease_id", json_data.get("id", "unknown"))
            
            metadata = base_metadata.copy()
            metadata.update({
                "data_type": "structured_json",
                "entity_id": disease_id,
                "chunk_index": chunk_idx
            })
            
            processed_chunks.append({
                "id": f"{base_metadata['source_file']}#json-{chunk_idx}",
                "text": block.strip(),
                "metadata": metadata
            })
            chunk_idx += 1
        except json.JSONDecodeError:
            # 途中で切れている不完全なJSONは通常のテキストとして処理するためスキップ
            continue

    # ----------------------------------------------------
    # B. 通常テキストセクション（見出し単位のチャンキング）
    # ----------------------------------------------------
    clean_body = re.sub(r'```json\s*\n(.*?)\n```', '', body, flags=re.DOTALL)
    sections = re.split(r'(^#{1,4}\s+.*$)', clean_body, flags=re.MULTILINE)
    
    current_header = "概要"
    if sections[0].strip():
        sections.insert(0, "# 概要\n")

    for i in range(0, len(sections)-1, 2):
        header = sections[i].strip().replace("#", "").strip()
        text = sections[i+1].strip() if i+1 < len(sections) else ""
        
        if text:
            metadata = base_metadata.copy()
            metadata.update({
                "data_type": "unstructured_text",
                "section_header": header,
                "chunk_index": chunk_idx
            })
            processed_chunks.append({
                "id": f"{base_metadata['source_file']}#text-{chunk_idx}",
                "text": f"[{header}]\n{text}",
                "metadata": metadata
            })
            chunk_idx += 1

    return processed_chunks

# ==========================================
# 4. メイン同期処理
# ==========================================
def sync_vault():
    print("🚀 BioNexus_vault フォルダ自動スキャンを開始します...")
    md_files = glob.glob(os.path.join(VAULT_DIR, "**/*.md"), recursive=True)
    
    for file_path in md_files:
        # .obsidian や .trash などの隠しフォルダはスキップ
        if ".obsidian" in file_path or ".trash" in file_path:
            continue
            
        chunks = parse_markdown_file(file_path)
        if not chunks:
            continue
            
        print(f"📄 処理中: {file_path}")
        texts = [c["text"] for c in chunks]
        embeddings = embedding_model.encode(texts).tolist()
        
        vectors_to_upsert = []
        for idx, chunk in enumerate(chunks):
            pinecone_metadata = chunk["metadata"]
            pinecone_metadata["text"] = chunk["text"]  # 本文を復元用に格納
            
            vectors_to_upsert.append({
                "id": chunk["id"],
                "values": embeddings[idx],
                "metadata": pinecone_metadata
            })
            
        if vectors_to_upsert:
            index.upsert(vectors=vectors_to_upsert)

if __name__ == "__main__":
    sync_vault()
```
