**2. スクリプトファイルを作成**（VS Codeで新規ファイルを作成し、`upload_to_pinecone.py` という名前で保存）：

```python
import os
from pinecone import Pinecone
from openai import OpenAI
from dotenv import load_dotenv
import glob

# 環境変数を読み込む（.envファイルがあれば）
load_dotenv()

# ----- 1. 設定 -----
# あなたのObsidian Vaultのパス（ここをあなたのパスに置き換えてください）
VAULT_PATH = r"C:\Users\user\OneDrive\BioNexus_vault"

# Pineconeの設定（あなたの情報を入力）
PINECONE_API_KEY = "あなたのAPIキー"  # ← ステップ1でコピーしたものを貼り付け
PINECONE_INDEX_NAME = "agrix-knowledge-base"
OPENAI_API_KEY = "あなたのOpenAI APIキー"  # ← 持っていなければ取得が必要

# ----- 2. 接続 -----
pc = Pinecone(api_key=PINECONE_API_KEY)
index = pc.Index(PINECONE_INDEX_NAME)

# OpenAIクライアント（埋め込み生成用）
openai_client = OpenAI(api_key=OPENAI_API_KEY)

# ----- 3. Obsidian Vaultから.mdファイルを読み込む -----
md_files = glob.glob(os.path.join(VAULT_PATH, "**/*.md"), recursive=True)
print(f"📁 {len(md_files)}個のMarkdownファイルが見つかりました。")

# ----- 4. 各ファイルを処理してPineconeにアップロード -----
for i, file_path in enumerate(md_files):
    try:
        # ファイルを読む
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 空のファイルはスキップ
        if len(content.strip()) < 50:
            continue
        
        # 埋め込みベクトルを生成（OpenAI API）
        response = openai_client.embeddings.create(
            model="text-embedding-3-small",
            input=content[:8000]  # 制限に引っかからないよう最初の8000文字だけ使う
        )
        vector = response.data[0].embedding
        
        # メタデータ（情報のタグ）を作成
        file_name = os.path.basename(file_path)
        folder_name = os.path.basename(os.path.dirname(file_path))
        
        # Pineconeにアップロード（IDとベクトル、メタデータをセットで）
        index.upsert(vectors=[(
            f"file_{i}",  # 一意なID
            vector,       # ベクトル
            {
                "filename": file_name,
                "folder": folder_name,
                "full_path": file_path,
                "content": content[:500]  # 検索結果表示用に最初の500文字
            }
        )])
        
        # 進捗を表示（100ファイルごと）
        if (i + 1) % 100 == 0:
            print(f"✅ {i+1}/{len(md_files)} ファイルをアップロードしました。")
            
    except Exception as e:
        print(f"⚠️ エラー: {file_path} - {e}")

print("🎉 全てのアップロードが完了しました！")
```

**3. スクリプトを実行**（ターミナルで）：
```bash
python upload_to_pinecone.py
```
