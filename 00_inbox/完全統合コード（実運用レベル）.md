## ■ コード（そのまま使用可）

```python
import numpy as np

# =========================
# 1. MBT55 モジュール
# =========================
class MBT55Engine:
    def __init__(self, decomposition_eff=0.95):
        self.decomposition_eff = decomposition_eff

    def process(self, waste):
        decomposed = waste * self.decomposition_eff
        residue = waste - decomposed
        return {
            "decomposed": decomposed,
            "residue": residue
        }


# =========================
# 2. GHG モジュール
# =========================
class GHGModel:
    def __init__(self, ef_conv=1.5, ef_mbt=0.5):
        self.ef_conv = ef_conv
        self.ef_mbt = ef_mbt

    def calculate(self, waste, adoption):
        base = waste * self.ef_conv
        mbt = waste * self.ef_mbt
        reduction = (base - mbt) * adoption

        return {
            "baseline": base,
            "after_mbt": mbt,
            "reduction": reduction
        }


# =========================
# 3. PBPE（金融モデル）
# =========================
class PBPEModel:
    def __init__(self, carbon_price=50, cost_reduction_rate=0.84):
        self.carbon_price = carbon_price
        self.cost_reduction_rate = cost_reduction_rate

    def evaluate(self, waste, ghg_reduction):
        carbon_value = ghg_reduction * self.carbon_price
        cost_saving = waste * self.cost_reduction_rate

        return {
            "carbon_value": carbon_value,
            "cost_saving": cost_saving,
            "total_value": carbon_value + cost_saving
        }


# =========================
# 4. 統合モデル
# =========================
class BioNexusModel:
    def __init__(self):
        self.mbt = MBT55Engine()
        self.ghg = GHGModel()
        self.pbpe = PBPEModel()

    def run(self, waste, adoption):
        mbt_result = self.mbt.process(waste)
        ghg_result = self.ghg.calculate(waste, adoption)
        finance_result = self.pbpe.evaluate(waste, ghg_result["reduction"])

        return {
            "MBT55": mbt_result,
            "GHG": ghg_result,
            "Finance": finance_result
        }


# =========================
# 5. CLI実行
# =========================
if __name__ == "__main__":
    model = BioNexusModel()

    waste = 1e6        # 100万トン
    adoption = 0.3     # 30%

    result = model.run(waste, adoption)

    print("\n=== BioNexus Simulation ===")
    for k, v in result.items():
        print(f"\n[{k}]")
        for kk, vv in v.items():
            print(f"{kk}: {vv}")
```

## ■ コード

```python
import streamlit as st
from bionexus_master_model import BioNexusModel

st.title("BioNexus Integrated Simulation")

model = BioNexusModel()

# 入力
waste = st.slider("Waste (tons)", 0, 10000000, 1000000)
adoption = st.slider("Adoption Rate", 0.0, 1.0, 0.3)

# 実行
result = model.run(waste, adoption)

# 出力
st.header("Results")

st.subheader("GHG")
st.metric("Reduction (tCO2e)", int(result["GHG"]["reduction"]))

st.subheader("Finance")
st.metric("Carbon Value ($)", int(result["Finance"]["carbon_value"]))
st.metric("Cost Saving ($)", int(result["Finance"]["cost_saving"]))

st.subheader("MBT55")
st.metric("Decomposed", int(result["MBT55"]["decomposed"]))
```
