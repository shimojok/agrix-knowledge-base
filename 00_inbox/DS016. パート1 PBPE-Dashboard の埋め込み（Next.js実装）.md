## 🧬 パート2: Module 1 を科学モデル呼び出しラッパーに変更

### ① Pythonブリッジ（子プロセス実行）を作成

```typescript
// lib/bridge/python-bridge.ts

import { execSync } from 'child_process';
import { existsSync } from 'fs';
import { join } from 'path';

// MBT-Biosecurity-Engine のリポジトリパス（環境変数で指定可）
const ENGINE_ROOT = process.env.MBT_ENGINE_PATH || '../MBT-Biosecurity-Engine';

export interface PythonModelInput {
  [key: string]: any;
}

export interface PythonModelOutput {
  [key: string]: any;
}

/**
 * Python科学モデルを呼び出す
 * @param modelName - モデルファイル名（例: 'Microbial_Simulation'）
 * @param input - 入力データ（Data_Model.json準拠）
 * @returns モデル出力（JSON）
 */
export function callPythonModel(modelName: string, input: PythonModelInput): PythonModelOutput {
  const modelPath = join(ENGINE_ROOT, '09_Code_Models', `${modelName}.py`);
  
  if (!existsSync(modelPath)) {
    throw new Error(`Python model not found: ${modelPath}`);
  }

  try {
    // Pythonスクリプトを子プロセスで実行
    const inputJson = JSON.stringify(input);
    const command = `python3 "${modelPath}" --input '${inputJson}'`;
    
    const output = execSync(command, {
      encoding: 'utf-8',
      maxBuffer: 1024 * 1024 * 10, // 10MB
    });

    // PythonスクリプトがJSONを出力する想定
    return JSON.parse(output);
  } catch (error: any) {
    console.error(`Python model execution failed: ${modelName}`, error.message);
    throw new Error(`Python model error: ${error.message}`);
  }
}

/**
 * 複数のPythonモデルを並列実行
 */
export async function callPythonModelsParallel(
  modelCalls: { name: string; input: PythonModelInput }[]
): Promise<PythonModelOutput[]> {
  // 同期的な実装だが、必要に応じてPromise.allに変更可能
  return modelCalls.map(({ name, input }) => callPythonModel(name, input));
}
```

### ③ 変換アダプター（AGRIX入出力 ↔ DataModel）

```typescript
// lib/agrix/module1/adapters/data-model-adapter.ts

import { SoilSensorInput } from '../types/mbt55.types';
import { HypercycleInput } from '../types/hypercycle.types';
import { DataModelInput, DataModelOutput } from '../types/data-model.types';
import { HypercycleOutput } from '../types/hypercycle.types';

/**
 * AGRIX入力 → DataModel入力 に変換
 */
export function toDataModelInput(
  input: HypercycleInput | SoilSensorInput,
  extra?: Partial<DataModelInput>
): DataModelInput {
  const base: DataModelInput = {
    soil: {
      pH: input.pH,
      temperature_C: input.temperature_c,
      moisture_percent: input.moisture_pct,
      Eh_mV: input.eh_mV,
      EC_dSm: (input as any).ec_dSm || 1.0,
      organicCarbon_gL: input.organicCarbon_gL,
      availableNitrogen_gL: input.availableNitrogen_gL,
    },
    mbt55: {
      dosage_Lha: input.mbt55Dosage_Lha,
      applicationDays: input.mbt55ApplicationDays,
      frequency: (input as any).mbt55Frequency || 2,
    },
    biomass: {
      input_tPerHa: (input as any).biomassInput_tPerHa || 0,
      type: 'MIXED',
    },
  };

  return { ...base, ...extra };
}

/**
 * DataModel出力 → AGRIX HypercycleOutput に変換
 */
export function fromDataModelOutput(output: DataModelOutput): HypercycleOutput {
  return {
    cyclePhase: output.hypercycle.phase,
    cycleEfficiency: output.hypercycle.efficiency,
    humificationRate: output.carbon.humificationRate,
    stableCarbonFraction: output.carbon.stableCarbonFraction,
    resourceConversionRate: output.decomposers.total * 0.7 + output.hypercycle.efficiency * 0.3,
    cnShift: output.hypercycle.cnShift,
    currentCNRatio: output.hypercycle.currentCN,
    humicAcidRatio: output.carbon.humificationRate * 0.6 + 0.3,
    statusMessage: output.messages.status,
    recommendations: output.messages.recommendations,
    calculatedAt: new Date().toISOString(),
  };
}
```
