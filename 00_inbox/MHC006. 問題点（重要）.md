```python
import numpy as np
from scipy.integrate import solve_ivp
import matplotlib.pyplot as plt

def model(t,y):
    Cp, Cm, Cs, E, H2, Na, No, Sr = y

    Xf = 0.5*Cm/(1+Cm)
    Xa = 0.5*Cs/(1+Cs)
    Xn = 0.3*Na
    Xs = 0.2*Sr
    Xm = 0.2*H2

    dCp = -0.3*Xf*Cp
    dCm = 0.3*Xf*Cp - 0.5*Xf*Cm - 0.4*Xa*Cm
    dCs = 0.6*0.5*Xf*Cm - 0.2*Xa*Cs

    dE = 0.5*0.5*Xf*Cm - 0.4*E*(1+No+Sr)
    dH2 = 0.3*E - 0.5*(Xm+Xs)*H2

    dNa = 0.2*Cm - 0.3*Xn*Na
    dNo = 0.3*Xn*Na - 0.1*No

    dSr = -0.2*Xs*Sr + 0.1*H2

    return [dCp,dCm,dCs,dE,dH2,dNa,dNo,dSr]

y0 = [10,2,1,1,0.5,1,1,1]

sol = solve_ivp(model,(0,48),y0,t_eval=np.linspace(0,48,300))

plt.plot(sol.t, sol.y[0], label="Cp")
plt.plot(sol.t, sol.y[2], label="Cs")
plt.plot(sol.t, sol.y[4], label="H2")
plt.legend()
plt.title("Currency Hypercycle")
plt.show()
```
