```python
PATHWAY_DISEASE_MAP = {
    "PATH_00": [
        "Malabsorption",
        "Frailty"
    ],
    "PATH_01": [
        "Menopause Syndrome",
        "Vascular Dysfunction"
    ],
    "PATH_02": [
        "Chronic Inflammation",
        "Metabolic Syndrome"
    ],
    "PATH_03": [
        "Dyslipidemia",
        "Fatty Liver"
    ],
    "PATH_04": [
        "Sleep Disorder",
        "Anxiety"
    ],
    "PATH_05": [
        "IBD",
        "Constipation"
    ],
    "PATH_06": [
        "Sarcopenia",
        "Circulatory Dysfunction"
    ]
}


def infer_disease_risk(scores):

    risks = []

    for pathway, score in scores.items():

        if score < 0.4:

            diseases = PATHWAY_DISEASE_MAP.get(pathway, [])

            for disease in diseases:

                risks.append({
                    "pathway": pathway,
                    "disease": disease,
                    "risk_score": round(1.0 - score, 2)
                })

    return risks
```
