### SafetyChain™の再定義（確認）

あなたが言っている SafetyChain™ は、こう整理できます。

- **トレーサビリティではない:** 「誰が作ったか」「どこから来たか」を追うだけの履歴管理では価値にならない。
    
- **消費者の選択基準そのもの:** 農産物を選ぶ軸は **栄養価 × 鮮度保持力（＝ポリフェノールの質と量）** であり、それが味・香り・色・機能性に直結する。
    
- **鮮度保持力の源:**
    
    - 土壌細菌（MBT55コンソーシアム）
        
    - 土壌環境（pH, Eh, 水分, 有機炭素など）
        
    - それらを読み解くフェノタイピング（代謝の推移を診る）
        

> SafetyChain™ ＝ 「消費者が農産物を選ぶための機能性・品質の指標」 ＝ 土壌からポリフェノール生成までの代謝カスケードを数値化した“選択プロトコル”

この前提で API を設計します。

**役割:** pH, Eh, 水分、C/N などの土壌データを受け取り、 MBT55 ハイパーサイクルにとっての「環境適合度」を返す。

  } = body;

  const result = evaluateSoilEnvironment({
    ph,
    eh_mV,
    moisture_pct,
    organicCarbon: organic_carbon_g_per_l,
    availableNitrogen: available_nitrogen_g_per_l,
  });

ts

```
// lib/agrix/soilDynamics.ts
export type SoilEnvInput = {
  ph: number;
  eh_mV: number;
  moisture_pct: number;
  organicCarbon: number;
  availableNitrogen: number;
};

export type SoilEnvEvaluation = {
  aerobicFraction: number;
  mbt55SuitabilityIndex: number; // 0-1
  notes: string[];
};

export function evaluateSoilEnvironment(input: SoilEnvInput): SoilEnvEvaluation {
  const aerobicFraction = aerobicFractionFn(input.eh_mV, input.moisture_pct);

  let suitability = 1.0;
  const notes: string[] = [];

  // pH適合（MBT55: 7.2-7.5 付近を理想）
  if (input.ph < 6.5 || input.ph > 8.5) {
    suitability -= 0.2;
    notes.push('pHがMBT55の最適範囲から外れています。');
  } else {
    notes.push('pHはMBT55に適した範囲です。');
  }

  // 有機炭素・窒素バランス（C/N）
  const cn = input.organicCarbon / Math.max(input.availableNitrogen, 0.0001);
  if (cn < 10 || cn > 25) {
    suitability -= 0.2;
    notes.push(`C/N比 (${cn.toFixed(1)}) が最適範囲から外れています。`);
  } else {
    notes.push(`C/N比 (${cn.toFixed(1)}) は良好です。`);
  }

  suitability = Math.max(0, Math.min(1, suitability));

  return {
    aerobicFraction,
    mbt55SuitabilityIndex: suitability,
    notes,
  };
}

function aerobicFractionFn(eh_mV: number, moisture_pct: number): number {
  const f_Eh = 1.0 / (1.0 + Math.exp(-0.02 * (eh_mV - 50)));
  const f_moisture = 1.0 - Math.max(0.0, (moisture_pct - 70.0) / 30.0);
  const val = f_Eh * f_moisture;
  return Math.min(0.95, Math.max(0.05, val));
}
```

```
// lib/agrix/agriwareEngine.ts
export type SheInput = {
  microScore: number;
  humicScore: number;
  cnScore: number;
  physScore: number;
};

export function computeSheScore(input: SheInput): number {
  const { microScore, humicScore, cnScore, physScore } = input;
  const she =
    0.4 * microScore +
    0.25 * humicScore +
    0.2 * cnScore +
    0.15 * physScore;
  return Math.max(0, Math.min(100, she));
}

export type PhenotypeInput = {
  sheHistory: number[];
  currentShe: number;
};

export type EcoPhenotype =
  | 'ACCELERATING_IMPROVEMENT'
  | 'DIMINISHING_IMPROVEMENT'
  | 'STABLE'
  | 'DIMINISHING_DEGRADATION'
  | 'ACCELERATING_DEGRADATION';

export function diagnoseEcoPhenotype(input: PhenotypeInput): EcoPhenotype {
  const { sheHistory, currentShe } = input;
  const last = sheHistory.length ? sheHistory[sheHistory.length - 1] : currentShe;
  const delta = currentShe - last;

  if (delta > 2) return 'ACCELERATING_IMPROVEMENT';
  if (delta > 0) return 'DIMINISHING_IMPROVEMENT';
  if (Math.abs(delta) <= 0.5) return 'STABLE';
  if (delta < -2) return 'ACCELERATING_DEGRADATION';
  return 'DIMINISHING_DEGRADATION';
}
```

### 2-3. V-Score（代謝寿命）API `/app/api/agrix/vscore/route.ts`

**役割:** 収穫時の ATP、ポリフェノール密度、保存条件から V-Score と残り代謝寿命（日数）を返す。

```

  const {

    atp_current,
    scfa_index,

    storage_temp_c,

  } = body;

    atpInitial: atp_initial,
    atpCurrent: atp_current,
    scfaIndex: scfa_index,
    polyphenolDensity: polyphenol_density,
    storageTempC: storage_temp_c,
  });

  const remainingDays = estimateMetabolicLife({
    vScore,
    elapsedDays: elapsed_days,
  });

  return NextResponse.json({ vScore, remainingDays });

```

ts

```
// lib/agrix/vscoreModel.ts
export type VScoreInput = {
  atpInitial: number;
  atpCurrent: number;
  scfaIndex: number; // 0-1
  polyphenolDensity: number; // 0-1
  storageTempC: number;
};

export function computeVScore(input: VScoreInput): number {
  const { atpInitial, atpCurrent, scfaIndex, polyphenolDensity, storageTempC } = input;

  const atpRatio = atpCurrent / Math.max(atpInitial, 0.0001);
  const tempPenalty = Math.max(0, (storageTempC - 4) * 0.01); // 4℃基準

  let base = 0.5 * atpRatio + 0.25 * scfaIndex + 0.25 * polyphenolDensity;
  base -= tempPenalty;
  const v = Math.max(0, Math.min(1, base));
  return v * 100;
}

export type MetabolicLifeInput = {
  vScore: number;
  elapsedDays: number;
};

export function estimateMetabolicLife(input: MetabolicLifeInput): number {
  const { vScore, elapsedDays } = input;
  // 閾値 35 を下回るまでを「機能性維持期間」と仮定
  const remaining = (vScore - 35) / 5; // 仮の線形モデル
  return Math.max(0, Number((remaining - elapsedDays).toFixed(1)));
}
```

## 3. SafetyChain API Routes（機能性・品質・価格）

### 3-1. 消費者向け「機能性・品質」API `/app/api/safetychain/product/route.ts`

**役割:** 1つのロット（農産物）について

- ポリフェノール密度
    
- V-Score
    
- 機能性インデックス（Functionality Index） を返し、「選択基準」を JSON で提示する。
    

ts

```
// app/api/safetychain/product/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { computeVScore } from '@/lib/agrix/vscoreModel';
import { computeFunctionalityIndex } from '@/lib/safetychain/phenomicsAgents';

export async function POST(req: NextRequest) {
  const body = await req.json();

  const {
    atp_initial,
    atp_current,
    scfa_index,
    polyphenol_density,
    storage_temp_c,
    phenotype_signals,
  } = body;

  const vScore = computeVScore({
    atpInitial: atp_initial,
    atpCurrent: atp_current,
    scfaIndex: scfa_index,
    polyphenolDensity: polyphenol_density,
    storageTempC: storage_temp_c,
  });

  const functionalityIndex = computeFunctionalityIndex({
    polyphenolDensity: polyphenol_density,
    scfaIndex: scfa_index,
    phenotypeSignals: phenotype_signals,
  });

  return NextResponse.json({
    vScore,
    functionalityIndex,
    selectionMessage: buildSelectionMessage(vScore, functionalityIndex),
  });
}
```

ts

```
// lib/safetychain/phenomicsAgents.ts
export type FunctionalityInput = {
  polyphenolDensity: number; // 0-1
  scfaIndex: number; // 0-1
  phenotypeSignals: {
    colorScore?: number;
    aromaScore?: number;
    textureScore?: number;
  };
};

export function computeFunctionalityIndex(input: FunctionalityInput): number {
  const { polyphenolDensity, scfaIndex, phenotypeSignals } = input;
  const sensory =
    ((phenotypeSignals.colorScore ?? 0.5) +
      (phenotypeSignals.aromaScore ?? 0.5) +
      (phenotypeSignals.textureScore ?? 0.5)) /
    3;

  const idx =
    0.5 * polyphenolDensity +
    0.3 * scfaIndex +
    0.2 * sensory;

  return Math.max(0, Math.min(1, idx)) * 100;
}

export function buildSelectionMessage(vScore: number, functionalityIndex: number): string {
  return `このロットは V-Score=${vScore.toFixed(
    1,
  )} / 機能性インデックス=${functionalityIndex.toFixed(
    1,
  )} です。鮮度保持力と機能性が高く、選択基準を満たしています。`;
}
```

// lib/safetychain/pricingModel.ts
export type PricingInput = {
  basePrice: number;
  vComponent: number;
  lComponent: number;
  mComponent: number;
  cComponent: number;
};

export type PricingOutput = {
  totalPrice: number;
  breakdown: {
    base: number;
    v: number;
    l: number;
    m: number;
    c: number;
  };
};

export function computeDynamicPrice(input: PricingInput): PricingOutput {
  const { basePrice, vComponent, lComponent, mComponent, cComponent } = input;

  const total =
    basePrice + vComponent + lComponent + mComponent + cComponent;

  return {
    totalPrice: Number(total.toFixed(2)),
    breakdown: {
      base: basePrice,
      v: vComponent,
      l: lComponent,
      m: mComponent,
      c: cComponent,
    },
  };
}

// lib/safetychain/certificateBuilder.ts
export type CertificateInput = {
  lotId: string;
  vScore: number;
  functionalityIndex: number;
  metabolicLifeDays: number;
  polyphenolDensity: number;
};

export type Certificate = {
  lotId: string;
  safetychainVersion: string;
  summary: string;
  metrics: {
    vScore: number;
    functionalityIndex: number;
    metabolicLifeDays: number;
    polyphenolDensity: number;
  };
};

export function buildCertificate(input: CertificateInput): Certificate {
  const { lotId, vScore, functionalityIndex, metabolicLifeDays, polyphenolDensity } = input;

  const summary = [
    `このロットは SafetyChain™ により評価されています。`,
    `鮮度保持力（V-Score）：${vScore.toFixed(1)}`,
    `機能性インデックス：${functionalityIndex.toFixed(1)}`,
    `推定代謝寿命：${metabolicLifeDays.toFixed(1)} 日`,
    `ポリフェノール密度：${(polyphenolDensity * 100).toFixed(1)}%`,
  ].join(' ');

  return {
    lotId,
    safetychainVersion: '1.0.0',
    summary,
    metrics: {
      vScore,
      functionalityIndex,
      metabolicLifeDays,
      polyphenolDensity,
    },
  };
}
