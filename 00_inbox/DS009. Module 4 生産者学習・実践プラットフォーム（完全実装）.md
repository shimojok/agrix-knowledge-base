export function runLearningEngine(input: LearningEngineInput): LearningEngineOutput {
  const { userId, moduleId, action, topicId, quizResult, timeSpentMinutes } = input;

  // 既存進捗を取得（ここでは新規作成）
  let progress: LearningProgress = {
    userId,
    moduleId,
    status: 'NOT_STARTED',
    topicProgress: [],
    quizResults: [],
    overallScore: 0,
    passStatus: false
  };

  // モジュールを取得
  const module = LEARNING_CATALOG.find(m => m.id === moduleId);
  if (!module) {
    throw new Error(`Module not found: ${moduleId}`);
  }

  // トピック進捗を初期化
  if (progress.topicProgress.length === 0) {
    progress.topicProgress = module.topics.map(t => ({
      topicId: t.id,
      completed: false,
      timeSpentMinutes: 0
    }));
  }

  // クイズ結果を初期化
  if (progress.quizResults.length === 0) {
    progress.quizResults = module.quizzes.map(q => ({
      quizId: q.id,
      answered: false,
      correct: false,
      attempts: 0
    }));
  }

  // アクション処理
  switch (action) {
    case 'START':
      progress.status = 'IN_PROGRESS';
      progress.startedAt = new Date().toISOString();
      break;

    case 'COMPLETE_TOPIC':
      if (topicId) {
        const topicProgress = progress.topicProgress.find(t => t.topicId === topicId);
        if (topicProgress) {
          topicProgress.completed = true;
          topicProgress.timeSpentMinutes += timeSpentMinutes || 5;
        }
      }
      break;

    case 'ANSWER_QUIZ':
      if (quizResult) {
        const quizResultEntry = progress.quizResults.find(q => q.quizId === quizResult.quizId);
        if (quizResultEntry) {
          const quiz = module.quizzes.find(q => q.id === quizResult.quizId);
          if (quiz) {
            quizResultEntry.answered = true;
            quizResultEntry.correct = quizResult.selectedAnswer === quiz.correctAnswer;
            quizResultEntry.attempts += 1;
            quizResultEntry.lastAttemptAt = new Date().toISOString();
          }
        }
      }
      break;

    case 'COMPLETE_MODULE':
      // 全トピック完了とクイズ合格率をチェック
      const allTopicsCompleted = progress.topicProgress.every(t => t.completed);
      const correctQuizzes = progress.quizResults.filter(q => q.correct).length;
      const totalQuizzes = progress.quizResults.length;
      const passRate = totalQuizzes > 0 ? (correctQuizzes / totalQuizzes) * 100 : 0;
      
      progress.overallScore = passRate;
      progress.passStatus = passRate >= module.completionCriteria.quizPassRate && allTopicsCompleted;
      
      if (progress.passStatus) {
        progress.status = 'COMPLETED';
        progress.completedAt = new Date().toISOString();
        // 修了証書発行
        progress.certificateId = `CERT-${userId}-${moduleId}-${Date.now()}`;
        progress.certificateIssuedAt = new Date().toISOString();
      }
      break;

    case 'CERTIFY':
      if (progress.status === 'COMPLETED') {
        progress.status = 'CERTIFIED';
      }
      break;
  }

  // 次に取るべきステップを生成
  const nextSteps = generateNextSteps(progress, module);

  // 達成状況
  const achievements = generateAchievements(progress, module);

  // 修了証書URL（あれば）
  const certificateUrl = progress.certificateId 
    ? `/api/agrix/module4/certificate/${progress.certificateId}`
    : undefined;

  return {
    progress,
    nextSteps,
    achievements,
    certificateUrl
  };
}

function generateNextSteps(progress: LearningProgress, module: LearningModule): string[] {
  const steps: string[] = [];

  if (progress.status === 'NOT_STARTED' || progress.status === 'IN_PROGRESS') {
    const incompleteTopics = progress.topicProgress.filter(t => !t.completed);
    if (incompleteTopics.length > 0) {
      const topic = module.topics.find(t => t.id === incompleteTopics[0].topicId);
      if (topic) {
        steps.push(`📖 「${topic.title}」を学習してください`);
      }
    }

    const unansweredQuizzes = progress.quizResults.filter(q => !q.answered);
    if (unansweredQuizzes.length > 0) {
      steps.push(`📝 クイズ「${unansweredQuizzes[0].quizId}」に回答してください`);
    }
  }

  if (progress.status === 'COMPLETED' && progress.passStatus) {
    steps.push('🎉 モジュールを修了しました！修了証書を発行できます。');
    steps.push('📚 次のモジュールに進む準備ができました。');
  }

  if (steps.length === 0) {
    steps.push('✅ 全ての学習が完了しました。おめでとうございます！');
  }

  return steps;
}

function generateAchievements(progress: LearningProgress, module: LearningModule): string[] {
  const achievements: string[] = [];

  const completedTopics = progress.topicProgress.filter(t => t.completed).length;
  const totalTopics = module.topics.length;

  if (completedTopics === totalTopics) {
    achievements.push(`🏆 全${totalTopics}トピックを完了`);
  } else if (completedTopics >= totalTopics / 2) {
    achievements.push(`📖 ${completedTopics}/${totalTopics}トピックを完了`);
  }

  const correctQuizzes = progress.quizResults.filter(q => q.correct).length;
  const totalQuizzes = progress.quizResults.length;

  if (correctQuizzes === totalQuizzes) {
    achievements.push('🏆 全クイズ正解！');
  } else if (correctQuizzes >= totalQuizzes * 0.8) {
    achievements.push(`📝 クイズ合格率 ${Math.round((correctQuizzes / totalQuizzes) * 100)}%`);
  }

  if (progress.status === 'CERTIFIED') {
    achievements.push('🎓 修了証書を取得済み');
  }

  if (achievements.length === 0) {
    achievements.push('🚀 学習を開始しましょう！');
  }

  return achievements;
}

export function runSimulatorEngine(input: SimulatorInput): SimulatorOutput {
  const { farmConfig, scenario, simulationPeriodYears } = input;
  const { fieldSize, cropType, currentSOC, currentYield, inputCost, sellingPrice } = farmConfig;

  // ----- 1. MBT55導入効果の計算 -----

  // 収量増加率（MBT55投与量と頻度に依存）
  const dosageEffect = Math.min(1, scenario.mbt55Dosage_Lha / 3.0);
  const frequencyEffect = Math.min(1, scenario.mbt55Frequency / 3);
  const humusEffect = Math.min(1, scenario.humusApplication_tPerHa / 5);
  const fertilizerEffect = Math.min(1, 
    (scenario.fertilizerApplication.liquid_LPerHa / 50 + 
     scenario.fertilizerApplication.solid_kgPerHa / 200) / 2
  );

  const yieldIncreaseBase = 5 + 20 * dosageEffect + 10 * frequencyEffect + 10 * humusEffect + 5 * fertilizerEffect;
  const yieldIncreaseRate = Math.min(50, yieldIncreaseBase);

  // SOC増加量
  const socIncrease = 0.1 + 0.3 * dosageEffect + 0.2 * humusEffect + 0.1 * fertilizerEffect;
  const projectedSOC = currentSOC + socIncrease * Math.min(simulationPeriodYears / 5, 1);

  // 炭素隔離量
  const carbonSequestration = socIncrease * 3.67 * 0.5 * simulationPeriodYears;

  // GHG削減量
  const ghgReduction = carbonSequestration * 0.6 + 0.5 * (scenario.mbt55Dosage_Lha / 3);

  // ----- 2. 経済効果計算 -----

  // MBT55導入後の収量
  const projectedYield = currentYield * (1 + yieldIncreaseRate / 100);

  // コスト
  const mbt55Cost = scenario.mbt55Dosage_Lha * 500 + scenario.mbt55Frequency * 200; // 円/ha（仮定）
  const humusCost = scenario.humusApplication_tPerHa * 2000;
  const fertilizerCost = scenario.fertilizerApplication.liquid_LPerHa * 100 + scenario.fertilizerApplication.solid_kgPerHa * 50;
  const additionalCost = mbt55Cost + humusCost + fertilizerCost;

  // 収入
  const revenueWithMBT55 = projectedYield * sellingPrice;
  const revenueWithoutMBT55 = currentYield * sellingPrice;

  // 純利益
  const netProfitWithMBT55 = revenueWithMBT55 - (inputCost + additionalCost);
  const netProfitWithoutMBT55 = revenueWithoutMBT55 - inputCost;

  // 利益率
  const profitMarginWithMBT55 = (netProfitWithMBT55 / revenueWithMBT55) * 100;
  const profitMarginWithoutMBT55 = (netProfitWithoutMBT55 / revenueWithoutMBT55) * 100;

  // ----- 3. 比較計算 -----

  const yieldImprovement = ((projectedYield - currentYield) / currentYield) * 100;
  const costReduction = ((inputCost - (inputCost + additionalCost)) / inputCost) * 100;
  const profitIncrease = ((netProfitWithMBT55 - netProfitWithoutMBT55) / Math.abs(netProfitWithoutMBT55)) * 100;
  const carbonBenefit = carbonSequestration;
  const ghgReductionRate = (ghgReduction / (ghgReduction + 1)) * 100;

  // ROI（投資収益率）
  const totalInvestment = additionalCost * fieldSize;
  const totalBenefit = (netProfitWithMBT55 - netProfitWithoutMBT55) * fieldSize * simulationPeriodYears;
  const roi = (totalBenefit / totalInvestment) * 100;

  // 投資回収期間（年）
  const annualBenefit = (netProfitWithMBT55 - netProfitWithoutMBT55) * fieldSize;
  const paybackPeriod = totalInvestment / Math.max(annualBenefit, 1);

  // 損益分岐点年数
  const breakEvenYear = Math.ceil(paybackPeriod);

  // haあたり正味利益増
  const netBenefitPerHa = netProfitWithMBT55 - netProfitWithoutMBT55;

  // ----- 4. 環境インパクト -----

  const socIncreaseTotal = socIncrease * fieldSize;
  const carbonCredits = carbonSequestration * 0.9;
  const creditValue = carbonCredits * 2000; // 円（仮定）

  // 保水力向上（簡易推定）

  // 生物多様性スコア（簡易推定）
  const biodiversityScore = 40 + 30 * dosageEffect + 20 * frequencyEffect + 10 * humusEffect;

  // ----- 5. 総合評価 -----

  const overallScore = (
    (yieldIncreaseRate / 50) * 25 +
    (profitIncrease / 100) * 20 +
    (carbonSequestration / 10) * 20 +
    (biodiversityScore / 100) * 15 +
    (roi / 100) * 20
  );

  // 推奨・リスク・機会
  const recommendation = generateSimulatorRecommendation(overallScore, yieldIncreaseRate, roi, paybackPeriod);
  const risks = generateSimulatorRisks(yieldIncreaseRate, paybackPeriod, scenario);
  const opportunities = generateSimulatorOpportunities(carbonSequestration, biodiversityScore, profitIncrease);

  return {
    withMBT55: {
      projectedSOC: Number(projectedSOC.toFixed(2)),
      projectedYield: Number(projectedYield.toFixed(2)),
      projectedCost: Number((inputCost + additionalCost).toFixed(0)),
      carbonSequestration: Number(carbonSequestration.toFixed(2)),
      ghgReduction: Number(ghgReduction.toFixed(2)),
      netProfit: Number(netProfitWithMBT55.toFixed(0)),
      profitMargin: Number(profitMarginWithMBT55.toFixed(1))
    },
    withoutMBT55: {
      projectedSOC: currentSOC,
      projectedYield: currentYield,
      projectedCost: inputCost,
      carbonSequestration: 0,
      ghgReduction: 0,
      netProfit: Number(netProfitWithoutMBT55.toFixed(0)),
      profitMargin: Number(profitMarginWithoutMBT55.toFixed(1))
    },
    comparison: {
      yieldImprovement: Number(yieldImprovement.toFixed(1)),
      costReduction: Number(Math.max(0, costReduction).toFixed(1)),
      profitIncrease: Number(profitIncrease.toFixed(1)),
      carbonBenefit: Number(carbonBenefit.toFixed(2)),
      ghgReductionRate: Number(ghgReductionRate.toFixed(1)),
      roi: Number(roi.toFixed(0)),
      paybackPeriod: Number(paybackPeriod.toFixed(1)),
      netBenefitPerHa: Number(netBenefitPerHa.toFixed(0)),
      breakEvenYear

    environmentalImpact: {
      socIncrease: Number(socIncreaseTotal.toFixed(2)),
      carbonCredits: Number(carbonCredits.toFixed(0)),

    },

}

/**

  if (soilData.pH < 6.0 || soilData.pH > 8.0) {

      description: `現在のpHは${soilData.pH.toFixed(1)}です。MBT55の最適範囲（6.5-7.5）から外れています。`
    });
  }
  if (soilData.moisture_pct < 30 || soilData.moisture_pct > 70) {
    limitingFactors.push({
      factor: '水分ストレス',
      severity: soilData.moisture_pct < 20 || soilData.moisture_pct > 80 ? 'HIGH' : 'MEDIUM',
      description: `現在の水分は${soilData.moisture_pct.toFixed(0)}%です。適正範囲（40-60%）への調整が必要です。`
    });
  }
  if (hypercycle.cycleEfficiency < 0.4) {
    limitingFactors.push({
      factor: 'ハイパーサイクル活性低下',
      severity: hypercycle.cycleEfficiency < 0.2 ? 'HIGH' : 'MEDIUM',
      description: `ハイパーサイクル効率が${(hypercycle.cycleEfficiency * 100).toFixed(0)}%と低く、微生物活性の向上が必要です。`
    });
  }

  // 栄養状態
  const nutrientStatus = {
    n: soilData.availableNitrogen_gL > 1.5 ? 'OPTIMAL' as const : soilData.availableNitrogen_gL > 0.8 ? 'OPTIMAL' as const : 'DEFICIENT' as const,
    p: soilData.organicCarbon_gL > 10 ? 'OPTIMAL' as const : 'DEFICIENT' as const,
    k: soilData.ec_dSm > 1.0 ? 'OPTIMAL' as const : 'DEFICIENT' as const,
    organicMatter: soilData.organicCarbon_gL > 15 ? 'HIGH' as const : soilData.organicCarbon_gL > 8 ? 'MEDIUM' as const : 'LOW' as const
  };

  // ----- 2. MBT55施用推奨 -----

  // 投与量（土壌状態と作物タイプに基づく）
  let baseDosage = 2.5;
  if (currentHealth < 40) baseDosage += 0.5;
  if (nutrientStatus.organicMatter === 'LOW') baseDosage += 0.5;
  if (goals.includes('炭素隔離')) baseDosage += 0.3;
  
  const dosage_Lha = Math.min(4.0, Math.max(2.0, baseDosage));

  // 頻度
  let frequency = 2;
  if (currentHealth < 40) frequency += 1;
  if (regionalData.growingSeasonDays > 180) frequency += 1;
  frequency = Math.min(4, Math.max(1, frequency));

  // タイミング
  const timing = regionalData.climate === 'COLD' 
    ? '春（地温10℃以上）と秋の収穫後' 
    : '生育初期と収穫後';

  // ----- 3. 3資材の最適配合 -----

  // 腐植質（SOC増加が目標の場合に増量）
  const humusAmount = goals.includes('炭素隔離')

    : Math.min(5, 2 + currentHealth / 30);

  const solidAmount = Math.min(300, 100 + (nutrientStatus.p === 'DEFICIENT' ? 100 : 0));

  const netProfitIncrease = 50000 + 30000 * (dosage_Lha / 3) + 20000 * (humusAmount / 5);

  const carbonCreditValue = ghgReduction * 2000;

  const totalInvestment = dosage_Lha * 500 + humusAmount * 2000 + liquidAmount * 100 + solidAmount * 50;
  const annualBenefit = netProfitIncrease + carbonCreditValue;
  const roi = (annualBenefit / totalInvestment) * 100;
  const paybackPeriod = totalInvestment / Math.max(annualBenefit, 1);

  // ----- 6. 推奨学習モジュール -----

    (socIncrease / 1) * 15 +

  );

  const confidenceLevel: PrescriptionOutput['confidenceLevel'] =

function generateActionPlan(health: number, factors: any[], experience: string): PrescriptionOutput['actionPlan'] {
  const phase1Actions: string[] = [];
  const phase2Actions: string[] = [];
  const phase3Actions: string[] = [];
  const phase4Actions: string[] = [];

  // 即時対応（pH・水分調整）
  for (const factor of factors) {
    if (factor.factor === 'pH不均衡' && factor.severity === 'HIGH') {
      phase1Actions.push('緊急pH調整（石灰資材または硫黄資材の施用）');
    }
    if (factor.factor === '水分ストレス' && factor.severity === 'HIGH') {
      phase1Actions.push('灌水計画の緊急見直し');
    }
  }

  // 短期対応（MBT55施用）
  phase2Actions.push('MBT55の初回施用（推奨量）');
  phase2Actions.push('発酵肥料の初回施用');

  if (health < 40) {
    phase2Actions.push('土壌微生物活性化のための有機物追加投入');
  }

  // 中期対応（継続的管理）
  phase3Actions.push('MBT55の定期施用（2-3回目）');
  phase3Actions.push('腐植質の追加投入（必要に応じて）');
  phase3Actions.push('土壌分析による効果検証');

  // 長期対応（システム構築）
  phase4Actions.push('MBT55ハイパーサイクルの定着確認');
  phase4Actions.push('炭素クレジット認証申請の準備');
  phase4Actions.push('持続可能な土壌管理体制の確立');

  if (experience === 'BEGINNER') {
    phase1Actions.push('📚 MBT55基礎学習モジュールの受講推奨');
  }

  return {
    phase1: { actions: phase1Actions.length > 0 ? phase1Actions : ['土壌サンプリングを実施し、詳細分析を依頼'], priority: 'HIGH' },
    phase2: { actions: phase2Actions, priority: 'HIGH' },
    phase3: { actions: phase3Actions, priority: 'MEDIUM' },
    phase4: { actions: phase4Actions, priority: 'MEDIUM' }
  };
}

function generateRecommendedLearning(experience: string, goals: string[]): string[] {
  const modules: string[] = [];

  if (experience === 'BEGINNER') {
    modules.push('mbt55-basics');
  }

  if (goals.includes('炭素隔離') || goals.includes('GHG削減')) {
    modules.push('mbt55-advanced');
  }

  if (modules.length === 0) {
    modules.push('mbt55-basics');
  }

  return modules;
}

function generatePrescriptionStatus(score: number, confidence: string): string {
  if (score >= 80) {
    return `✅ 総合評価は非常に良好です（${score.toFixed(0)}点）。処方の確度は${confidence}です。`;
  }
  if (score >= 60) {
    return `📈 総合評価は良好です（${score.toFixed(0)}点）。処方の確度は${confidence}です。`;
  }
  if (score >= 40) {
    return `📊 総合評価は改善の余地があります（${score.toFixed(0)}点）。処方の確度は${confidence}です。`;
  }
  return `⚠️ 総合評価が低めです（${score.toFixed(0)}点）。土壌状態の再評価を推奨します。`;
}

function generatePrescriptionSummary(goals: string[], yieldInc: number, socInc: number, roi: number): string {
  const goalSummary = goals.length > 0 ? `「${goals.join('・')}」` : '総合的な土壌改良';
  return `この処方は${goalSummary}を目標に設計されています。収量${yieldInc.toFixed(0)}%向上、SOC${socInc.toFixed(2)}t/ha増加、ROI${roi.toFixed(0)}%が見込まれます。`;
}

export interface Module4Output {
  timestamp: string;
  version: string;
  
  // 学習進捗（あれば）
  learning?: {
    progress: any;
    nextSteps: string[];
    achievements: string[];
    certificateUrl?: string;
  };

  simulator: any;
  
  // 処方提案
  prescription: any;

    keyRecommendations: string[];
    nextAction: string;
  };
}

export function runModule4Full(input: Module4Input): Module4Output {
  // ----- 1. シミュレーション実行 -----
  const simulatorInput = {
    farmConfig: {
      fieldSize: input.farmerProfile.fieldSize,
      cropType: input.farmerProfile.cropType,
      currentSOC: input.farmConfig.currentSOC,
      currentYield: input.farmConfig.currentYield,
      inputCost: input.farmConfig.inputCost,
      sellingPrice: input.farmConfig.sellingPrice,
    },
    scenario: input.scenario || {
      mbt55Dosage_Lha: 2.8,
      mbt55Frequency: 2,
      humusApplication_tPerHa: 3.0,
      fertilizerApplication: { liquid_LPerHa: 30, solid_kgPerHa: 150 },
    },
    simulationPeriodYears: 5,
  };
  const simulator = runSimulatorEngine(simulatorInput);

  // ----- 2. 処方提案実行 -----
  const prescriptionInput = {
    fieldId: `FIELD-${Date.now()}`,
    soilData: input.soilData,
    hypercycle: input.hypercycle,
    simulator,
    farmerProfile: input.farmerProfile,
    regionalData: input.regionalData,
  };
  const prescription = runPrescriptionEngine(prescriptionInput);

  // ----- 3. 学習（あれば）-----
  let learning = undefined;
  if (input.learningAction) {
    const learningInput = {
      userId: input.farmerProfile.userId,
      moduleId: input.learningAction.moduleId,
      action: input.learningAction.action,
      topicId: input.learningAction.topicId,
      quizResult: input.learningAction.quizResult,
      timeSpentMinutes: input.learningAction.timeSpentMinutes,
    };
    learning = runLearningEngine(learningInput);
  }

  // ----- 4. まとめ -----
  const overallScore = (simulator.overallScore * 0.4 + prescription.overallScore * 0.6);
  let status: Module4Output['summary']['status'] = 'FAIR';
  if (overallScore >= 80) status = 'EXCELLENT';
  else if (overallScore >= 65) status = 'GOOD';
  else if (overallScore >= 45) status = 'FAIR';
  else status = 'POOR';

  const keyRecommendations = [
    `MBT55投与量: ${prescription.mbt55Recommendation.dosage_Lha.toFixed(1)} L/ha、頻度: ${prescription.mbt55Recommendation.frequency}回/シーズン`,
    `腐植質投入: ${prescription.materialRecommendation.humus.amount_tPerHa.toFixed(1)} t/ha`,
    `発酵肥料: 液肥${prescription.materialRecommendation.fertilizer.liquid_LPerHa.toFixed(0)}L/ha + 固形肥${prescription.materialRecommendation.fertilizer.solid_kgPerHa.toFixed(0)}kg/ha`,
  ];

  if (prescription.materialRecommendation.feed.recommended) {
    keyRecommendations.push(`発酵飼料: ${prescription.materialRecommendation.feed.amount_kgPerHeadDay}kg/頭/日`);
  }

  const nextAction = prescription.actionPlan.phase1.actions[0] || '処方に従って即時対応を開始してください。';

  return {
    timestamp: new Date().toISOString(),
    version: '1.0.0',
    learning,
    simulator,
    prescription,
    summary: {
      overallScore: Number(overallScore.toFixed(0)),
      status,
      keyRecommendations,
      nextAction,
    },
  };
}

// app/api/agrix/module4/orchestrate/route.ts

import { NextRequest, NextResponse } from 'next/server';
import { runModule4Full } from '@/lib/agrix/module4';
import { runModule1Full } from '@/lib/agrix/module1';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    
    // 必須フィールドチェック
    const requiredFields = [
      'soilData', 'farmerProfile', 'regionalData', 'farmConfig'
    ];
    for (const field of requiredFields) {
      if (!body[field]) {
        return NextResponse.json(
          { error: `Missing required field: ${field}` },
          { status: 400 }
        );
      }
    }

    // Module1実行
    const module1Result = runModule1Full(body.soilData);

    // Module4実行
    const module4Input = {
      soilData: body.soilData,
      hypercycle: module1Result.hypercycle,
      farmerProfile: body.farmerProfile,
      regionalData: body.regionalData,
      farmConfig: body.farmConfig,
      scenario: body.scenario,
      learningAction: body.learningAction,
    };

    const result = runModule4Full(module4Input);

    return NextResponse.json({
      module1: {
        agrixHealthScore: module1Result.agrixHealthScore,
        healthStatus: module1Result.healthStatus,
        hypercycle: module1Result.hypercycle,
      },
      module4: result,
    });
  } catch (error) {
    return NextResponse.json(
      { error: 'Internal server error', details: String(error) },
      { status: 500 }
    );
  }
}
