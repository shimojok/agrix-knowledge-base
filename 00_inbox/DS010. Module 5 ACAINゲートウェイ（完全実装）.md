// lib/agrix/module5/engines/safetychain.gateway.ts

import { SafetyChainGatewayOutput, SafetyChainRequest } from '../types/safetychain-gateway.types';
import { Module1Output } from '../../module1/orchestrate/full';
import { Module2Output } from '../../module2/orchestrate/full';
import { Module3Output } from '../../module3/orchestrate/full';

/**
 * SafetyChain™ Gateway
 * 消費者・流通業者向けの機能性・品質証明データを生成
 */
export function generateSafetyChainGateway(
  module1: Module1Output,
  module2: Module2Output,
  module3: Module3Output,
  request: SafetyChainRequest
): SafetyChainGatewayOutput {
  const { lotId, includeBlockchain, language } = request;

  // --- 機能性指標（Module3から） ---
  const vScore = module3.ghgAccounting.globalContribution.perHaContribution * 10 + 50;
  const grade = vScore >= 85 ? 'PLATINUM' : vScore >= 70 ? 'GOLD' : vScore >= 55 ? 'SILVER' : 'STANDARD';
  const metabolicLifeDays = module1.hypercycle.cycleEfficiency * 14 + 3;

  // --- ポリフェノール密度（Module2の肥料品質から推定） ---
  const polyphenolDensity = module2.fertilizer.cascadeEffect.qualityImprovement * 0.7 + 0.3;
  const functionalityIndex = (polyphenolDensity * 0.5 + module1.hypercycle.cycleEfficiency * 0.3 + module1.biosecurity.biomeResilienceScore * 0.2) * 100;
  const functionalityGrade = functionalityIndex >= 80 ? 'A' : functionalityIndex >= 60 ? 'B' : functionalityIndex >= 40 ? 'C' : 'D';

  // --- 証明書 ---
  const certificateId = `SC-${lotId}-${Date.now()}`;
  const qrCodeUrl = `https://safetychain.agrix.io/cert/${certificateId}`;
  const verificationUrl = `https://safetychain.agrix.io/verify/${certificateId}`;

  // --- 消費者向けメッセージ（言語対応） ---
  const messages = {
    ja: {
      title: `${grade}グレード認証農産物`,
      description: `この農産物はSafetyChain™によって${grade}グレードに認証されました。鮮度保持力と機能性が高く評価されています。`,
      highlights: [
        `V-Score: ${vScore.toFixed(0)}（${grade}）`,
        `代謝寿命: ${metabolicLifeDays.toFixed(0)}日`,
        `機能性インデックス: ${functionalityIndex.toFixed(0)}（${functionalityGrade}）`,
        `ポリフェノール密度: ${(polyphenolDensity * 100).toFixed(0)}%`,
        `MBT55土壌健康スコア: ${module1.agrixHealthScore.toFixed(0)}`
      ],
      recommendation: grade === 'PLATINUM' ? '最高品質の農産物です。特別な日におすすめします。' : '品質と鮮度が保証された農産物です。'
    },
    en: {
      title: `${grade} Grade Certified Produce`,
      description: `This produce is SafetyChain™ certified at ${grade} grade. High freshness retention and functionality.`,
      highlights: [
        `V-Score: ${vScore.toFixed(0)} (${grade})`,
        `Metabolic Life: ${metabolicLifeDays.toFixed(0)} days`,
        `Functionality Index: ${functionalityIndex.toFixed(0)} (${functionalityGrade})`,
        `Polyphenol Density: ${(polyphenolDensity * 100).toFixed(0)}%`,
        `MBT55 Soil Health Score: ${module1.agrixHealthScore.toFixed(0)}`
      ],
      recommendation: grade === 'PLATINUM' ? 'Premium quality. Recommended for special occasions.' : 'Quality and freshness guaranteed.'
    }
  };

  const lang = (language || 'ja') as 'ja' | 'en';
  const msg = messages[lang] || messages.ja;

  return {
    lotId,
    productName: `AGRIX ${module2.humus.production.productionEfficiency > 0.7 ? 'Premium' : 'Standard'} Crop`,
    harvestDate: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000).toISOString(),
    issuedAt: new Date().toISOString(),
    validUntil: new Date(Date.now() + metabolicLifeDays * 24 * 60 * 60 * 1000).toISOString(),
    functionality: {
      vScore: Number(vScore.toFixed(1)),
      grade,
      metabolicLifeDays: Number(metabolicLifeDays.toFixed(0)),
      polyphenolDensity: Number(polyphenolDensity.toFixed(3)),
      functionalityIndex: Number(functionalityIndex.toFixed(1)),
      functionalityGrade
    },
    origin: {
      farmId: 'FARM-AGRIX-001',
      region: 'Global',
      soilType: 'LOAM',
      sheScore: Number(module1.agrixHealthScore.toFixed(0)),
      mbt55CycleEfficiency: Number(module1.hypercycle.cycleEfficiency.toFixed(3))
    },
    certificate: {
      id: certificateId,
      qrCodeUrl,
      verificationUrl,
      securityLevel: grade === 'PLATINUM' || grade === 'GOLD' ? 'HIGH' : 'MEDIUM',
      blockchainHash: includeBlockchain ? `0x${Array.from({ length: 64 }, () => Math.floor(Math.random() * 16).toString(16)).join('')}` : undefined
    },
    consumerMessage: {
      title: msg.title,
      description: msg.description,
      highlights: msg.highlights,
      recommendation: msg.recommendation
    },
    version: '1.0.0',
    generatedAt: new Date().toISOString()
  };
}

// lib/agrix/module5/engines/pbpe.gateway.ts

import { PBPEGatewayOutput, PBPERequest } from '../types/pbpe-gateway.types';
import { Module1Output } from '../../module1/orchestrate/full';
import { Module2Output } from '../../module2/orchestrate/full';
import { Module3Output } from '../../module3/orchestrate/full';

/**
 * PBPE Marketplace Gateway
 * 投資家・トレーダー向けの資産評価・取引データを生成
 */
export function generatePBPEGateway(
  module1: Module1Output,
  module2: Module2Output,
  module3: Module3Output,
  request: PBPERequest
): PBPEGatewayOutput {
  const { assetId, includeTradingData, includeCarbonCredits, currency = 'USD' } = request;

  // --- 資産評価 ---
  const carbonValue = module3.sequestration.creditPotential.marketValue || 0;
  const cropValue = module2.economicValue.fertilizerValue + module2.economicValue.humusValue * 0.5;
  const biodiversityValue = module1.hypercycle.cycleEfficiency * 100;
  const totalAssetValue = cropValue + carbonValue + biodiversityValue;
  const netAssetValue = totalAssetValue * 0.85; // 簡易負債控除

  // --- 炭素クレジット ---
  const issuable = module3.sequestration.creditPotential.issuableCredits || 0;
  const verified = issuable * 0.6;
  const sold = verified * 0.2;
  const available = verified - sold;

  // --- 動的価格（4レイヤー） ---
  const basePrice = 300;
  const vPremium = module3.soc.trend.annualIncrease_tPerHa * 20;
  const lPremium = module2.circularEconomy.wasteReductionRate * 50;
  const mPremium = (1 - module1.biosecurity.diseaseOutbreakRisk) * 40;
  const cPremium = module3.sequestration.stableCarbon.annualSequestration_tCO2 * 5;
  const totalPrice = basePrice + vPremium + lPremium + mPremium + cPremium;

  const breakdown = [
    { label: '市場基準価格', amount: basePrice, percentage: (basePrice / totalPrice) * 100 },
    { label: '機能性プレミアム (V)', amount: Number(vPremium.toFixed(0)), percentage: (vPremium / totalPrice) * 100 },
    { label: 'ロス削減プレミアム (L)', amount: Number(lPremium.toFixed(0)), percentage: (lPremium / totalPrice) * 100 },
    { label: '医療費削減プレミアム (m)', amount: Number(mPremium.toFixed(0)), percentage: (mPremium / totalPrice) * 100 },
    { label: '炭素隔離プレミアム (C)', amount: Number(cPremium.toFixed(0)), percentage: (cPremium / totalPrice) * 100 },
  ];

  // --- 取引情報 ---
  const liquidityScore = module2.mscIntegratedScore * 0.7 + 20;
  const riskScore = 100 - module1.agrixHealthScore;

  return {
    assetValuation: {
      cropValue: Number(cropValue.toFixed(0)),
      carbonValue: Number(carbonValue.toFixed(0)),
      biodiversityValue: Number(biodiversityValue.toFixed(0)),
      totalAssetValue: Number(totalAssetValue.toFixed(0)),
      netAssetValue: Number(netAssetValue.toFixed(0)),
      valuationDate: new Date().toISOString(),
      valuationMethodology: 'AGRIX Asset Valuation Model v1.0 - 複合資産評価法'
    },
    carbonCredits: {
      issuable: Number(issuable.toFixed(0)),
      verified: Number(verified.toFixed(0)),
      sold: Number(sold.toFixed(0)),
      available: Number(available.toFixed(0)),
      marketValue: Number(carbonValue.toFixed(0)),
      pricePerCredit: Number((carbonValue / Math.max(issuable, 1)).toFixed(2)),
      scheme: module3.sequestration.creditPotential.scheme as 'VERRA' | 'GOLD_STANDARD' | 'J_CREDIT' | 'CUSTOM',
      vintage: module3.sequestration.creditPotential.vintage
    },
    dynamicPrice: {
      basePrice,
      vPremium: Number(vPremium.toFixed(0)),
      lPremium: Number(lPremium.toFixed(0)),
      mPremium: Number(mPremium.toFixed(0)),
      cPremium: Number(cPremium.toFixed(0)),
      totalPrice: Number(totalPrice.toFixed(0)),
      currency,
      breakdown
    },
    trading: {
      isTradable: true,
      lastTradePrice: Number((totalPrice * 0.95).toFixed(0)),
      lastTradeDate: new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString(),
      dailyVolume: Math.round(available * 0.1),
      marketCap: Math.round(totalAssetValue * 0.5),
      liquidityScore: Number(liquidityScore.toFixed(0)),
      riskScore: Number(Math.min(100, Math.max(0, riskScore)).toFixed(0))
    },
    version: '1.0.0',
    generatedAt: new Date().toISOString()
  };
}

  const { farmId, cropType, areaHa, coverageType, includeMitigation } = request;

  // --- リスク評価 ---
  const diseaseRisk = (1 - module1.biosecurity.pathogenSuppressionIndex) * 100;
  const pestRisk = (1 - module1.biocontrol.pestResistanceScore) * 100;
  const weatherRisk = 30 + (1 - module3.soc.trend.annualIncrease_tPerHa / 0.5) * 30;
  const yieldRisk = 20 + (1 - module1.hypercycle.cycleEfficiency) * 50;
  const priceRisk = 40 - module1.hypercycle.cycleEfficiency * 20;
  const carbonShortfallRisk = (1 - module3.sequestration.stableCarbon.permanenceScore) * 100;

  // 総合リスクスコア
  const overallRiskScore = (
    diseaseRisk * 0.20 +
    pestRisk * 0.15 +
    weatherRisk * 0.15 +
    yieldRisk * 0.20 +
    priceRisk * 0.15 +
    carbonShortfallRisk * 0.15
  );

  let riskLevel: InsuranceGatewayOutput['riskAssessment']['riskLevel'] = 'MEDIUM';
  if (overallRiskScore < 30) riskLevel = 'LOW';
  else if (overallRiskScore < 50) riskLevel = 'MEDIUM';
  else if (overallRiskScore < 70) riskLevel = 'HIGH';
  else riskLevel = 'CRITICAL';

  // リスク要因
  const riskFactors: InsuranceGatewayOutput['riskAssessment']['riskFactors'] = [];
  if (diseaseRisk > 50) {
    riskFactors.push({ factor: '病害リスク', severity: diseaseRisk > 70 ? 'HIGH' : 'MEDIUM', description: '病原菌抑制率が低く、病害発生リスクが高い' });
  }
  if (yieldRisk > 50) {
    riskFactors.push({ factor: '収量変動リスク', severity: yieldRisk > 70 ? 'HIGH' : 'MEDIUM', description: 'ハイパーサイクル効率が不安定で収量変動が予想される' });
  }
  if (carbonShortfallRisk > 50) {
    riskFactors.push({ factor: '炭素不足リスク', severity: carbonShortfallRisk > 70 ? 'HIGH' : 'MEDIUM', description: '炭素隔離の永続性が不十分' });
  }

  // --- 保険料提案 ---
  const basePremium = areaHa * 50;
  const riskDiscount = Math.max(0, 1 - overallRiskScore / 100);
  const discountRate = riskDiscount * 30;
  const adjustedPremium = basePremium * (1 - discountRate / 100);
  const totalAnnualPremium = adjustedPremium * areaHa;

  const discountReason: string[] = [];
  if (module1.biosecurity.biosecurityLevel === 'HIGH') discountReason.push('バイオセキュリティ高評価による割引');
  if (module1.biocontrol.biocontrolEfficacy > 0.7) discountReason.push('バイオコントロール効果による割引');
  if (module3.soc.trend.trajectory === 'ACCELERATING') discountReason.push('SOC増加トレンドによる割引');

  // --- 補償推奨 ---
  const coverageOptions = {
    BASIC: ['病害補償', '収量補償'],
    STANDARD: ['病害補償', '収量補償', '気象災害補償'],
    PREMIUM: ['病害補償', '収量補償', '気象災害補償', '炭素クレジット補償', '価格変動補償']
  };
  const recommendedCoverage = coverageOptions[coverageType as keyof typeof coverageOptions] || coverageOptions.STANDARD;

  const coverageAmount = areaHa * 200 * (coverageType === 'PREMIUM' ? 1.5 : coverageType === 'STANDARD' ? 1.0 : 0.7);
  const deductible = coverageAmount * 0.1;

  // --- リスク軽減対策 ---
  const mitigationActions: string[] = [];
  if (diseaseRisk > 40) mitigationActions.push('MBT55投与量を増やし、バイオセキュリティを強化する');
  if (yieldRisk > 40) mitigationActions.push('ハイパーサイクル効率向上のため、有機物投入量を増加する');
  if (carbonShortfallRisk > 40) mitigationActions.push('腐植質生産を強化し、炭素隔離の永続性を高める');
  mitigationActions.push('定期モニタリングを実施し、リスク指標を追跡する');

  const expectedRiskReduction = Math.min(30, mitigationActions.length * 8);
  const costBenefitRatio = (expectedRiskReduction / 100) * 5;

  return {
    riskAssessment: {
      overallRiskScore: Number(overallRiskScore.toFixed(1)),
      riskLevel,
      confidenceLevel: module1.agrixHealthScore > 60 ? 'HIGH' : 'MEDIUM',
      categories: {
        diseaseRisk: Number(diseaseRisk.toFixed(1)),
        pestRisk: Number(pestRisk.toFixed(1)),
        weatherRisk: Number(weatherRisk.toFixed(1)),
        yieldRisk: Number(yieldRisk.toFixed(1)),
        priceRisk: Number(priceRisk.toFixed(1)),
        carbonShortfallRisk: Number(carbonShortfallRisk.toFixed(1))
      },
      riskFactors
    },
    premium: {
      basePremium: Number(basePremium.toFixed(0)),
      adjustedPremium: Number(adjustedPremium.toFixed(0)),
      discountRate: Number(discountRate.toFixed(0)),
      discountReason: discountReason.length > 0 ? discountReason : ['標準保険料率を適用'],
      totalAnnualPremium: Number(totalAnnualPremium.toFixed(0)),
      currency: 'USD'
    },
    coverageRecommendation: {
      recommendedCoverage,
      coverageAmount: Number(coverageAmount.toFixed(0)),
      deductible: Number(deductible.toFixed(0)),
      policyTermDays: 365
    },
    mitigation: {
      actions: mitigationActions,
      expectedRiskReduction: Number(expectedRiskReduction.toFixed(0)),
      costBenefitRatio: Number(costBenefitRatio.toFixed(2))
    },
    version: '1.0.0',
    generatedAt: new Date().toISOString(),
    validUntil: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString()
  };
}
