```python
import networkx as nx
import plotly.graph_objects as go


def create_network(graph_data):

    G = nx.DiGraph()

    for edge in graph_data:

        substrate = edge["substrate"]
        metabolite = edge["final_metabolite"]

        G.add_edge(substrate, metabolite)

    pos = nx.spring_layout(G)

    edge_x = []
    edge_y = []

    for edge in G.edges():

        x0, y0 = pos[edge[0]]
        x1, y1 = pos[edge[1]]

        edge_x.extend([x0, x1, None])
        edge_y.extend([y0, y1, None])

    edge_trace = go.Scatter(
        x=edge_x,
        y=edge_y,
        line=dict(width=1),
        hoverinfo='none',
        mode='lines'
    )

    node_x = []
    node_y = []

    for node in G.nodes():

        x, y = pos[node]

        node_x.append(x)
        node_y.append(y)

    node_trace = go.Scatter(
        x=node_x,
        y=node_y,
        mode='markers+text',
        text=list(G.nodes()),
        textposition="top center"
    )

    fig = go.Figure(
        data=[edge_trace, node_trace]
    )

    return fig
```
