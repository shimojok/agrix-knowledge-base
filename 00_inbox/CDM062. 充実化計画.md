"""
Carbon flow modeling for MBT55 ecological hypercycle.
Implements carbon cycling efficiency, humification rates, and sequestration quantification.
Based on theoretical documents from MBT Hypercycle research.
"""

import numpy as np
from typing import Dict, List, Tuple
from dataclasses import dataclass
from enum import Enum


class CarbonPool(Enum):
    """Carbon pools in soil ecosystem."""
    MICROBIAL_BIOMASS = "microbial_biomass"
    DISSOLVED_ORGANIC = "dissolved_organic"
    HUMUS_STABLE = "humus_stable"
    HUMUS_ACTIVE = "humus_active"
    PARTICULATE = "particulate"
    RESPIRED = "respired"


@dataclass
class CarbonFlow:
    """Carbon flow between pools with MBT55 enhancement factors."""
    source: CarbonPool
    target: CarbonPool
    base_rate: float  # kg C/ha/day
    mbt_enhancement: float  # MBT55 enhancement factor
    temperature_q10: float
    moisture_response: float


class CarbonSequestrationModel:
    """
    Models carbon flow with MBT55-enhanced cycling efficiency.
    Based on theoretical equations from MBT Hypercycle documentation.
    """
    
    def __init__(self, apply_mbt_enhancement: bool = True):
        self.apply_mbt_enhancement = apply_mbt_enhancement
        
        # MBT55 enhancement factors from theoretical documents
        self.mbt_factors = {
            'humification_rate': 2.8,      # 2.8x faster humification
            'carbon_retention': 1.5,       # 1.5x better retention
            'microbial_efficiency': 1.8,   # 1.8x higher efficiency
            'decomposition_rate': 10.0,    # 10x faster decomposition (24h vs 3-6mo)
            'methane_reduction': 0.82,     # 82% methane reduction
            'fire_risk_reduction': 0.42    # 42% fire risk reduction
        }
        
        # Initialize carbon pools (kg C/ha)
        self.carbon_pools = self._initialize_pools()
        self.flow_network = self._initialize_flows()
        
    def _initialize_pools(self) -> Dict[CarbonPool, float]:
        """Initialize carbon pools based on typical agricultural soils."""
        return {
            CarbonPool.MICROBIAL_BIOMASS: 500.0,
            CarbonPool.DISSOLVED_ORGANIC: 1000.0,
            CarbonPool.HUMUS_STABLE: 5000.0,
            CarbonPool.HUMUS_ACTIVE: 2000.0,
            CarbonPool.PARTICULATE: 3000.0,
            CarbonPool.RESPIRED: 0.0
        }
    
    def _initialize_flows(self) -> List[CarbonFlow]:
        """Initialize carbon flows with MBT55 enhancement."""
        flows = []
        
        # From theoretical document: CCE = (C_stable/C_input) × (t_retention/t_decomp)
        base_decomposition_rate = 1.0  # kg C/ha/day
        
        if self.apply_mbt_enhancement:
            decomposition_rate = base_decomposition_rate * self.mbt_factors['decomposition_rate']
            humification_rate = base_decomposition_rate * self.mbt_factors['humification_rate']
        else:
            decomposition_rate = base_decomposition_rate
            humification_rate = base_decomposition_rate * 0.43  # Conventional: 43% of MBT55
            
        # Decomposition: Particulate → Dissolved Organic
        flows.append(CarbonFlow(
            source=CarbonPool.PARTICULATE,
            target=CarbonPool.DISSOLVED_ORGANIC,
            base_rate=decomposition_rate * 0.8,
            mbt_enhancement=self.mbt_factors['decomposition_rate'],
            temperature_q10=2.0,
            moisture_response=1.5
        ))
        
        # Microbial assimilation
        flows.append(CarbonFlow(
            source=CarbonPool.DISSOLVED_ORGANIC,
            target=CarbonPool.MICROBIAL_BIOMASS,
            base_rate=decomposition_rate * 0.45,
            mbt_enhancement=self.mbt_factors['microbial_efficiency'],
            temperature_q10=1.8,
            moisture_response=1.2
        ))
        
        # Humification (MBT55 significantly enhances)
        flows.append(CarbonFlow(
            source=CarbonPool.MICROBIAL_BIOMASS,
            target=CarbonPool.HUMUS_STABLE,
            base_rate=humification_rate * 0.35,
            mbt_enhancement=self.mbt_factors['humification_rate'],
            temperature_q10=1.5,
            moisture_response=1.8
        ))
        
        # Respiration (MBT55 reduces)
        respiration_reduction = 1.0 / self.mbt_factors['carbon_retention']
        flows.append(CarbonFlow(
            source=CarbonPool.MICROBIAL_BIOMASS,
            target=CarbonPool.RESPIRED,
            base_rate=decomposition_rate * 0.55 * respiration_reduction,
            mbt_enhancement=respiration_reduction,
            temperature_q10=2.2,
            moisture_response=1.3
        ))
        
        return flows
    
    def calculate_carbon_cycling_efficiency(self, time_period_days: int = 365) -> Dict:
        """
        Calculate Carbon Cycling Efficiency (CCE) based on theoretical formula:
        CCE = (C_stable / C_input) × (t_retention / t_decomp)
        
        From MBT Hypercycle document:
        - Traditional: CCE = 0.3 (t_decomp = 3 years)
        - MBT55: CCE = 0.8 (t_decomp = 0.3 years)
        """
        
        # Calculate based on theoretical values
        if self.apply_mbt_enhancement:
            # MBT55 values from theoretical document
            t_decomp = 0.3  # years (vs 3 years conventional)
            t_retention = 10.0  # years (humus stability)
            c_stable_ratio = 0.8  # 80% of input becomes stable carbon
        else:
            # Conventional values
            t_decomp = 3.0  # years
            t_retention = 5.0  # years
            c_stable_ratio = 0.3  # 30% of input becomes stable carbon
        
        # Theoretical CCE calculation
        cce = c_stable_ratio * (t_retention / t_decomp)
        
        # Calculate actual sequestration from flows
        daily_sequestration = self._calculate_daily_sequestration()
        annual_sequestration = daily_sequestration * 365
        
        # Convert to CO₂ equivalents (1 kg C = 3.67 kg CO₂)
        annual_co2e = annual_sequestration * 3.67 / 1000  # t CO₂e/ha/yr
        
        # Theoretical improvement from document: 186% increase
        conventional_baseline = annual_co2e / 2.86 if self.apply_mbt_enhancement else annual_co2e
        
        return {
            'carbon_cycling_efficiency': cce,
            'decomposition_time_years': t_decomp,
            'retention_time_years': t_retention,
            'stable_carbon_ratio': c_stable_ratio,
            'annual_sequestration_t_co2e': annual_co2e,
            'improvement_vs_conventional': 1.86 if self.apply_mbt_enhancement else 0.0,
            'theoretical_values': {
                'traditional_CCE': 0.3,
                'mbt55_CCE': 0.8,
                'improvement_factor': 2.67
            }
        }
    
    def _calculate_daily_sequestration(self) -> float:
        """Calculate daily carbon sequestration."""
        daily_rate = 0.0
        
        for flow in self.flow_network:
            if flow.target == CarbonPool.HUMUS_STABLE:
                rate = flow.base_rate
                if self.apply_mbt_enhancement:
                    rate *= flow.mbt_enhancement
                daily_rate += rate
        
        return daily_rate
    
    def calculate_humus_formation(self, microbial_biomass: float, msi: float = 0.7) -> float:
        """
        Calculate humus formation rate based on theoretical equation:
        dH/dt = k_h × (1 - e^(-λ × MSI)) × C_microbial
        
        From theoretical document:
        - k_h: humification rate constant (MBT55: 0.35, Conventional: 0.12)
        - λ: sensitivity coefficient (0.35)
        - MSI: Microbial Synergy Index (>0.7 for MBT55)
        - C_microbial: microbial biomass carbon
        """
        
        if self.apply_mbt_enhancement:
            k_h = 0.35  # MBT55 humification constant
            lambda_val = 0.35  # Sensitivity coefficient
        else:
            k_h = 0.12  # Conventional humification constant
            lambda_val = 0.25  # Lower sensitivity
            
        # Humus formation rate (kg C/ha/day)
        humus_rate = k_h * (1 - np.exp(-lambda_val * msi)) * microbial_biomass
        
        # Apply MBT55 enhancement if applicable
        if self.apply_mbt_enhancement:
            humus_rate *= self.mbt_factors['humification_rate']
        
        return humus_rate
    
    def project_10_year_impact(self, area_ha: float = 1.0) -> Dict:
        """
        Project 10-year carbon sequestration impact based on theoretical data.
        
        From theoretical document table:
        | Metric | Conventional | MBT55 | Improvement |
        |--------|--------------|-------|-------------|
        | Carbon Sequestration | 38.2 tCO₂e/ha | 109.5 tCO₂e/ha | +186% |
        | Humus Generated | 12.1 t/ha | 35.8 t/ha | +196% |
        """
        
        if self.apply_mbt_enhancement:
            annual_seq_t_co2e = 10.95  # tCO₂e/ha/yr (109.5/10 years)
            annual_humus_t = 3.58  # t/ha/yr (35.8/10 years)
            methane_reduction = 0.82  # 82% reduction
            fire_risk_reduction = 0.42  # 42% reduction
        else:
            annual_seq_t_co2e = 3.82  # tCO₂e/ha/yr (38.2/10 years)
            annual_humus_t = 1.21  # t/ha/yr (12.1/10 years)
            methane_reduction = 0.10  # 10% reduction
            fire_risk_reduction = 0.05  # 5% reduction
        
        # 10-year totals
        total_seq = annual_seq_t_co2e * 10 * area_ha
        total_humus = annual_humus_t * 10 * area_ha
        
        # Additional benefits
        methane_reduced = annual_seq_t_co2e * 0.1 * methane_reduction * 10 * area_ha
        fire_prevention = annual_seq_t_co2e * 0.15 * fire_risk_reduction * 10 * area_ha
        
        # Net Climate Benefit (NCB)
        ncb = total_seq + methane_reduced + fire_prevention
        
        return {
            '10_year_sequestration_t_co2e': total_seq,
            '10_year_humus_formation_t': total_humus,
            'methane_reduction_t_co2e': methane_reduced,
            'fire_prevention_benefit_t_co2e': fire_prevention,
            'net_climate_benefit_t_co2e': ncb,
            'annual_rates': {
                'sequestration': annual_seq_t_co2e,
                'humus_formation': annual_humus_t,
                'methane_reduction': methane_reduction,
                'fire_risk_reduction': fire_risk_reduction
            },
            'improvement_factors': {
                'sequestration': 1.86,  # +186%
                'humus': 1.96,  # +196%
                'methane': 7.2,  # 72 percentage points
                'fire_risk': 0.37  # 37 percentage points
            }
        }
    
    def generate_carbon_credits(self, 
                               area_ha: float, 
                               duration_years: int = 10,
                               verification_level: str = 'high') -> Dict:
        """
        Generate carbon credit calculations based on theoretical performance.
        
        Args:
            area_ha: Area in hectares
            duration_years: Project duration
            verification_level: 'basic', 'standard', 'high'
            
        Returns:
            Carbon credit calculation with verification requirements
        """
        
        # Theoretical sequestration rates
        if self.apply_mbt_enhancement:
            base_rate = 10.95  # tCO₂e/ha/yr from theoretical data
        else:
            base_rate = 3.82   # tCO₂e/ha/yr from theoretical data
        
        # Verification adjustment (conservativeness)
        verification_factors = {
            'basic': 0.7,
            'standard': 0.85,
            'high': 0.95
        }
        
        adjustment = verification_factors.get(verification_level, 0.85)
        
        # Calculate total credits
        annual_credits = base_rate * adjustment
        total_credits = annual_credits * duration_years * area_ha
        
        # Permanence risk adjustment (MBT55 reduces reversibility risk)
        if self.apply_mbt_enhancement:
            permanence_risk = 0.05  # 5% risk (vs 15% conventional)
        else:
            permanence_risk = 0.15
        
        risk_adjusted_credits = total_credits * (1 - permanence_risk)
        
        # Co-benefits from theoretical document
        co_benefits = {
            'water_retention_ml_per_year': 300 * area_ha,  # 300 ML/ha/yr
            'biodiversity_units': 15 * area_ha,  # Biodiversity units
            'soil_health_improvement': 25 * area_ha,  # SHE score improvement
            'yield_increase_percent': 12 * area_ha  # % yield increase
        }
        
        return {
            'total_credits_t_co2e': risk_adjusted_credits,
            'annual_rate_t_co2e_ha': annual_credits,
            'project_duration_years': duration_years,
            'area_ha': area_ha,
            'verification_level': verification_level,
            'permanence_risk_factor': permanence_risk,
            'co_benefits': co_benefits,
            'monitoring_requirements': self._get_monitoring_requirements(verification_level),
            'theoretical_basis': {
                'data_source': 'MBT Hypercycle Theoretical Document',
                'improvement_over_conventional': '186% carbon sequestration',
                'validation_status': 'Theoretically validated, field validation pending'
            }
        }
    
    def _get_monitoring_requirements(self, level: str) -> List[str]:
        """Get monitoring requirements based on verification level."""
        requirements = {
            'basic': [
                'Annual soil sampling (0-30cm depth)',
                'Basic soil carbon analysis (Walkley-Black method)',
                'Land management practice records'
            ],
            'standard': [
                'Semi-annual soil sampling',
                'Full soil carbon fractionation (SOC, POC, MAOC)',
                'Microbial biomass assessment (fumigation-extraction)',
                'Remote sensing verification (NDVI, soil moisture)'
            ],
            'high': [
                'Quarterly soil sampling with isotopic analysis (δ¹³C)',
                'Full microbial community profiling (16S/ITS sequencing)',
                'Continuous sensor monitoring (temperature, moisture, CO₂ flux)',
                'Third-party verification with peer review',
                'Long-term stability assessment (10+ years)'
            ]
        }
        return requirements.get(level, requirements['standard'])
