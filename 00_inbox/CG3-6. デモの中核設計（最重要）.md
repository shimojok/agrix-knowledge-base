# ③ Pythonデモ（そのまま使える最小構成）

まずはCLIで十分です（即動くことが重要）

```python
import numpy as np

def simulate(mbt55=True):
    # 初期値
    D0 = 0.4
    Y0 = 5.0
    C = 2.0
    C0 = 100
    demand = 100
    
    # MBT55効果
    M = 1.0 if mbt55 else 0.3
    
    # モデル
    D = D0 * np.exp(-1.2 * M)
    Y = Y0 * (1 - D) + 0.5 * C
    cost = C0 - 20 * M
    price = demand / Y
    co2 = 2 - (1.5 * M + 0.3 * C)
    
    return {
        "Yield": Y,
        "Cost": cost,
        "Price": price,
        "CO2": co2
    }

# 実行比較
print("MBT55 OFF:", simulate(False))
print("MBT55 ON :", simulate(True))
```
