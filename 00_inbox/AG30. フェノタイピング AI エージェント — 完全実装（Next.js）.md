# 2. **Microbe-Agent（ハイパーサイクル回転儀）**

`app/api/agents/microbe/route.ts`

ts

```
import { NextRequest, NextResponse } from "next/server";
import { SoilSignals, MetabolicCurrency } from "@/app/lib/agents/types";

export async function POST(req: NextRequest) {
  const body = (await req.json()) as { soil: SoilSignals };
  const { ph, eh_mV, moisture_pct, co2_flux_mg_m2_h } = body.soil;

  const aerobicFraction =
    1 / (1 + Math.exp(-0.02 * (eh_mV - 50))) *
    (1 - Math.max(0, (moisture_pct - 70) / 30));

  const omegaCycle = co2_flux_mg_m2_h * aerobicFraction;

  const currency: MetabolicCurrency = {
    lactate_mmol_l: 2.5 * aerobicFraction,
    scfa_pool_mmol_l: 5.0 * aerobicFraction,
    electron_balance_index: Math.tanh(eh_mV / 200),
  };

  return NextResponse.json({
    omegaCycle,
    metabolicCurrency: currency,
  });
}
```

```
import { NextRequest, NextResponse } from "next/server";
import { PlantSignals, MetabolicCurrency } from "@/app/lib/agents/types";

export async function POST(req: NextRequest) {
  const body = (await req.json()) as {
    plant: PlantSignals;
    currency: MetabolicCurrency;
  };

  const { ndvi, pri, polyphenol_density_index } = body.plant;
  const { scfa_pool_mmol_l, electron_balance_index } = body.currency;

  const functionalConversion =
    polyphenol_density_index *
    (0.6 * ndvi + 0.4 * pri) *
    (1 + 0.3 * electron_balance_index);

  const phiFunctional = functionalConversion * (1 + scfa_pool_mmol_l / 10);

  return NextResponse.json({ phiFunctional });
}
```

  const pathogenPressure = Math.log10(pathogen_load_cfu_g + 1);

# 5. **Inertia-Agent（鮮度保持力）**

`app/api/agents/inertia/route.ts`

ts

```
import { NextRequest, NextResponse } from "next/server";
import { PostHarvestSignals, MetabolicCurrency } from "@/app/lib/agents/types";

export async function POST(req: NextRequest) {
  const body = (await req.json()) as {
    postharvest: PostHarvestSignals;
    currency: MetabolicCurrency;
  };

  const {
    atp_initial,
    atp_current,
    respiration_rate_mg_co2_kg_h,
    ethylene_ppm,
    elapsed_days,
  } = body.postharvest;

  const { scfa_pool_mmol_l } = body.currency;

  const atpRatio = atp_current / atp_initial;
  const stressFactor = respiration_rate_mg_co2_kg_h * (1 + ethylene_ppm / 10);

  const inertiaRaw =
    atpRatio * (1 + scfa_pool_mmol_l / 10) / (1 + stressFactor / 100);

  const tauInertiaDays = Math.max(0, inertiaRaw * 14 - elapsed_days);

  return NextResponse.json({
    vScore: inertiaRaw,
    tauInertiaDays,
  });
}
```

# 6. **Phenotyping-Orchestrator（4体統合）**

`app/api/phenotyping/route.ts`

ts

```
import { NextRequest, NextResponse } from "next/server";
import {
  SoilSignals,
  PlantSignals,
  PathogenSignals,
  PostHarvestSignals,
} from "@/app/lib/agents/types";

export async function POST(req: NextRequest) {
  const body = (await req.json()) as {
    soil: SoilSignals;
    plant: PlantSignals;
    pathogen: PathogenSignals;
    postharvest: PostHarvestSignals;
  };

  const base = process.env.NEXT_PUBLIC_BASE_URL!;

  const micro = await fetch(`${base}/api/agents/microbe`, {
    method: "POST",
    body: JSON.stringify({ soil: body.soil }),
  }).then((r) => r.json());

  const cascade = await fetch(`${base}/api/agents/cascade`, {
    method: "POST",
    body: JSON.stringify({
      plant: body.plant,
      currency: micro.metabolicCurrency,
    }),
  }).then((r) => r.json());

  const stability = await fetch(`${base}/api/agents/stability`, {
    method: "POST",
    body: JSON.stringify({
      pathogen: body.pathogen,
      currency: micro.metabolicCurrency,
    }),
  }).then((r) => r.json());

  const inertia = await fetch(`${base}/api/agents/inertia`, {
    method: "POST",
    body: JSON.stringify({
      postharvest: body.postharvest,
      currency: micro.metabolicCurrency,
    }),
  }).then((r) => r.json());

  const functionality = cascade.phiFunctional;
  const inertiaScore = inertia.vScore;
  const security = stability.sigmaResilience;

  const vIndex = Math.cbrt(functionality * inertiaScore * security);

  return NextResponse.json({
    micro,
    cascade,
    stability,
    inertia,
    vIndex,
  });
}
```
