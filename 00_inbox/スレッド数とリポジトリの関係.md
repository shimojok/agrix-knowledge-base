## ① 概要

## ④ Pythonコード（コア）

```python
import numpy as np
from scipy.integrate import odeint

def model(y, t, params):
    X = y[:4]
    S = y[4]

    r = params["r"]
    A = params["A"]
    k = params["k"]
    Ks = params["Ks"]

    dX = []
    for i in range(4):
        interaction = sum(A[i][j] * X[j] for j in range(4))
        growth = X[i] * (r[i] + interaction)
        dX.append(growth)

    dS = -sum(k[i] * X[i] * S / (Ks + S) for i in range(4))

    return dX + [dS]

# 初期値
y0 = [1,1,1,1,100]
t = np.linspace(0, 24, 100)

params = {
    "r":[0.3,0.2,0.25,0.15],
    "A":np.ones((4,4))*0.05,
    "k":[0.1,0.1,0.1,0.1],
    "Ks":10
}

result = odeint(model, y0, t, args=(params,))
```

    S_final = result[-1,4]

    decomposition_eff = 1 - S_final/100

    cost_reduction = decomposition_eff * 0.84

    carbon_reduction = decomposition_eff * 100  # 仮係数

    carbon_value = carbon_reduction * carbon_price

    return {
