## 🧱 プラットフォーム全体アーキテクチャ（4層構造）

### Layer 1: **AGRIX-OS（農業生態系制御層）**

- バイオセキュリティ（病原菌制御）

- 機能性インデックス（ポリフェノール密度）
- 鮮度保持力（V-Score/代謝寿命）
- SafetyChain証明書（消費者選択基準）
- 4レイヤー動的価格（V+L+m+C）

### Layer 4: **PBPE Marketplace（金融・取引層）**
- 農作物取引＝排出権取引
- 炭素クレジット（Carbon Credit）連携
- ロス削減価値の資産化

```typescript
// lib/agrix/hypercycleModel.ts

export type HypercycleInput = {
  ph: number;
  eh_mV: number;
  moisture_pct: number;
  temperature_c: number;
  organicCarbon_gL: number;
  availableNitrogen_gL: number;
  mbt55Dosage_Lha: number;  // MBT55投入量
};

export type HypercycleOutput = {
  mbt55ActivityIndex: number;      // 0-1
  degradationRate24h: number;      // 0-1（24時間有機物分解率）
  cnShift: number;                 // C/N変化量（初期→24h後）
  hypercycleScore: number;         // 0-100（総合評価）
  status: 'ACTIVE' | 'SUBOPTIMAL' | 'INACTIVE';
};

export function evaluateHypercycle(input: HypercycleInput): HypercycleOutput {
  // MBT55の活性は pH 7.2-7.5、Eh 100-200mV、水分40-60% が最適
  const pHScore = 1 - Math.abs(input.ph - 7.35) / 2;
  const ehScore = 1 - Math.abs(input.eh_mV - 150) / 200;
  const moistureScore = 1 - Math.abs(input.moisture_pct - 50) / 30;
  const tempScore = 1 - Math.abs(input.temperature_c - 28) / 20;

  const activityIndex = (pHScore * 0.3 + ehScore * 0.25 + moistureScore * 0.25 + tempScore * 0.2);
  
  // 分解率は活性指数と投入量に依存
  const degradationRate24h = Math.min(0.95, 0.5 * activityIndex + 0.3 * (input.mbt55Dosage_Lha / 3) + 0.2);
  
  // C/Nシフト（MBT55は高C/N比を15-18に引き下げる）
  const initialCN = input.organicCarbon_gL / Math.max(input.availableNitrogen_gL, 0.001);
  const targetCN = 16.5;
  const cnShift = Math.max(0, initialCN - targetCN);
  
  const hypercycleScore = (activityIndex * 0.4 + degradationRate24h * 0.4 + Math.min(cnShift / 20, 1) * 0.2) * 100;
  
  return {
    mbt55ActivityIndex: Math.max(0, Math.min(1, activityIndex)),
    degradationRate24h,
    cnShift,
    hypercycleScore,
    status: hypercycleScore >= 70 ? 'ACTIVE' : hypercycleScore >= 40 ? 'SUBOPTIMAL' : 'INACTIVE'
  };
}
```

### ② MSC（MBT Sustainable Cycle）演算エンジン

```typescript
// lib/msc/carbonSequestration.ts

export type MSCInput = {
  hypercycle: HypercycleOutput;
  soilData: SoilData;
  cropType: string;
  areaHa: number;
};

export type MSCOutput = {
  carbonSequestration: {
    tCO2PerHaPerYear: number;
    totalTCO2: number;
    humificationRate: number;  // 腐植化率
  };
  biocontrol: {
    pathogenSuppressionRate: number;  // 病原菌抑制率
    diseaseIncidenceReduction: number; // 病害発生低減率
  };
  yieldImprovement: {
    yieldIncreaseRate: number;  // 収量増加率（%）
    qualityScore: number;       // 品質スコア
  };
  totalEconomicValue: number;  // 総合経済価値（円/ha）
};

export function evaluateMSC(input: MSCInput): MSCOutput {
  const { hypercycle, soilData, areaHa } = input;
  
  // ① 炭素隔離（腐植質生成）
  const baseSequestration = 2.0; // tCO2/ha/year 基準
  const humificationRate = 0.4 * hypercycle.degradationRate24h + 0.3 * hypercycle.mbt55ActivityIndex + 0.3;
  const sequestration = baseSequestration * humificationRate * (1 + soilData.organicCarbon_gL / 20);
  
  // ② バイオコントロール（病原菌抑制）
  const pathogenSuppressionRate = 0.3 + 0.5 * hypercycle.mbt55ActivityIndex;
  const diseaseIncidenceReduction = 30 + 40 * hypercycle.mbt55ActivityIndex; // %
  
  // ③ 収量・品質向上
  const baseYieldIncrease = 5; // %（基準）
  const yieldIncreaseRate = baseYieldIncrease + 15 * hypercycle.hypercycleScore / 100;
  const qualityScore = 60 + 35 * (hypercycle.hypercycleScore / 100);
  
  // ④ 総合経済価値
  const carbonValue = sequestration * 5000; // 5,000円/tCO2
  const yieldValue = (yieldIncreaseRate / 100) * 1000000; // 収量増加分（仮）
  const qualityPremium = (qualityScore / 100) * 200000; // 品質プレミアム
  const lossReduction = diseaseIncidenceReduction / 100 * 300000; // ロス削減
  
  const totalEconomicValue = carbonValue + yieldValue + qualityPremium + lossReduction;
  
  return {
    carbonSequestration: {
      tCO2PerHaPerYear: Number(sequestration.toFixed(2)),
      totalTCO2: Number((sequestration * areaHa).toFixed(2)),
      humificationRate: Number(humificationRate.toFixed(3))
    },
    biocontrol: {
      pathogenSuppressionRate: Number(pathogenSuppressionRate.toFixed(3)),
      diseaseIncidenceReduction: Number(diseaseIncidenceReduction.toFixed(1))
    },
    yieldImprovement: {
      yieldIncreaseRate: Number(yieldIncreaseRate.toFixed(1)),
      qualityScore: Number(qualityScore.toFixed(1))
    },
    totalEconomicValue: Number(totalEconomicValue.toFixed(0))
  };
}
```

### ③ SafetyChain™ 演算エンジン（統合版）

```typescript
// lib/safetychain/integratedEngine.ts

import { evaluateHypercycle } from '@/lib/agrix/hypercycleModel';
import { evaluateMSC } from '@/lib/msc/carbonSequestration';

export type SafetyChainInput = {
  soilData: SoilData;
  harvestData: {
    atpInitial: number;
    atpCurrent: number;
    elapsedDays: number;
    storageTempC: number;
  };
  phenotypeSignals: {
    colorScore?: number;
    aromaScore?: number;
    textureScore?: number;
    ndvi?: number;
    pri?: number;
  };
  areaHa: number;
  cropType: string;
};

export type SafetyChainOutput = {
  // AGRIX連携
  hypercycle: HypercycleOutput;
  msc: MSCOutput;
  
  // V-Score（鮮度保持力）
  vScore: number;
  grade: 'Platinum' | 'Gold' | 'Silver' | 'Standard';
  metabolicLifeDays: number;
  
  // 機能性（ポリフェノール）
  polyphenolDensity: number;  // 0-1
  functionalityIndex: number; // 0-100
  functionalityGrade: 'A' | 'B' | 'C' | 'D';
  
  // 証明書
  certificate: {
    lotId: string;
    issuedAt: string;
    validUntil: string;
    summary: string;
  };
  
  // 価格
  pricing: {
    basePrice: number;
    vComponent: number;
    lComponent: number;
    mComponent: number;
    cComponent: number;
    totalPrice: number;
  };
};

export function evaluateSafetyChain(input: SafetyChainInput): SafetyChainOutput {
  // 1. AGRIXハイパーサイクル評価
  const hypercycle = evaluateHypercycle({
    ph: input.soilData.pH,
    eh_mV: input.soilData.Eh,
    moisture_pct: input.soilData.moisture,
    temperature_c: input.soilData.temperature,
    organicCarbon_gL: input.soilData.SOC,
    availableNitrogen_gL: input.soilData.TN,
    mbt55Dosage_Lha: input.soilData.mbt55Dosage || 2.5
  });
  
  // 2. MSC評価
  const msc = evaluateMSC({
    hypercycle,
    soilData: input.soilData,
    cropType: input.cropType,
    areaHa: input.areaHa
  });
  
  // 3. V-Score（鮮度保持力）
  const atpRatio = input.harvestData.atpCurrent / Math.max(input.harvestData.atpInitial, 0.0001);
  const protection = 0.5 * (msc.yieldImprovement.qualityScore / 100) + 0.5 * (hypercycle.mbt55ActivityIndex);
  const tempFactor = Math.pow(2, (input.harvestData.storageTempC - 4) / 10);
  const kAtp = -Math.log(Math.max(atpRatio, 0.0001)) / Math.max(input.harvestData.elapsedDays, 0.001);
  const adjustedK = kAtp * tempFactor * (1 - 0.3 * protection);
  
  const hoursToThreshold = Math.log(0.35) / -Math.max(adjustedK, 0.0001);
  const metabolicLifeDays = Math.max(0, (hoursToThreshold - input.harvestData.elapsedDays * 24) / 24);
  
  const vScoreRaw = 0.5 * atpRatio + 0.3 * protection + 0.2 * (1 - Math.min(input.harvestData.elapsedDays / 14, 1));
  const vScore = Math.max(0, Math.min(100, vScoreRaw * 100));
  
  let grade: SafetyChainOutput['grade'] = 'Standard';
  if (vScore >= 85) grade = 'Platinum';
  else if (vScore >= 70) grade = 'Gold';
  else if (vScore >= 55) grade = 'Silver';
  
  // 4. ポリフェノール密度・機能性インデックス
  const polyphenolDensity = 0.4 * hypercycle.mbt55ActivityIndex + 
                            0.3 * (input.phenotypeSignals.ndvi || 0.5) +
                            0.3 * (input.phenotypeSignals.pri || 0.5);
  const sensory = ((input.phenotypeSignals.colorScore || 0.5) + 
                   (input.phenotypeSignals.aromaScore || 0.5) + 
                   (input.phenotypeSignals.textureScore || 0.5)) / 3;
  const functionalityIndex = 0.5 * polyphenolDensity + 0.3 * (msc.yieldImprovement.qualityScore / 100) + 0.2 * sensory;
  
  // 5. 動的価格（4レイヤー）
  const basePrice = 300; // 基準市場価格
  const vComponent = vScore * 1.0;
  const lComponent = (msc.biocontrol.diseaseIncidenceReduction / 100) * 50;
  const mComponent = (1 - (input.harvestData.atpCurrent / input.harvestData.atpInitial)) * 100;
  const cComponent = msc.carbonSequestration.tCO2PerHaPerYear * 5;
  const totalPrice = basePrice + vComponent + lComponent + mComponent + cComponent;
  
  // 6. 証明書
  const lotId = `SC-${Date.now()}-${input.cropType.slice(0, 3).toUpperCase()}`;
  const issuedAt = new Date().toISOString();
  const validUntil = new Date(Date.now() + metabolicLifeDays * 24 * 60 * 60 * 1000).toISOString();
  
  return {
    hypercycle,
    msc,
    vScore,
    grade,
    metabolicLifeDays: Number(metabolicLifeDays.toFixed(1)),
    polyphenolDensity: Number(polyphenolDensity.toFixed(3)),
    functionalityIndex: Number((functionalityIndex * 100).toFixed(1)),
    functionalityGrade: functionalityIndex >= 0.8 ? 'A' : functionalityIndex >= 0.6 ? 'B' : functionalityIndex >= 0.4 ? 'C' : 'D',
    certificate: {
      lotId,
      issuedAt,
      validUntil,
      summary: `SafetyChain™ 認証済み: ${grade}グレード, 機能性${functionalityIndex >= 0.7 ? '優良' : '標準'}`
    },
    pricing: {
      basePrice,
      vComponent: Number(vComponent.toFixed(0)),
      lComponent: Number(lComponent.toFixed(0)),
      mComponent: Number(mComponent.toFixed(0)),
      cComponent: Number(cComponent.toFixed(0)),
      totalPrice: Number(totalPrice.toFixed(0))
    }
  };
}
```

### ④ PBPE Marketplace 演算エンジン

```typescript
// lib/pbpe/marketplaceEngine.ts

export type MarketOrder = {
  orderId: string;
  type: 'BUY' | 'SELL';
  assetType: 'CROP' | 'CARBON_CREDIT' | 'SAFETY_CHAIN_CERT';
  quantity: number;
  price: number;
  qualityScore: number;  // 0-100（SafetyChainスコア連携）
  carbonSequestration: number; // tCO2
};

export type MarketMatch = {
  buyOrder: MarketOrder;
  sellOrder: MarketOrder;
  matchPrice: number;
  matchQuantity: number;
  totalValue: number;
  premiumBreakdown: {
    qualityPremium: number;
    carbonPremium: number;
    safetyPremium: number;
  };
};

export function matchOrders(buyOrders: MarketOrder[], sellOrders: MarketOrder[]): MarketMatch[] {
  // シンプルなマッチングエンジン
  const matches: MarketMatch[] = [];
  const sortedBuys = [...buyOrders].sort((a, b) => b.price - a.price);
  const sortedSells = [...sellOrders].sort((a, b) => a.price - b.price);
  
  for (const buy of sortedBuys) {
    for (const sell of sortedSells) {
      if (buy.price >= sell.price && buy.quantity > 0 && sell.quantity > 0) {
        const matchQuantity = Math.min(buy.quantity, sell.quantity);
        const matchPrice = (buy.price + sell.price) / 2;
        
        // 品質プレミアム（SafetyChainスコアに基づく）
        const avgQuality = (buy.qualityScore + sell.qualityScore) / 2;
        const qualityPremium = avgQuality > 70 ? (avgQuality - 70) * 5 : 0;
        const carbonPremium = (buy.carbonSequestration + sell.carbonSequestration) / 2 * 10;
        const safetyPremium = avgQuality > 80 ? 50 : 0;
        
        matches.push({
          buyOrder: { ...buy, quantity: matchQuantity },
          sellOrder: { ...sell, quantity: matchQuantity },
          matchPrice: matchPrice + qualityPremium + carbonPremium + safetyPremium,
          matchQuantity,
          totalValue: (matchPrice + qualityPremium + carbonPremium + safetyPremium) * matchQuantity,
          premiumBreakdown: {
            qualityPremium,
            carbonPremium,
            safetyPremium
          }
        });
        
        buy.quantity -= matchQuantity;
        sell.quantity -= matchQuantity;
      }
    }
  }
  
  return matches;
}

export function calculatePBPEIndex(orders: MarketOrder[]): {
  index: number;
  trend: 'RISING' | 'STABLE' | 'FALLING';
  components: {
    qualityComponent: number;
    carbonComponent: number;
    safetyComponent: number;
  };
} {
  const avgQuality = orders.reduce((sum, o) => sum + o.qualityScore, 0) / orders.length;
  const avgCarbon = orders.reduce((sum, o) => sum + o.carbonSequestration, 0) / orders.length;
  const avgPrice = orders.reduce((sum, o) => sum + o.price, 0) / orders.length;
  
  const qualityComponent = avgQuality * 0.5;
  const carbonComponent = avgCarbon * 20;
  const safetyComponent = avgQuality > 70 ? 30 : 0;
  const index = avgPrice + qualityComponent + carbonComponent + safetyComponent;
  
  return {
    index: Number(index.toFixed(2)),
    trend: index > 100 ? 'RISING' : index < 90 ? 'FALLING' : 'STABLE',
    components: {
      qualityComponent: Number(qualityComponent.toFixed(2)),
      carbonComponent: Number(carbonComponent.toFixed(2)),
      safetyComponent: Number(safetyComponent.toFixed(2))
    }
  };
}
```
