## 📂 フェーズ3：リポジトリ・Obsidian Vaultを読み込み、Pineconeに格納するPythonスクリプト

```python
# scripts/ingest_knowledge.py
import os
import re
from pathlib import Path
from dotenv import load_dotenv
from github import Github, GithubException
from langchain.text_splitter import MarkdownHeaderTextSplitter, RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_pinecone import PineconeVectorStore
from pinecone import Pinecone, ServerlessSpec
import tempfile
import shutil
import git
from typing import List, Dict, Any

load_dotenv()

# ----- 設定 -----
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
PINECONE_INDEX_NAME = os.getenv("PINECONE_INDEX_NAME")
OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")
GITHUB_TOKEN = os.getenv("GITHUB_TOKEN")

# 知識源リポジトリリスト
REPOSITORIES = [
    {"url": "https://github.com/shimojok/AGRIX-OS.git", "branch": "main"},
    {"url": "https://github.com/shimojok/M3-Core-Engine.git", "branch": "main"},
    {"url": "https://github.com/shimojok/M3-BioSynergy-Core.git", "branch": "main"},
    {"url": "https://github.com/shimojok/M3-BioSynergy.git", "branch": "main"},
    {"url": "https://github.com/shimojok/BioNexus-Core.git", "branch": "main"},
    {"url": "https://github.com/shimojok/BioNexus-Enterprise.git", "branch": "main"},
]

# Obsidian Vaultのパス（ローカル）
OBSIDIAN_VAULT_PATH = os.getenv("OBSIDIAN_VAULT_PATH", "~/Documents/Obsidian/AGRIX_Vault")

# 処理対象ファイル拡張子
TARGET_EXTENSIONS = [".md", ".py", ".txt", ".json", ".csv", ".js", ".ts", ".ipynb"]

# ----- Pinecone初期化 -----
pc = Pinecone(api_key=PINECONE_API_KEY)

# インデックスが存在しない場合のみ作成
if PINECONE_INDEX_NAME not in pc.list_indexes().names():
    pc.create_index(
        name=PINECONE_INDEX_NAME,
        dimension=1536,
        metric="cosine",
        spec=ServerlessSpec(cloud="aws", region="us-east-1")
    )

embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
vector_store = PineconeVectorStore(index_name=PINECONE_INDEX_NAME, embedding=embeddings)

# ----- テキスト抽出・分割関数 -----
def extract_text_from_file(file_path: str) -> str:
    """ファイルからテキストを抽出する"""
    ext = Path(file_path).suffix
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
    except UnicodeDecodeError:
        return ""  # バイナリファイルはスキップ
    
    if ext == '.md':
        # Markdownファイルは見出しを維持
        return content
    elif ext == '.py' or ext == '.js' or ext == '.ts':
        # コードファイルは言語指定付きでコードブロック化
        lang = ext[1:]  # .py -> py
        return f"```{lang}\n{content}\n```"
    elif ext == '.json':
        # JSONは整形
        try:
            import json
            return json.dumps(json.loads(content), indent=2)
        except:
            return content
    else:
        return content

def split_documents(text: str, file_path: str, repo_name: str) -> List[Dict[str, Any]]:
    """テキストをチャンクに分割し、メタデータを付与する"""
    # メタデータ用のベース情報
    base_metadata = {
        "source": file_path,
        "repository": repo_name,
        "file_type": Path(file_path).suffix
    }
    
    # Markdownファイルは見出しで分割
    if file_path.endswith('.md'):
        headers_to_split_on = [
            ("#", "Header_1"),
            ("##", "Header_2"),
            ("###", "Header_3"),
        ]
        splitter = MarkdownHeaderTextSplitter(headers_to_split_on)
        try:
            md_splits = splitter.split_text(text)
            chunks = []
            for split in md_splits:
                metadata = {**base_metadata, **split.metadata}
                chunks.append({
                    "page_content": split.page_content,
                    "metadata": metadata
                })
            return chunks
        except:
            pass  # 分割失敗時は通常のテキスト分割にフォールバック
    
    # 通常テキスト分割（再帰的文字分割）
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=2000,
        chunk_overlap=200,
        separators=["\n\n", "\n", "。", ".", " ", ""]
    )
    splits = splitter.split_text(text)
    return [
        {
            "page_content": split,
            "metadata": base_metadata
        }
        for split in splits
    ]

# ----- リポジトリ処理関数 -----
def process_github_repo(repo_url: str, branch: str):
    """GitHubリポジトリをクローンして処理する"""
    repo_name = repo_url.split('/')[-1].replace('.git', '')
    print(f"Processing repository: {repo_name}")
    
    with tempfile.TemporaryDirectory() as tmpdir:
        try:
            # リポジトリをクローン
            repo_path = os.path.join(tmpdir, repo_name)
            git.Repo.clone_from(repo_url, repo_path, branch=branch)
            
            # ファイルを再帰的に処理
            for root, dirs, files in os.walk(repo_path):
                # .git ディレクトリはスキップ
                if '.git' in dirs:
                    dirs.remove('.git')
                
                for file in files:
                    file_path = os.path.join(root, file)
                    # 対象拡張子のみ処理
                    if Path(file_path).suffix in TARGET_EXTENSIONS:
                        try:
                            text = extract_text_from_file(file_path)
                            if not text.strip():
                                continue
                            
                            # 相対パスを取得
                            rel_path = os.path.relpath(file_path, repo_path)
                            
                            # チャンクに分割
                            chunks = split_documents(text, rel_path, repo_name)
                            
                            # Pineconeに格納
                            if chunks:
                                vector_store.add_texts(
                                    texts=[chunk["page_content"] for chunk in chunks],
                                    metadatas=[chunk["metadata"] for chunk in chunks]
                                )
                                print(f"  Added {len(chunks)} chunks from {rel_path}")
                        except Exception as e:
                            print(f"  Error processing {file}: {e}")
        except Exception as e:
            print(f"Error cloning {repo_url}: {e}")

def process_obsidian_vault(vault_path: str):
    """Obsidian Vaultを処理する"""
    vault_expanded = os.path.expanduser(vault_path)
    if not os.path.exists(vault_expanded):
        print(f"Obsidian Vault not found: {vault_expanded}")
        return
    
    print(f"Processing Obsidian Vault: {vault_expanded}")
    
    for root, dirs, files in os.walk(vault_expanded):
        # Obsidianの設定ディレクトリはスキップ
        if '.obsidian' in dirs:
            dirs.remove('.obsidian')
        
        for file in files:
            if file.endswith('.md'):
                file_path = os.path.join(root, file)
                try:
                    text = extract_text_from_file(file_path)
                    if not text.strip():
                        continue
                    
                    # Obsidianノート内の[[リンク]]をテキストとして保持
                    text = re.sub(r'\[\[(.*?)\]\]', r'\1', text)
                    
                    rel_path = os.path.relpath(file_path, vault_expanded)
                    chunks = split_documents(text, rel_path, "obsidian-vault")
                    
                    if chunks:
                        vector_store.add_texts(
                            texts=[chunk["page_content"] for chunk in chunks],
                            metadatas=[chunk["metadata"] for chunk in chunks]
                        )
                        print(f"  Added {len(chunks)} chunks from {rel_path}")
                except Exception as e:
                    print(f"  Error processing {file}: {e}")

# ----- メイン実行 -----
if __name__ == "__main__":
    print("🚀 Starting knowledge ingestion to Pinecone...")
    
    # 1. GitHubリポジトリの処理
    for repo in REPOSITORIES:
        process_github_repo(repo["url"], repo["branch"])
    
    # 2. Obsidian Vaultの処理
    process_obsidian_vault(OBSIDIAN_VAULT_PATH)
    
    print("✅ Knowledge ingestion completed!")
    print(f"Total vectors in index: {pc.Index(PINECONE_INDEX_NAME).describe_index_stats()}")
```

## 🤖 フェーズ4：マルチAIエージェントにPinecone検索機能を実装（LangGraph拡張）

```python
# lib/agents/knowledge_retriever.py
from langchain_pinecone import PineconeVectorStore
from langchain_openai import OpenAIEmbeddings
import os
from typing import List, Dict, Any

def get_relevant_knowledge(query: str, top_k: int = 5) -> List[Dict[str, Any]]:
    """Pineconeから関連知識を検索する"""
    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    vector_store = PineconeVectorStore(
        index_name=os.getenv("PINECONE_INDEX_NAME"),
        embedding=embeddings
    )
    
    results = vector_store.similarity_search(query, k=top_k)
    return [
        {
            "content": doc.page_content,
            "metadata": doc.metadata,
            "score": doc.metadata.get("score", 1.0)
        }
        for doc in results
    ]

# lib/agents/orchestrator/agent.py（拡張部分）
def research_task(task_description: str, context: dict) -> dict:
    """タスク実行前にPineconeで関連知識を検索し、コンテキストに追加する"""
    # タスクの説明から検索クエリを生成
    query = f"{task_description} {context.get('domain', 'MBT55')} {context.get('topic', '')}"
    
    # 関連知識を取得
    relevant_knowledge = get_relevant_knowledge(query, top_k=8)
    
    # コンテキストに追加
    context["pinecone_knowledge"] = relevant_knowledge
    
    # 検証AI用に知識の要約を生成
    knowledge_summary = "\n\n".join([
        f"【{k['metadata'].get('repository', 'unknown')}】{k['content'][:500]}..."
        for k in relevant_knowledge
    ])
    context["knowledge_summary"] = knowledge_summary
    
    return context
```
