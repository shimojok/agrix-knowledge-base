### 修正するメソッド（`issue_pbpe_credit`）

```python
def issue_pbpe_credit(self,
                      mbt55_result: Dict[str, float],
                      agrix_result: Dict[str, float],
                      scope3_result: Dict[str, float],
                      area_ha: float = 1.0,
                      owner: str = "unknown",
                      kpi_id: Optional[str] = None) -> Dict[str, Any]:
    """
    PBPEクレジット発行（Registry連携付き）
    """
    # 1. 既存の統合計算
    integrated = self.integrate_engines(mbt55_result, agrix_result, scope3_result, area_ha)
    
    # 2. KPI-ID生成（なければ作成）
    if kpi_id is None:
        import uuid
        kpi_id = f"KPI-{uuid.uuid4().hex[:8].upper()}"
    
    credit_id = integrated["credit_id"]
    total_pbpe = integrated["weighted_total"]["total_pbpe"]
    components = integrated["components"]
    
    # 3. Registryに記録（新規追加）
    registry_result = registry_engine.record_issuance(
        credit_id=credit_id,
        kpi_id=kpi_id,
        owner=owner,
        amount_pbpe=total_pbpe,
        components=components,
        metadata={
            "area_ha": area_ha,
            "mbt55": mbt55_result,
            "agrix": agrix_result,
            "scope3": scope3_result
        }
    )
    
    # 4. 結果にRegistry情報を追加
    return {
        "credit": integrated,
        "credit_id": credit_id,
        "kpi_id": kpi_id,
        "total_pbpe": total_pbpe,
        "owner": owner,
        "registry_entry": registry_result.get("registry_entry"),
        "registry_id": registry_result.get("registry_entry", {}).get("registry_id"),
        "block_hash": registry_result.get("block", {}).get("hash"),
        "status": "issued" if registry_result.get("success") else "failed"
    }
```

## 📝 2. 修正：`backend/routers/credits.py`

発行APIエンドポイントでRegistry連携を有効化します。

### 修正するエンドポイント

```python
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from engines.pbpe_engine import pbpe_engine

router = APIRouter(prefix="/credits", tags=["credits"])

class IssueCreditRequest(BaseModel):
    mbt55: Dict[str, float]
    agrix: Dict[str, float]
    scope3: Dict[str, float]
    area_ha: float = 1.0
    owner: str = "unknown"
    kpi_id: Optional[str] = None

@router.post("/issue")
def issue_credit(request: IssueCreditRequest):
    """
    PBPEクレジットを発行し、レジストリに記録する
    """
    try:
        result = pbpe_engine.issue_pbpe_credit(
            mbt55_result=request.mbt55,
            agrix_result=request.agrix,
            scope3_result=request.scope3,
            area_ha=request.area_ha,
            owner=request.owner,
            kpi_id=request.kpi_id
        )
        
        return {
            "status": "success",
            "credit_id": result["credit_id"],
            "kpi_id": result["kpi_id"],
            "total_pbpe": result["total_pbpe"],
            "owner": result["owner"],
            "registry_id": result.get("registry_id"),
            "block_hash": result.get("block_hash"),
            "registry_entry": result.get("registry_entry")
        }
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```
