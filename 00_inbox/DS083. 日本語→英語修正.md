## ✅ 課題2: MBT55 ON/OFFで数値が変わらない問題

### 問題の本質

| 問題 | 原因 |
|---|---|
| `mbtEnabled` が各ページに伝わらない | `layout.tsx` の状態を子コンポーネントに渡す仕組みがない |

### 解決策：**URLクエリパラメータ方式**（最もシンプル）

`layout.tsx` でトグルをクリックしたときにURLを変更し、各ページでそのURLパラメータを読み取る方法です。

```tsx
// frontend/app/layout.tsx - トグル部分のみ修正
<button
  onClick={() => {
    const newState = !mbtEnabled;
    setMbtEnabled(newState);
    // URLを更新（クエリパラメータで状態を保持）
    const url = new URL(window.location.href);
    url.searchParams.set('mbt', newState ? 'on' : 'off');
    window.history.pushState({}, '', url.toString());
  }}
  style={{
    width: '100%',
    padding: '10px',
    borderRadius: '8px',
    border: 'none',
    fontWeight: 'bold',
    cursor: 'pointer',
    backgroundColor: mbtEnabled ? '#22c55e' : '#475569',
    color: 'white',
    transition: 'all 0.2s',
  }}
>
  {mbtEnabled ? '🟢 ON' : '⚫ OFF'}
</button>
```

### Step 2: `kpis/page.tsx` をClient Componentに変更

`frontend/app/dashboard/kpis/page.tsx` をClient Componentに変更し、URLパラメータを読み取るようにします：

```tsx
// frontend/app/dashboard/kpis/page.tsx
'use client'

import { useEffect, useState } from 'react'

async function fetchSummary() {
  const res = await fetch('http://localhost:8000/dashboard/summary');
  if (!res.ok) throw new Error('API error');
  return res.json();
}

export default function KPIsPage() {
  const [summary, setSummary] = useState<any>(null);
  const [mbtEnabled, setMbtEnabled] = useState(true);

  // URLからmbt状態を取得
  useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const mbtParam = params.get('mbt');
    if (mbtParam === 'on') setMbtEnabled(true);
    else if (mbtParam === 'off') setMbtEnabled(false);

    // データ取得
    fetchSummary().then(data => setSummary(data));
  }, []);

  // URL変更を監視
  useEffect(() => {
    const handlePopState = () => {
      const params = new URLSearchParams(window.location.search);
      const mbtParam = params.get('mbt');
      if (mbtParam === 'on') setMbtEnabled(true);
      else if (mbtParam === 'off') setMbtEnabled(false);
    };
    window.addEventListener('popstate', handlePopState);
    return () => window.removeEventListener('popstate', handlePopState);
  }, []);

  if (!summary) return <div style={{ color: '#f1f5f9' }}>Loading...</div>;

  // MBT ON/OFFで数値を切り替え
  const displayData = mbtEnabled ? {
    ghg_reduction: summary.ghg_reduction_tco2e_per_year,
    pbpe_issued: summary.pbpe_issued_per_year,
    // ... 他のKPIも同様
  } : {
    ghg_reduction: 0, // MBT OFF時はゼロまたはベースライン値
    pbpe_issued: 0,
    // ... 他のKPIも同様
  };

  const cardStyle = {
    backgroundColor: '#1e293b',
    border: '1px solid #334155',
    borderRadius: '12px',
    padding: '16px',
    boxShadow: '0 1px 3px rgba(0,0,0,0.5)',
  };

  return (
    <div style={{ padding: "0" }}>
      <h1 style={{ fontSize: "24px", fontWeight: "bold", marginBottom: "8px", color: "#f1f5f9" }}>
        Global KPIs {mbtEnabled ? '(MBT ON)' : '(MBT OFF)'}
      </h1>
      <p style={{ color: "#94a3b8", marginBottom: "24px" }}>
        Planetary balance sheet — GHG, Soil, Water, Health, Stability, PBPE
      </p>

      {/* 6 KPI Cards */}
      <div style={{
        display: 'grid',
        gridTemplateColumns: 'repeat(auto-fit, minmax(280px, 1fr))',
        gap: '16px',
        marginBottom: '24px',
      }}>
        <div style={cardStyle}>
          <div style={{ color: "#94a3b8" }}>GHG Reduction</div>
          <div style={{ fontSize: "28px", fontWeight: "bold", color: "#f1f5f9" }}>
            {displayData.ghg_reduction.toLocaleString()} tCO₂e
          </div>
          <div style={{ color: "#64748b" }}>Units: {summary.units.toLocaleString()}</div>
        </div>
        {/* 残りの5つのKPIカードも同様に displayData を使用 */}
      </div>
    </div>
  );
}
```
