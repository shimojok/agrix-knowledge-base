### Planetary Metabolism OS 完全統合版 — 全体像

### 2. 統合ダッシュボード UI（`app/planet/metabolism-os/page.tsx`）

tsx

```
'use client';

import { useEffect, useState } from 'react';
import SoilSummaryCard from '@/app/agrrix/components/SoilSummaryCard';
import SoilTimeSeriesChart from '@/app/agrrix/components/SoilTimeSeriesChart';
import SafetyChainFlowCard from '@/app/agrrix/components/SafetyChainFlowCard';

export default function PlanetaryMetabolismOSPage() {
  const [soil, setSoil] = useState<any | null>(null);
  const [phenotyping, setPhenotyping] = useState<any | null>(null);
  const [certificate, setCertificate] = useState<any | null>(null);

  useEffect(() => {
    (async () => {
      const soilRes = await fetch('/api/soil').then(r => r.json());

      const phenRes = await fetch('/api/phenotyping', {
        method: 'POST',
        body: JSON.stringify({
          soil: {
            ph: soilRes.latest.ph,
            eh_mV: soilRes.latest.eh_mV,
            moisture_pct: soilRes.latest.moisture_pct,
            co2_flux_mg_m2_h: soilRes.latest.co2_flux_mg_m2_h,
          },
          plant: soilRes.latest.plantSignals,
          pathogen: soilRes.latest.pathogenSignals,
          postharvest: soilRes.latest.postHarvestSignals,
        }),
      }).then(r => r.json());

      const certRes = await fetch('/api/safetychain/certificate', {
        method: 'POST',
        body: JSON.stringify({
          lotId: soilRes.lotId,
          soil: {
            sheScore: soilRes.latest.she_score,
            ecoPhenotype: soilRes.latest.eco_phenotype,
          },
          phenotyping: {
            omegaCycle: phenRes.micro.omegaCycle,
            phiFunctional: phenRes.cascade.phiFunctional,
            sigmaResilience: phenRes.stability.sigmaResilience,
            vScore: phenRes.inertia.vScore,
            vIndex: phenRes.vIndex,
          },
          quality: {
            polyphenolDensity: soilRes.latest.polyphenol_density,
            metabolicLifeDays: phenRes.inertia.tauInertiaDays,
          },
          pricing: soilRes.latest.pricing,
        }),
      }).then(r => r.json());

      setSoil(soilRes);
      setPhenotyping(phenRes);
      setCertificate(certRes);
    })();
  }, []);

  if (!soil || !phenotyping || !certificate)
    return <div>Loading Planetary Metabolism OS...</div>;

  const latest = soil.snapshots[soil.snapshots.length - 1];

  return (
    <main className="min-h-screen p-8 bg-gray-100 space-y-8">
      <h1 className="text-3xl font-bold">Planetary Metabolism OS</h1>
      <p className="text-gray-700">
        土壌生態 × 代謝フェノタイピング × SafetyChain™ × Certificate を統合した代謝OSビュー。
      </p>

      <div className="grid grid-cols-2 gap-6">
        <div className="space-y-4">
          <h2 className="text-xl font-semibold">AGRIX Soil</h2>
          <SoilSummaryCard
            latestShe={latest.she_score}
            latestPhenotype={latest.eco_phenotype}
          />
          <SoilTimeSeriesChart snapshots={soil.snapshots} />
        </div>

        <div className="space-y-4">
          <h2 className="text-xl font-semibold">SafetyChain™ Flow</h2>
          <SafetyChainFlowCard
            data={{
              soil,
              phenotyping,
              pricing: soil.latest.pricing,
              vIndex: phenotyping.vIndex,
            }}
          />
        </div>
      </div>

      <section className="grid grid-cols-2 gap-6">
        <div className="bg-white p-4 rounded shadow">
          <h2 className="font-semibold mb-2">フェノタイピング AI Agents</h2>
          <p>ω-cycle: {phenotyping.micro.omegaCycle.toFixed(2)}</p>
          <p>Φ-functional: {phenotyping.cascade.phiFunctional.toFixed(3)}</p>
          <p>σ-resilience: {phenotyping.stability.sigmaResilience.toFixed(3)}</p>
          <p>V-Score: {phenotyping.inertia.vScore.toFixed(3)}</p>
          <p>V-Index: {phenotyping.vIndex.toFixed(3)}</p>
        </div>

        <div className="bg-white p-4 rounded shadow flex flex-col items-center">
          <h2 className="font-semibold mb-2">SafetyChain™ Certificate</h2>
          <img
            src={certificate.qrCode}
            alt="SafetyChain Certificate QR"
            className="w-40 h-40"
          />
        </div>
      </section>
    </main>
  );
}
```

- **MBT55 由来の代謝圧（ω-cycle）**
