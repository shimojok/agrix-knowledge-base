```python
import streamlit as st
import pandas as pd

from core_model import run_soil_carbon_model
from mrv import compare_model_vs_observed, generate_mrv_report
from planetary_os_adapter import export_soil_module_state

# -----------------------
# Language dictionary
# -----------------------
LANG = {
    "ja": {
        "title": "🌏 土壌・炭素・収益 ダッシュボード",
        "overview": "概要",
        "soil": "土壌・微生物",
        "carbon": "炭素隔離",
        "finance": "グリーンプレミアム",
        "mrv": "MRV（検証）",
        "export": "エクスポート",
        "years": "シミュレーション年数",
        "area": "プロジェクト面積（ha）",
        "initial_soc": "初期SOC（tC/ha）",
        "carbon_price": "炭素価格（USD/tCO₂e）",
        "yield_baseline": "基準収量（t/ha）",
        "yield_price": "作物価格（USD/t）",
        "mbt_dose": "MBT55 投入量（プロジェクト）",
        "upload": "観測データをアップロード（CSV/Excel）",
        "mrv_metrics": "MRV 指標",
        "export_title": "Planetary OS Soil Module Export",
        "export_desc": "Planetary OS Soil Module（国際標準）形式の JSON を生成します。",
        "open_browser": "ブラウザで開く",
    },
    "en": {
        "title": "🌍 Soil–Carbon–Finance Dashboard",
        "overview": "Overview",
        "soil": "Soil & Microbes",
        "carbon": "Carbon",
        "finance": "Green Premium",
        "mrv": "MRV",
        "export": "Export",
        "years": "Simulation Years",
        "area": "Project Area (ha)",
        "initial_soc": "Initial SOC (tC/ha)",
        "carbon_price": "Carbon Credit Price (USD/tCO₂e)",
        "yield_baseline": "Baseline Yield (t/ha)",
        "yield_price": "Crop Price (USD/t)",
        "mbt_dose": "MBT55 Application Level (Project)",
        "upload": "Upload observed data (CSV/Excel)",
        "mrv_metrics": "MRV Metrics",
        "export_title": "Planetary OS Soil Module Export",
        "export_desc": "Generate JSON in the official Planetary OS Soil Module format.",
        "open_browser": "Open in Browser",
    }
}

# -----------------------
# Language toggle
# -----------------------
if "lang" not in st.session_state:
    st.session_state["lang"] = "ja"

col1, col2 = st.columns(2)
if col1.button("日本語"):
    st.session_state["lang"] = "ja"
if col2.button("English"):
    st.session_state["lang"] = "en"

L = LANG[st.session_state["lang"]]

# -----------------------
# UI
# -----------------------
st.set_page_config(page_title=L["title"], layout="wide")
st.title(L["title"])

# Sidebar
years = st.sidebar.slider(L["years"], 5, 30, 15)
area_ha = st.sidebar.number_input(L["area"], 1.0, 1_000_000.0, 765.0)
initial_soc = st.sidebar.number_input(L["initial_soc"], 10.0, 200.0, 50.0)
carbon_price = st.sidebar.number_input(L["carbon_price"], 10.0, 500.0, 210.0)
yield_baseline = st.sidebar.number_input(L["yield_baseline"], 1.0, 20.0, 7.7)
yield_price = st.sidebar.number_input(L["yield_price"], 50.0, 2000.0, 500.0)
mbt_dose = st.sidebar.slider(L["mbt_dose"], 0.0, 3.0, 1.5)

# Common params
params = dict(
    years=years,
    area_ha=area_ha,
    initial_soc=initial_soc,
    carbon_price=carbon_price,
    yield_baseline=yield_baseline,
    yield_price=yield_price,
)

baseline_params = {**params, "mbt_dose": 0.0}
project_params = {**params, "mbt_dose": mbt_dose}

# Run model
baseline = run_soil_carbon_model(baseline_params)
project = run_soil_carbon_model(project_params)

df_b = pd.DataFrame(baseline)
df_p = pd.DataFrame(project)

df_diff = df_p.copy()
df_diff["sequestration_diff"] = df_p["sequestration_tco2e_ha_yr"] - df_b["sequestration_tco2e_ha_yr"]
df_diff["yield_diff"] = df_p["yield_t_ha"] - df_b["yield_t_ha"]

# Tabs
tab_overview, tab_soil, tab_carbon, tab_finance, tab_mrv, tab_export = st.tabs(
    [L["overview"], L["soil"], L["carbon"], L["finance"], L["mrv"], L["export"]]
)

# Overview
with tab_overview:
    st.subheader(L["overview"])
    chart_df = pd.DataFrame({
        "Year": df_b["year"],
        "Baseline": df_b["sequestration_tco2e_ha_yr"],
        "MBT55": df_p["sequestration_tco2e_ha_yr"],
    }).set_index("Year")
    st.line_chart(chart_df)

# Soil
with tab_soil:
    st.subheader(L["soil"])
    soil_df = pd.DataFrame({
        "Year": df_b["year"],
        "SOC Baseline": df_b["soc_tC_ha"],
        "SOC MBT55": df_p["soc_tC_ha"],
    }).set_index("Year")
    st.line_chart(soil_df)

# Carbon
with tab_carbon:
    st.subheader(L["carbon"])
    st.line_chart(chart_df)

# Finance
with tab_finance:
    st.subheader(L["finance"])
    st.write(df_diff[["year", "sequestration_diff", "yield_diff"]])

# MRV
with tab_mrv:
    st.subheader(L["mrv"])
    uploaded = st.file_uploader(L["upload"], type=["csv", "xlsx"])
    if uploaded:
        if uploaded.name.endswith(".csv"):
            obs = pd.read_csv(uploaded)
        else:
            obs = pd.read_excel(uploaded)

        model_for_mrv = df_p.rename(columns={
            "soc_tC_ha": "soc_tC_ha_model",
            "sequestration_tco2e_ha_yr": "sequestration_tco2e_ha_yr_model",
            "yield_t_ha": "yield_t_ha_model",
        })

        obs_for_mrv = obs.rename(columns={
            "soc_tC_ha": "soc_tC_ha_obs",
            "sequestration_tco2e_ha_yr": "sequestration_tco2e_ha_yr_obs",
            "yield_t_ha": "yield_t_ha_obs",
        })

        stats = compare_model_vs_observed(model_for_mrv, obs_for_mrv)
        st.write(L["mrv_metrics"])
        st.json(stats)

        report = generate_mrv_report(model_for_mrv, obs_for_mrv, stats)
        st.json(report)

# Export
with tab_export:
    st.subheader(L["export_title"])
    st.caption(L["export_desc"])

    export_json = export_soil_module_state(df_b, df_p, project_params)
    st.json(export_json)
```
