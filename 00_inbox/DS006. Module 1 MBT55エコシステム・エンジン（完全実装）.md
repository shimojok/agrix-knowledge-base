```typescript
// lib/agrix/module1/engines/hypercycle.engine.ts

import { 
  HypercycleInput, 
  HypercycleOutput,
  DecomposerActivitySummary 
} from '../types/hypercycle.types';
import { SoilSensorInput } from '../types/mbt55.types';

/**
 * Agent-1: ハイパーサイクル・アナライザー
 * MBT55の土壌生態制御（炭素・窒素サイクル）を数値化する
 */
export function runHypercycleEngine(input: HypercycleInput): HypercycleOutput {
  const {
    pH,
    eh_mV,
    moisture_pct,
    temperature_c,
    organicCarbon_gL,
    availableNitrogen_gL,
    mbt55Dosage_Lha,
    mbt55ApplicationDays,
    biomassInput_tPerHa,
    decomposerActivity,
  } = input;

  // ----- 1. 環境適合度スコア -----
  // MBT55最適条件: pH 7.2-7.5, Eh 100-200mV, 水分40-60%, 温度25-35℃
  const pHScore = 1 - Math.abs(pH - 7.35) / 2.0;
  const ehScore = 1 - Math.abs(eh_mV - 150) / 200;
  const moistureScore = 1 - Math.abs(moisture_pct - 50) / 30;
  const tempScore = 1 - Math.abs(temperature_c - 28) / 20;
  
  const environmentalScore = Math.max(0, Math.min(1, 
    (pHScore * 0.25 + ehScore * 0.20 + moistureScore * 0.25 + tempScore * 0.30)
  ));

  // ----- 2. 分解菌活性統合 -----
  const decomposerTotal = decomposerActivity?.total ?? 0.5;
  const decomposerScore = Math.max(0, Math.min(1, decomposerTotal));

  // ----- 3. MBT55投与効果 -----
  // 適正投与量: 2.5-3.5 L/ha、経過日数に応じて活性化
  const dosageScore = Math.min(1, mbt55Dosage_Lha / 3.0);
  const daysScore = Math.min(1, mbt55ApplicationDays / 14); // 14日で最大活性
  
  // バイオマス投入効果（MSC連携）
  const biomassScore = Math.min(1, biomassInput_tPerHa / 5.0);

  // ----- 4. ハイパーサイクル効率（総合） -----
  const cycleEfficiency = (
    environmentalScore * 0.30 +
    decomposerScore * 0.25 +
    dosageScore * 0.20 +
    daysScore * 0.15 +
    biomassScore * 0.10
  );

  // ----- 5. 腐植化率（炭素隔離の中核指標） -----
  // セルロース分解菌の活性が特に重要
  const cellulaseActivity = decomposerActivity?.cellulase ?? 0.5;
  const humificationRate = Math.min(1, 
    0.3 * cycleEfficiency +
    0.3 * cellulaseActivity +
    0.2 * (biomassInput_tPerHa / 10) +
    0.2 * (mbt55ApplicationDays / 30)
  );

  // ----- 6. 資源変換率（有機物→代謝産物） -----
  const resourceConversionRate = Math.min(1,
    0.4 * cycleEfficiency +
    0.3 * (decomposerActivity?.protease ?? 0.5) +
    0.3 * (decomposerActivity?.amylase ?? 0.5)
  );

  // ----- 7. C/N比シフト計算 -----
  const initialCN = organicCarbon_gL / Math.max(availableNitrogen_gL, 0.001);
  const targetCN = 16.5; // MBT55の目標値
  const cnShift = Math.max(0, initialCN - targetCN);
  const currentCN = Math.max(targetCN, initialCN - cnShift * Math.min(1, cycleEfficiency * 1.2));

  // ----- 8. サイクルフェーズ判定 -----
  const phaseScore = cycleEfficiency * 0.6 + humificationRate * 0.4;
  let cyclePhase: HypercycleOutput['cyclePhase'] = 'INITIATING';
  if (phaseScore < 0.3) cyclePhase = 'INITIATING';
  else if (phaseScore < 0.55) cyclePhase = 'ACCELERATING';
  else if (phaseScore < 0.8) cyclePhase = 'MATURE';
  else if (phaseScore < 0.95) cyclePhase = 'DECLINING';
  else cyclePhase = 'STALLED';

  // ----- 9. 安定炭素比率（長期隔離可能率） -----
  const stableCarbonFraction = Math.min(1,
    0.4 * humificationRate +
    0.3 * (decomposerActivity?.cellulase ?? 0.5) +
    0.3 * (biomassInput_tPerHa / 8)
  );

  // ----- 10. 診断メッセージ -----
  const statusMessage = generateStatusMessage(cyclePhase, cycleEfficiency, humificationRate);
  const recommendations = generateRecommendations(cyclePhase, cycleEfficiency, pH, moisture_pct, mbt55Dosage_Lha);

  return {
    cyclePhase,
    cycleEfficiency: Number(Math.max(0, Math.min(1, cycleEfficiency)).toFixed(3)),
    humificationRate: Number(Math.max(0, Math.min(1, humificationRate)).toFixed(3)),
    humicAcidRatio: Number((0.5 + 0.4 * humificationRate).toFixed(3)), // 腐植化率から推定
    stableCarbonFraction: Number(stableCarbonFraction.toFixed(3)),
    resourceConversionRate: Number(Math.max(0, Math.min(1, resourceConversionRate)).toFixed(3)),
    cnShift: Number(cnShift.toFixed(1)),
    currentCNRatio: Number(currentCN.toFixed(1)),
    statusMessage,
    recommendations,
    calculatedAt: new Date().toISOString(),
  };
}

// ヘルパー関数
function generateStatusMessage(phase: string, efficiency: number, humification: number): string {
  if (phase === 'INITIATING') {
    return 'MBT55ハイパーサイクルが起動中です。微生物活性が徐々に上昇しています。';
  }
  if (phase === 'ACCELERATING') {
    return `ハイパーサイクルが加速しています（効率${(efficiency * 100).toFixed(0)}%）。C/N比の低下が進行中です。`;
  }
  if (phase === 'MATURE') {
    return `ハイパーサイクルが成熟期に達しました（効率${(efficiency * 100).toFixed(0)}%）。腐植質生成が安定的に進行しています。`;
  }
  if (phase === 'DECLINING') {
    return 'ハイパーサイクルが下降期です。追加のMBT55投与またはバイオマス投入を検討してください。';
  }
  return 'ハイパーサイクルが停滞しています。環境条件（pH・水分・温度）を再評価してください。';
}

function generateRecommendations(phase: string, efficiency: number, pH: number, moisture: number, dosage: number): string[] {
  const recs: string[] = [];
  
  if (pH < 6.5 || pH > 8.0) {
    recs.push(`pH (${pH.toFixed(1)})が最適範囲（6.5-8.0）から外れています。pH調整を推奨します。`);
  }
  if (moisture < 35 || moisture > 70) {
    recs.push(`水分 (${moisture.toFixed(0)}%)が最適範囲（35-70%）から外れています。灌水または排水を調整してください。`);
  }
  if (dosage < 2.0) {
    recs.push(`MBT55投与量 (${dosage.toFixed(1)} L/ha) が推奨範囲（2.5-3.5 L/ha）を下回っています。増量を検討してください。`);
  }
  if (phase === 'INITIATING' || phase === 'ACCELERATING') {
    recs.push('ハイパーサイクル加速のため、有機物（作物残渣・堆肥）の追加投入を推奨します。');
  }
  if (phase === 'DECLINING' || phase === 'STALLED') {
    recs.push('ハイパーサイクル再活性化のため、MBT55の追加投与とバイオマス投入を同時に行うことを推奨します。');
  }
  if (recs.length === 0) {
    recs.push('現在のハイパーサイクルは良好な状態です。現状の管理を継続してください。');
  }
  return recs;
}
```

```typescript
import { BiosecurityInput, BiosecurityOutput } from '../types/biosecurity.types';

/**
 * Agent-2: バイオセキュリティ・モニター
 * 病原菌制御・微生物叢バランスを数値化する
 */
export function runBiosecurityEngine(input: BiosecurityInput): BiosecurityOutput {
  const {
    pH,
    eh_mV,
    moisture_pct,
    temperature_c,
    pathogenCounts,
    beneficialMicrobes,
    mbt55Dosage_Lha,
    mbt55ApplicationDays,
    historicalPathogenTrend,
  } = input;

  // ----- 1. 病原菌総負荷 -----
  const totalPathogens = 
    (pathogenCounts.fusarium || 0) +
    (pathogenCounts.pythium || 0) +
    (pathogenCounts.phytophthora || 0) +
    (pathogenCounts.rhizoctonia || 0);
  
  // 基準値（一般的な農業土壌の平均値から設定）
  const baselinePathogenLoad = 500; // cfu/g（仮定値）
  const pathogenLoadRatio = Math.min(1, totalPathogens / baselinePathogenLoad);

  // ----- 2. 有用菌/病原菌比 -----
  const beneficialToPathogenRatio = beneficialMicrobes / Math.max(totalPathogens, 1);
  const ratioScore = Math.min(1, beneficialToPathogenRatio / 5); // 5:1で理想

  // ----- 3. MBT55による病原菌抑制効果 -----
  // 投与量と経過日数から抑制率を算出
  const dosageEffect = Math.min(1, mbt55Dosage_Lha / 3.0);
  const timeEffect = Math.min(1, mbt55ApplicationDays / 21); // 3週間で最大効果
  const suppressionBase = 0.3 + 0.5 * dosageEffect * timeEffect;
  const pathogenSuppressionIndex = Math.min(0.95, suppressionBase);

  // 実際の病原菌低減率
  const pathogenLoadReduction = pathogenSuppressionIndex * 100;

  // ----- 4. 環境要因によるリスク補正 -----
  // pHが酸性（<6.0）またはアルカリ性（>8.0）でリスク上昇
  const pHRisk = pH < 6.0 ? 0.2 : pH > 8.0 ? 0.15 : 0;
  // 水分過多（>70%）でリスク上昇（嫌気性病原菌の増殖）
  const moistureRisk = moisture_pct > 70 ? 0.15 : 0;
  // 高温（>30℃）でリスク上昇
  const tempRisk = temperature_c > 30 ? 0.1 : 0;
  
  const riskFactor = pHRisk + moistureRisk + tempRisk;
  const diseaseOutbreakRisk = Math.min(1, 
    (1 - pathogenSuppressionIndex) * 0.6 + 
    (1 - ratioScore) * 0.3 + 
    riskFactor
  );

  // ----- 5. 微生物叢の回復力（レジリエンス） -----
  // 多様性が高く、有用菌比率が高いほど回復力が高い
  const diversityEstimate = 0.5 + 0.3 * (1 - pathogenLoadRatio) + 0.2 * ratioScore;
  const biomeResilienceScore = Math.min(1, 
    0.4 * diversityEstimate +
    0.3 * (1 - diseaseOutbreakRisk) +
    0.3 * (beneficialMicrobes / Math.max(beneficialMicrobes + totalPathogens, 1))
  );

  // ----- 6. 多様性安定性 -----
  const diversityStability = Math.min(1,
    0.5 * biomeResilienceScore +
    0.5 * (1 - Math.abs(historicalPathogenTrend?.at(-1) ?? 0 - (historicalPathogenTrend?.at(-2) ?? 0)) / 100)
  );

  // ----- 7. バイオセキュリティレベル判定 -----
  let biosecurityLevel: BiosecurityOutput['biosecurityLevel'] = 'LOW';
  const combinedScore = pathogenSuppressionIndex * 0.4 + ratioScore * 0.3 + biomeResilienceScore * 0.3;
  if (combinedScore >= 0.7) biosecurityLevel = 'HIGH';
  else if (combinedScore >= 0.4) biosecurityLevel = 'MEDIUM';
  else biosecurityLevel = 'LOW';

  // ----- 8. リスク要因とアラート -----
  const riskFactors: string[] = [];
  const alerts: string[] = [];
  const recommendedActions: string[] = [];

  if (diseaseOutbreakRisk > 0.6) {
    riskFactors.push('病害発生リスクが高い');
    alerts.push('⚠️ 高リスク: 即時対応が必要です');
    recommendedActions.push('MBT55の追加投与（緊急）');
    recommendedActions.push('バイオコントロール対策の強化');
  }
  if (beneficialToPathogenRatio < 2) {
    riskFactors.push('有用菌/病原菌比が低い');
    recommendedActions.push('MBT55投与量の増加を検討');
    recommendedActions.push('有機物投入による微生物活性化');
  }
  if (biosecurityLevel === 'LOW') {
    alerts.push('🔴 バイオセキュリティが脆弱です');
    recommendedActions.push('総合的な土壌生態制御の見直し');
  }
  if (biosecurityLevel === 'HIGH' && alerts.length === 0) {
    recommendedActions.push('現状の管理を継続してください');
  }

  return {
    pathogenSuppressionIndex: Number(Math.max(0, Math.min(1, pathogenSuppressionIndex)).toFixed(3)),
    pathogenLoadReduction: Number(Math.max(0, Math.min(100, pathogenLoadReduction)).toFixed(1)),
    biomeResilienceScore: Number(Math.max(0, Math.min(1, biomeResilienceScore)).toFixed(3)),
    beneficialToPathogenRatio: Number(beneficialToPathogenRatio.toFixed(1)),
    diversityStability: Number(diversityStability.toFixed(3)),
    diseaseOutbreakRisk: Number(Math.max(0, Math.min(1, diseaseOutbreakRisk)).toFixed(3)),
    biosecurityLevel,
    riskFactors,
    alerts,
    recommendedActions,
    calculatedAt: new Date().toISOString(),
  };
}
```

```typescript
  const biocontrolEfficacy = Math.min(1,
    0.30 * cropVigor +
    0.20 * scfaScore +
    0.20 * enzymeScore +
    0.20 * mbt55Effect +
    0.10 * leafHealth
  );

  // ----- 6. 病害抵抗性スコア -----
  const pestResistanceScore = Math.min(1,
    0.4 * cropVigor +
    0.3 * (1 - (1 - biocontrolEfficacy)) +
    0.3 * scfaScore
  );

  // ----- 7. 病害抑制率 -----
  const diseaseSuppressionRate = Math.min(1,
    0.5 * biocontrolEfficacy +
    0.3 * mbt55Effect +
    0.2 * enzymeScore
  );

  // ----- 8. 有用菌相（Pseudomonas / Bacillus） -----
  // SCFAと酵素活性から推定
  const pseudomonasAbundance = Math.min(1, 0.3 + 0.4 * scfaScore + 0.3 * enzymeScore);
  const bacillusAbundance = Math.min(1, 0.2 + 0.5 * enzymeBetaGlucosidase + 0.3 * scfaScore);
  const beneficialMicrobeRatio = Math.min(1, 0.4 * pseudomonasAbundance + 0.4 * bacillusAbundance + 0.2);

  // ----- 9. 獲得抵抗性誘導レベル -----
  const inducedResistanceLevel = Math.min(1,
    0.3 * mbt55Effect +
    0.3 * (scfaScore) +
    0.4 * (1 - (1 - cropVigor))
  );

  // ----- 10. 推奨アクション -----
  let recommendedAction = '';
  let actionPriority: 'HIGH' | 'MEDIUM' | 'LOW' = 'MEDIUM';
  let details = '';

  if (biocontrolEfficacy < 0.4) {
    recommendedAction = '⚠️ バイオコントロール効果が不十分です。MBT55投与量の増加と有機物投入を推奨します。';
    actionPriority = 'HIGH';
    details = '病害リスクが高まっています。早急な対策が必要です。';
  } else if (biocontrolEfficacy < 0.7) {
    recommendedAction = 'バイオコントロール効果は標準的です。SCFA前駆体の追加投入でさらなる向上が期待できます。';
    actionPriority = 'MEDIUM';
    details = '現在の管理を維持しつつ、微量栄養素の補給を検討してください。';
  } else {
    recommendedAction = '✅ バイオコントロール効果は優良です。現状の管理を継続してください。';
    actionPriority = 'LOW';
    details = '病害抑制が安定的に機能しています。モニタリングを継続してください。';
  }

  return {
    pestResistanceScore: Number(Math.max(0, Math.min(1, pestResistanceScore)).toFixed(3)),
    diseaseSuppressionRate: Number(Math.max(0, Math.min(1, diseaseSuppressionRate)).toFixed(3)),
    beneficialMicrobeRatio: Number(beneficialMicrobeRatio.toFixed(3)),
    pseudomonasAbundance: Number(pseudomonasAbundance.toFixed(3)),
    bacillusAbundance: Number(bacillusAbundance.toFixed(3)),
    biocontrolEfficacy: Number(Math.max(0, Math.min(1, biocontrolEfficacy)).toFixed(3)),
    inducedResistanceLevel: Number(Math.max(0, Math.min(1, inducedResistanceLevel)).toFixed(3)),
    recommendedAction,
    actionPriority,
    details,
    calculatedAt: new Date().toISOString(),
  };
}
```

```typescript
import { SoilSensorInput, DecomposerActivity, MBT55Microbiome } from '../types/mbt55.types';

/**
 * Agent-4: 分解菌活性トラッカー
 * 4大分解菌（プロテアーゼ・アミラーゼ・リパーゼ・セルラーゼ）の活性を数値化する
 */
export function runDecomposerEngine(input: SoilSensorInput): MBT55Microbiome {
  const {
    pH,
    eh_mV,
    moisture_pct,
    temperature_c,
    organicCarbon_gL,
    availableNitrogen_gL,
    mbt55Dosage_Lha,
    mbt55ApplicationDays,
    biomassInput_tPerHa,
  } = input;

  // ----- 各分解菌の最適環境からの乖離を計算 -----
  
  // 1. タンパク質分解菌（プロテアーゼ）
  const proteaseActivity = calculateDecomposerActivity({
    pH,
    pHOptimal: { min: 6.5, max: 7.5 },
    temperature: temperature_c,
    tempOptimal: { min: 25, max: 35 },
    moisture: moisture_pct,
    moistureOptimal: { min: 40, max: 65 },
    substrateAvailability: Math.min(1, availableNitrogen_gL / 2.0), // 窒素が基質
    mbt55Effect: mbt55Dosage_Lha / 3.0,
    timeEffect: mbt55ApplicationDays / 14,
    role: 'タンパク質→アミノ酸→SCFA 変換',
  });

  // 2. デンプン分解菌（アミラーゼ）
  const amylaseActivity = calculateDecomposerActivity({
    pH,
    pHOptimal: { min: 6.0, max: 7.0 },
    temperature: temperature_c,
    tempOptimal: { min: 28, max: 38 },
    moisture: moisture_pct,
    moistureOptimal: { min: 35, max: 60 },
    substrateAvailability: Math.min(1, organicCarbon_gL / 15), // 炭素が基質
    mbt55Effect: mbt55Dosage_Lha / 3.0,
    timeEffect: mbt55ApplicationDays / 10,
    role: 'デンプン→糖類→有機酸 変換',
  });

  // 3. 油脂分解菌（リパーゼ）
  const lipaseActivity = calculateDecomposerActivity({
    pH,
    pHOptimal: { min: 6.5, max: 8.0 },
    temperature: temperature_c,
    tempOptimal: { min: 25, max: 35 },
    moisture: moisture_pct,
    moistureOptimal: { min: 40, max: 70 },
    substrateAvailability: Math.min(1, biomassInput_tPerHa / 3.0), // バイオマス中の脂質
    mbt55Effect: mbt55Dosage_Lha / 3.0,
    timeEffect: mbt55ApplicationDays / 12,
    role: '油脂→脂肪酸→エネルギー 変換',
  });

  // 4. セルロース分解菌（セルラーゼ）※炭素隔離の鍵
  const cellulaseActivity = calculateDecomposerActivity({
    pH,
    pHOptimal: { min: 6.0, max: 7.5 },
    temperature: temperature_c,
    tempOptimal: { min: 25, max: 40 },
    moisture: moisture_pct,
    moistureOptimal: { min: 45, max: 70 },
    substrateAvailability: Math.min(1, biomassInput_tPerHa / 4.0), // セルロース基質
    mbt55Effect: mbt55Dosage_Lha / 3.0,
    timeEffect: mbt55ApplicationDays / 21, // セルロース分解は遅い
    role: 'セルロース→糖類→腐植質 変換（炭素隔離の鍵）',
  });

  // ----- 総合活性 -----
  const totalActivity = (
    proteaseActivity.activity * 0.25 +
    amylaseActivity.activity * 0.25 +
    lipaseActivity.activity * 0.20 +
    cellulaseActivity.activity * 0.30
  );

  // ----- 多様性指数（シャノン指数の簡易版） -----
  const activities = [
    proteaseActivity.activity,
    amylaseActivity.activity,
    lipaseActivity.activity,
    cellulaseActivity.activity,
  ];
  const diversityIndex = calculateDiversityIndex(activities);

  // ----- 安定性スコア -----
  const stabilityScore = Math.min(1,
    0.4 * (1 - Math.abs(proteaseActivity.activity - amylaseActivity.activity)) +
    0.4 * (1 - Math.abs(lipaseActivity.activity - cellulaseActivity.activity)) +
    0.2 * Math.min(totalActivity * 1.5, 1)
  );

  return {
    aerobicRatio: 0.55,
    anaerobicRatio: 0.45,
    decomposers: {
      protease: proteaseActivity,
      amylase: amylaseActivity,
      lipase: lipaseActivity,
      cellulase: cellulaseActivity,
    },
    totalActivity: Number(Math.max(0, Math.min(1, totalActivity)).toFixed(3)),
    diversityIndex: Number(diversityIndex.toFixed(3)),
    stabilityScore: Number(stabilityScore.toFixed(3)),
  };
}

// 分解菌活性計算ヘルパー
function calculateDecomposerActivity(params: {
  pH: number;
  pHOptimal: { min: number; max: number };
  temperature: number;
  tempOptimal: { min: number; max: number };
  moisture: number;
  moistureOptimal: { min: number; max: number };
  substrateAvailability: number;
  mbt55Effect: number;
  timeEffect: number;
  role: string;
}): DecomposerActivity {
  const { pH, pHOptimal, temperature, tempOptimal, moisture, moistureOptimal, substrateAvailability, mbt55Effect, timeEffect, role } = params;

  // pH適合度
  const pHFit = 1 - Math.abs(pH - (pHOptimal.min + pHOptimal.max) / 2) / ((pHOptimal.max - pHOptimal.min) / 2 + 0.5);
  
  // 温度適合度
  const tempCenter = (tempOptimal.min + tempOptimal.max) / 2;
  const tempRange = (tempOptimal.max - tempOptimal.min) / 2 + 5;
  const tempFit = 1 - Math.abs(temperature - tempCenter) / tempRange;
  
  // 水分適合度
  const moistureCenter = (moistureOptimal.min + moistureOptimal.max) / 2;
  const moistureRange = (moistureOptimal.max - moistureOptimal.min) / 2 + 10;
  const moistureFit = 1 - Math.abs(moisture - moistureCenter) / moistureRange;

  // 環境適合度（平均）
  const environmentFit = Math.max(0, Math.min(1, (pHFit + tempFit + moistureFit) / 3));

  // 活性 = 環境適合度 × 基質供給 × MBT55効果 × 時間効果
  const activity = Math.max(0, Math.min(1,
    environmentFit * 0.40 +
    substrateAvailability * 0.25 +
    mbt55Effect * 0.20 +
    timeEffect * 0.15
  ));

  return {
    role,
    activity: Number(activity.toFixed(3)),
    optimalCondition: {
      pH: { min: pHOptimal.min, max: pHOptimal.max },
      temperature: { min: tempOptimal.min, max: tempOptimal.max },
      moisture: { min: moistureOptimal.min, max: moistureOptimal.max },
    },
    currentConditionFit: Number(environmentFit.toFixed(3)),
    substrateAvailability: Number(Math.max(0, Math.min(1, substrateAvailability)).toFixed(3)),
  };
}

// 多様性指数計算
function calculateDiversityIndex(activities: number[]): number {
  const total = activities.reduce((a, b) => a + b, 0) + 0.001;
  const proportions = activities.map(a => a / total);
  let shannon = 0;
  for (const p of proportions) {
    if (p > 0) shannon -= p * Math.log(p);
  }
  // 最大値 log(4) ≈ 1.386 で正規化
  return Math.min(1, shannon / 1.386);
}
```

```typescript
  };
```
