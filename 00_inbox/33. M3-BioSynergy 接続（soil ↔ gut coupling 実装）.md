```text
coupling/gut_soil_coupling.py
```

内容：

```python
from dataclasses import dataclass


@dataclass
class SoilState:
    carbon_flow: float
    nitrogen_flow: float
    microbial_diversity: float


@dataclass
class GutState:
    scfa_production: float
    peptide_generation: float
    metabolic_resilience: float


def couple_soil_gut(soil: SoilState):

    scfa = soil.carbon_flow * 0.8

    peptide = soil.microbial_diversity * 0.6

    resilience = (
        soil.carbon_flow +
        soil.nitrogen_flow +
        soil.microbial_diversity
    ) / 3

    return GutState(
        scfa_production=round(scfa, 2),
        peptide_generation=round(peptide, 2),
        metabolic_resilience=round(resilience, 2)
    )
```
