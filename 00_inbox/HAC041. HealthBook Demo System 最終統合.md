```python
import streamlit as st
import plotly.graph_objects as go

from core.demo_orchestrator import (
    DemoOrchestrator
)

engine = DemoOrchestrator()

scores = engine.generate_scores()

diseases = engine.infer_disease()

substrates = engine.recommend_substrates()

strains = engine.recommend_strains()

st.title("HealthBook Metabolic OS")

st.header("1. PATHWAY SCORES")

categories = list(scores.keys())

values = list(scores.values())

fig = go.Figure()

fig.add_trace(go.Scatterpolar(
    r=values + [values[0]],
    theta=categories + [categories[0]],
    fill='toself'
))

fig.update_layout(
    polar=dict(
        radialaxis=dict(
            visible=True,
            range=[0, 1]
        )
    ),
    showlegend=False
)

st.plotly_chart(
    fig,
    use_container_width=True
)

st.header("2. Predicted Disease Risks")

for disease in diseases:

    st.markdown(f"""
    - {disease}
    """)

st.header("3. Recommended Substrates")

for substrate in substrates:

    st.markdown(f"""
    - {substrate}
    """)

st.header("4. Recommended MBT55 Strains")

for strain in strains:

    st.markdown(f"""
    - {strain}
    """)
```
