## ① Physical Layer（物理層）

ts

```
physical: {
  soilType: string,        // 砂質・ローム・粘土

  moisture: number,        // %

```

## ② Chemical Layer（化学層）

ts

```
chemical: {
  pH: number,
  ec: number,              // dS/m
  cnRatio: number,

    n: number,
    p: number,
    k: number,
    ca: number,
    mg: number,
    fe: number,

    pb: number,
    as: number,
    pesticideResidue: boolean,
  }
}
```

## ③ Carbon & Humus Layer（炭素・腐植層）

ts

```
carbon: {
  soc: number,             // 有機炭素量
  som: number,             // 有機物量
  humus: {
    humicAcid: number,
    fulvicAcid: number,
    humin: number,
    humificationRate: number,
  }

```

## ④ Biological Layer（生物層）

ts

```
biological: {
  microbialBiomassC: number,
  microbialActivity: number,   // 呼吸量
  omDegradationRate: number,   // 有機物分解率
  diversityIndex: number,      // Shannon
  pathogens: number,           // 病原菌数
}
```

ts

```
mbt55: {

    methaneRisk: number,

  }

```

## 📌 `/app/api/soil/analyze/route.ts`

ts

```
import { NextResponse } from "next/server";

export async function POST(req: Request) {
  const soil = await req.json();

  const hypercycleScore =
    (soil.mbt55.activityIndex +
     soil.biological.omDegradationRate +
     soil.carbon.humus.humificationRate) / 3;

  return NextResponse.json({
    hypercycleScore,
    mbt55Status: soil.mbt55.activityIndex > 0.7 ? "optimal" : "suboptimal",
    ghgRisk: soil.mbt55.ghgRisk
  });
}
```

## 📌 `/app/api/soil/hypercycle/route.ts`

ts

```
import { NextResponse } from "next/server";

export async function POST(req: Request) {
  const soil = await req.json();

  const score = {
    carbon: soil.carbon.soc * soil.carbon.humus.humificationRate,
    microbial: soil.biological.microbialActivity * soil.biological.diversityIndex,
    nutrients: soil.chemical.nutrients.n + soil.chemical.nutrients.k,
    mbt55: soil.mbt55.activityIndex * soil.mbt55.degradation24h
  };

  return NextResponse.json({
    hypercycleIndex: score,
    status: score.mbt55 > 50 ? "active" : "inactive"
  });
}
```
