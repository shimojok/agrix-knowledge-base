## 🛠️ ステップ1専用スクリプト：`sync_agrix_os.py`

`AGRIX-OS` リポジトリのルート、または `scripts/sync_agrix_os.py` として配置してください。

Python

```
import os
import re
import glob
import json
import sys
from typing import List, Dict, Any
from pinecone import Pinecone
from sentence_transformers import SentenceTransformer

# ==========================================
# 1. 環境変数とPineconeの初期化
# ==========================================
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
PINECONE_INDEX_NAME = os.getenv("PINECONE_INDEX_NAME", "bionexus-hypercycle")
TARGET_NAMESPACE = "ns-production-core"  # 最優先の「精製済みコア」名前空間

if not PINECONE_API_KEY:
    print("❌ エラー: 環境変数 'PINECONE_API_KEY' が設定されていません。")
    sys.exit(1)

print("📥 埋め込みモデル（無料・ローカル実行）をロード中...")
embedding_model = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')

pc = Pinecone(api_key=PINECONE_API_KEY)
index = pc.Index(PINECONE_INDEX_NAME)

# ==========================================
# 2. 拡張子別の堅牢なパースロジック（エラーハンドリング付き）
# ==========================================
def parse_file_safe(file_path: str) -> List[Dict[str, Any]]:
    """ファイルのパースを行い、エラーが発生した場合はスキップしてログを出す"""
    ext = os.path.splitext(file_path)[1].lower()
    base_name = os.path.basename(file_path)
    
    base_metadata = {
        "source_repo": "AGRIX-OS",
        "file_path": file_path.replace("\\", "/"),
        "file_name": base_name,
        "platform_layer": "production_infrastructure"
    }
    
    chunks = []

    try:
        # ---- A. Markdown のパース（フロントマター剥がし ＆ 見出しチャンキング） ----
        if ext == ".md":
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()
            
            # フロントマター (YAML) の簡易パースを試みる
            body = content
            frontmatter = {}
            if content.startswith("---"):
                match = re.match(r'^---\s*\n(.*?)\n---\s*\n', content, re.DOTALL)
                if match:
                    body = content[match.end():]
                    # 簡単なYAMLキー・バリューの抽出（エラー防止のため簡易実装）
                    for line in match.group(1).split("\n"):
                        if ":" in line:
                            k, v = line.split(":", 1)
                            frontmatter[k.strip()] = v.strip().strip('"').strip("'")

            # 見出し（# 〜 ####）で分割
            sections = re.split(r'(^#{1,4}\s+.*$)', body, flags=re.MULTILINE)
            if sections[0].strip():
                sections.insert(0, "# 概要\n")

            for i in range(0, len(sections)-1, 2):
                header = sections[i].strip().replace("#", "").strip()
                text = sections[i+1].strip() if i+1 < len(sections) else ""
                if text:
                    meta = base_metadata.copy()
                    meta.update({
                        "data_type": "unstructured_text",
                        "section_header": header,
                        **frontmatter # 抽出できたフロントマターをマージ
                    })
                    chunks.append({
                        "id": f"{base_name}#md-{i}",
                        "text": f"[{header}]\n{text}",
                        "metadata": meta
                    })

        # ---- B. Python コードのパース ----
        elif ext == ".py":
            with open(file_path, 'r', encoding='utf-8') as f:
                code = f.read()
            
            meta = base_metadata.copy()
            meta.update({"data_type": "python_code"})
            chunks.append({
                "id": f"{base_name}#code",
                "text": f"### Python Code in AGRIX-OS: {base_name} ###\n{code}",
                "metadata": meta
            })

    except Exception as e:
        # 5,100ノート処理時のための布石：エラーが起きてもファイル単位でスキップして次に進む
        print(f"⚠️ 警告: ファイルの解析に失敗しました (スキップします): {file_path}. エラー: {e}")
        return []

    return chunks

# ==========================================
# 3. メイン実行エンジン
# ==========================================
def main():
    print(f"🚀 AGRIX-OS ➔ Pinecone [{TARGET_NAMESPACE}] 同期を開始します...")
    
    # 対象ファイルのスキャン (.md と .py)
    targets = ["**/*.md", "**/*.py"]
    file_paths = []
    for t in targets:
        file_paths.extend(glob.glob(t, recursive=True))

    success_count = 0
    error_count = 0

    for path in file_paths:
        # 不要なディレクトリの除外
        if any(x in path for x in ["node_modules", "venv", ".git", ".obsidian", ".trash"]):
            continue
            
        chunks = parse_file_safe(path)
        if not chunks:
            continue
            
        try:
            texts = [c["text"] for c in chunks]
            embeddings = embedding_model.encode(texts).tolist()
            
            vectors_to_upsert = []
            for idx, chunk in enumerate(chunks):
                pinecone_metadata = chunk["metadata"]
                pinecone_metadata["text"] = chunk["text"]  # RAG復元用
                
                vectors_to_upsert.append({
                    "id": f"agrix-os#{chunk['id']}",
                    "values": embeddings[idx],
                    "metadata": pinecone_metadata
                })
                
            if vectors_to_upsert:
                # 💡 ここで明示的に namespace を指定してアップサートします
                index.upsert(vectors=vectors_to_upsert, namespace=TARGET_NAMESPACE)
                success_count += 1
                print(f"✅ 同期成功: {path} ({len(vectors_to_upsert)} chunks)")
                
        except Exception as e:
            print(f"❌ ベクトル化・アップサートに失敗: {path}. エラー: {e}")
            error_count += 1

    print(f"\n🏁 同期ジョブが完了しました。")
    print(f" └ 成功ファイル数: {success_count}")
    print(f" └ 失敗ファイル数: {error_count}")

if __name__ == "__main__":
    main()
```

## 🏃‍♂️ 次のアクション（Kazさんの作業ステップ）

1. **スクリプトの配置：**
    
    ローカルの `AGRIX-OS` リポジトリ内に上記の `sync_agrix_os.py` を作成して保存してください。
    
2. **ローカルでのテスト実行（推奨）：**
    
    お使いのPCのターミナル（またはPowerShell）で、PineconeのAPIキーを環境変数にセットし、スクリプトを実行してみます。
    
    Bash
    
    ```
    # Windows (PowerShell) の場合
    $env:PINECONE_API_KEY="あなたのキー"
    python sync_agrix_os.py
    
    # Mac/Linux の場合
    export PINECONE_API_KEY="あなたのキー"
    python sync_agrix_os.py
    ```
