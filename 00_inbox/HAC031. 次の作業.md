
内容：

```python
import streamlit as st
import plotly.graph_objects as go

scores = {
    "PATH_00": 0.72,
    "PATH_01": 0.55,
    "PATH_02": 0.48,
    "PATH_03": 0.61,
    "PATH_04": 0.44,
    "PATH_05": 0.38,
    "PATH_06": 0.66,
}

categories = list(scores.keys())
values = list(scores.values())

fig = go.Figure()

fig.add_trace(go.Scatterpolar(
    r=values + [values[0]],
    theta=categories + [categories[0]],
    fill='toself',
    name='Metabolic Pathways'
))

fig.update_layout(
    polar=dict(
        radialaxis=dict(
            visible=True,
            range=[0, 1]
        )),
    showlegend=False
)

st.title("PATH Radar")
st.plotly_chart(fig, use_container_width=True)
```
