```typescript
// lib/agrix/module3/engines/soc.engine.ts

import { SOCInput, SOCOutput } from '../types/soil-organic-carbon.types';

/**
 * SOC計測・予測エンジン
 * 土壌有機炭素の現在値・将来予測・増加トレンドを計算する
 */
export function runSOCEngine(input: SOCInput): SOCOutput {
  const { soilData, hypercycle, humus, measurementConfig, historicalSOC } = input;
  const { baselineSOC, measurementDepth_cm, measurementMethod } = measurementConfig;

  // ----- 1. 現在のSOC計算 -----
  // Module1/2のデータから推定
  const humusEffect = humus.production.outputHumus_tPerHa * 0.08; // 腐植質→SOC変換率
  const cycleEffect = hypercycle.humificationRate * 0.3;
  const baselineEffect = baselineSOC * (1 + 0.01 * soilData.organicCarbon_gL / 10);
  
  const currentSOC_percent = Math.min(5, 
    baselineEffect * 0.5 +
    humusEffect * 0.3 +
    cycleEffect * 0.2
  );

  // SOC（t/ha）への換算（深度×仮比重を考慮）
  const bulkDensity = 1.2 + 0.1 * (1 - humus.soilImpact.aggregateStability); // g/cm3
  const depth_m = measurementDepth_cm / 100;
  const soc_tPerHa = currentSOC_percent / 100 * bulkDensity * depth_m * 10000; // t/ha

  // ----- 2. 将来予測（S字カーブモデル）-----
  const maxPotentialSOC = Math.min(6, baselineSOC * 2.0 + humusEffect * 1.5);
  const growthRate = 0.15 + 0.1 * hypercycle.cycleEfficiency + 0.05 * humus.soilImpact.aggregateStability;
  
  const projectSOC = (years: number): { soc_percent: number; soc_tPerHa: number } => {
    // シグモイド成長曲線
    const progress = years / 20; // 20年で飽和を仮定
    const sigmoid = 1 / (1 + Math.exp(-8 * (progress - 0.5)));
    const soc = currentSOC_percent + (maxPotentialSOC - currentSOC_percent) * sigmoid * Math.min(1, years / 10);
    const tPerHa = soc / 100 * bulkDensity * depth_m * 10000;
    return {
      soc_percent: Number(soc.toFixed(2)),
      soc_tPerHa: Number(tPerHa.toFixed(1)),
    };
  };

  // ----- 3. 増加トレンド計算 -----
  const annualIncrease_percent = (projectSOC(1).soc_percent - currentSOC_percent);
  const annualIncrease_tPerHa = (projectSOC(1).soc_tPerHa - soc_tPerHa);
  
  // 将来の増加率から軌道を判定
  const futureGrowth = projectSOC(5).soc_percent - currentSOC_percent;
  const growthAcceleration = futureGrowth / 5 - annualIncrease_percent;
  const trajectory: SOCOutput['trend']['trajectory'] = 
    growthAcceleration > 0.02 ? 'ACCELERATING' :
    growthAcceleration < -0.02 ? 'DECLINING' : 'STABLE';

  // ----- 4. 4per1000目標達成度 -----
  // 4per1000 = 年間0.4%増加が目標
  const perHeqTarget = currentSOC_percent * 0.004 * 20; // 20年後の目標増加量
  const perHeqAchievement = (projectSOC(20).soc_percent - currentSOC_percent) / perHeqTarget * 100;

  // ----- 5. 診断メッセージ -----
  const confidenceLevel: SOCOutput['confidenceLevel'] = 
    measurementMethod === 'LAB' ? 'HIGH' :
    measurementMethod === 'SPECTROSCOPY' ? 'MEDIUM' : 'LOW';

  const statusMessage = generateSOCStatus(currentSOC_percent, perHeqAchievement, trajectory);
  const recommendations = generateSOCRecommendations(perHeqAchievement, trajectory, currentSOC_percent);

  return {
    current: {
      soc_percent: Number(currentSOC_percent.toFixed(2)),
      soc_tPerHa: Number(soc_tPerHa.toFixed(1)),
      socTonnesPerHa: Number(soc_tPerHa.toFixed(1)),
      bulkDensity: Number(bulkDensity.toFixed(2)),
      measurementDepth_cm,
    },
    baseline: {
      soc_percent: Number(baselineSOC.toFixed(2)),
      soc_tPerHa: Number((baselineSOC / 100 * bulkDensity * depth_m * 10000).toFixed(1)),
      year: 2026,
    },
    projection: {
      year1: projectSOC(1),
      year5: projectSOC(5),
      year10: projectSOC(10),
      year20: projectSOC(20),
    },
    trend: {
      annualIncrease_percent: Number(annualIncrease_percent.toFixed(3)),
      annualIncrease_tPerHa: Number(annualIncrease_tPerHa.toFixed(2)),
      cumulativeIncrease_5yr: Number((projectSOC(5).soc_tPerHa - soc_tPerHa).toFixed(1)),
      cumulativeIncrease_10yr: Number((projectSOC(10).soc_tPerHa - soc_tPerHa).toFixed(1)),
      cumulativeIncrease_20yr: Number((projectSOC(20).soc_tPerHa - soc_tPerHa).toFixed(1)),
      trajectory,
    },
    targetComparison: {
      perHeqTarget: Number(perHeqTarget.toFixed(2)),
      perHeqAchievement: Number(Math.min(100, perHeqAchievement).toFixed(1)),
      gapToTarget: Number(Math.max(0, perHeqTarget - (projectSOC(20).soc_percent - currentSOC_percent)).toFixed(2)),
    },
    statusMessage,
    recommendations,
    confidenceLevel,
    calculatedAt: new Date().toISOString(),
  };
}

function generateSOCStatus(soc: number, achievement: number, trajectory: string): string {
  if (soc >= 4.0 && achievement >= 80) {
    return `✅ SOCは${soc.toFixed(1)}%と高水準です。4per1000目標の${achievement.toFixed(0)}%を達成しています。`;
  }
  if (soc >= 3.0 && achievement >= 50) {
    return `📈 SOCは${soc.toFixed(1)}%で順調に増加中です（${trajectory}）。`;
  }
  if (soc >= 2.0) {
    return `📊 SOCは${soc.toFixed(1)}%です。さらなる増加の余地があります。`;
  }
  return `⚠️ SOCが${soc.toFixed(1)}%と低水準です。早急な土壌改良が必要です。`;
}

function generateSOCRecommendations(achievement: number, trajectory: string, soc: number): string[] {
  const recs: string[] = [];
  if (achievement < 50) {
    recs.push('腐植質投入量を増やし、SOC増加を加速させてください。');
    recs.push('カバークロップの導入や不耕起栽培を検討してください。');
  }
  if (trajectory === 'DECLINING') {
    recs.push('SOCが減少傾向です。有機物投入とMBT55施用を強化してください。');
  }
  if (soc < 2.5) {
    recs.push('土壌有機物が不足しています。堆肥・腐植質の大量投入を推奨します。');
  }
  if (recs.length === 0) {
    recs.push('現状のSOC管理は良好です。継続してください。');
  }
  return recs;
}
```

```typescript
// lib/agrix/module3/engines/sequestration.engine.ts

import { SequestrationInput, SequestrationOutput } from '../types/sequestration.types';

/**
 * 炭素隔離エンジン
 * 腐植質由来の安定炭素隔離量とクレジットポテンシャルを計算する
 */
export function runSequestrationEngine(input: SequestrationInput): SequestrationOutput {
  const { soc, humus, hypercycle, regionalConfig, projectConfig } = input;
  const { climateZone, soilType, landUseHistory } = regionalConfig;
  const { durationYears, verificationStandard } = projectConfig;

  // ----- 1. 安定炭素比率計算 -----
  // 腐植質の質と土壌タイプから安定炭素率を推定
  const baseStableFraction = 0.3 + 0.4 * humus.quality.stabilityIndex;
  const soilFactor = soilType === 'CLAY' ? 1.2 : soilType === 'LOAM' ? 1.0 : soilType === 'SANDY' ? 0.7 : 0.9;
  const climateFactor = climateZone === 'TEMPERATE' ? 1.0 : climateZone === 'TROPICAL' ? 0.8 : 0.6;
  const landUseFactor = landUseHistory === 'FOREST' ? 1.1 : landUseHistory === 'GRASSLAND' ? 1.0 : 0.8;
  
  const stableFraction = Math.min(0.95, baseStableFraction * soilFactor * climateFactor * landUseFactor);

  // ----- 2. 永続性スコア -----
  const permanenceScore = Math.min(1,
    0.3 * stableFraction +
    0.3 * humus.quality.stabilityIndex +
    0.2 * hypercycle.stableCarbonFraction +
    0.2 * (1 - Math.exp(-durationYears / 50))
  );

  // ----- 3. 炭素隔離量計算 -----
  // 年間SOC増加量から炭素隔離量を換算
  const annualSOCIncrease = soc.trend.annualIncrease_tPerHa;
  const totalSOCIncrease = annualSOCIncrease * durationYears;
  
  // SOC → CO2換算（3.67倍）
  const totalSequestration_tCO2 = totalSOCIncrease * 3.67 * stableFraction;
  const annualSequestration_tCO2 = annualSOCIncrease * 3.67 * stableFraction;
  const perHaSequestration_tCO2 = totalSequestration_tCO2 / durationYears;

  // ----- 4. MRVデータ -----
  const baselineEmissions = landUseHistory === 'CROPLAND' ? 2.5 : landUseHistory === 'GRASSLAND' ? 1.5 : 0.5;
  const actualSequestration = totalSequestration_tCO2;
  const additionality = Math.min(1, (actualSequestration - baselineEmissions * durationYears) / (actualSequestration + 0.01));

  // 認証スキーム別の価格設定
  const priceMap = {
    'VERRA': 12,
    'GOLD_STANDARD': 15,
    'J_CREDIT': 8,
    'CUSTOM': 10,
  };
  const pricePerCredit = priceMap[verificationStandard as keyof typeof priceMap] || 10;

  // ----- 5. クレジット発行ポテンシャル -----
  const issuableCredits = totalSequestration_tCO2 * 0.9; // 10%バッファー
  const marketValue = issuableCredits * pricePerCredit;

  // ----- 6. 診断メッセージ -----
  const statusMessage = generateSequestrationStatus(stableFraction, permanenceScore, totalSequestration_tCO2);
  const recommendations = generateSequestrationRecommendations(stableFraction, permanenceScore);

  return {
    stableCarbon: {
      totalSequestration_tCO2: Number(totalSequestration_tCO2.toFixed(1)),
      annualSequestration_tCO2: Number(annualSequestration_tCO2.toFixed(2)),
      perHaSequestration_tCO2: Number(perHaSequestration_tCO2.toFixed(2)),
      stableFraction: Number(stableFraction.toFixed(3)),
      permanenceScore: Number(permanenceScore.toFixed(3)),
    },
    mrv: {
      monitoringPeriod: `${durationYears} years`,
      baselineEmissions: Number(baselineEmissions.toFixed(1)),
      actualSequestration: Number(actualSequestration.toFixed(1)),
      additionality: Number(additionality.toFixed(3)),
      verificationStatus: 'PENDING',
      nextVerificationDate: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000).toISOString(),
    },
    creditPotential: {
      issuableCredits: Number(issuableCredits.toFixed(0)),
      marketValue: Number(marketValue.toFixed(0)),
      pricePerCredit,
      scheme: verificationStandard,
      vintage: `2026-${2026 + durationYears}`,
    },
    statusMessage,
    recommendations,
    calculatedAt: new Date().toISOString(),
  };
}
```

```typescript
/**
 * メタン削減エンジン
 * 発酵飼料によるメタン排出削減効果を計算する
 */
export function runMethaneEngine(input: MethaneInput): MethaneOutput {
  const { feed, livestockConfig, baselineConfig } = input;
  const { livestockType, headCount, averageWeight_kg, manureManagement, feedingPeriod_days } = livestockConfig;

  // ----- 1. ベースライン排出量 -----
  // 家畜種別の基準排出量（kg CH4/頭/年）
  const baselineEmissionMap = {
    'RUMINANT': 120,
    'PIG': 8,
    'POULTRY': 0.8,
    'AQUACULTURE': 0.3,
  };
  const baselineEmission = baselineEmissionMap[livestockType as keyof typeof baselineEmissionMap] || 50;

  // ----- 2. 発酵飼料による削減効果 -----
  const qualityEffect = feed.production.fermentationQuality;
  const scfaEffect = feed.nutritionalValue.scfaContent / 200; // 200mmol/kgを基準
  
  // 削減率（家畜種別で補正）
  let reductionRateBase = 0.15 + 0.4 * qualityEffect + 0.1 * scfaEffect;
  if (livestockType === 'RUMINANT') reductionRateBase *= 1.1; // 反芻動物は効果大
  else if (livestockType === 'PIG') reductionRateBase *= 0.9;
  else reductionRateBase *= 0.7;
  
  const reductionRate = Math.min(0.6, reductionRateBase);

  // ----- 3. 腸内発酵メタン削減 -----
  const entericReduction = baselineEmission * reductionRate * (feedingPeriod_days / 365);
  const entericCO2e = entericReduction * 28; // CH4 → CO2換算（GWP100 = 28）

  // ----- 4. 糞尿管理メタン削減 -----
  const manureEmissionFactor = manureManagement === 'LIQUID' ? 0.3 : 
                               manureManagement === 'SOLID' ? 0.2 :
                               manureManagement === 'DIGESTER' ? 0.1 : 0.25;
  const manureBaseline = baselineEmission * 0.4; // 糞尿由来は全体の40%と仮定
  const manureReduction = manureBaseline * reductionRate * 0.7 * (feedingPeriod_days / 365);
  const manureCO2e = manureReduction * 28;

  // ----- 5. 総合削減量 -----
  const totalReduction_tCO2e = (entericCO2e + manureCO2e) * headCount / 1000; // tCO2e

  // ----- 6. 環境インパクト -----
  const avoidedEmissions = totalReduction_tCO2e;
  const carbonFootprintReduction = reductionRate * 100;

  // 持続可能な開発目標（SDGs）貢献度
  const sdg13 = Math.min(100, 50 + 40 * reductionRate);
  const sdg12 = Math.min(100, 40 + 50 * qualityEffect);

  // ----- 7. 診断メッセージ -----
  const statusMessage = generateMethaneStatus(reductionRate, totalReduction_tCO2e, livestockType);
  const recommendations = generateMethaneRecommendations(reductionRate, qualityEffect, livestockType);

  return {
    reduction: {
      entericMethaneReduction: {
        kgCO2ePerHeadYear: Number(entericCO2e.toFixed(1)),
        totalReduction: Number((entericCO2e * headCount / 1000).toFixed(2)),
        reductionRate: Number((reductionRate * 100).toFixed(1)),
      },
      manureMethaneReduction: {
        kgCO2ePerHeadYear: Number(manureCO2e.toFixed(1)),
        totalReduction: Number((manureCO2e * headCount / 1000).toFixed(2)),
        reductionRate: Number((reductionRate * 0.7 * 100).toFixed(1)),
      },
      total: {
        totalReduction_tCO2e: Number(totalReduction_tCO2e.toFixed(2)),
        reductionRateOverall: Number((reductionRate * 100).toFixed(1)),
      },
    },
    co2Equivalent: {
      enteric_tCO2e: Number((entericCO2e * headCount / 1000).toFixed(2)),
      manure_tCO2e: Number((manureCO2e * headCount / 1000).toFixed(2)),
      total_tCO2e: Number(totalReduction_tCO2e.toFixed(2)),
    },
    environmentalImpact: {
      avoidedEmissions: Number(avoidedEmissions.toFixed(2)),
      carbonFootprintReduction: Number(carbonFootprintReduction.toFixed(1)),
      sustainableDevelopmentGoals: {
        sdg13: Number(sdg13.toFixed(0)),
        sdg12: Number(sdg12.toFixed(0)),
      },
    },
    statusMessage,
    recommendations,
    calculatedAt: new Date().toISOString(),

}
```

```typescript
import { GHGAccountingInput, GHGAccountingOutput } from '../types/ghg-accounting.types';

/**
 * 総合GHGアカウンティングエンジン
 * Scope 1/2/3 を統合し、ネットゼロ評価とグローバル貢献度を計算する
 */
export function runGHGAccountingEngine(input: GHGAccountingInput): GHGAccountingOutput {
  const { soc, sequestration, methane, additionalEmissions, scopeConfig, farmConfig } = input;
  const { areaHa, cropType, yield_tPerHa } = farmConfig;

  // ----- 1. Scope別排出量計算 -----
  // Scope1: 直接排出（農場内）
  const entericEmissions = methane.co2Equivalent.enteric_tCO2e;
  const manureEmissions = methane.co2Equivalent.manure_tCO2e;
  const soilN2O = additionalEmissions.fertilizerN2O * areaHa * 0.3; // 簡易推定
  
  const scope1Total = (entericEmissions + manureEmissions + soilN2O) * areaHa;

  // Scope2: 間接排出（電力・肥料）
  const electricityEmissions = additionalEmissions.energyCO2 * areaHa * 0.5;
  const fertilizerEmissions = additionalEmissions.fertilizerN2O * areaHa * 0.7;
  const scope2Total = electricityEmissions + fertilizerEmissions;

  // Scope3: バリューチェーン
  const transportEmissions = additionalEmissions.transportCO2 * areaHa;
  const processingEmissions = additionalEmissions.processingCO2 * areaHa;
  const packagingEmissions = 0.5 * areaHa;
  const wasteEmissions = 0.3 * areaHa;
  const scope3Total = transportEmissions + processingEmissions + packagingEmissions + wasteEmissions;

  // 総排出量（Gross）
  const grossEmissions = scope1Total + scope2Total + scope3Total;

  // 削減効果
  const sequestrationOffset = sequestration.stableCarbon.annualSequestration_tCO2 * areaHa;
  const methaneReduction = methane.co2Equivalent.total_tCO2e * areaHa;
  const totalReduction = sequestrationOffset + methaneReduction;

  // 純排出量（Net）
  const netEmissions = Math.max(0, grossEmissions - totalReduction);

  // ----- 2. 削減率 -----
  const baselineEmissions = grossEmissions * 1.5; // 簡易ベースライン
  const reductionRate = ((baselineEmissions - netEmissions) / baselineEmissions) * 100;

  // ----- 3. ネットゼロ評価 -----
  let netZeroStatus: GHGAccountingOutput['netZero']['status'] = 'EMITTING';
  const residualEmissions = netEmissions;
  const offsetRequired = netEmissions;

  if (netEmissions <= 0) {
    netZeroStatus = 'CARBON_NEGATIVE';
  } else if (netEmissions < grossEmissions * 0.1) {
    netZeroStatus = 'NET_ZERO';
  }

  // ネットゼロ到達予測年（単純線形予測）
  const annualReduction = totalReduction / 1; // 年間削減量
  const yearsToZero = netEmissions / Math.max(annualReduction, 0.01);
  const netZeroTargetYear = Math.min(2050, 2026 + Math.ceil(yearsToZero));

  // ----- 4. グローバル貢献度 -----
  const perHaContribution = netEmissions > 0 ? -netEmissions / areaHa : sequestrationOffset / areaHa;
  const totalContribution = perHaContribution * areaHa;
  const percentageOf51B = (totalContribution / 5.1e9) * 100; // 5.1億トン基準

  // パリ協定整合性評価
  let parisAgreementAlignment = '';
  if (percentageOf51B > 1.0) {
    parisAgreementAlignment = '✅ パリ協定1.5℃目標に大きく貢献（世界貢献度の1%以上）';
  } else if (percentageOf51B > 0.1) {
    parisAgreementAlignment = '📈 パリ協定目標に貢献中（世界貢献度の0.1-1%）';
  } else {
    parisAgreementAlignment = '📊 さらなるスケール拡大が必要（世界貢献度0.1%未満）';
  }

  // ----- 5. 診断メッセージ -----
  const statusMessage = generateGHGStatus(netZeroStatus, reductionRate, totalContribution);
  const recommendations = generateGHGRecommendations(netZeroStatus, reductionRate, scope1Total, scope2Total, scope3Total);

  return {
    scopeEmissions: {
      scope1: {
        direct: Number(scope1Total.toFixed(2)),
        sources: {
          enteric: Number(entericEmissions.toFixed(2)),
          manure: Number(manureEmissions.toFixed(2)),
          soilN2O: Number(soilN2O.toFixed(2)),
        },
      },
      scope2: {
        indirect: Number(scope2Total.toFixed(2)),
        sources: {
          electricity: Number(electricityEmissions.toFixed(2)),
          fertilizer: Number(fertilizerEmissions.toFixed(2)),
        },
      },
      scope3: {
        valueChain: Number(scope3Total.toFixed(2)),
        sources: {
          transport: Number(transportEmissions.toFixed(2)),
          processing: Number(processingEmissions.toFixed(2)),
          packaging: Number(packagingEmissions.toFixed(2)),
          waste: Number(wasteEmissions.toFixed(2)),
        },
      },
      total: {
        grossEmissions: Number(grossEmissions.toFixed(2)),
        netEmissions: Number(netEmissions.toFixed(2)),
      },
    },
    reduction: {
      sequestrationOffset: Number(sequestrationOffset.toFixed(2)),
      methaneReduction: Number(methaneReduction.toFixed(2)),
      totalReduction: Number(totalReduction.toFixed(2)),
      reductionRate: Number(reductionRate.toFixed(1)),
    },
    netZero: {
      status: netZeroStatus,
      residualEmissions: Number(residualEmissions.toFixed(2)),
      offsetRequired: Number(offsetRequired.toFixed(2)),
      netZeroTargetYear,
    },
    globalContribution: {
      perHaContribution: Number(perHaContribution.toFixed(2)),
      totalContribution: Number(totalContribution.toFixed(2)),
      percentageOf51B: Number(percentageOf51B.toFixed(4)),
      parisAgreementAlignment,
    },
    statusMessage,
    recommendations,
    calculatedAt: new Date().toISOString(),
  };
}

  }
```

## ⑤ Next.js APIエンドポイント

```typescript
// app/api/agrix/module3/orchestrate/route.ts

import { NextRequest, NextResponse } from 'next/server';
import { runModule3Full } from '@/lib/agrix/module3';
import { runModule1Full } from '@/lib/agrix/module1';
import { runModule2Full } from '@/lib/agrix/module2';

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    
    // 必須フィールドチェック
    const requiredFields = [
      'soilData', 'socConfig', 'regionalConfig', 'projectConfig',
      'livestockConfig', 'additionalEmissions', 'farmConfig', 'scopeConfig'
    ];
    for (const field of requiredFields) {
      if (!body[field]) {
        return NextResponse.json(
          { error: `Missing required field: ${field}` },
          { status: 400 }
        );
      }
    }

    // ----- Module1実行 -----
    const module1Result = runModule1Full(body.soilData);
    
    // ----- Module2実行（Module1の出力を使用）-----
    const module2Input = {
      hypercycle: module1Result.hypercycle,
      soilData: body.soilData,
      decomposerActivity: module1Result.decomposer,
      productionConfig: body.productionConfig || {
        humus: { inputBiomass_tPerHa: 5.0, retentionDays: 75, turningFrequency: 1.5, moistureTarget: 55 },
        fertilizer: { type: 'BOTH', inputMaterial: 'MIXED', fermentationDays: 18, aerationRate: 0.4, temperatureTarget: 30, phTarget: 6.8 },
        feed: { livestockType: body.livestockConfig.livestockType, feedBase: 'SILAGE', fermentationDays: 28, inoculationRate: 2.8, moistureTarget: 55, compactionDensity: 550 },
      },
      feedStock: body.feedStock || {
        carbonSource: 38, nitrogenSource: 1.5, mineralContent: 8, moistureContent: 60,
        crudeProtein: 14, crudeFiber: 22, starch: 18, fat: 3, ash: 8,
      },
    };
    const module2Result = runModule2Full(module2Input);
    
    // ----- Module3実行 -----
    const module3Input = {
      soilData: body.soilData,
      hypercycle: module1Result.hypercycle,
      humus: module2Result.humus,
      fertilizer: module2Result.fertilizer,
      feed: module2Result.feed,
      socConfig: body.socConfig,
      regionalConfig: body.regionalConfig,
      projectConfig: body.projectConfig,
      livestockConfig: body.livestockConfig,
      additionalEmissions: body.additionalEmissions,
      farmConfig: body.farmConfig,
      scopeConfig: body.scopeConfig,
    };
    
    const result = runModule3Full(module3Input);
    
    return NextResponse.json({
      module1: {
        agrixHealthScore: module1Result.agrixHealthScore,
        healthStatus: module1Result.healthStatus,
      },
      module2: {
        mscIntegratedScore: module2Result.mscIntegratedScore,
        mscStatus: module2Result.mscStatus,
        economicValue: module2Result.economicValue,
      },
      module3: result,
    });
  } catch (error) {
    return NextResponse.json(
      { error: 'Internal server error', details: String(error) },
      { status: 500 }
    );
  }
}
```
