### 1. V-Score API（鮮度保持力＝代謝寿命）

**パス:** `app/api/safetychain/vscore/route.ts`

ts

```
import { NextRequest, NextResponse } from 'next/server';

type VScoreInput = {
  atpInitial: number;        // 収穫時 ATP 相対値 (0–1)
  atpCurrent: number;        // 現在 ATP 相対値 (0–1)
  elapsedHours: number;      // 収穫後経過時間 (h)
  scfaIndex: number;         // SCFA 保護指数 (0–1)
  polyphenolDensity: number; // ポリフェノール密度 (0–1)
  sheScore: number;          // SHE™ スコア (0–100)
  storageTempC: number;      // 保存温度 (℃)
};

type VScoreOutput = {
  vScore: number;            // 0–100
  grade: 'Platinum' | 'Gold' | 'Silver' | 'Standard';
  metabolicLifeDays: number; // 残り代謝寿命（日）
};

function computeVScore(input: VScoreInput): VScoreOutput {
  const {
    atpInitial,
    atpCurrent,
    elapsedHours,
    scfaIndex,
    polyphenolDensity,
    sheScore,
    storageTempC,
  } = input;

  // ATP 減衰率
  const atpRatio = atpCurrent / Math.max(atpInitial, 1e-6);
  const baseK = -Math.log(Math.max(atpRatio, 1e-6)) / Math.max(elapsedHours, 1e-3);

  // 保護因子による k_atp 補正
  const protection =
    0.4 * scfaIndex +
    0.3 * polyphenolDensity +
    0.3 * (sheScore / 100);

  const tempFactor = Math.pow(2, (storageTempC - 4) / 10); // Q10≈2
  const kAtp = baseK * tempFactor * (1 - 0.3 * protection); // MBT55 で約30%低減を想定

  // 閾値 V=35 を下回るまでの時間（概算）
  const targetRatio = 0.35;
  const hoursToThreshold = Math.log(targetRatio) / -Math.max(kAtp, 1e-6);
  const metabolicLifeDays = Math.max(0, (hoursToThreshold - elapsedHours) / 24);

  // V-Score を ATP 比＋保護因子から統合
  const vScoreRaw =
    0.5 * atpRatio +
    0.3 * protection +
    0.2 * (1 - Math.min(elapsedHours / (24 * 14), 1)); // 14日で減衰

  const vScore = Math.max(0, Math.min(100, vScoreRaw * 100));

  let grade: VScoreOutput['grade'] = 'Standard';
  if (vScore >= 85) grade = 'Platinum';
  else if (vScore >= 70) grade = 'Gold';
  else if (vScore >= 55) grade = 'Silver';

  return {
    vScore,
    grade,
    metabolicLifeDays: Number(metabolicLifeDays.toFixed(1)),
  };
}

export async function POST(req: NextRequest) {
  try {
    const body = (await req.json()) as VScoreInput;
    const result = computeVScore(body);
    return NextResponse.json(result);
  } catch (e) {
    return NextResponse.json(
      { error: 'Invalid payload', details: String(e) },
      { status: 400 },
    );
  }
}
```

### 2. 動的価格決定 API（4レイヤー）

**パス:** `app/api/safetychain/price/route.ts`

ts

```
import { NextRequest, NextResponse } from 'next/server';

type PriceInput = {
  baseMarketPrice: number; // 従来市場価格 (JPY/kg 等)
  vFunctionality: number;  // 機能性プレミアム (V-Score 由来)
  lLossAvoidance: number;  // ロス削減価値
  mMedicalSavings: number; // 医療費削減価値
  cCarbon: number;         // 炭素隔離価値
};

type PriceOutput = {
  totalPrice: number;
  breakdown: {
    baseMarketPrice: number;
    vFunctionality: number;
    lLossAvoidance: number;
    mMedicalSavings: number;
    cCarbon: number;
  };
};

export async function POST(req: NextRequest) {
  try {
    const body = (await req.json()) as PriceInput;

    const totalPrice =
      body.baseMarketPrice +
      body.vFunctionality +
      body.lLossAvoidance +
      body.mMedicalSavings +
      body.cCarbon;

    const result: PriceOutput = {
      totalPrice: Number(totalPrice.toFixed(2)),
      breakdown: {
        baseMarketPrice: body.baseMarketPrice,
        vFunctionality: body.vFunctionality,
        lLossAvoidance: body.lLossAvoidance,
        mMedicalSavings: body.mMedicalSavings,
        cCarbon: body.cCarbon,
      },
    };

    return NextResponse.json(result);
  } catch (e) {
    return NextResponse.json(
      { error: 'Invalid payload', details: String(e) },
      { status: 400 },
    );
  }
}
```

### 3. フェノタイピング入力 API（ポリフェノール生成ポテンシャル）

**パス:** `app/api/safetychain/phenomics/route.ts`

```
import { NextRequest, NextResponse } from 'next/server';

type PhenomicsInput = {
  soilMicroScore: number;     // 微生物活性 (0–100)
  soilHumicScore: number;     // 腐植スコア (0–100)
  cnBalanceScore: number;     // C/N バランス (0–100)
  physChemScore: number;      // pH・Eh 等 (0–100)
  ndvi: number;               // NDVI (0–1)
  pri: number;                // PRI (0–1)
};

type PhenomicsOutput = {
  sheScore: number;             // SHE™ スコア
  polyphenolPotentialIndex: number; // ポリフェノール生成ポテンシャル (0–1)
};

function computeSheScore(input: PhenomicsInput): number {
  const { soilMicroScore, soilHumicScore, cnBalanceScore, physChemScore } = input;
  const she =
    0.40 * soilMicroScore +
    0.25 * soilHumicScore +
    0.20 * cnBalanceScore +
    0.15 * physChemScore;
  return Math.max(0, Math.min(100, she));
}

function computePolyphenolPotential(input: PhenomicsInput, sheScore: number): number {
  const canopySignal = 0.6 * input.ndvi + 0.4 * input.pri;
  const soilSignal = sheScore / 100;
  const potential = 0.5 * canopySignal + 0.5 * soilSignal;
  return Math.max(0, Math.min(1, potential));
}

export async function POST(req: NextRequest) {
  try {
    const body = (await req.json()) as PhenomicsInput;
    const sheScore = computeSheScore(body);
    const polyphenolPotentialIndex = computePolyphenolPotential(body, sheScore);

    const result: PhenomicsOutput = {
      sheScore: Number(sheScore.toFixed(1)),
      polyphenolPotentialIndex: Number(polyphenolPotentialIndex.toFixed(3)),
    };

    return NextResponse.json(result);
  } catch (e) {
    return NextResponse.json(
      { error: 'Invalid payload', details: String(e) },
      { status: 400 },
    );
  }
}
```

**パス:** `app/api/safetychain/soil/route.ts`

ts

```
import { NextRequest, NextResponse } from 'next/server';

type SoilInput = {
  eh_mV: number;          // 酸化還元電位
  moisturePct: number;    // 水分 (%)
  organicCarbon_gL: number;
  nitrogenAvailable_gL: number;
  mbt55Dose_Lha: number;
};

type SoilOutput = {
  aerobicFraction: number;      // 好気性画分 (0.05–0.95)
  carbonSequestrationRate: number; // tCO2/ha/year (概算)
};

function aerobicFraction(eh_mV: number, moisturePct: number): number {
  const fEh = 1.0 / (1.0 + Math.exp(-0.02 * (eh_mV - 50)));
  const fMoisture = 1.0 - Math.max(0.0, (moisturePct - 70.0) / 30.0);
  const val = fEh * fMoisture;
  return Math.min(0.95, Math.max(0.05, val));
}

function estimateCarbonSequestrationRate(input: SoilInput, aerobicFrac: number): number {
  const baseRate = 2.0; // tCO2/ha/year の基準
  const doseFactor = 1 + 0.3 * (input.mbt55Dose_Lha / 3.0); // 3L/ha を基準
  const moistureFactor = 1 + 0.2 * (Math.min(input.moisturePct, 80) - 50) / 30;
  const rate = baseRate * aerobicFrac * doseFactor * moistureFactor;
  return Number(rate.toFixed(2));
}

export async function POST(req: NextRequest) {
  try {
    const body = (await req.json()) as SoilInput;
    const af = aerobicFraction(body.eh_mV, body.moisturePct);
    const csr = estimateCarbonSequestrationRate(body, af);

    const result: SoilOutput = {
      aerobicFraction: Number(af.toFixed(3)),
      carbonSequestrationRate: csr,
    };

    return NextResponse.json(result);
  } catch (e) {
    return NextResponse.json(
      { error: 'Invalid payload', details: String(e) },
      { status: 400 },
    );
  }
}
```
