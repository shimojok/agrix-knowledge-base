```typescript
// lib/gates-dashboard/gatesCalculator.ts

export type GatesDashboardInput = {
  // 現在のAGRIX導入実績
  currentAdoptionHa: number;       // 現在の導入面積（ha）
  currentAvgGHGReduction: number;  // 現在の平均削減量（tCO2/ha/yr）
  
  // 拡張シナリオ
  targetAdoptionHa: number;        // 目標導入面積（例：5億ha）
  targetYear: number;              // 目標年（例：2050）
  
  // 経済データ
  conventionalCostPerHa: number;   // 従来農業コスト（USD/ha）
  agrixCostPerHa: number;          // AGRIX導入コスト（USD/ha）
  carbonCreditPrice: number;       // 炭素クレジット価格（USD/tCO2）
  yieldIncreaseRate: number;       // 収量増加率（%）
  qualityPremiumRate: number;      // 品質プレミアム（USD/ha）
  insuranceReduction: number;      // 保険料低減（USD/ha）
};

export type GatesDashboardOutput = {
  // 🎯 5.1Bトン削減達成評価
  ghgReduction: {
    currentAnnualReduction: number;        // 現在の年間削減量（tCO2）
    projectedAnnualReduction: number;      // 目標年の年間削減量（tCO2）
    targetAchieved: boolean;               // 5.1Bトン達成フラグ
    percentageOfGlobalEmissions: number;   // 世界排出量に対する割合（%）
    parisAgreementContribution: string;    // パリ協定貢献度（例：1.5℃目標のXX%）
  };
  
  // 💰 ネガティブ・グリーンプレミアム証明
  greenPremium: {
    conventionalTotalCost: number;         // 従来農業総コスト（USD/ha）
    agrixTotalCost: number;                // AGRIX総コスト（USD/ha）
    agrixTotalRevenue: number;             // AGRIX総収益（USD/ha）
    netBenefit: number;                    // 正味利益（USD/ha）※これが＞0ならネガティブ・グリーンプレミアム
    greenPremiumStatus: 'NEGATIVE' | 'ZERO' | 'POSITIVE';
    paybackPeriodYears: number;            // 投資回収期間（年）
    roi: number;                           // 投資利益率（%）
  };
  
  // 📈 スケール設計
  scaling: {
    adoptionCurve: { year: number; ha: number }[];
    cumulativeGHGReduction: number;        // 累積削減量
    totalEconomicImpact: number;           // 総経済インパクト（USD）
    jobsCreated: number;                   // 創出雇用数（推定）
  };
};

export function calculateGatesDashboard(input: GatesDashboardInput): GatesDashboardOutput {
  const {
    currentAdoptionHa,
    currentAvgGHGReduction,
    targetAdoptionHa,
    targetYear,
    conventionalCostPerHa,
    agrixCostPerHa,
    carbonCreditPrice,
    yieldIncreaseRate,
    qualityPremiumRate,
    insuranceReduction,
  } = input;

  // 1. GHG削減計算
  const currentReduction = currentAdoptionHa * currentAvgGHGReduction;
  // 拡張後の平均削減量（技術向上を仮定）
  const futureAvgReduction = currentAvgGHGReduction * 1.2;
  const projectedReduction = targetAdoptionHa * futureAvgReduction;
  const targetAchieved = projectedReduction >= 5.1e9; // 5.1Bトン

  // 2. ネガティブ・グリーンプレミアム計算
  const conventionalCost = conventionalCostPerHa;
  const agrixCost = agrixCostPerHa;
  const carbonRevenue = futureAvgReduction * carbonCreditPrice; // tCO2/ha × 価格
  const yieldRevenue = conventionalCostPerHa * (yieldIncreaseRate / 100);
  const qualityRevenue = qualityPremiumRate;
  const insuranceSavings = insuranceReduction;
  
  const totalRevenue = carbonRevenue + yieldRevenue + qualityRevenue + insuranceSavings;
  const netBenefit = totalRevenue - (agrixCost - conventionalCost);
  
  const greenPremiumStatus = netBenefit > 0 ? 'NEGATIVE' : netBenefit === 0 ? 'ZERO' : 'POSITIVE';
  const paybackPeriod = (agrixCost - conventionalCost) / (totalRevenue / 1); // 簡易計算
  const roi = ((totalRevenue - (agrixCost - conventionalCost)) / (agrixCost - conventionalCost)) * 100;

  // 3. スケール設計（指数関数的拡張を想定）
  const adoptionCurve = generateAdoptionCurve(2026, targetYear, currentAdoptionHa, targetAdoptionHa);
  const cumulativeReduction = adoptionCurve.reduce((sum, year) => sum + year.ha * futureAvgReduction, 0);
  const totalEconomicImpact = cumulativeReduction * carbonCreditPrice * 0.5 + // 炭素収益
                              adoptionCurve.reduce((sum, y) => sum + y.ha * (yieldRevenue + qualityRevenue + insuranceSavings), 0);

  return {
    ghgReduction: {
      currentAnnualReduction: currentReduction,
      projectedAnnualReduction: projectedReduction,
      targetAchieved,
      percentageOfGlobalEmissions: Number((projectedReduction / 51e9 * 100).toFixed(1)), // 世界排出量510億t/年
      parisAgreementContribution: targetAchieved 
        ? `1.5℃目標達成に十分な貢献（${(projectedReduction / 51e9 * 100).toFixed(1)}%削減）` 
        : `継続的な拡大が必要`,
    },
    greenPremium: {
      conventionalTotalCost: conventionalCost,
      agrixTotalCost: agrixCost,
      agrixTotalRevenue: totalRevenue,
      netBenefit: Number(netBenefit.toFixed(0)),
      greenPremiumStatus,
      paybackPeriodYears: Number(Math.max(0.5, paybackPeriod).toFixed(1)),
      roi: Number(roi.toFixed(0)),
    },
    scaling: {
      adoptionCurve,
      cumulativeGHGReduction: Number((cumulativeReduction / 1e9).toFixed(1)), // 10億t単位
      totalEconomicImpact: Number((totalEconomicImpact / 1e9).toFixed(2)), // 10億USD単位
      jobsCreated: Math.round(targetAdoptionHa * 0.05), // 簡易雇用推定
    },
  };
}

function generateAdoptionCurve(startYear: number, endYear: number, startHa: number, endHa: number) {
  const years = endYear - startYear;
  const curve = [];
  for (let i = 0; i <= years; i++) {
    const year = startYear + i;
    const progress = i / years;
    // S字カーブ（シグモイド）を仮定
    const sigmoid = 1 / (1 + Math.exp(-12 * (progress - 0.5)));
    const ha = startHa + (endHa - startHa) * sigmoid;
    curve.push({ year, ha: Math.round(ha) });
  }
  return curve;
}
```
