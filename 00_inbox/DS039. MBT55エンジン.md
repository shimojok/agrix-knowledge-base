# backend/engines/mbt55_engine.py
"""
MBT55 Ecological Hypercycle Engine
Based on M³-BioSynergy Theory and Planetary Metabolic Hypercycle (PMH) Model

Sources:
- MB001: Wilkinson's Planetary Ecology Framework
- MB002: M³-BioSynergy Theory (120菌種共生ネットワーク)
- MB003: Humus Formation Mechanism
- MB004: Chemical Fertilizer Reduction & Heavy Metal Detoxification
- MB005: PMH Mathematical Model
"""

import math
from typing import Dict, Tuple, Optional

class MBT55Engine:
    """
    MBT55微生物群による生態学的ハイパーサイクル・エンジン
    
    核心的特徴：
    - 120菌種の共生ネットワーク（好気性55% + 嫌気性45%）
    - 80-100℃高温発酵・24時間処理
    - 自己組織化的ハイパーサイクル制御
    """
    
    def __init__(self):
        # 菌種構成パラメータ（MB002, MB004より）
        self.microbial_diversity = 120  # 菌種数
        self.aerobic_ratio = 0.55       # 好気性菌比率
        self.anaerobic_ratio = 0.45     # 嫌気性菌比率
        
        # 処理パラメータ
        self.temperature_range = (80, 100)  # 発酵温度（℃）
        self.processing_time_hours = 24     # 処理時間
        
        # 数理モデルパラメータ（MB005より）
        self.alpha_1 = 0.35   # 炭素代謝係数
        self.alpha_2 = 0.28   # 窒素代謝係数
        self.beta_1 = 0.05    # 微生物死滅率
        
        # ハイパーサイクル安定性パラメータ
        self.hypercycle_threshold = 0.85  # MEI閾値（MB002）
        
    def calculate_ghg_reduction(self, 
                                 waste_amount_ton: float,
                                 waste_type: str = "mixed") -> Dict[str, float]:
        """
        温室効果ガス削減量の計算（MB001, MB003, MB004に基づく）
        
        Args:
            waste_amount_ton: 処理廃棄物量（トン）
            waste_type: 廃棄物種別（food_waste, livestock, mixed等）
        
        Returns:
            GHG削減量詳細（CH4, N2O, CO2, 合計）
        """
        # 従来処理との比較に基づく削減係数（MB001より）
        reduction_factors = {
            "CH4": 0.94,      # メタン94%削減
            "N2O": 0.78,      # 亜酸化窒素78%削減
            "CO2_manufacturing": 0.60,  # 製造時CO2 60%削減（化学肥料代替効果）
        }
        
        # ベースライン排出量（従来技術、MB001 Table参照）
        baseline_emissions = {
            "food_waste": {"CH4": 2.5, "N2O": 0.8, "CO2": 0.3},
            "livestock": {"CH4": 3.2, "N2O": 1.1, "CO2": 0.4},
            "mixed": {"CH4": 2.8, "N2O": 0.9, "CO2": 0.35},
        }
        
        base = baseline_emissions.get(waste_type, baseline_emissions["mixed"])
        
        # 削減量計算
        ch4_reduction = base["CH4"] * reduction_factors["CH4"] * waste_amount_ton
        n2o_reduction = base["N2O"] * reduction_factors["N2O"] * waste_amount_ton
        co2_reduction = base["CO2"] * reduction_factors["CO2_manufacturing"] * waste_amount_ton
        
        # CO2換算（GWP: CH4=28, N2O=265）
        total_co2e = (ch4_reduction * 28) + (n2o_reduction * 265) + co2_reduction
        
        return {
            "ch4_reduction_ton": round(ch4_reduction, 2),
            "n2o_reduction_ton": round(n2o_reduction, 2),
            "co2_reduction_ton": round(co2_reduction, 2),
            "total_co2e_reduction": round(total_co2e, 2),
            "unit": "tCO₂e"
        }
    
    def calculate_carbon_sequestration(self, 
                                       area_ha: float,
                                       application_years: int = 10) -> Dict[str, float]:
        """
        炭素隔離量の計算（MB001, MB003に基づく）
        
        Args:
            area_ha: 施用面積（ヘクタール）
            application_years: 施用期間（年）
        
        Returns:
            炭素隔離量詳細
        """
        # MB001より：正味炭素隔離量 310 tCO₂e/ha（10年間）
        base_sequestration_per_ha_10y = 310
        
        # MB003より：腐植化係数（HF）40%
        humification_factor = 0.40
        
        # MB001より：従来堆肥比 +186%
        improvement_ratio = 2.86  # 1 + 1.86
        
        total_sequestration = base_sequestration_per_ha_10y * area_ha * (application_years / 10)
        
        return {
            "total_co2e_sequestered": round(total_sequestration, 2),
            "per_ha_per_year": round(base_sequestration_per_ha_10y / 10, 2),
            "humification_factor": humification_factor,
            "improvement_over_conventional": f"+{(improvement_ratio - 1) * 100:.0f}%",
            "unit": "tCO₂e",
            "period_years": application_years
        }
    
    def calculate_fertilizer_replacement(self,
                                         crop_type: str = "rice",
                                         waste_input_ton: float = 1.0) -> Dict[str, float]:
        """
        化学肥料代替率の計算（MB004に基づく）
        
        Args:
            crop_type: 作物種別（rice, tomato, soybean等）
            waste_input_ton: 投入廃棄物量（トン）
        
        Returns:
            代替率と栄養供給量
        """
        # MB004 Table参照の作物別削減実績
        crop_replacement = {
            "rice": {"N": 60, "P": 60, "K": 50, "overall": 57},
            "tomato": {"N": 70, "P": 60, "K": 50, "overall": 60},
            "soybean": {"N": 50, "P": 100, "K": 50, "overall": 67},
            "default": {"N": 55, "P": 65, "K": 50, "overall": 57},
        }
        
        params = crop_replacement.get(crop_type, crop_replacement["default"])
        
        # 栄養含有量（MB004より：N85%・P90%変換効率）
        nutrient_conversion = {
            "N_efficiency": 0.85,
            "P_efficiency": 0.90,
            "K_efficiency": 0.70,
        }
        
        # 標準的な栄養含有量（廃棄物1トンあたり）
        base_nutrients = {"N_kg": 25, "P_kg": 8, "K_kg": 15}
        
        available_nutrients = {
            "N_kg": base_nutrients["N_kg"] * nutrient_conversion["N_efficiency"] * waste_input_ton,
            "P_kg": base_nutrients["P_kg"] * nutrient_conversion["P_efficiency"] * waste_input_ton,
            "K_kg": base_nutrients["K_kg"] * nutrient_conversion["K_efficiency"] * waste_input_ton,
        }
        
        return {
            "crop_type": crop_type,
            "replacement_rate_percent": {
                "N": params["N"],
                "P": params["P"],
                "K": params["K"],
                "overall": params["overall"]
            },
            "available_nutrients_kg": available_nutrients,
            "waste_input_ton": waste_input_ton
        }
    
    def calculate_heavy_metal_detoxification(self,
                                             heavy_metal_type: str,
                                             initial_concentration_mg_kg: float,
                                             treatment_hours: int = 24) -> Dict[str, float]:
        """
        重金属無害化率の計算（MB004, 特許2009-180255に基づく）
        
        Args:
            heavy_metal_type: 重金属種別（Cd, Pb, As）
            initial_concentration_mg_kg: 初期濃度（mg/kg）
            treatment_hours: 処理時間（時間）
        
        Returns:
            無害化率と最終濃度
        """
        # MB004よりの固定化率
        detox_rates = {
            "Cd": 0.95,   # 95%削減
            "Pb": 0.90,   # 90%削減
            "As": 0.85,   # 85%削減
        }
        
        rate = detox_rates.get(heavy_metal_type.upper(), 0.80)
        
        # 処理時間による補正（24時間基準）
        time_factor = min(1.0, treatment_hours / 24)
        effective_rate = rate * time_factor
        
        final_concentration = initial_concentration_mg_kg * (1 - effective_rate)
        
        return {
            "heavy_metal": heavy_metal_type.upper(),
            "detoxification_rate_percent": round(effective_rate * 100, 1),
            "initial_concentration_mg_kg": initial_concentration_mg_kg,
            "final_concentration_mg_kg": round(final_concentration, 2),
            "treatment_hours": treatment_hours,
            "mechanism": "sulfide_precipitation" if heavy_metal_type == "Cd" else "phosphate_precipitation"
        }
    
    def calculate_hypercycle_stability(self,
                                       microbial_diversity: Optional[int] = None,
                                       temperature_c: float = 85.0) -> Dict[str, float]:
        """
        生態学的ハイパーサイクル安定度の計算（MB002, MB005に基づく）
        
        Args:
            microbial_diversity: 微生物多様度（菌種数）
            temperature_c: 処理温度（℃）
        
        Returns:
            安定度指数とMEI（微生物創発係数）
        """
        diversity = microbial_diversity or self.microbial_diversity
        
        # MEI（微生物創発係数）計算（MB002より）
        # MEI = Σγ_ij / √n（γ_ijは菌種間相互作用強度）
        # 簡易実装：多様度が高いほど高MEI
        mei = min(0.98, 0.3 + 0.55 * (diversity / 120))
        
        # 温度適応性（MB001より：45℃でも性能±15%以内）
        optimal_temp = 90
        temp_factor = max(0.85, 1.0 - 0.007 * abs(temperature_c - optimal_temp))
        
        # 安定度指数（>0.85で超高速分解発現、MB002）
        stability_index = mei * temp_factor
        
        hypercycle_active = stability_index >= self.hypercycle_threshold
        
        return {
            "microbial_emergence_index_mei": round(mei, 3),
            "temperature_factor": round(temp_factor, 3),
            "hypercycle_stability_index": round(stability_index, 3),
            "threshold": self.hypercycle_threshold,
            "hypercycle_active": hypercycle_active,
            "status": "ACTIVE" if hypercycle_active else "INACTIVE"
        }
    
    def calculate_humus_formation(self,
                                  organic_carbon_input_ton: float,
                                  soil_type: str = "default") -> Dict[str, float]:
        """
        腐植質形成量の計算（MB003に基づく）
        
        Args:
            organic_carbon_input_ton: 投入有機炭素量（トン）
            soil_type: 土壌種別
        
        Returns:
            腐植質形成量と炭素固定量
        """
        # MB003より：腐植化係数（HF）= 35-40%
        humification_factor = 0.38  # 平均値
        
        # 土壌別補正係数
        soil_factors = {
            "andosol": 1.2,
            "gray_lowland": 1.0,
            "sand": 0.7,
            "default": 1.0,
        }
        
        soil_factor = soil_factors.get(soil_type, 1.0)
        
        # 腐植質形成量
        humus_formed_ton = organic_carbon_input_ton * humification_factor * soil_factor
        
        # 腐植質の炭素含量（約58%）
        carbon_sequestered = humus_formed_ton * 0.58 * 3.67  # 3.67 = 44/12（C→CO2換算）
        
        # MB003より：半減期15年以上
        half_life_years = 15
        
        return {
            "humus_formed_ton": round(humus_formed_ton, 2),
            "carbon_sequestered_co2e_ton": round(carbon_sequestered, 2),
            "humification_factor": humification_factor,
            "estimated_half_life_years": half_life_years,
            "soil_type": soil_type
        }
    
    def get_engine_status(self) -> Dict[str, any]:
        """エンジンの現在設定状態を返す"""
        return {
            "engine_name": "MBT55 Ecological Hypercycle Engine",
            "version": "1.0.0",
            "microbial_diversity": self.microbial_diversity,
            "aerobic_ratio": self.aerobic_ratio,
            "anaerobic_ratio": self.anaerobic_ratio,
            "temperature_range_c": self.temperature_range,
            "processing_time_hours": self.processing_time_hours,
            "hypercycle_threshold": self.hypercycle_threshold,
            "theoretical_basis": [
                "M³-BioSynergy Theory (MB002)",
                "Planetary Metabolic Hypercycle Model (MB005)",
                "Wilkinson's Planetary Ecology Framework (MB001)",
                "Humus Formation Mechanism (MB003)",
                "Chemical Fertilizer Reduction Data (MB004)"
            ]
        }


# エンジンのインスタンス作成
