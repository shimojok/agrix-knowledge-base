### 2. Bond エンティティとエンジンの初期実装

`bond_engine.py` に以下を貼り付けてください。

python

```
from dataclasses import dataclass
import math

@dataclass
class PBPESovereignBond:
    bond_id: str
    face_value: float          # 額面
    coupon_rate: float         # 年利（ベース）
    maturity_years: int        # 満期（年）
    pbpe_backing: float        # PBPE による裏付け価値
    pbpe_stability: float      # PBPE 安定性指数（0〜1）

class SovereignBondEngine:
    """
    PBPE Sovereign Bond の価格と利回りを計算するエンジン。
    PBPE 安定性指数に応じて割引率を調整する。
    """

    def price(self, bond: PBPESovereignBond) -> float:
        r = self._pbpe_adjusted_rate(bond)
        price = 0.0

        # クーポンの現在価値
        for t in range(1, bond.maturity_years + 1):
            coupon = bond.face_value * bond.coupon_rate
            price += coupon / ((1 + r) ** t)

        # 額面＋PBPE裏付けの現在価値
        terminal_value = bond.face_value + bond.pbpe_backing
        price += terminal_value / ((1 + r) ** bond.maturity_years)

        return price

    def yield_to_maturity(self, bond: PBPESovereignBond, market_price: float) -> float:
        """
        単純なニュートン法で YTM を近似計算。
        """
        y = 0.03  # 初期推定
        for _ in range(20):
            f = self._npv(bond, y) - market_price
            df = self._npv_derivative(bond, y)
            if abs(df) < 1e-8:
                break
            y = y - f / df
        return max(0.0, y)

    def _npv(self, bond: PBPESovereignBond, rate: float) -> float:
        npv = 0.0
        for t in range(1, bond.maturity_years + 1):
            coupon = bond.face_value * bond.coupon_rate
            npv += coupon / ((1 + rate) ** t)
        terminal_value = bond.face_value + bond.pbpe_backing
        npv += terminal_value / ((1 + rate) ** bond.maturity_years)
        return npv

    def _npv_derivative(self, bond: PBPESovereignBond, rate: float) -> float:
        d = 0.0
        for t in range(1, bond.maturity_years + 1):
            coupon = bond.face_value * bond.coupon_rate
            d += -t * coupon / ((1 + rate) ** (t + 1))
        terminal_value = bond.face_value + bond.pbpe_backing
        d += -bond.maturity_years * terminal_value / ((1 + rate) ** (bond.maturity_years + 1))
        return d

    def _pbpe_adjusted_rate(self, bond: PBPESovereignBond) -> float:
        """
        PBPE 安定性指数に応じて割引率を調整。
        安定性が高いほど割引率が下がり、価格が上がる。
        """
        base_rate = 0.05          # ベースの市場金利
        adjustment = 0.03 * bond.pbpe_stability
        return max(0.001, base_rate - adjustment)
```

```
from pbpe_core.sovereign_bond.bond_engine import PBPESovereignBond, SovereignBondEngine

bond = PBPESovereignBond(
    bond_id="JP-PBPE-2035",
    face_value=1000,
    coupon_rate=0.03,
    maturity_years=10,
    pbpe_backing=200,
    pbpe_stability=0.8
)

engine = SovereignBondEngine()
price = engine.price(bond)
ytm = engine.yield_to_maturity(bond, market_price=price)

print("Price:", price)
print("YTM:", ytm)
```
