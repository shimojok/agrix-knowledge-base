import { useState } from "react";
import PBPEScoreCard from "../../components/marketplace/PBPEScoreCard";

export default function PBPEValuePage() {
  const [score, setScore] = useState(null);

  async function calcPBPE() {
    const res = await fetch("http://localhost:8000/pbpe/value/calc", {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({
        project_id: "UI-TEST",
        domain: "agriculture",
        country: "JP",
        time_horizon_years: 10,
        inputs: { financial: {}, social: {}, ecological: {}, governance: {} },
        weights: { financial: 0.3, social: 0.3, ecological: 0.25, governance: 0.15 }
      })
    });
    const data = await res.json();
    setScore(data.pbpe_score);
  }

  return (
    <div>
      <h2>PBPE Value Calculator</h2>
      <button onClick={calcPBPE}>Calculate PBPE</button>
      {score && <PBPEScoreCard score={score} />}
    </div>
  );
}
```

### 🧮 初期実装（PBPE Bond Pricing）

python

```
from dataclasses import dataclass
import datetime as dt
import math

@dataclass
class PBPESovereignBond:
    bond_id: str
    face_value: float
    coupon_rate: float
    maturity_years: int
    pbpe_backing: float
    pbpe_stability: float

class SovereignBondEngine:

    def price(self, bond: PBPESovereignBond):
        r = self._pbpe_adjusted_rate(bond)
        price = 0

        for t in range(1, bond.maturity_years + 1):
            price += (bond.face_value * bond.coupon_rate) / ((1 + r) ** t)

        price += bond.pbpe_backing / ((1 + r) ** bond.maturity_years)
        return price

    def _pbpe_adjusted_rate(self, bond: PBPESovereignBond):
        base_rate = 0.03
        delta = 0.02 * bond.pbpe_stability
        return max(0.001, base_rate - delta)
```

from dataclasses import dataclass

@dataclass
class PBPECurrencyUnit:
    pbpe_value: float
    reserve_ratio: float

class PBPECurrencyModel:

    def issue(self, unit: PBPECurrencyUnit):
        issued = (1 - unit.reserve_ratio) * unit.pbpe_value
        reserve = unit.reserve_ratio * unit.pbpe_value
        return {
            "issued_amount": issued,
            "reserve": reserve
        }

    def fx_stability(self, pbpe_value, stability_index):
        return pbpe_value * (0.5 + stability_index)
```
