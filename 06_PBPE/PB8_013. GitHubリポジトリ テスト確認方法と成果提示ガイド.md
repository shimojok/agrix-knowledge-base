```typescript
import { render, screen, fireEvent } from '@testing-library/react';
import ImpactSimulator from '../ImpactSimulator';

describe('ImpactSimulator', () => {
  test('renders simulator with default values', () => {
    render(<ImpactSimulator />);
    expect(screen.getByText('Impact Simulator')).toBeInTheDocument();
    expect(screen.getByText('Farm Area (ha)')).toBeInTheDocument();
  });

  test('updates results when parameters change', async () => {
    render(<ImpactSimulator />);
    const areaInput = screen.getByLabelText('Farm Area (ha)');
    fireEvent.change(areaInput, { target: { value: '20' } });
    
    // 結果が更新されることを確認
    expect(await screen.findByText('Total Revenue')).toBeInTheDocument();
  });

  test('MBT55 toggle affects results', () => {
    render(<ImpactSimulator />);
    const toggle = screen.getByLabelText('MBT55 Applied');
    fireEvent.click(toggle);
    
    // MBT55適用時のROIが表示されることを確認
    expect(screen.getByText(/ROI/)).toBeInTheDocument();
  });
});
```
