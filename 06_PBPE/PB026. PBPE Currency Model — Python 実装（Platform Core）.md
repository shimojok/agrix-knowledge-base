# 💱 2. Currency Model 初期実装（PBPE 通貨発行モデル）

以下を `model.py` に貼り付けてください。

python

```
from dataclasses import dataclass

@dataclass
class PBPECurrencyUnit:
    """
    PBPE 通貨単位の基本構造。
    pbpe_value: PBPE Value Engine による価値（0〜1）
    reserve_ratio: 裏付け資産比率（0〜1）
    """
    pbpe_value: float
    reserve_ratio: float


class PBPECurrencyModel:
    """
    PBPE 通貨モデル。
    PBPE Value と PBPE Stability に基づき、通貨発行量と安定性を計算する。
    """

    def issue(self, unit: PBPECurrencyUnit):
        """
        PBPE Value に基づく通貨発行量。
        PBPE Value が高いほど発行量が増える。
        reserve_ratio に応じて裏付け資産が確保される。
        """
        issued_amount = (1 - unit.reserve_ratio) * unit.pbpe_value
        reserve_amount = unit.reserve_ratio * unit.pbpe_value

        return {
            "issued_amount": issued_amount,
            "reserve_amount": reserve_amount
        }

    def fx_stability(self, pbpe_value: float, stability_index: float):
        """
        PBPE Value と Stability Index に基づく為替安定化モデル。
        stability_index（0〜1）が高いほど通貨価値が安定する。
        """
        return pbpe_value * (0.5 + stability_index)

    def convert(self, amount: float, pbpe_value: float):
        """
        PBPE Value に基づく通貨換算。
        PBPE Value が高いほど換算レートが上がる。
        """
        rate = 1 + pbpe_value
        return amount * rate
```
