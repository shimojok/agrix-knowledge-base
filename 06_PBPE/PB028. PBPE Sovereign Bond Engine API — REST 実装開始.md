### 1. API ファイル作成

### 2. Bond API コード貼り付け

`routes_pbpe_bond.py` に以下を貼り付けます。

python

```
from fastapi import APIRouter
from pbpe_core.sovereign_bond.bond_engine import (
    PBPESovereignBond,
    SovereignBondEngine
)

router = APIRouter()
engine = SovereignBondEngine()

@router.post("/pbpe/bond/price")
def bond_price(payload: dict):
    bond = PBPESovereignBond(
        bond_id=payload["bond_id"],
        face_value=payload["face_value"],
        coupon_rate=payload["coupon_rate"],
        maturity_years=payload["maturity_years"],
        pbpe_backing=payload["pbpe_backing"],
        pbpe_stability=payload["pbpe_stability"]
    )
    price = engine.price(bond)
    return {
        "bond_id": bond.bond_id,
        "price": price
    }

@router.post("/pbpe/bond/ytm")
def bond_ytm(payload: dict):
    bond = PBPESovereignBond(
        bond_id=payload["bond_id"],
        face_value=payload["face_value"],
        coupon_rate=payload["coupon_rate"],
        maturity_years=payload["maturity_years"],
        pbpe_backing=payload["pbpe_backing"],
        pbpe_stability=payload["pbpe_stability"]
    )
    market_price = payload["market_price"]
    ytm = engine.yield_to_maturity(bond, market_price)
    return {
        "bond_id": bond.bond_id,
        "ytm": ytm
    }
```
