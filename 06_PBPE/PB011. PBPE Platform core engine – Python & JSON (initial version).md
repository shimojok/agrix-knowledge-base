This module provides the core orchestration layer for the PBPE Platform:

from __future__ import annotations
import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Dict, List, Optional


# ---------------------------------------------------------------------------
# Configuration models
# ---------------------------------------------------------------------------

@dataclass
class ModuleConfig:
    """Configuration for a single PBPE module (Marketplace, ValueEngine, CurrencyModel, etc.)."""
    name: str
    enabled: bool = True
    version: str = "0.1.0"
    endpoint: Optional[str] = None
    params: Dict[str, Any] = field(default_factory=dict)


@dataclass
class PlatformConfig:
    """Top‑level PBPE Platform configuration."""
    platform_name: str
    environment: str
    modules: List[ModuleConfig] = field(default_factory=list)
    logging_level: str = "INFO"


# ---------------------------------------------------------------------------

class PBPEEngine:
    """
    PBPEEngine is the central coordinator:

        self._initialized = False

        path = Path(path)
        with path.open("r", encoding="utf-8") as f:
            raw = json.load(f)

        modules = [
            ModuleConfig(
                name=m["name"],
                enabled=m.get("enabled", True),
                version=m.get("version", "0.1.0"),
                endpoint=m.get("endpoint"),
                params=m.get("params", {}),
            )
            for m in raw.get("modules", [])
        ]

        config = PlatformConfig(
            platform_name=raw.get("platform_name", "PBPE Platform"),
            environment=raw.get("environment", "development"),
            modules=modules,
            logging_level=raw.get("logging_level", "INFO"),
        )
        return cls(config)

        Route a value flow between modules.
