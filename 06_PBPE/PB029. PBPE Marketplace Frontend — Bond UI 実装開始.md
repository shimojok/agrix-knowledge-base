## 🖥️ 5. Bond UI ページコード（API 連携）

`bond.tsx` に貼り付け：

tsx

```
import { useState } from "react";
import AppLayout from "../../components/layout/AppLayout";
import BondPriceCard from "../../components/marketplace/BondPriceCard";
import BondYTMCard from "../../components/marketplace/BondYTMCard";

export default function BondPage() {
  const [price, setPrice] = useState<number | null>(null);
  const [ytm, setYtm] = useState<number | null>(null);

  async function calcPrice() {
    const res = await fetch("http://localhost:8000/pbpe/bond/price", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        bond_id: "JP-PBPE-2035",
        face_value: 1000,
        coupon_rate: 0.03,
        maturity_years: 10,
        pbpe_backing: 200,
        pbpe_stability: 0.8
      })
    });
    const data = await res.json();
    setPrice(data.price);
  }

  async function calcYTM() {
    if (price === null) return;

    const res = await fetch("http://localhost:8000/pbpe/bond/ytm", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        bond_id: "JP-PBPE-2035",
        face_value: 1000,
        coupon_rate: 0.03,
        maturity_years: 10,
        pbpe_backing: 200,
        pbpe_stability: 0.8,
        market_price: price
      })
    });
    const data = await res.json();
    setYtm(data.ytm);
  }

  return (
    <AppLayout>
      <h2>PBPE Sovereign Bond Model</h2>

      <button onClick={calcPrice}>Calculate Bond Price</button>
      {price !== null && <BondPriceCard price={price} />}

      {price !== null && (
        <button onClick={calcYTM} style={{ marginTop: "20px" }}>
          Calculate YTM
        </button>
      )}
      {ytm !== null && <BondYTMCard ytm={ytm} />}
    </AppLayout>
  );
}
```
