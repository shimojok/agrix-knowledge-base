### 2. PBPE Currency Model API コード貼り付け

`routes_pbpe_currency.py` に以下を貼り付けます。

python

```
from fastapi import APIRouter
from pbpe_core.currency_model.model import PBPECurrencyUnit, PBPECurrencyModel

router = APIRouter()
model = PBPECurrencyModel()

@router.post("/pbpe/currency/issue")
def issue_currency(payload: dict):
    unit = PBPECurrencyUnit(
        pbpe_value=payload["pbpe_value"],
        reserve_ratio=payload["reserve_ratio"]
    )
    result = model.issue(unit)
    return {
        "issued_amount": result["issued_amount"],
        "reserve_amount": result["reserve_amount"]
    }

@router.post("/pbpe/currency/stability")
def currency_stability(payload: dict):
    pbpe_value = payload["pbpe_value"]
    stability_index = payload["stability_index"]
    result = model.fx_stability(pbpe_value, stability_index)
    return {
        "fx_stability": result
    }

@router.post("/pbpe/currency/convert")
def convert_currency(payload: dict):
    amount = payload["amount"]
    pbpe_value = payload["pbpe_value"]
    result = model.convert(amount, pbpe_value)
    return {
        "converted_amount": result
    }
```
