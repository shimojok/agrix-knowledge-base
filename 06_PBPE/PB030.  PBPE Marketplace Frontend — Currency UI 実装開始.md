## 🖥️ 6. Currency UI ページコード（API 連携）

`currency.tsx` に貼り付け：

tsx

```
import { useState } from "react";
import AppLayout from "../../components/layout/AppLayout";
import CurrencyIssueCard from "../../components/marketplace/CurrencyIssueCard";
import CurrencyStabilityCard from "../../components/marketplace/CurrencyStabilityCard";
import CurrencyConvertCard from "../../components/marketplace/CurrencyConvertCard";

export default function CurrencyPage() {
  const [issue, setIssue] = useState<any>(null);
  const [stability, setStability] = useState<number | null>(null);
  const [converted, setConverted] = useState<number | null>(null);

  async function calcIssue() {
    const res = await fetch("http://localhost:8000/pbpe/currency/issue", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        pbpe_value: 0.85,
        reserve_ratio: 0.3
      })
    });
    const data = await res.json();
    setIssue(data);
  }

  async function calcStability() {
    const res = await fetch("http://localhost:8000/pbpe/currency/stability", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        pbpe_value: 0.85,
        stability_index: 0.9
      })
    });
    const data = await res.json();
    setStability(data.fx_stability);
  }

  async function calcConvert() {
    const res = await fetch("http://localhost:8000/pbpe/currency/convert", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        amount: 100,
        pbpe_value: 0.85
      })
    });
    const data = await res.json();
    setConverted(data.converted_amount);
  }

  return (
    <AppLayout>
      <h2>PBPE Currency Model</h2>

      <button onClick={calcIssue}>Calculate Issuance</button>
      {issue && <CurrencyIssueCard issued={issue.issued_amount} reserve={issue.reserve_amount} />}

      <button onClick={calcStability} style={{ marginTop: "20px" }}>
        Calculate FX Stability
      </button>
      {stability !== null && <CurrencyStabilityCard stability={stability} />}

      <button onClick={calcConvert} style={{ marginTop: "20px" }}>
        Convert Currency
      </button>
      {converted !== null && <CurrencyConvertCard converted={converted} />}
    </AppLayout>
  );
}
```
