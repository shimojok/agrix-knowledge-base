### Step 1: ナレッジ抽出用RAG検索スクリプトの準備

まずは、設計に必要な数式・アルゴリズム・境界条件をナレッジベースから一括で引き出すための専用抽出スクリプト `scripts/extract_hypercycle_scheme.py` を作成します。

プロジェクト直下の **`scripts/extract_hypercycle_scheme.py`** として、以下のコードを配置してください。

```Python
# scripts/extract_hypercycle_scheme.py
import sys
import os

# プロジェクトルートをPythonパスに追加
sys.path.append(os.path.abspath(os.path.join(os.path.dirname(__file__), "..")))

from src.repositories.rag_knowledge_repository import RAGKnowledgeRepository

def main():
    print("=== BioNexus Hypercycle Scheme Knowledge Extraction ===")
    rag_repo = RAGKnowledgeRepository()
    
    # スキーマ設計に必要な高度な数理・ロジック抽出クエリ
    queries = [
        "菌相相互作用マトリックス gamma_ij 微生物ギルド 相互作用 方程式",
        "ミカエリス・メンテン 分解速度定数 Km Vmax 温度依存性 Arrhenius",
        "最大エントロピー生産原理 MEP 自由エネルギー 散逸構造",
        "NCS 炭素隔離 10年 30年 試算モデル 腐植質形成率 HFR"
    ]
    
    for i, query in enumerate(queries, 1):
        print(f"\n==========================================")
        print(f"🔍 [Extraction Query {i}]: {query}")
        print(f"==========================================")
        try:
            results = rag_repo.search_knowledge(query, top_k=2)
            if not results:
                print("⚠️ 該当するナレッジが見つかりませんでした。")
                continue
                
            for idx, res in enumerate(results, 1):
                source = getattr(res, 'source', getattr(res, 'metadata', {}).get('source', 'Unknown'))
                print(f"\n--- Result [{idx}] From: {source} ---")
                content = getattr(res, 'content', str(res))
                # テキストの冒頭と末尾を表示して構造を確認
                print(content[:500] + "\n...\n" + content[-300:])
        except Exception as e:
            print(f"❌ 抽出エラーが発生しました: {e}")

if __name__ == "__main__":
    main()
```
