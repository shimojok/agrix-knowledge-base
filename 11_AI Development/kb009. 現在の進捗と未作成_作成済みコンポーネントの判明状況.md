import os
import json
from typing import Optional, List
from src.models.hamada_matrix import HamadaMatrixSchema, DiseaseRiskProfile

class HamadaMatrixRepository:
    def __init__(self, json_path: Optional[str] = None):
        if json_path is None:
            base_dir = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
            json_path = os.path.join(base_dir, "..", "agrix-knowledge-base", "data", "hamada_137_risk_matrix.json")
            json_path = os.path.abspath(json_path)

        self.json_path = json_path
        self.data: Optional[HamadaMatrixSchema] = None
        self._load_data()

    def _load_data(self):
        if not os.path.exists(self.json_path):
            # ダミーフォールバックデータの生成 (ファイル不在時の安全策)
            dummy_diseases = [
                DiseaseRiskProfile(
                    disease_id=i,
                    disease_name=f"疾患サンプル_{i}",
                    category="代謝性疾患" if i % 2 == 0 else "循環器疾患",
                    base_risk_score=0.5,
                    risk_factors={"野菜不足": 0.2, "運動不足": 0.3, "MBT55摂取": -0.4}
                ) for i in range(1, 138)
            ]
            self.data = HamadaMatrixSchema(version="2.0-fallback", total_diseases=137, diseases=dummy_diseases)
            return

        with open(self.json_path, "r", encoding="utf-8") as f:
            raw_data = json.load(f)
            self.data = HamadaMatrixSchema(**raw_data)

    def get_disease_by_id(self, disease_id: int) -> Optional[DiseaseRiskProfile]:
        if not self.data:
            return None
        for d in self.data.diseases:
            if d.disease_id == disease_id:
                return d
        return None

    def search_diseases(self, keyword: str) -> List[DiseaseRiskProfile]:
        if not self.data:
            return []
        return [d for d in self.data.diseases if keyword.lower() in d.disease_name.lower() or keyword.lower() in d.category.lower()]

import os
import sys
from dotenv import load_dotenv
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_chroma import Chroma

load_dotenv()

class RAGKnowledgeRepository:
    def __init__(self):
        self.base_dir = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
        self.db_dir = os.path.join(self.base_dir, "chroma_db")
        self.model_name = os.getenv("EMBEDDING_MODEL", "intfloat/multilingual-e5-small")
        
        self.embeddings = HuggingFaceEmbeddings(model_name=self.model_name)
        
        if os.path.exists(self.db_dir):
            self.db = Chroma(persist_directory=self.db_dir, embedding_function=self.embeddings)
        else:
            self.db = None

    def search_knowledge(self, query: str, folder_filter: str = None, top_k: int = 4):
        if not self.db:
            return []
        
        formatted_query = f"query: {query}"
        search_kwargs = {"k": top_k}
        if folder_filter:
            search_kwargs["filter"] = {"folder": folder_filter}

        return self.db.similarity_search(formatted_query, **search_kwargs)

from typing import Dict, List

class RepoDependencyAnalyzer:
    """16リポジトリの依存関係およびデータフローを解析するアナライザー"""
    def __init__(self):
        self.repo_map = {
            "agrix-knowledge-base": ["PBPE-Marketplace", "PBPE-Finance", "HealthBook-AI"],
            "HealthBook-AI": ["M3-Core-Engine", "hamada_137_risk_matrix"],
            "M3-Core-Engine": ["phytochemicals_diet"],
            "PBPE-Marketplace": ["PBPE-Finance"]
        }

    def analyze_impact(self, modified_repo: str) -> List[str]:
        impacted = set()
        def dfs(repo):
            for target, deps in self.repo_map.items():
                if repo in deps and target not in impacted:
                    impacted.add(target)
                    dfs(target)
        dfs(modified_repo)
        return list(impacted)
```

import os
import sys
from typing import Dict, Any

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
sys.path.append(BASE_DIR)

from src.repositories.hamada_matrix_repository import HamadaMatrixRepository
from src.analyzer.repo_dependency_analyzer import RepoDependencyAnalyzer

class BioNexusSimEngine:
    def __init__(self):
        self.hamada_repo = HamadaMatrixRepository()
        self.analyzer = RepoDependencyAnalyzer()

    def run_metabolic_simulation(self, disease_id: int, interventions: Dict[str, float]) -> Dict[str, Any]:
        """
        浜田137マトリックスと介入因子（漢方・MBT55等）に基づくシミュレーション演算
        """
        disease = self.hamada_repo.get_disease_by_id(disease_id)
        if not disease:
            return {"error": f"Disease ID {disease_id} not found."}

        current_risk = disease.base_risk_score
        delta_risk = 0.0

        for factor, weight in interventions.items():
            if factor in disease.risk_factors:
                effect = disease.risk_factors[factor] * weight
                delta_risk += effect

        simulated_risk = max(0.0, min(1.0, current_risk + delta_risk))
        risk_reduction_rate = ((current_risk - simulated_risk) / current_risk) * 100 if current_risk > 0 else 0.0

        # パラメータ試算 (SCFA, ATP, GLP-1)
        mbt_weight = interventions.get("MBT55", 0.0)
        scfa_boost = mbt_weight * 1.45
        atp_boost = mbt_weight * 1.28
        glp1_signal = mbt_weight * 1.35

        return {
            "disease_id": disease.disease_id,
            "disease_name": disease.disease_name,
            "base_risk": round(current_risk, 3),
            "simulated_risk": round(simulated_risk, 3),
            "risk_reduction_percent": round(risk_reduction_rate, 2),
            "metrics": {
                "scfa_boost_rate": round(scfa_boost, 2),
                "atp_boost_rate": round(atp_boost, 2),
                "glp1_signal_level": round(glp1_signal, 2)
            }
        }

if __name__ == "__main__":
    engine = BioNexusSimEngine()
    res = engine.run_metabolic_simulation(1, {"MBT55": 0.8, "野菜不足": -0.5})
    print("--- [BioNexus Sim Engine Test Output] ---")
    print(res)
```
