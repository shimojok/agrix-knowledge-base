```Python
# src/models/mbt_hypercycle.py
from typing import List, Dict, Optional
from pydantic import BaseModel, Field

class MicrobialGuildConfig(BaseModel):
    """菌群（ギルド）および相互作用パラメータ"""
    guild_id: str = Field(..., description="ギルド識別子 (例: Decomposers, Transformers, Regulators)")
    species_count: int = Field(default=40, description="所属菌種数 (合計120菌種)")
    activity_rate: float = Field(..., description="分解・代謝活性度 (0.0 - 1.0)")

class HypercycleMatrix(BaseModel):
    """菌種・ギルド間相互作用マトリックス (gamma_ij)"""
    interaction_matrix: Dict[str, Dict[str, float]] = Field(
        ..., description="ギルド間相利共生・競合係数マトリックス"
    )

class SubstrateInput(BaseModel):
    """投入バイオマス・基質境界条件"""
    total_carbon_kg: float = Field(..., description="投入全炭素量 (kg)")
    total_nitrogen_kg: float = Field(..., description="投入全窒素量 (kg)")
    moisture_content: float = Field(..., description="含水率 (%)")
    temperature_celsius: float = Field(..., description="環境/反応温度 (℃)")
    retention_hours: float = Field(default=24.0, description="滞留・分解時間 (h)")

class HypercycleSimulationResult(BaseModel):
    """MBTハイパーサイクルシミュレーション計算結果"""
    cce: float = Field(..., description="炭素循環効率 (Carbon Cycle Efficiency: CCE)")
    hfr: float = Field(..., description="腐植質形成率 (Humus Formation Rate: HFR)")
    ncs_ton_co2e: float = Field(..., description="正味炭素隔離量 (Net Carbon Sequestration: NCS, tCO2e/yr)")
    ch4_reduction_rate: float = Field(..., description="メタン排出削減率 (%)")
    n2o_reduction_rate: float = Field(..., description="一酸化二窒素排出削減率 (%)")
    energy_conversion_efficiency: float = Field(..., description="エネルギー変換効率 (eta)")
```
