1. PBPE経済モデルの数式をコード化（PBPE-Finance拡張）

既存の src/pbpe_engine.py に、BAU vs PBPE の比較と 10年シナリオを追加する形で拡張します。

`python

src/pbpe_engine.py

import numpy as np
import numpy_financial as npf


class PBPEEngine:
    def init(self, inputs: dict):
        self.inputs = inputs
        # BAU側の基準値（必要なら外部から渡してもよい）
        self.inputs.setdefault("bauyieldloss_pct", 0.25)      # 25% disease loss
        self.inputs.setdefault("pbpeyieldgainpct", self.inputs["yieldincrease_pct"])
        self.inputs.setdefault("pbpeyieldloss_pct", 0.10)     # 10% loss after PBPE
        self.inputs.setdefault("baucostfactor", 1.0)
        self.inputs.setdefault("pbpecostreduction_pct", 0.4)  # 40% total cost reduction
        self.inputs.setdefault("bau_ch4", 1.0)                  # tCO2e/ha baseline
        self.inputs.setdefault("bau_n2o", 1.0)
        self.inputs.setdefault("pbpech4reduction_pct", 0.5)   # 50%削減
        self.inputs.setdefault("pbpen2oreduction_pct", 0.4)   # 40%削減
        self.inputs.setdefault("deltactcperha", 2.0)        # ΔC tCO2/ha/yr

    # ----------------------------
    # GOLD（既存）
    # ----------------------------
    def compute_gold(self):
        y = self.inputs["yield_base"]
        dy = self.inputs["yieldincreasepct"]
        p = self.inputs["priceperton"]
        ha = self.inputs["ha"]
        fert = self.inputs["fertilizercostsavings"]

        base = y  ha  p
        new = y  (1 + dy)  ha * p

        return (new - base) + fert * ha

    # ----------------------------
    # GREEN（既存）
    # ----------------------------
    def compute_green(self):
        production = self.inputs["yieldbase"]  (1 + self.inputs["yieldincrease_pct"])  self.inputs["ha"]
        return production  self.inputs["co2reductionperton"]  self.inputs["carbonprice"]

    # ----------------------------
    # BLUE（既存）
    # ----------------------------
    def compute_blue(self, year):
        production = self.inputs["yieldbase"]  (1 + self.inputs["yieldincrease_pct"])  self.inputs["ha"]
        return (production  self.inputs["healthcarecostavoidance"]) / ((1 + self.inputs["healthdiscountrate"]) * year)

    # ----------------------------
    # PBPE Value（統合）
    # ----------------------------
    def computepbpevalue(self, year):
        return self.computegold() + self.computegreen() + self.compute_blue(year)

    # ----------------------------
    # BAU vs PBPE：Yieldモデル
    # ----------------------------
    def yield_bau(self):
        y0 = self.inputs["yield_base"]
        loss = self.inputs["bauyieldloss_pct"]
        ha = self.inputs["ha"]
        return y0  (1 - loss)  ha

    def yield_pbpe(self):
        y0 = self.inputs["yield_base"]
        gain = self.inputs["pbpeyieldgain_pct"]
        loss = self.inputs["pbpeyieldloss_pct"]
        ha = self.inputs["ha"]
        return y0  (1 + gain)  (1 - loss) * ha

    # ----------------------------
    # BAU vs PBPE：Costモデル
    # ----------------------------
    def cost_bau(self):
        # BAUコストは Fert + Pesticide + Irrigation + Waste をまとめた係数として扱う
        basecost = self.inputs["baucost_factor"] * self.inputs["ha"]
        return base_cost

    def cost_pbpe(self):
        basecost = self.inputs["baucost_factor"] * self.inputs["ha"]
        reduction = self.inputs["pbpecostreduction_pct"]
        return base_cost * (1 - reduction)

    # ----------------------------
    # BAU vs PBPE：GHGモデル
    # ----------------------------
    def ghg_bau(self):
        ch4 = self.inputs["bau_ch4"]
        n2o = self.inputs["bau_n2o"]
        return ch4 + n2o  # CO2は別途扱うなら追加

    def ghg_pbpe(self):
        ch4 = self.inputs["bauch4"] * (1 - self.inputs["pbpech4reductionpct"])
        n2o = self.inputs["baun2o"] * (1 - self.inputs["pbpen2oreductionpct"])
        deltac = self.inputs["deltactcper_ha"] * self.inputs["ha"]
        return ch4 + n2o - delta_c

    # ----------------------------
    # CASHFLOW（既存）
    # ----------------------------
    def compute_cashflows(self):
        cf = []
        for t in range(1, self.inputs["years"] + 1):
            cf.append(self.computepbpevalue(t))
        return cf

    # ----------------------------
    # NPV / IRR（既存）
    # ----------------------------
    def computenpvirr(self):
        cf = self.compute_cashflows()
        npv = sum([c / ((1 + self.inputs["discount_rate"])  (i + 1)) for i, c in enumerate(cf)])
        irr = npf.irr([-self.inputs["initial_investment"]] + cf)
        return npv, irr

    # ----------------------------
    # 10年シナリオ：BAU vs PBPE
    # ----------------------------
    def scenario_10y(self):
        years = 10
        r = self.inputs["discount_rate"]
        price = self.inputs["priceperton"]

        bauyield = self.yieldbau()
        pbpeyield = self.yieldpbpe()

        baurevenue = bauyield * price
        pbperevenue = pbpeyield * price

        baucost = self.costbau()
        pbpecost = self.costpbpe()

        bau_cf = []
        pbpe_cf = []
        for t in range(1, years + 1):
            baucft = baurevenue - baucost
            pbpecft = pbperevenue - pbpecost + self.computepbpevalue(t)
            baucf.append(baucf_t)
            pbpecf.append(pbpecf_t)

        baunpv = sum([c / ((1 + r)  (i + 1)) for i, c in enumerate(baucf)])
        pbpenpv = sum([c / ((1 + r)  (i + 1)) for i, c in enumerate(pbpecf)])

        bauirr = npf.irr([-self.inputs["initialinvestment"]] + bau_cf)
        pbpeirr = npf.irr([-self.inputs["initialinvestment"]] + pbpe_cf)

        # Scope3：PBPE側のみ
        scope3pbpe = self.scope3value(coffeetons=self.inputs.get("scope3coffee_tons", 400000))

        return {
            "BAU": {"CF": baucf, "NPV": baunpv, "IRR": bau_irr},
            "PBPE": {"CF": pbpecf, "NPV": pbpenpv, "IRR": pbpeirr, "Scope3": scope3pbpe}
        }

    # 既存の simulatefoodprice / capitalmultiplier / scope3value / esgmarketcapimpact / investormetrics はそのまま
`

2. PBPE Marketplace の KPI をこの数式に完全連動させる

Next.js 側（pbpe-marketplace）の /dashboard/kpis は、PBPE-Finance の PBPEEngine から以下を取得して表示する構造にすると矛盾がなくなります。

バックエンド（例：APIルート）

`ts
// pages/api/pbpe/kpis.ts（Next.js API Route の例）

import type { NextApiRequest, NextApiResponse } from "next";
import { PBPEEngine } from "../path/to/pbpeenginewrapper"; // Python側をHTTPで叩く or Node版に移植

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const inputs = {
    years: 10,
    discount_rate: 0.08,
    initial_investment: 500000,
    ha: 100,
    yield_base: 1.5,
    yieldincreasepct: 0.2,
    fertilizercostsavings: 200,
    priceperton: 3000,
    co2reductionper_ton: 1.0,
    carbon_price: 80,
    healthcarecostavoidance: 50,
    healthdiscountrate: 0.03,
    scope3coffeetons: 400000
  };

  const engine = new PBPEEngine(inputs);
  const [npv, irr] = engine.computenpvirr();
  const scenario = engine.scenario_10y();
  const investor = engine.investor_metrics();
  const multiplierlayers = engine.capitalmultiplier(1000000);

  res.status(200).json({
    npv,
    irr,
    scenario,
    investor,
    multiplier_layers
  });
}
`

フロントエンド（KPI表示の例）

投資家向けに見せるべきポイント

- BAU 10年 NPV：低い／不安定  
- PBPE 10年 NPV：大幅に高い（PBPE Value＋コスト削減＋GHG価値）  
- BAU IRR vs PBPE IRR：PBPE側が圧倒的に高い  
- Scope3 Value：PBPE導入により、企業の Scope3 削減価値が数億〜数十億 USD 規模になる
