### 🧩 1. Fetcher Layer（投稿収集）

✔ LinkedIn（HTMLスクレイピング）


```python
fetcher/linkedin.py
import requests
from bs4 import BeautifulSoup
import json

def fetchlinkedinposts(url):
    html = requests.get(url).text
    soup = BeautifulSoup(html, "html.parser")

    posts = []
    for div in soup.findall("div", class="feed-shared-update-v2"):
        text = div.get_text(strip=True)
        posts.append({
            "source": "LinkedIn",
            "text": text,
            "url": url
        })

    return posts
`
```

```python
import feedparser

def fetch_gatesnotes():
    feed = feedparser.parse("https://www.gatesnotes.com/rss")
    posts = []

    for entry in feed.entries:
        posts.append({
            "source": "GatesNotes",
            "title": entry.title,
            "text": entry.summary,
            "url": entry.link
        })

    return posts
```

### ✔ Threads（Puppeteer経由）

```python
fetcher/threads.py
from pyppeteer import launch

async def fetch_threads(username):
    browser = await launch()
    page = await browser.newPage()
    await page.goto(f"https://www.threads.net/@{username}")

    posts = await page.evaluate("""
        () => Array.from(document.querySelectorAll('div[role="article"]'))
            .map(div => div.innerText)
    """)

    await browser.close()

    return [{"source": "Threads", "text": p} for p in posts]
`
```

### 🧩 2. Normalizer Layer（正規化）

```python
normalizer/normalize.py
import uuid
from datetime import datetime

def normalize(posts):
    normalized = []

    for p in posts:
        normalized.append({
            "id": str(uuid.uuid4()),
            "source": p.get("source"),
            "text": p.get("text"),
            "url": p.get("url", None),
            "timestamp": datetime.utcnow().isoformat()
        })

    return normalized
`
```

### 🧩 3. Router Layer（BioNexus 関連度スコアリング）

✔ Pinecone + SentenceTransformer による embedding

```python
router/scorer.py
from sentence_transformers import SentenceTransformer
import numpy as np

model = SentenceTransformer("all-MiniLM-L6-v2")

agrix-knowledge-base の embedding をロード
import json
with open("config/agrix_embeddings.json") as f:
    AGRIX_EMB = np.array(json.load(f))

def semantic_similarity(text):
    emb = model.encode(text)
    sims = np.dot(AGRIXEMB, emb) / (np.linalg.norm(AGRIXEMB, axis=1) * np.linalg.norm(emb))
    return float(np.max(sims))

def domain_match(text):
    keywords = ["GHG", "soil", "carbon", "microbiome", "PBPE", "MBT55", "AGRIX"]
    score = sum(k.lower() in text.lower() for k in keywords) / len(keywords)
    return score

def impact_potential(text):
    impact_terms = ["climate", "agriculture", "health", "AI", "food security"]
    score = sum(t.lower() in text.lower() for t in impactterms) / len(impactterms)
    return score

def score_post(post):
    s = semantic_similarity(post["text"])
    d = domain_match(post["text"])
    i = impact_potential(post["text"])

    total = (0.5s + 0.3d + 0.2i)  100

    rank = "A" if total >= 80 else "B" if total >= 60 else "C" if total >= 40 else "D"

    return {
        post,
        "semantic": s,
        "domain": d,
        "impact": i,
        "score": total,
        "rank": rank
    }
`
```

```python
publisher/github_sync.py
import subprocess
import json
import os

def savetogithub(data, filename):
    path = f"../agrix-knowledge-base/{filename}"
    with open(path, "w") as f:
        json.dump(data, f, indent=2)

    subprocess.run(["git", "add", path])
    subprocess.run(["git", "commit", "-m", f"Add {filename}"])
    subprocess.run(["git", "push"])
```

### 🧩 6. main.py（全パイプラインを連結）

```python
main.py
import asyncio
from fetcher.linkedin import fetchlinkedinposts
from fetcher.gatesnotes import fetch_gatesnotes
from fetcher.threads import fetch_threads
from normalizer.normalize import normalize
from router.scorer import score_post
from responder.composer import compose_response
from publisher.githubsync import saveto_github

async def run_pipeline():
    posts = []

    posts += fetchlinkedinposts("https://www.linkedin.com/in/billgates/")
    posts += fetch_gatesnotes()
    posts += await fetch_threads("thisisbillgates")

    normalized = normalize(posts)
    scored = [score_post(p) for p in normalized]

    responses = []
    for p in scored:
        if p["rank"] in ["A", "B"]:
            responses.append({
                "post": p,
                "response": compose_response(p)
            })

    savetogithub(responses, "daily_responses.json")

asyncio.run(run_pipeline())
`
```
