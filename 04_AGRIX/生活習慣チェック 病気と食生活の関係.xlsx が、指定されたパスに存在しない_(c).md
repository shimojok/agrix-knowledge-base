エラーの原因は、元データであるExcelファイル **`生活習慣チェック 病気と食生活の関係.xlsx`** が、指定されたパス `C:\Users\user\OneDrive\agrix-knowledge-base\` の直下に存在しないためです。

ファイルが上位ディレクトリ `C:\Users\user\OneDrive\agrix-knowledge-base\` ではなく、作業中の `C:\Users\user\bionexus-engine\` フォルダ側、あるいはOneDrive内の別フォルダ（`agrix-knowledge-base` 内のサブフォルダなど）に置かれている可能性が高いです。

### 解決手順

ファイルの位置を自動検索して生成する修正版のスクリプトを実行します。

以下のコードをコピーして、ターミナルにそのまま貼り付けてEnterキーを押してください。

PowerShell

```
@'
import json
import os
import pandas as pd
from dotenv import load_dotenv

load_dotenv()

# Excelファイルの場所候補リスト
possible_paths = [
    r"C:\Users\user\bionexus-engine\生活習慣チェック 病気と食生活の関係.xlsx",
    r"C:\Users\user\OneDrive\agrix-knowledge-base\生活習慣チェック 病気と食生活の関係.xlsx",
    r"生活習慣チェック 病気と食生活の関係.xlsx"
]

excel_file = None
for p in possible_paths:
    if os.path.exists(p):
        excel_file = p
        break

if not excel_file:
    # 親ディレクトリも含めて探索
    search_dirs = [r"C:\Users\user\bionexus-engine", r"C:\Users\user\OneDrive\agrix-knowledge-base"]
    for d in search_dirs:
        if os.path.exists(d):
            for root, dirs, files in os.walk(d):
                for f in files:
                    if "生活習慣チェック" in f and f.endswith(".xlsx"):
                        excel_file = os.path.join(root, f)
                        break
                if excel_file:
                    break
        if excel_file:
            break

if not excel_file:
    raise FileNotFoundError("『生活習慣チェック 病気と食生活の関係.xlsx』が見つかりませんでした。ファイルの配置場所を確認してください。")

print(f"読み込み対象ファイル: {excel_file}")

output_dir = os.getenv("DATA_DIR", r"C:\Users\user\OneDrive\agrix-knowledge-base\data")
output_path = os.path.join(output_dir, "hamada_137_risk_matrix.json")

df_raw = pd.read_excel(excel_file, sheet_name='Sheet1')

headers = df_raw.iloc[0].values
headers[0] = "disease_name"

df_data = df_raw.iloc[1:].copy()
df_data.columns = headers
df_data = df_data.dropna(subset=['disease_name']).reset_index(drop=True)

risk_columns = [col for col in headers if pd.notna(col) and col not in ['disease_name', '合計']]

matrix_list = []
for idx, row in df_data.iterrows():
    disease_name = str(row['disease_name']).strip()
    active_risks = []
    for col in risk_columns:
        val = row[col]
        if pd.notna(val) and str(val).strip() in ['●', '○', '1', 'true', 'True']:
            active_risks.append(col)
            
    matrix_list.append({
        "disease_id": f"D{idx+1:03d}",
        "disease_name_ja": disease_name,
        "risk_factors": active_risks,
        "risk_factor_count": len(active_risks)
    })

full_json = {
    "title": "hamada_137_risk_matrix",
    "version": "1.0.0",
    "description": "浜田プロトコル：137疾患と38項目の食生活・生活習慣リスク因子対応マトリックス（完全反映版）",
    "total_diseases": len(matrix_list),
    "total_risk_factor_categories": len(risk_columns),
    "risk_factor_definitions": risk_columns,
    "diseases": matrix_list
}

os.makedirs(output_dir, exist_ok=True)
with open(output_path, "w", encoding="utf-8") as f:
    json.dump(full_json, f, ensure_ascii=False, indent=2)

print("--------------------------------------------------")
print(f"成功: 全{len(matrix_list)}疾患の完全データを格納しました。")
print(f"保存先: {output_path}")
print("--------------------------------------------------")
'@ | Out-File -Encoding utf8 generate_full_matrix.py

python generate_full_matrix.py
```

この修正により、ExcelファイルがPC内のどの場所（`bionexus-engine`内または`agrix-knowledge-base`内のサブフォルダ）にあっても自動探索して読み込み、`hamada_137_risk_matrix.json` へ完全な137疾患データを書き出します。
