from typing import Dict, Any

from bionexus_sim_engine import BioNexusSimEngine, UserBiomarkerInput

class ExecutiveQueryHandler:
    def __init__(self, matrix_json_path: str = "data/hamada_137_risk_matrix.json"):
        self.repo = HamadaMatrixRepository(matrix_json_path)
        self.matrix_data = self.repo.load_matrix()
        self.engine = BioNexusSimEngine(self.matrix_data)

    def ask(self, query_text: str, target_diseases: list, phyto_intake: Dict[str, float]) -> Dict[str, Any]:
        sim_input = UserBiomarkerInput(
            user_id="exec-user-001",
            target_disease_ids=target_diseases,
            phytochemical_intake=phyto_intake
        )
        results = self.engine.run_simulation(sim_input)
        
        summary = {
            "query": query_text,
            "evaluated_diseases_count": len(results),
            "top_recommendations": [],
            "detailed_results": [r.model_dump() for r in results]
        }
        
        for res in results:
            if res.mitigation_percentage > 0:
                summary["top_recommendations"].append(
                    f"疾患 [{res.disease_name}] に対し、リスクが {res.mitigation_percentage}% 低減されました。"
                )

        return summary

if __name__ == "__main__":

import streamlit as st
from src.repositories.hamada_matrix_repository import HamadaMatrixRepository

def load_repository_data():
    repo = HamadaMatrixRepository("data/hamada_137_risk_matrix.json")
    return repo.load_matrix()

try:
    matrix_models = load_repository_data()
    engine = BioNexusSimEngine(matrix_models)

    sim_input = UserBiomarkerInput(
        user_id="dashboard-user",
        target_disease_ids=["DIS-137-001"],

    )

    results = engine.run_simulation(sim_input)

    col1, col2, col3 = st.columns(3)
    if results:
        res = results[0]
        col1.metric("対象疾患", res.disease_name)
        col2.metric("ベースラインリスク", res.baseline_risk_score)
        col3.metric("調整後リスク", res.adjusted_risk_score, delta=f"-{res.mitigation_percentage}%")

    st.subheader("詳細シミュレーション結果")

### 3. `src/repositories/rag_knowledge_repository.py` （統合RAGリポジトリ・パス修正版）

```Python
import json
from typing import List, Dict, Any
from src.models.hamada_matrix import HamadaDiseaseRiskModel
from src.analyzer.repo_dependency_analyzer import RepoDependencyAnalyzer

class LocalRAGKnowledgeRepository:
    def __init__(
        self, 
        matrix_path: str = "data/hamada_137_risk_matrix.json", 
        repo_skeleton_path: str = "data/repo_skeleton.json"
    ):
        self.matrix_path = matrix_path
        self.repo_skeleton_path = repo_skeleton_path

    def search_disease_knowledge(self, disease_id: str) -> Dict[str, Any]:
        with open(self.matrix_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        
        for item in data:
            if item.get("disease_id") == disease_id:
                return item
        return {}

    def analyze_repo_dependencies(self) -> Dict[str, List[str]]:
        analyzer = RepoDependencyAnalyzer(self.repo_skeleton_path)
        return analyzer.build_dependency_graph()
```
