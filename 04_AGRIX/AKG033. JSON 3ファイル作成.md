**2. Pydanticモデルとバリデーション（models/phytochemical.json）**

```python
from pydantic import BaseModel, Field, field_validator
from typing import List

class PhytochemicalDietModel(BaseModel):
    id: str = Field(..., description="一意の識別ID")
    name: str = Field(..., min_length=1, description="ファイトケミカル名称")
    category: str = Field(..., min_length=1, description="成分カテゴリ")
    target_disease: str = Field(..., min_length=1, description="対象・関連疾患")
    food_sources: List[str] = Field(..., min_items=1, description="主な含まれる食材")
    dietary_advice: str = Field(..., min_length=10, description="推奨される食生活アドバイス")

    @field_validator("id")
    @classmethod
    def validate_id_format(cls, v: str) -> str:
        if not v.startswith("phyto-"):
            raise ValueError("IDは 'phyto-' で始まる必要があります。")
        return v

```

    def load_all_from_file(self) -> List[PhytochemicalDietModel]:
        """ファイルから全件読み込み・Validation適用"""
        with open(self.file_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return [PhytochemicalDietModel(**item) for item in data]
