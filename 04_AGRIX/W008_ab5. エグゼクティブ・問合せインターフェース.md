### 1. `ask_executive.py`（エグゼクティブ・問合せインターフェース）

自然言語問い合わせを解析し、`BioNexusSimEngine` と `HamadaMatrixRepository` を呼び出してエグゼクティブ向けサマリーを返すCLI/Pythonモジュールです。

```python
import sys
from typing import Dict, Any
from models.hamada_matrix import HamadaDiseaseRiskModel
from repositories.hamada_matrix_repository import HamadaMatrixRepository
from bionexus_sim_engine import BioNexusSimEngine, UserBiomarkerInput

class ExecutiveQueryHandler:
    def __init__(self, matrix_json_path: str):
        self.repo = HamadaMatrixRepository(matrix_json_path)
        self.matrix_data = self.repo.load_matrix()
        self.engine = BioNexusSimEngine(self.matrix_data)

    def ask(self, query_text: str, target_diseases: list, phyto_intake: Dict[str, float]) -> Dict[str, Any]:
        """エグゼクティブからの問い合わせに対する統合診断プロセッサ"""
        # 入力パラメータの標準化
        sim_input = UserBiomarkerInput(
            user_id="exec-user-001",
            target_disease_ids=target_diseases,
            phytochemical_intake=phyto_intake
        )
        
        # シミュレーション実行
        results = self.engine.run_simulation(sim_input)
        
        # エグゼクティブ向けサマリー構築
        summary = {
            "query": query_text,
            "evaluated_diseases_count": len(results),
            "top_recommendations": [],
            "detailed_results": [r.model_dump() for r in results]
        }
        
        for res in results:
            if res.mitigation_percentage > 0:
                summary["top_recommendations"].append(
                    f"疾患 [{res.disease_name}] に対し、リスクが {res.mitigation_percentage}% 低減されました。"
                )

        return summary

if __name__ == "__main__":
    # ローカル検証用実行ブロック
    handler = ExecutiveQueryHandler("data/hamada_137_risk_matrix.json")
    demo_res = handler.ask(
        query_text="高血圧および代謝性疾患に対するファイトケミカルの介入効果",
        target_diseases=["DIS-137-001"],
        phyto_intake={"リコピン": 20.0, "ケルセチン": 15.0}
    )
    print("=== Executive Response ===")
    print(demo_res)

```

### 2. Streamlit フロントエンドダッシュボード (`app.py`)

ローカルで `streamlit run app.py` を実行することで、シミュレーション結果とリスクマトリックスをリアルタイムに可視化するUIです。

```python
import streamlit as st
import json
from repositories.hamada_matrix_repository import HamadaMatrixRepository
from bionexus_sim_engine import BioNexusSimEngine, UserBiomarkerInput

st.set_page_config(page_title="BioNexus Executive Dashboard", layout="wide")

st.title("🌱 BioNexus Executive Engine Dashboard")
st.caption("agrix-knowledge-base & 浜田137疾病リスクマトリックス 統合可視化UI")

# サイドバー：ファイトケミカル調整
st.sidebar.header("📊 介入パラメータ設定")
lycopene = st.sidebar.slider("リコピン摂取量 (mg)", 0.0, 50.0, 15.0)
quercetin = st.sidebar.slider("ケルセチン摂取量 (mg)", 0.0, 50.0, 10.0)

# メイン領域：データロード（デモ用ダミーパス設定可能）
@st.cache_data
def load_data():
    # 実際運用時は data/hamada_137_risk_matrix.json のパスを指定
    return [
        {
            "disease_id": "DIS-137-001",
            "disease_name": "高血圧症",
            "category": "循環器疾患",
            "target_phytochemicals": ["リコピン", "ケルセチン"],
            "risk_correlation_score": 0.85,
            "metabolic_pathways": [{"pathway_name": "PDH", "effect_type": "upregulate", "description": "ピルビン酸脱水素酵素活性化"}],
            "dietary_interventions": ["トマト調理品", "タマネギサラダ"]
        }
    ]

raw_matrix = load_data()
from models.hamada_matrix import HamadaDiseaseRiskModel
matrix_models = [HamadaDiseaseRiskModel(**item) for item in raw_matrix]

engine = BioNexusSimEngine(matrix_models)

# シミュレーション実行
sim_input = UserBiomarkerInput(
    user_id="dashboard-user",
    target_disease_ids=["DIS-137-001"],
    phytochemical_intake={"リコピン": lycopene, "ケルセチン": quercetin}
)

results = engine.run_simulation(sim_input)

# メトリクス表示
col1, col2, col3 = st.columns(3)
if results:
    res = results[0]
    col1.metric("対象疾患", res.disease_name)
    col2.metric("ベースラインリスク", res.baseline_risk_score)
    col3.metric("調整後リスク", res.adjusted_risk_score, delta=f"-{res.mitigation_percentage}%")

st.divider()
st.subheader("詳細シミュレーション結果")
st.json([r.model_dump() for r in results])

```
