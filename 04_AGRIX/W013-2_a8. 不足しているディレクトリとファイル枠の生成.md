from pydantic import BaseModel, Field, field_validator
from typing import List, Literal

class MetabolicPathwayImpact(BaseModel):
    pathway_name: str = Field(..., description="代謝経路名 (例: PDH, SCFA, Mitochondrial ATP)")
    effect_type: Literal["upregulate", "downregulate", "modulate"] = Field(..., description="作用タイプ")
    description: str = Field(..., description="メカニズムの概要")

class HamadaDiseaseRiskModel(BaseModel):
    disease_id: str = Field(..., description="疾患ID (例: DIS-137-001)")
    disease_name: str = Field(..., description="疾患・状態名称")
    category: str = Field(..., description="疾患分類")
    target_phytochemicals: List[str] = Field(..., min_items=1, description="有効なファイトケミカル一覧")
    risk_correlation_score: float = Field(..., ge=-1.0, le=1.0, description="リスク相関スコア (-1.0〜+1.0)")
    metabolic_pathways: List[MetabolicPathwayImpact] = Field(default_factory=list, description="関連する代謝経路")
    dietary_interventions: List[str] = Field(..., description="推奨食生活介入策")

    @field_validator("disease_id")
    @classmethod
    def validate_disease_id(cls, v: str) -> str:
        if not v.startswith("DIS-"):
            raise ValueError("disease_id は 'DIS-' で始まる必要があります。")
        return v

import json
from typing import List, Optional
from src.models.hamada_matrix import HamadaDiseaseRiskModel

class HamadaMatrixRepository:
    def __init__(self, json_path: str = "../agrix-knowledge-base/data/hamada_137_risk_matrix.json"):
        self.json_path = json_path
        self._cache: Optional[List[HamadaDiseaseRiskModel]] = None

    def load_matrix(self) -> List[HamadaDiseaseRiskModel]:
        """Vault内のJSONデータから浜田137リスクマトリックスをロード"""
        if self._cache is not None:
            return self._cache

        with open(self.json_path, "r", encoding="utf-8") as f:
            raw_data = json.load(f)

        self._cache = [HamadaDiseaseRiskModel(**item) for item in raw_data]
        return self._cache

    def get_by_disease_id(self, disease_id: str) -> Optional[HamadaDiseaseRiskModel]:
        matrix = self.load_matrix()
        for record in matrix:
            if record.disease_id == disease_id:
                return record
        return None

### 3. シミュレーション演算コア

**配置先:** `C:\Users\user\bionexus-engine\bionexus_sim_engine.py`

```Python
from typing import List, Dict
from pydantic import BaseModel, Field
from src.models.hamada_matrix import HamadaDiseaseRiskModel

class UserBiomarkerInput(BaseModel):
    user_id: str
    target_disease_ids: List[str] = Field(..., description="評価対象の疾患IDリスト")
    phytochemical_intake: Dict[str, float] = Field(..., description="ファイトケミカル摂取量 (mg/日)")

class SimulationResult(BaseModel):
    disease_id: str
    disease_name: str
    baseline_risk_score: float
    adjusted_risk_score: float
    mitigation_percentage: float
    impacted_pathways: List[str]

class BioNexusSimEngine:
    def __init__(self, hamada_matrix: List[HamadaDiseaseRiskModel]):
        self.matrix_map = {item.disease_id: item for item in hamada_matrix}

    def run_simulation(self, user_input: UserBiomarkerInput) -> List[SimulationResult]:
        results = []
        for disease_id in user_input.target_disease_ids:
            disease_data = self.matrix_map.get(disease_id)
            if not disease_data:
                continue

            base_score = disease_data.risk_correlation_score
            mitigation_factor = 0.0

            for phyto_name, amount in user_input.phytochemical_intake.items():
                if phyto_name in disease_data.target_phytochemicals:
                    mitigation_factor += min(amount / 100.0, 0.3)

            adjusted_score = max(base_score * (1.0 - mitigation_factor), 0.05)
            mitigation_pct = round((1.0 - (adjusted_score / base_score)) * 100, 2) if base_score > 0 else 0.0
            pathways = [p.pathway_name for p in disease_data.metabolic_pathways]

            results.append(SimulationResult(
                disease_id=disease_id,
                disease_name=disease_data.disease_name,
                baseline_risk_score=base_score,
                adjusted_risk_score=round(adjusted_score, 3),
                mitigation_percentage=mitigation_pct,
                impacted_pathways=pathways
            ))
        return results
```
