```Python
"""
BioNexus Core Simulation Engine (Phase 2)
Integrates AGRIX (Soil/Carbon), HealthBook (Microbiome/TCM/ATP), and PBPE (Economic Tokenomics)
"""

import math

class AGRIXEngine:
    """AGRIX: Soil Carbon Sequestration & GHG Reduction Model"""
    def __init__(self, area_ha=1.0, mbt55_input_ton=1.0):
        self.area_ha = area_ha
        self.mbt55_input_ton = mbt55_input_ton

    def calculate(self):
        # Base metrics based on Nairobi empirical data & MBT55 models
        soc_sequestration_rate = 5.8  # tCO2e/ha/year
        ch4_reduction_rate = 0.82     # 82%
        n2o_reduction_rate = 0.72     # 72%
        
        total_co2_sequestered = self.area_ha * soc_sequestration_rate * (1 + 0.1 * math.log1p(self.mbt55_input_ton))
        net_co2_balance = -total_co2_sequestered  # Negative indicates net sink
        
        return {
            "area_ha": self.area_ha,
            "mbt55_input_ton": self.mbt55_input_ton,
            "net_co2_balance_tco2e": round(net_co2_balance, 2),
            "ch4_reduction_percent": ch4_reduction_rate * 100,
            "n2o_reduction_percent": n2o_reduction_rate * 100,
            "soil_organic_carbon_gain_t": round(total_co2_sequestered * 0.27, 2)
        }

class HealthBookEngine:
    """HealthBook: Gut Microbiome, TCM Biotransformation, Endocrine & Cellular ATP Model"""
    def __init__(self, mbt_probiotics_dose_g=10.0, tcm_herb_input_g=5.0):
        self.mbt_probiotics_dose_g = mbt_probiotics_dose_g
        self.tcm_herb_input_g = tcm_herb_input_g

    def calculate(self):
        # Biotransformation kinetics of TCM macro-molecules via MBT55 multi-species microbial hypercycle
        conversion_efficiency = math.tanh(self.mbt_probiotics_dose_g / 5.0)  # Synergistic saturation
        
        # Short-Chain Fatty Acids (SCFA) Flux (mM/day)
        scfa_butyrate_mM = round((12.5 + self.tcm_herb_input_g * 2.1) * conversion_efficiency, 2)
        scfa_acetate_mM = round((25.0 + self.tcm_herb_input_g * 3.5) * conversion_efficiency, 2)
        
        # Endocrine response (GLP-1 activation index)
        glp1_signal_score = round(min(100.0, (scfa_butyrate_mM * 2.4 + scfa_acetate_mM * 0.8)), 1)
        
        # Cellular Energy / Mitochondrial ATP Efficiency (% increase over baseline)
        atp_elevation_pct = round(min(45.0, scfa_butyrate_mM * 1.8), 1)
        
        # Gut Redox Potential (ORP / Eh in mV)
        gut_orp_mV = round(-250.0 - (conversion_efficiency * 50.0), 1) # Anaerobic reductive environment

        return {
            "mbt_probiotics_dose_g": self.mbt_probiotics_dose_g,
            "tcm_herb_input_g": self.tcm_herb_input_g,
            "biotransformation_efficiency": round(conversion_efficiency * 100, 1),
            "scfa_butyrate_mM": scfa_butyrate_mM,
            "scfa_acetate_mM": scfa_acetate_mM,
            "glp1_endocrine_score": glp1_signal_score,
            "mitochondrial_atp_gain_percent": atp_elevation_pct,
            "gut_redox_orp_mV": gut_orp_mV
        }

class PBPEEngine:
    """PBPE: Physical-Biophysical Marketplace & Circular Tokenomics Model"""
    def __init__(self, agrix_results, healthbook_results, carbon_price_usd=35.0):
        self.agrix_results = agrix_results
        self.healthbook_results = healthbook_results
        self.carbon_price_usd = carbon_price_usd

    def calculate(self):
        co2_credits = abs(self.agrix_results["net_co2_balance_tco2e"])
        carbon_credit_value_usd = co2_credits * self.carbon_price_usd
        
        # Health value tokenization (based on medical cost reduction & ATP productivity)
        health_index = (self.healthbook_results["glp1_endocrine_score"] + self.healthbook_results["mitochondrial_atp_gain_percent"]) / 2.0
        health_token_minted = round(health_index * 12.5, 2)
        
        total_ecosystem_value_usd = round(carbon_credit_value_usd + (health_token_minted * 2.0), 2)
        roi_estimate_pct = round((total_ecosystem_value_usd / max(1.0, (self.agrix_results["mbt55_input_ton"] * 150 + self.healthbook_results["mbt_probiotics_dose_g"] * 5))) * 100, 1)

        return {
            "carbon_credits_tco2e": co2_credits,
            "carbon_value_usd": round(carbon_credit_value_usd, 2),
            "health_tokens_minted": health_token_minted,
            "total_ecosystem_value_usd": total_ecosystem_value_usd,
            "estimated_roi_percent": roi_estimate_pct
        }

def run_bionexus_simulation(area_ha=10.0, mbt55_ton=5.0, probiotics_g=15.0, tcm_g=10.0):
    agrix = AGRIXEngine(area_ha=area_ha, mbt55_input_ton=mbt55_ton).calculate()
    hb = HealthBookEngine(mbt_probiotics_dose_g=probiotics_g, tcm_herb_input_g=tcm_g).calculate()
    pbpe = PBPEEngine(agrix, hb).calculate()
    
    return {
        "AGRIX_Soil_Carbon": agrix,
        "HealthBook_Microbiome_ATP": hb,
        "PBPE_Ecosystem_Economy": pbpe
    }

if __name__ == "__main__":
    import json
    res = run_bionexus_simulation()
    print(json.dumps(res, indent=2, ensure_ascii=False))
```

- **概要**: 上記シミュレーション計算結果と ChromaDB 内のベクトルインデックスを結合し、意思決定者向けに統合回答を生成するインタラクティブ CLI スクリプトです[cite: 9]。
    
```Python
"""
BioNexus Executive Engine (ask_executive.py)
Unified RAG + Simulation + Executive Briefing Interface
"""

import os
import sys
import json
from dotenv import load_dotenv

from langchain_huggingface import HuggingFaceEmbeddings
from langchain_chroma import Chroma
from langchain_openai import ChatOpenAI
from langchain_ollama import OllamaLLM
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

# 自作シミュレーションモジュールのインポート設定
BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.append(BASE_DIR)
sys.path.append(os.path.join(BASE_DIR, "scripts"))

from bionexus_sim_engine import run_bionexus_simulation

load_dotenv()

DB_DIR = os.path.join(BASE_DIR, "chroma_db")
VAULT_DIR = os.getenv("OBSIDIAN_VAULT_PATH", r"C:\Users\user\agrix-knowledge-base")
EMBEDDING_MODEL = "intfloat/multilingual-e5-small"
OPENAI_MODEL = "gpt-4o-mini"
OLLAMA_MODEL = "llama3"

def get_llm():
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key and openai_api_key != "your_openai_api_key_here":
        try:
            llm = ChatOpenAI(model=OPENAI_MODEL, temperature=0.2)
            llm.invoke("Test")
            return llm
        except Exception:
            pass
    try:
        llm = OllamaLLM(model=OLLAMA_MODEL, temperature=0.2)
        llm.invoke("Test")
        return llm
    except Exception as e:
        print(f"【エラー】LLM 接続不可: {e}")
        sys.exit(1)

def format_docs(docs):
    formatted = []
    for i, doc in enumerate(docs, 1):
        source = doc.metadata.get("rel_path", doc.metadata.get("source", "不明"))
        folder = doc.metadata.get("folder", "不明")
        formatted.append(f"--- [参照資料 {i} ({folder})]: {source} ---\n{doc.page_content}")
    return "\n\n".join(formatted)

def main():
    print("============================================================")
    print("      BioNexus Executive Engine (RAG + Simulation)          ")
    print("============================================================")

    if not os.path.exists(DB_DIR):
        print(f"【エラー】ChromaDB ({DB_DIR}) が存在しません。先に update_rag_index.py を実行してください。")
        return

    embeddings = HuggingFaceEmbeddings(model_name=EMBEDDING_MODEL)
    db = Chroma(persist_directory=DB_DIR, embedding_function=embeddings)
    llm = get_llm()

    template = """You are BioNexus Executive Engine. You must ALWAYS reply in Japanese.

以下の【リアルタイムシミュレーション結果】と【ナレッジベース参照資料】を統合し、
最高意思決定者（例: ビル・ゲイツ氏ら）向けに、定量的かつ論理的なエグゼクティブ・ブリーフィングを作成してください。

【制約事項】
1. 回答はすべて**日本語**で記述すること。
2. AGRIX (炭素/土壌)、HealthBook (漢方/MBT Probiotics/腸内代謝産物/ATP/内分泌)、PBPE (経済トークン) の相互作用を明確に示すこと。

【リアルタイムシミュレーション計算結果】
{sim_results}

【ナレッジベース参照資料】
{context}

【ユーザーの質問】
{question}

【エグゼクティブ・ブリーフィング回答（日本語）】:
"""
    prompt = ChatPromptTemplate.from_template(template)

    # デフォルト・シミュレーションパラメータ (100haモデル)
    sim_data = run_bionexus_simulation(area_ha=100.0, mbt55_ton=50.0, probiotics_g=20.0, tcm_g=10.0)
    sim_json_str = json.dumps(sim_data, indent=2, ensure_ascii=False)

    print("\n[現在ロード中の標準シミュレーション条件 (100haモデル)]")
    print(f" - Net CO2 Balance: {sim_data['AGRIX_Soil_Carbon']['net_co2_balance_tco2e']} tCO2e")
    print(f" - SCFA (酪酸) 産出量: {sim_data['HealthBook_Microbiome_ATP']['scfa_butyrate_mM']} mM/day")
    print(f" - ミトコンドリア ATP 向上率: +{sim_data['HealthBook_Microbiome_ATP']['mitochondrial_atp_gain_percent']}%")
    print(f" - 総エコシステム価値: ${sim_data['PBPE_Ecosystem_Economy']['total_ecosystem_value_usd']}")

    print("\n質問を入力してください（終了: 'q'）:\n")

    while True:
        question = input("Executive Query > ").strip()
        if not question:
            continue
        if question.lower() == 'q':
            print("セッションを終了します。")
            break

        print("\n[BioNexus Executive Engine: 統合推論＆RAGコンテキスト抽出中...]")
        try:
            formatted_query = f"query: {question}"
            docs = db.similarity_search(formatted_query, k=4)
            context_str = format_docs(docs)

            chain = prompt | llm | StrOutputParser()
            response = chain.invoke({
                "sim_results": sim_json_str,
                "context": context_str,
                "question": question
            })

            print("\n" + "=" * 60)
            print("【BioNexus Executive Briefing】")
            print(response)
            print("=" * 60 + "\n")
        except Exception as e:
            print(f"【生成エラー】: {e}\n")

if __name__ == "__main__":
    main()
```
