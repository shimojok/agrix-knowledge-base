# 🟦 Step 3 — 全 UI コンポーネントを PBPECard に置き換え

例：SnapshotPanel.tsx

tsx

```
"use client";

import PBPECard from "../../components/PBPECard";
import { useEffect, useState } from "react";
import { Snapshot } from "../../../lib/types";
import { fetchSnapshot } from "../../../lib/api/dashboard";

export default function SnapshotPanel() {
  const [snapshot, setSnapshot] = useState<Snapshot | null>(null);

  useEffect(() => {
    fetchSnapshot("user001").then(setSnapshot);
  }, []);

  return (
    <PBPECard title="Snapshot">
      {!snapshot && <p>Loading...</p>}
      {snapshot && (
        <pre className="text-sm text-gray-700">
          {JSON.stringify(snapshot, null, 2)}
        </pre>
      )}
    </PBPECard>
  );
}
```

# 🟦 Step 5 — PBPE KPI ダッシュボードとの整合性

PBPE KPI ダッシュボードは「数値の強調」が重要。

### KPIPanel.tsx（デザイン統合版）

tsx

```
export default function KPIPanel() {
  const [kpis, setKpis] = useState<PBPEKPI | null>(null);

  useEffect(() => {
    fetch("http://localhost:8000/pbpe/kpis/user001")
      .then((res) => res.json())
      .then(setKpis);
  }, []);

  return (
    <PBPECard title="PBPE KPIs">
      {!kpis && <p>Loading...</p>}

      {kpis && (
        <div className="grid grid-cols-2 gap-4">
          <div className="text-center">
            <div className="text-3xl font-bold text-[var(--pbpe-blue)]">
              {kpis.phenotype_score}
            </div>
            <p className="text-gray-600">Phenotype</p>
          </div>

          <div className="text-center">
            <div className="text-3xl font-bold text-[var(--pbpe-blue)]">
              {kpis.estrobolome_score}
            </div>
            <p className="text-gray-600">Estrobolome</p>
          </div>

          <div className="text-center">
            <div className="text-3xl font-bold text-[var(--pbpe-blue)]">
              {kpis.atp}
            </div>
            <p className="text-gray-600">ATP</p>
          </div>

          <div className="text-center">
            <div className="text-3xl font-bold text-[var(--pbpe-blue)]">
              {kpis.inflammation}
            </div>
            <p className="text-gray-600">Inflammation</p>
          </div>
        </div>
      )}
    </PBPECard>
  );
}
```
