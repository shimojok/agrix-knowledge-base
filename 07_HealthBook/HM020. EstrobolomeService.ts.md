import { Hb003, EstrobolomeMIResult } from "../models/Hb003";

export class EstrobolomeService {
  evaluate(hb003: Hb003): EstrobolomeMIResult {
    const { lifestyle, microbiome, hormones } = hb003;

    const fiberScore =
      lifestyle.fiberIntakeLevel === "high"
        ? 80
        : lifestyle.fiberIntakeLevel === "medium"
        ? 50
        : 20;

    const fermentedScore =
      lifestyle.fermentedFoodFrequency === "daily"
        ? 80
        : lifestyle.fermentedFoodFrequency === "sometimes"
        ? 50
        : 20;

    const processedPenalty = lifestyle.processedFoodScore * 0.5;

    const scfaScore = microbiome.scfaProductionScore;
    const estrogenBase =
      hormones.estrogenStatus === "low"
        ? 40
        : hormones.estrogenStatus === "high"
        ? 60
        : 50;

    const estrobolomeActivity =
      (fiberScore + fermentedScore + scfaScore + estrogenBase) / 4 -
      processedPenalty * 0.2;

    const recycling = estrobolomeActivity * 0.85;
    const mitoResponse = estrobolomeActivity * 0.7;
    const painCorr = estrobolomeActivity * 0.4;

    const riskFlags: string[] = [];
    if (processedPenalty > 30) riskFlags.push("high_processed_food");
    if (fiberScore < 40) riskFlags.push("low_fiber");
    if (scfaScore < 40) riskFlags.push("low_scfa");
    if (hormones.estrogenStatus === "low") riskFlags.push("low_estrogen");

    const recommendations = [
      {
        code: "increase_fiber",
        title: "食物繊維を増やす",
        description: "野菜・海藻・豆類を増やし、腸内細菌のSCFA産生を高めます。",
        priority: "high",
      },
      {
        code: "add_fermented_foods",
        title: "発酵食品を毎日少量",
        description: "味噌・納豆・ヨーグルトなどを継続的に摂取します。",
        priority: "medium",
      },
    ];

    return {
      estrobolomeActivityScore: Math.max(0, Math.min(100, Math.round(estrobolomeActivity))),
      estrogenRecyclingEfficiency: Math.max(0, Math.min(100, Math.round(recycling))),
      mitochondrialEstrogenResponseScore: Math.max(0, Math.min(100, Math.round(mitoResponse))),
      painCorrelationScore: Math.max(0, Math.min(100, Math.round(painCorr))),
      riskFlags,
      recommendations,
    };
  }
}

import { Hb003 } from "../models/Hb003";
import { AcetylCoAMIResult } from "../models/AcetylCoAMI";

export class AcetylCoAService {
  evaluate(hb003: Hb003): AcetylCoAMIResult {
    const { lifestyle, microbiome, medicationAndStress } = hb003;

    const glucose = 60; // 後で食事詳細が入ったら分解
    const lipids = lifestyle.dietPattern === "high_processed" ? 40 : 60;
    const amino = 50; // タンパク質入力が増えたら調整
    const scfa = microbiome.scfaProductionScore;

    const coldPenalty =
      medicationAndStress.coldSensitivity === "high"
        ? 25
        : medicationAndStress.coldSensitivity === "medium"
        ? 15
        : 5;

    const drugPenalty =
      medicationAndStress.antihypertensiveUse || medicationAndStress.magnesiumDepletingDrugs
        ? 20
        : 0;

    const baseIndex =
      (glucose + lipids + amino + scfa) / 4 - coldPenalty * 0.3 - drugPenalty * 0.4;

    const mitoATP = baseIndex * 0.8;
    const tcaEff = baseIndex * 0.7;
    const stressIndex =
      (lifestyle.processedFoodScore + (medicationAndStress.chronicStressLevel === "high" ? 40 : 20)) /
      2;

    const flags: string[] = [];
    if (scfa < 40) flags.push("low_scfa");
    if (lifestyle.processedFoodScore > 60) flags.push("high_processed_food");
    if (drugPenalty > 0) flags.push("magnesium_depleting_drugs");

    const recommendations = [
      {
        code: "support_scfa",
        title: "短鎖脂肪酸産生を支える",
        description: "水溶性食物繊維とMBT55系プロバイオティクスを組み合わせます。",
        priority: "high",
      },
    ];

    return {
      acetylCoAAvailabilityIndex: clamp(baseIndex),
      glucoseContributionScore: clamp(glucose),
      lipidContributionScore: clamp(lipids),
      aminoAcidContributionScore: clamp(amino),
      scfaContributionScore: clamp(scfa),
      mitochondrialATPCapacityScore: clamp(mitoATP),
      tcaCycleEfficiencyScore: clamp(tcaEff),
      metabolicStressIndex: clamp(stressIndex),
      flags,
      recommendations,
    };
  }
}

function clamp(v: number): number {
  return Math.max(0, Math.min(100, Math.round(v)));
}

import { Hb003, MBT55MappingResult } from "../models/Hb003";

export class MBT55Service {
  map(hb003: Hb003): MBT55MappingResult {
    const groups = hb003.microbiome.mbt55Composition.map((g) => ({
      groupId: g.groupId,
      name: g.name,
      functions: {
        estrogenMetabolism: g.estrogenMetabolismRole,
        scfaProduction: g.betaGlucuronidaseActivity === "high" ? "high" : "medium",
        antiInflammatory: g.antiInflammatoryPotential,
        cofactorGeneration: "medium",
      },
    }));

    const estroSupport = avgScore(groups.map((g) => scoreEstrogen(g.functions.estrogenMetabolism)));
    const antiInflammatory = avgScore(
      groups.map((g) => scoreLevel(g.functions.antiInflammatory)),
    );
    const metabolicSupport = avgScore(groups.map((g) => scoreLevel(g.functions.scfaProduction)));

    return {
      mbt55Groups: groups,
      overallEstrobolomeSupportScore: clamp(estroSupport),
      overallAntiInflammatoryScore: clamp(antiInflammatory),
      overallMetabolicSupportScore: clamp(metabolicSupport),
    };
  }
}

function scoreEstrogen(role: "none" | "support" | "core"): number {
  if (role === "core") return 80;
  if (role === "support") return 60;
  return 20;
}

function scoreLevel(level: "low" | "medium" | "high"): number {
  if (level === "high") return 80;
  if (level === "medium") return 50;
  return 20;
}

function avgScore(arr: number[]): number {
  if (!arr.length) return 0;
  return arr.reduce((a, b) => a + b, 0) / arr.length;
}

function clamp(v: number): number {
  return Math.max(0, Math.min(100, Math.round(v)));
}

import { Hb003 } from "../models/Hb003";
import { EstrobolomeService } from "./EstrobolomeService";
import { AcetylCoAService } from "./AcetylCoAService";
import { MBT55Service } from "./MBT55Service";
import { MetabolicOSSummary } from "../models/MetabolicOSSummary";

export class MetabolicOSService {
  evaluate(hb003: Hb003): {
    estrobolomeMi: Hb003["estrobolomeMi"];
    acetylCoaMi: Hb003["acetylCoaMi"];
    mbt55Mapping: Hb003["mbt55Mapping"];
    summary: MetabolicOSSummary;
  } {
    const estroService = new EstrobolomeService();
    const acetylService = new AcetylCoAService();
    const mbtService = new MBT55Service();

    const estro = estroService.evaluate(hb003);
    const acetyl = acetylService.evaluate(hb003);
    const mbt = mbtService.map(hb003);

    const metabolicAge =
      (100 - acetyl.mitochondrialATPCapacityScore + acetyl.metabolicStressIndex) / 2;
    const inflammationLevel = acetyl.metabolicStressIndex;
    const painCorr =
      (estro.painCorrelationScore + hb003.painProfile.thumbCmJointPainScore * 10) / 2;

    const keyInsights: string[] = [];
    if (estro.estrogenRecyclingEfficiency < 50)
      keyInsights.push("エストロゲン再循環効率が低下しています。腸内環境とエストロボロームの支援が必要です。");
    if (acetyl.acetylCoAAvailabilityIndex < 50)
      keyInsights.push("アセチルCoA生成能力が低く、ATP生成と修復が制限されています。");
    if (mbt.overallEstrobolomeSupportScore < 50)
      keyInsights.push("MBT55構成のエストロボローム支援が十分ではありません。");

    return {
      estrobolomeMi: estro,
      acetylCoaMi: acetyl,
      mbt55Mapping: mbt,
      summary: {
        metabolicAge: Math.round(metabolicAge),
        inflammationLevel: clamp(inflammationLevel),
        painMetabolicCorrelationScore: clamp(painCorr),
        keyInsights,
      },
    };
  }
}

function clamp(v: number): number {
  return Math.max(0, Math.min(100, Math.round(v)));
}

import { Hb003, EstrobolomeMIResult, AcetylCoAMIResult, MBT55MappingResult, MetabolicOSSummary } from "../types";

const BASE_URL = "/api/v1/mi";

export interface MetabolicOSResponse {
  timestamp: string;
  userId: string;
  estrobolomeMi: EstrobolomeMIResult;
  acetylCoaMi: AcetylCoAMIResult;
  mbt55Mapping: MBT55MappingResult;
  summary: MetabolicOSSummary;
}

export async function evaluateMetabolicOS(hb003: Hb003): Promise<MetabolicOSResponse> {
  const res = await fetch(`${BASE_URL}/metabolic-os/summary`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(hb003),
  });
  if (!res.ok) {
    throw new Error(`MetabolicOS evaluation failed: ${res.status}`);
  }
  return res.json();
}
