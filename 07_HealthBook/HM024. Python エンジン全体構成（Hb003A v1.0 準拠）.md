```python
from .utils import clamp
from ..models.hb003 import Hb003, EstrobolomeMI, Recommendation


def evaluate_estrobolome(hb: Hb003) -> EstrobolomeMI:
  ls = hb.lifestyle
  mb = hb.microbiome
  hm = hb.hormones

  fiber_score = {"low": 20, "medium": 50, "high": 80}[ls.fiber_intake_level]
  fermented_score = {"rare": 20, "sometimes": 50, "daily": 80}[ls.fermented_food_frequency]
  processed_penalty = ls.processed_food_score * 0.5
  scfa_score = mb.scfa_production_score
  estrogen_base = {
    "low": 40,
    "high": 60,
    "normal": 50,
    "unknown": 50,
  }[hm.estrogen_status]

  activity = (fiber_score + fermented_score + scfa_score + estrogen_base) / 4 - processed_penalty * 0.2
  recycling = activity * 0.85
  mito_resp = activity * 0.7
  pain_corr = activity * 0.4

  risk_flags = []
  if processed_penalty > 30:
    risk_flags.append("high_processed_food")
  if fiber_score < 40:
    risk_flags.append("low_fiber")
  if scfa_score < 40:
    risk_flags.append("low_scfa")
  if hm.estrogen_status == "low":
    risk_flags.append("low_estrogen")

  recommendations = [
    Recommendation(
      code="increase_fiber",
      title="食物繊維を増やす",
      description="野菜・海藻・豆類を増やし、腸内細菌のSCFA産生を高めます。",
      priority="high",
    ),
    Recommendation(
      code="add_fermented_foods",
      title="発酵食品を毎日少量",
      description="味噌・納豆・ヨーグルトなどを継続的に摂取します。",
      priority="medium",
    ),
  ]

  return EstrobolomeMI(
    estrobolome_activity_score=clamp(activity),
    estrogen_recycling_efficiency=clamp(recycling),
    mitochondrial_estrogen_response_score=clamp(mito_resp),
    pain_correlation_score=clamp(pain_corr),
    risk_flags=risk_flags,
    recommendations=recommendations,
  )
```

from .utils import clamp
from ..models.hb003 import Hb003, AcetylCoAMI, Recommendation


def evaluate_acetyl_coa(hb: Hb003) -> AcetylCoAMI:
  ls = hb.lifestyle
  mb = hb.microbiome
  ms = hb.medication_and_stress

  glucose = 60
  lipids = 40 if ls.diet_pattern == "high_processed" else 60
  amino = 50
  scfa = mb.scfa_production_score

  cold_penalty = {"low": 5, "medium": 15, "high": 25}[ms.cold_sensitivity]
  drug_penalty = 20 if (ms.antihypertensive_use or ms.magnesium_depleting_drugs) else 0

  base_index = (glucose + lipids + amino + scfa) / 4 - cold_penalty * 0.3 - drug_penalty * 0.4
  mito_atp = base_index * 0.8
  tca_eff = base_index * 0.7
  stress_index = (ls.processed_food_score + (40 if ms.chronic_stress_level == "high" else 20)) / 2

  flags = []
  if scfa < 40:
    flags.append("low_scfa")
  if ls.processed_food_score > 60:
    flags.append("high_processed_food")
  if drug_penalty > 0:
    flags.append("magnesium_depleting_drugs")

  recommendations = [
    Recommendation(
      code="support_scfa",
      title="短鎖脂肪酸産生を支える",
      description="水溶性食物繊維とMBT55系プロバイオティクスを組み合わせます。",
      priority="high",
    )
  ]

  return AcetylCoAMI(
    acetyl_coa_availability_index=clamp(base_index),
    glucose_contribution_score=clamp(glucose),
    lipid_contribution_score=clamp(lipids),
    amino_acid_contribution_score=clamp(amino),
    scfa_contribution_score=clamp(scfa),
    mitochondrial_atp_capacity_score=clamp(mito_atp),
    tca_cycle_efficiency_score=clamp(tca_eff),
    metabolic_stress_index=clamp(stress_index),
    flags=flags,
    recommendations=recommendations,
  )

def clamp(v: float) -> int:
  return max(0, min(100, round(v)))

### 📁 `python/mi_os/engines/snapshot_builder.py`

```python
from ..models.hb003 import Hb003
from .estrobolome_engine import evaluate_estrobolome
from .acetylcoa_engine import evaluate_acetyl_coa
from .mbt55_engine import map_mbt55
from .metabolic_os import build_summary


def build_snapshot(hb: Hb003) -> Hb003:
  estro = evaluate_estrobolome(hb)
  acetyl = evaluate_acetyl_coa(hb)
  mbt = map_mbt55(hb)
  summary = build_summary(hb, estro, acetyl, mbt)

  hb.estrobolome_mi = estro
  hb.acetyl_coa_mi = acetyl
  hb.mbt55_mapping = mbt
  hb.summary = summary
  return hb
```
