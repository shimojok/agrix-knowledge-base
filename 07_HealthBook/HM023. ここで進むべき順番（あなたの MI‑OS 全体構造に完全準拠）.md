# 📁 `frontend/src/app/mi/mbt55/MBT55Form.tsx`

```tsx
"use client";

import { useState } from "react";
import { Hb003, MBT55Group } from "@/src/lib/types";
import { evaluateMBT55 } from "@/src/lib/api/miMBT55";

export default function MBT55Form() {
  const [hb003, setHb003] = useState<Partial<Hb003>>({});
  const [groups, setGroups] = useState<MBT55Group[]>([]);
  const [loading, setLoading] = useState(false);

  const addGroup = () => {
    setGroups([
      ...groups,
      {
        groupId: crypto.randomUUID(),
        name: "",
        functions: {
          estrogenMetabolism: "none",
          scfaProduction: "low",
          antiInflammatory: "low",
          cofactorGeneration: "low",
        },
      },
    ]);
  };

  const updateGroup = (index: number, field: string, value: any) => {
    const updated = [...groups];
    updated[index].functions[field] = value;
    setGroups(updated);
  };

  const submit = async () => {
    setLoading(true);
    const result = await evaluateMBT55({
      ...hb003,
      microbiome: {
        ...hb003.microbiome,
        mbt55Composition: groups,
      },
    } as Hb003);
    localStorage.setItem("mbt55Result", JSON.stringify(result));
    setLoading(false);
  };

  return (
    <div className="border p-4 rounded mb-6">
      <h2 className="text-xl font-semibold mb-2">MBT55 菌群入力</h2>

      <button
        className="px-4 py-2 bg-green-600 text-white rounded mb-4"
        onClick={addGroup}
      >
        菌群を追加
      </button>

      {groups.map((g, i) => (
        <div key={g.groupId} className="border p-3 mb-3 rounded">
          <label className="block">菌群名</label>
          <input
            type="text"
            className="border p-1 w-full"
            onChange={(e) => {
              const updated = [...groups];
              updated[i].name = e.target.value;
              setGroups(updated);
            }}
          />

          <label className="block mt-2">エストロゲン代謝</label>
          <select
            onChange={(e) => updateGroup(i, "estrogenMetabolism", e.target.value)}
          >
            <option value="none">なし</option>
            <option value="support">サポート</option>
            <option value="core">中核</option>
          </select>

          <label className="block mt-2">SCFA産生</label>
          <select
            onChange={(e) => updateGroup(i, "scfaProduction", e.target.value)}
          >
            <option value="low">低い</option>
            <option value="medium">中</option>
            <option value="high">高い</option>
          </select>

          <label className="block mt-2">抗炎症作用</label>
          <select
            onChange={(e) => updateGroup(i, "antiInflammatory", e.target.value)}
          >
            <option value="low">低い</option>
            <option value="medium">中</option>
            <option value="high">高い</option>
          </select>

          <label className="block mt-2">補酵素生成</label>
          <select
            onChange={(e) => updateGroup(i, "cofactorGeneration", e.target.value)}
          >
            <option value="low">低い</option>
            <option value="medium">中</option>
            <option value="high">高い</option>
          </select>
        </div>
      ))}

      <button
        className="mt-4 px-4 py-2 bg-blue-600 text-white rounded"
        onClick={submit}
        disabled={loading}
      >
        {loading ? "計算中…" : "MBT55 機能マッピングを計算"}
      </button>
    </div>
  );
}
```
