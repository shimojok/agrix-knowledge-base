import { Snapshot, EngineResults, CausalMap, DiseaseRisk } from "../types";

export async function fetchSnapshot(userId: string): Promise<Snapshot> {
  const res = await fetch(`http://localhost:8000/snapshot/${userId}`);
  return res.json();
}

export async function fetchEngineResults(userId: string): Promise<EngineResults> {
  const res = await fetch(`http://localhost:8000/engine/results/${userId}`);
  return res.json();
}

export async function fetchCausalMap(userId: string): Promise<CausalMap> {
  const res = await fetch(`http://localhost:8000/causal/${userId}`);
  return res.json();
}

export async function fetchDiseaseRisk(userId: string): Promise<DiseaseRisk> {
  const res = await fetch(`http://localhost:8000/disease/risk/${userId}`);

}

## 🟦 Step 3 — UI コンポーネントに型を適用

### 📄 SnapshotPanel.tsx（型適用版）

tsx

```
"use client";

import { useEffect, useState } from "react";
import { Snapshot } from "../../../lib/types";
import { fetchSnapshot } from "../../../lib/api/dashboard";

export default function SnapshotPanel() {
  const [snapshot, setSnapshot] = useState<Snapshot | null>(null);

  useEffect(() => {
    fetchSnapshot("user001").then(setSnapshot);
  }, []);

  return (
    <div className="border rounded-lg p-4 shadow-sm">
      <h2 className="text-xl font-semibold mb-2">Snapshot</h2>

      {!snapshot && <p>Loading...</p>}

      {snapshot && (
        <pre className="text-sm text-gray-700">
          {JSON.stringify(snapshot, null, 2)}
        </pre>
      )}
    </div>
  );
}
```
