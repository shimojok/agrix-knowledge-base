# 🧬 **3. エンジン側の拡張（DiseaseMatrixEngine / KampoEngine）**

データ構造が拡張されたので、 エンジン側も **v2 データ構造に対応する拡張テンプレート**を生成します。

## 📁 `engines/disease_matrix/disease_engine_v2.py`

python

```
class DiseaseMatrixEngineV2:
    def __init__(self, disease_matrix):
        self.matrix = disease_matrix["diseases"]

    def match(self, causal_risk, phenotype_score):
        results = []
        for d in self.matrix:
            threshold = d["risk_thresholds"]["disease_risk_causal"]
            if causal_risk >= threshold or phenotype_score >= d["risk_thresholds"]["phenotype_score"]:
                results.append({
                    "disease_id": d["id"],
                    "name": d["name"],
                    "system": d["system"],
                    "recommended_kampo": d["recommended_kampo"],
                    "contraindicated_kampo": d["contraindicated_kampo"]
                })
        return results
```

## 📁 `engines/oriental_medicine/kampo_engine_v2.py`

python

```
import json
from pathlib import Path

class KampoEngineV2:
    def __init__(self, formulas_dir, herbs_dir):
        self.formulas_dir = Path(formulas_dir)
        self.herbs_dir = Path(herbs_dir)

    def load_formula(self, formula_id):
        file = self.formulas_dir / f"{formula_id}.json"
        return json.load(open(file))

    def load_herb(self, herb_id):
        file = self.herbs_dir / f"{herb_id}.json"
        return json.load(open(file))

    def recommend_for_disease(self, disease_entry):
        formulas = disease_entry["recommended_kampo"]
        results = []
        for fid in formulas:
            formula = self.load_formula(fid)
            herbs = [self.load_herb(hid) for hid in formula["herbs"]]
            results.append({
                "formula": formula,
                "herbs": herbs
            })
        return results
```
