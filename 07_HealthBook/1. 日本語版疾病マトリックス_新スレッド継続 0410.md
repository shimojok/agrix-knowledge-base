## 2. 推論エンジン多言語対応拡張 (`engine.py` 修正)

日本語マトリックスを読み込み、クエリ言語に応じて結果を切り替える機能を追加します。

```python
# engine.py への追加コード（既存クラスの拡張）

import json
import os
from typing import Dict, List, Optional, Union

class KampoInferenceEngine:
    def __init__(self, language: str = "ja"):
        """
        Args:
            language: 'ja' or 'en' (default: 'ja')
        """
        self.language = language
        self.base_path = os.path.dirname(os.path.abspath(__file__))
        
        # 言語別リソースパス
        self.disease_matrix_path = {
            "en": os.path.join(self.base_path, "en", "disease_matrix_137.json"),
            "ja": os.path.join(self.base_path, "ja", "disease_matrix_137.json")
        }
        self.kampo_lib_path = {
            "en": os.path.join(self.base_path, "en", "kampo_library_mbt_293.json"),
            "ja": os.path.join(self.base_path, "ja", "kampo_library_mbt_293.json")
        }
        
        # データロード
        self.disease_matrix = self._load_json(self.disease_matrix_path[language])
        self.kampo_library = self._load_json(self.kampo_lib_path[language])
        
        # インデックス構築
        self.disease_index = {d["disease_id"]: d for d in self.disease_matrix["disease_matrix"]}
        self.kampo_index = {k["formula_id"]: k for k in self.kampo_library["kampo_formulas"]}
        
    def _load_json(self, path: str) -> dict:
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)
    
    def set_language(self, language: str):
        """言語切替（再初期化）"""
        self.__init__(language=language)
    
    def get_disease_name(self, disease_id: str) -> str:
        """疾病IDから言語別疾病名を取得"""
        disease = self.disease_index.get(disease_id)
        if not disease:
            return disease_id
        return disease.get("disease_name_ja" if self.language == "ja" else "disease_name_en", disease_id)
    
    def get_kampo_name(self, formula_id: str) -> str:
        """処方IDから言語別漢方名を取得"""
        kampo = self.kampo_index.get(formula_id)
        if not kampo:
            return formula_id
        return kampo.get("name_ja" if self.language == "ja" else "name_en", formula_id)
    
    def infer_by_disease(self, disease_id: str) -> Dict:
        """
        疾病IDから推奨漢方処方を推論
        
        Returns:
            {
                "disease_id": "D001",
                "disease_name": "感冒（風邪症候群）",
                "recommended": [
                    {"id": "F252", "name": "葛根湯", "evidence": "A", "dosage": "標準"},
                    ...
                ],
                "contraindicated": [...]
            }
        """
        disease = self.disease_index.get(disease_id)
        if not disease:
            return {"error": f"Disease {disease_id} not found"}
        
        result = {
            "disease_id": disease_id,
            "disease_name": self.get_disease_name(disease_id),
            "disease_name_en": disease.get("disease_name_en", ""),
            "category": disease.get("category", ""),
            "icd11_code": disease.get("icd11_code", ""),
            "recommended": [],
            "contraindicated": []
        }
        
        # 推奨処方
        for fid in disease.get("recommended_kampo", []):
            kampo = self.kampo_index.get(fid)
            if kampo:
                result["recommended"].append({
                    "id": fid,
                    "name": self.get_kampo_name(fid),
                    "evidence": disease.get("evidence_level", "C"),
                    "dosage": kampo.get("standard_dosage", "標準")
                })
        
        # 禁忌処方
        for fid in disease.get("contraindicated_kampo", []):
            kampo = self.kampo_index.get(fid)
            if kampo:
                result["contraindicated"].append({
                    "id": fid,
                    "name": self.get_kampo_name(fid),
                    "reason": kampo.get("contraindications", [{}])[0].get("description_ja" if self.language == "ja" else "description_en", "") if kampo.get("contraindications") else ""
                })
        
        return result
    
    def infer_by_symptoms(self, symptoms: List[str], top_k: int = 5) -> List[Dict]:
        """
        症状リストから適合漢方処方を推論（簡易版）
        
        Args:
            symptoms: 症状名リスト（日本語/英語）
            top_k: 上位候補数
        """
        # 症状-処方マッチング（簡易キーワード検索）
        candidates = []
        for fid, kampo in self.kampo_index.items():
            indications = kampo.get("indications_ja" if self.language == "ja" else "indications_en", [])
            if not indications:
                continue
            
            score = 0
            matched_indications = []
            for symptom in symptoms:
                for ind in indications:
                    if symptom.lower() in ind.lower():
                        score += 1
                        matched_indications.append(ind)
            
            if score > 0:
                candidates.append({
                    "formula_id": fid,
                    "formula_name": self.get_kampo_name(fid),
                    "match_score": score,
                    "matched_indications": list(set(matched_indications))[:3],
                    "evidence_level": kampo.get("evidence_level", "C")
                })
        
        candidates.sort(key=lambda x: x["match_score"], reverse=True)
        return candidates[:top_k]

# 使用例
if __name__ == "__main__":
    # 日本語エンジン
    engine_ja = KampoInferenceEngine(language="ja")
    result = engine_ja.infer_by_disease("D001")
    print(f"【日本語】{result['disease_name']} → 推奨: {[r['name'] for r in result['recommended']]}")
    
    # 英語エンジンに切替
    engine_ja.set_language("en")
    result_en = engine_ja.infer_by_disease("D001")
    print(f"【English】{result_en['disease_name']} → Recommended: {[r['name'] for r in result_en['recommended']]}")
    
    # 症状ベース推論
    symptoms = ["咳", "鼻水", "悪寒"]
    candidates = engine_ja.infer_by_symptoms(symptoms)
    print(f"症状 {symptoms} への適合処方: {[c['formula_name'] for c in candidates]}")
```

import unittest
import sys

        """英語疾病推論テスト"""

        self.assertEqual(self.engine_en.get_kampo_name("F252"), "Kakkonto")
