```
# ai_agents/explainability/explainability_engine.py

class ExplainabilityEngine:
    """
    HealthBook-MI-Probiome-OS
    Explainability Engine (v1.0)
    -----------------------------------------
    - エンジン結果を因果的に説明
    - 医療機関・保険会社向けの説明責任モード
    - Causal Map と連携して「理由」を生成
    """

    def __init__(self, causal_map):
        self.causal_map = causal_map

    # ---------------------------------------------------------
    # 1. 因果連鎖の説明（MBT55 → Estrobolome → ATP → Inflammation）
    # ---------------------------------------------------------
    def explain_causal_chain(self, causal_results):
        return {
            "mbt55_effect": self._explain_mbt55(causal_results),
            "estrobolome_effect": self._explain_estrobolome(causal_results),
            "acetylcoa_effect": self._explain_acetylcoa(causal_results),
            "atp_effect": self._explain_atp(causal_results),
            "inflammation_effect": self._explain_inflammation(causal_results),
            "disease_risk_reason": self._explain_disease_risk(causal_results)
        }

    def _explain_mbt55(self, c):
        if c["estrobolome_score_causal"] < 40:
            return "MBT55 activity is low, reducing SCFA and bile acid modulation."
        return "MBT55 activity supports SCFA and bile acid balance."

    def _explain_estrobolome(self, c):
        if c["estrobolome_score_causal"] < 30:
            return "Estrobolome instability detected, affecting hormone metabolism."
        return "Estrobolome activity is stable."

    def _explain_acetylcoa(self, c):
        if c["acetylcoa_index_causal"] < 40:
            return "Low Acetyl-CoA production reduces mitochondrial energy flux."
        return "Acetyl-CoA production is adequate."

    def _explain_atp(self, c):
        if c["atp_causal"] < 40:
            return "ATP production is insufficient, increasing inflammatory susceptibility."
        return "ATP production is stable."

    def _explain_inflammation(self, c):
        if c["inflammation_causal"] > 70:
            return "High inflammation load detected due to low ATP or high oxidative stress."
        return "Inflammation load is within normal range."

    def _explain_disease_risk(self, c):
        if c["disease_risk_causal"] > 70:
            return "Disease risk is elevated due to combined metabolic and inflammatory factors."
        return "Disease risk remains moderate."

    # ---------------------------------------------------------
    # 2. 医療機関向け説明（Clinical Accountability）
    # ---------------------------------------------------------
    def clinical_explanation(self, snapshot, engine_results, causal_results):
        return {
            "summary": "Clinical explanation of metabolic, hormonal, and inflammatory status.",
            "causal_chain": self.explain_causal_chain(causal_results),
            "clinical_implications": self._clinical_implications(engine_results)
        }

    def _clinical_implications(self, engine_results):
        phenotype = engine_results["phenotype"]["phenotype"]
        inflammation = engine_results["inflammation"]["inflammation_load"]

        if phenotype == "Severe metabolic deviation":
            return "Severe metabolic deviation requires clinical follow-up."
        if inflammation > 70:
            return "High inflammation load suggests systemic inflammatory risk."
        return "Metabolic status is stable."

    # ---------------------------------------------------------
    # 3. 保険会社向け説明（Underwriting Accountability）
    # ---------------------------------------------------------
    def underwriting_explanation(self, decision, causal_results):
        reason = decision["reason"]
        causal_chain = self.explain_causal_chain(causal_results)

        return {
            "decision": decision["decision"],
            "reason": reason,
            "causal_factors": causal_chain
        }
```
