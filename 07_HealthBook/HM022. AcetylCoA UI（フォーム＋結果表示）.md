### 📁 `frontend/src/app/mi/acetyl-coa/AcetylCoAForm.tsx`

tsx

```
"use client";

import { useState } from "react";
import { Hb003 } from "@/src/lib/types";
import { evaluateAcetylCoA } from "@/src/lib/api/miAcetylCoA";

export default function AcetylCoAForm() {
  const [hb003, setHb003] = useState<Partial<Hb003>>({});
  const [loading, setLoading] = useState(false);

  const submit = async () => {
    setLoading(true);
    const result = await evaluateAcetylCoA(hb003 as Hb003);
    localStorage.setItem("acetylCoAResult", JSON.stringify(result));
    setLoading(false);
  };

  return (
    <div className="border p-4 rounded mb-6">
      <h2 className="text-xl font-semibold mb-2">入力フォーム</h2>

      <label className="block mt-2">食事パターン</label>
      <select
        onChange={(e) =>
          setHb003({
            ...hb003,
            lifestyle: {
              ...hb003.lifestyle,
              dietPattern: e.target.value,
            },
          })
        }
      >
        <option value="balanced">バランス型</option>
        <option value="high_processed">加工食品多め</option>
        <option value="high_fiber">高食物繊維</option>
      </select>

      <label className="block mt-2">加工食品スコア（0–100）</label>
      <input
        type="number"
        min={0}
        max={100}
        onChange={(e) =>
          setHb003({
            ...hb003,
            lifestyle: {
              ...hb003.lifestyle,
              processedFoodScore: Number(e.target.value),
            },
          })
        }
      />

      <label className="block mt-2">SCFA産生スコア（0–100）</label>
      <input
        type="number"
        min={0}
        max={100}
        onChange={(e) =>
          setHb003({
            ...hb003,
            microbiome: {
              ...hb003.microbiome,
              scfaProductionScore: Number(e.target.value),
            },
          })
        }
      />

      <label className="block mt-2">ストレスレベル</label>
      <select
        onChange={(e) =>
          setHb003({
            ...hb003,
            medicationAndStress: {
              ...hb003.medicationAndStress,
              chronicStressLevel: e.target.value as any,
            },
          })
        }
      >
        <option value="low">低い</option>
        <option value="medium">中</option>
        <option value="high">高い</option>
      </select>

      <label className="block mt-2">冷え感受性</label>
      <select
        onChange={(e) =>
          setHb003({
            ...hb003,
            medicationAndStress: {
              ...hb003.medicationAndStress,
              coldSensitivity: e.target.value as any,
            },
          })
        }
      >
        <option value="low">低い</option>
        <option value="medium">中</option>
        <option value="high">高い</option>
      </select>

      <button
        className="mt-4 px-4 py-2 bg-blue-600 text-white rounded"
        onClick={submit}
        disabled={loading}
      >
        {loading ? "計算中…" : "MI‑AcetylCoA を計算"}
      </button>
    </div>
  );
}

### 📁 `frontend/src/app/mi/acetyl-coa/AcetylCoAResult.tsx`

tsx

```
"use client";

import { useEffect, useState } from "react";
import { AcetylCoAMIResult } from "@/src/lib/types";

export default function AcetylCoAResult() {
  const [result, setResult] = useState<AcetylCoAMIResult | null>(null);

  useEffect(() => {
    const stored = localStorage.getItem("acetylCoAResult");
    if (stored) setResult(JSON.parse(stored));
  }, []);

  if (!result) return null;

  return (
    <div className="border p-4 rounded">
      <h2 className="text-xl font-semibold mb-2">結果</h2>

      <p>アセチルCoA生成能力: {result.acetylCoAAvailabilityIndex}</p>
      <p>ミトコンドリアATP能力: {result.mitochondrialATPCapacityScore}</p>
      <p>TCAサイクル効率: {result.tcaCycleEfficiencyScore}</p>
      <p>代謝ストレス指数: {result.metabolicStressIndex}</p>

      <h3 className="mt-4 font-semibold">推奨事項</h3>
      <ul>
        {result.recommendations.map((r) => (
          <li key={r.code}>
            <strong>{r.title}</strong> — {r.description}
          </li>
        ))}
      </ul>
    </div>
  );
}
