# 📁 frontend/src/app/mi/estrobolome/EstrobolomeForm.tsx

tsx

```
"use client";

import { useState } from "react";
import { evaluateEstrobolome } from "@/src/lib/api/miEstrobolome";
import { Hb003 } from "@/src/lib/types";

export default function EstrobolomeForm() {
  const [hb003, setHb003] = useState<Partial<Hb003>>({});
  const [loading, setLoading] = useState(false);

  const submit = async () => {
    setLoading(true);
    const result = await evaluateEstrobolome(hb003 as Hb003);
    localStorage.setItem("estrobolomeResult", JSON.stringify(result));
    setLoading(false);
  };

  return (
    <div className="border p-4 rounded mb-6">
      <h2 className="text-xl font-semibold mb-2">入力フォーム</h2>

      <label>食物繊維レベル</label>
      <select
        onChange={(e) =>
          setHb003({
            ...hb003,
            lifestyle: {
              ...hb003.lifestyle,
              fiberIntakeLevel: e.target.value as any,
            },
          })
        }
      >
        <option value="low">低い</option>
        <option value="medium">中</option>
        <option value="high">高い</option>
      </select>

      <label className="mt-4 block">発酵食品頻度</label>
      <select
        onChange={(e) =>
          setHb003({
            ...hb003,
            lifestyle: {
              ...hb003.lifestyle,
              fermentedFoodFrequency: e.target.value as any,
            },
          })
        }
      >
        <option value="rare">ほぼ食べない</option>
        <option value="sometimes">時々</option>
        <option value="daily">毎日</option>
      </select>

      <button
        className="mt-4 px-4 py-2 bg-blue-600 text-white rounded"
        onClick={submit}
        disabled={loading}
      >
        {loading ? "計算中…" : "Estrobolome-MI を計算"}
      </button>
    </div>
  );
}
```

# 📁 frontend/src/app/mi/estrobolome/EstrobolomeResult.tsx

tsx

```
"use client";

import { useEffect, useState } from "react";
import { EstrobolomeMIResult } from "@/src/lib/types";

export default function EstrobolomeResult() {
  const [result, setResult] = useState<EstrobolomeMIResult | null>(null);

  useEffect(() => {
    const stored = localStorage.getItem("estrobolomeResult");
    if (stored) setResult(JSON.parse(stored));
  }, []);

  if (!result) return null;

  return (
    <div className="border p-4 rounded">
      <h2 className="text-xl font-semibold mb-2">結果</h2>

      <p>Estrobolome 活性: {result.estrobolomeActivityScore}</p>
      <p>エストロゲン再循環効率: {result.estrogenRecyclingEfficiency}</p>
      <p>ミトコンドリア応答: {result.mitochondrialEstrogenResponseScore}</p>
      <p>痛み関連度: {result.painCorrelationScore}</p>

      <h3 className="mt-4 font-semibold">推奨事項</h3>
      <ul>
        {result.recommendations.map((r) => (
          <li key={r.code}>
            <strong>{r.title}</strong> — {r.description}
          </li>
        ))}
      </ul>
    </div>
  );
}
```
