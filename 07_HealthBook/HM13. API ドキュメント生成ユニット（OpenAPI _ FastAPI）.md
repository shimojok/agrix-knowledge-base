# 🧬 **1. API ドキュメント生成ユニット（OpenAPI / FastAPI）**

FastAPI は自動で OpenAPI を生成しますが、 HealthBook‑MI‑Probiome‑OS では **独自の API ドキュメント生成ユニット**を追加し、 医療機関・保険会社向けの「説明責任モード」を提供します。

## 📁 `interfaces/api_docs/api_docs_generator.py`

python

```
# interfaces/api_docs/api_docs_generator.py

from fastapi import FastAPI

class APIDocsGenerator:
    """
    HealthBook-MI-Probiome-OS
    API Documentation Generator
    -----------------------------------------
    - FastAPI の OpenAPI を拡張
    - 医療機関・保険会社向けの説明責任モード
    """

    def __init__(self, app: FastAPI):
        self.app = app

    def generate_summary(self):
        return {
            "title": self.app.title,
            "version": self.app.version if hasattr(self.app, "version") else "1.0",
            "routes": [
                {
                    "path": route.path,
                    "name": route.name,
                    "methods": list(route.methods)
                }
                for route in self.app.routes
            ]
        }

    def generate_medical_doc(self):
        return {
            "purpose": "Medical Evaluation API",
            "input": {
                "responses": "200問診の回答",
                "nutrition": "栄養データ（carb, fat, protein）",
                "mbt_data": "MBT55代謝データ（SCFA, Bile Acid）",
                "estro_data": "エストロボローム関連データ",
                "mito_data": "ミトコンドリア関連データ",
                "lifestyle": "生活習慣データ"
            },
            "output": {
                "phenotype": "代謝偏位（Phenotype）",
                "estrobolome": "ホルモン代謝指標",
                "acetylcoa": "エネルギー代謝指標",
                "inflammation": "炎症負荷",
                "explanation": "AIエージェントによる説明"
            }
        }

    def generate_insurance_doc(self):
        return {
            "purpose": "Insurance Underwriting API",
            "output": {
                "decision": "引受判断（Accept / Postpone / Decline）",
                "explanation": "AIエージェントによる説明",
                "risk_profile": "代謝・炎症・ホルモンの総合リスク"
            }
        }

    def generate_probiotics_doc(self):
        return {
            "purpose": "MBT55 Probiotics Screening API",
            "output": {
                "metabolites": "生成される代謝産物",
                "mbt_meta_recommendations": "推奨MBT_META菌株",
                "engine_results": "代謝OSの結果"
            }
        }

# tests/integration/test_medical_pipeline.py

from platform.orchestrator.snapshot_builder import SnapshotBuilder
from platform.orchestrator.engine_orchestrator import EngineOrchestrator
from platform.pipelines.medical_pipeline import MedicalPipeline

from engines.phenotyping.questionnaire_engine import PhenotypingEngine
from engines.estrobolome.estrobolome_engine import EstrobolomeEngine
from engines.acetylcoa_atp.acetylcoa_engine import AcetylCoAEngine
from engines.inflammation_immune.inflammation_engine import InflammationEngine

def test_medical_pipeline():
    snapshot_builder = SnapshotBuilder("data/questionnaire/questionnaire_200_jp.json")

    orchestrator = EngineOrchestrator({
        "phenotyping": PhenotypingEngine("data/questionnaire/questionnaire_200_jp.json"),
        "estrobolome": EstrobolomeEngine(),
        "acetylcoa": AcetylCoAEngine(),
        "inflammation": InflammationEngine()
    })

    pipeline = MedicalPipeline(snapshot_builder, orchestrator)

    responses = {"1": 1, "2": 0}
    nutrition = {"carb": 50, "fat": 40, "protein": 30}
    mbt_data = {"scfa": 60, "bile_acid": 40}
    estro_data = {"beta_glucuronidase": 20}
    mito_data = {"oxidative_stress": 30}
    lifestyle = {"sleep_quality": 70, "stress": 40}

    result = pipeline.run(responses, nutrition, mbt_data, estro_data, mito_data, lifestyle)

    assert "phenotype" in result
    assert "estrobolome" in result
    assert "acetylcoa" in result
    assert "inflammation" in result
```
