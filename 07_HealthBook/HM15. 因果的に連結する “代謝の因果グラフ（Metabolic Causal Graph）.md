```
# platform/orchestrator/causal_map.py

class CausalMap:
    """
    HealthBook-MI-Probiome-OS
    Metabolic Causal Graph (v1.0)
    -----------------------------------------
    代謝 → ホルモン → エネルギー → 炎症 → 疾病リスク
    MBT55 → Estrobolome → AcetylCoA → ATP → Inflammation → Disease
    """

    def __init__(self):
        pass

    # ---------------------------------------------------------
    # 1. MBT55 → Estrobolome
    # ---------------------------------------------------------
    def mbt55_to_estrobolome(self, scfa, bile_acid, beta_glucuronidase):
        """
        MBT55 の SCFA と胆汁酸はエストロボロームを強化し、
        β-グルクロニダーゼはホルモン不安定化を引き起こす。
        """
        score = (scfa * 0.4) + (bile_acid * 0.3) - (beta_glucuronidase * 0.5)
        return max(0, min(score, 100))

    # ---------------------------------------------------------
    # 2. Estrobolome → AcetylCoA
    # ---------------------------------------------------------
    def estrobolome_to_acetylcoa(self, estro_score, carb, fat, protein, scfa):
        """
        ホルモンバランスが整うとミトコンドリア代謝が安定し、
        Acetyl-CoA 生成効率が向上する。
        """
        base_flux = (
            carb * 0.25 +
            fat * 0.25 +
            protein * 0.20 +
            scfa * 0.30
        )
        modifier = (estro_score / 100) * 0.2  # ホルモン安定性による増幅
        return min(base_flux * (1 + modifier), 100)

    # ---------------------------------------------------------
    # 3. AcetylCoA → ATP
    # ---------------------------------------------------------
    def acetylcoa_to_atp(self, acetylcoa_index):
        """
        Acetyl-CoA が高いほど ATP 生成効率が高まる。
        """
        return min(acetylcoa_index * 0.9, 100)

    # ---------------------------------------------------------
    # 4. ATP → Inflammation
    # ---------------------------------------------------------
    def atp_to_inflammation(self, atp, oxidative_stress):
        """
        ATP が低いと炎症負荷が上昇し、
        酸化ストレスが高いと炎症が増幅される。
        """
        anti = atp * 0.5
        pro = oxidative_stress * 0.8
        load = max(0, min(100, pro - anti))
        return load

    # ---------------------------------------------------------
    # 5. Inflammation → Disease Risk
    # ---------------------------------------------------------
    def inflammation_to_disease_risk(self, inflammation, phenotype_score):
        """
        炎症負荷と代謝偏位（Phenotype）が疾病リスクを決定する。
        """
        risk = (inflammation * 0.6) + (phenotype_score * 0.4)
        return min(risk, 100)

    # ---------------------------------------------------------
    # 6. 全体因果連鎖の実行
    # ---------------------------------------------------------
    def run(self, snapshot, engine_results):
        """
        Snapshot と Engine 結果を統合し、
        因果連鎖モデルを通して最終疾病リスクを算出する。
        """

        # 1. MBT55 → Estrobolome
        estro_score = self.mbt55_to_estrobolome(
            snapshot["mbt55"]["scfa"],
            snapshot["mbt55"]["bile_acid"],
            snapshot["estrobolome"]["beta_glucuronidase"]
        )

        # 2. Estrobolome → AcetylCoA
        acetylcoa_index = self.estrobolome_to_acetylcoa(
            estro_score,
            snapshot["nutrition"]["carb"],
            snapshot["nutrition"]["fat"],
            snapshot["nutrition"]["protein"],
            snapshot["mbt55"]["scfa"]
        )

        # 3. AcetylCoA → ATP
        atp = self.acetylcoa_to_atp(acetylcoa_index)

        # 4. ATP → Inflammation
        inflammation = self.atp_to_inflammation(
            atp,
            snapshot["mitochondria"]["oxidative_stress"]
        )

        # 5. Inflammation → Disease Risk
        phenotype_score = engine_results["phenotype"]["total_score"]
        disease_risk = self.inflammation_to_disease_risk(inflammation, phenotype_score)

        return {
            "estrobolome_score_causal": estro_score,
            "acetylcoa_index_causal": acetylcoa_index,
            "atp_causal": atp,
            "inflammation_causal": inflammation,
            "disease_risk_causal": disease_risk
        }
```
