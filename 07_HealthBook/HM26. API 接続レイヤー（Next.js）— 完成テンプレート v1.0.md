# 🧬 **2. medical.ts（Medical Pipeline API 接続）**

ts

```
// ui/lib/api/medical.ts

export async function fetchMedicalEvaluation(payload: any) {
  const res = await fetch("http://localhost:8000/medical/evaluate", {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify(payload)
  });

  if (!res.ok) {
    throw new Error("Failed to fetch medical evaluation");
  }

  return res.json();
}
```

# 🟦 **UI コンポーネントへの接続例（Dashboard）**

例：`SnapshotCard.tsx` に API を接続する場合

tsx

```
"use client";

import { useEffect, useState } from "react";
import { fetchSnapshot } from "../../lib/api/dashboard";

export default function SnapshotCard() {
  const [snapshot, setSnapshot] = useState<any>(null);

  useEffect(() => {
    fetchSnapshot("user001").then(setSnapshot);
  }, []);

  return (
    <div className="border rounded-lg p-4 shadow-sm">
      <h2 className="text-xl font-semibold mb-2">Metabolic Snapshot</h2>

      {!snapshot && <p>Loading...</p>}

      {snapshot && (
        <pre className="text-sm text-gray-700">
          {JSON.stringify(snapshot, null, 2)}
        </pre>
      )}
    </div>
  );
}
```

# 🟦 **Medical UI への接続例（QuestionnaireForm）**

tsx

```
"use client";

import { useState } from "react";
import { fetchMedicalEvaluation } from "../../lib/api/medical";

export default function QuestionnaireForm() {
  const [result, setResult] = useState<any>(null);

  async function handleSubmit() {
    const payload = {
      responses: { "1": 1, "2": 0 },
      nutrition: { carb: 50, fat: 40, protein: 30 },
      mbt_data: { scfa: 60, bile_acid: 40 },
      estro_data: { beta_glucuronidase: 20 },
      mito_data: { oxidative_stress: 30 },
      lifestyle: { sleep_quality: 70, stress: 40 }
    };

    const res = await fetchMedicalEvaluation(payload);
    setResult(res);
  }

  return (
    <div className="border rounded-lg p-4 shadow-sm">
      <h2 className="text-xl font-semibold mb-2">Questionnaire (200 items)</h2>

      <button
        onClick={handleSubmit}
        className="px-4 py-2 bg-blue-600 text-white rounded"
      >
        Run Evaluation
      </button>

      {result && (
        <pre className="mt-4 text-sm text-gray-700">
          {JSON.stringify(result, null, 2)}
        </pre>
      )}
    </div>
  );
}
```
