### 2. `rag_engine.py`（ローカルRAG検索エンジン）

python

```
import os
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import FAISS
from langchain.embeddings import HuggingFaceEmbeddings

def build_rag_db(txt_dir: str = "txt"):
    docs = []
    for file in os.listdir(txt_dir):
        path = os.path.join(txt_dir, file)
        if not os.path.isfile(path):
            continue
        with open(path, "r", encoding="utf-8") as f:
            docs.append(f.read())

    splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200
    )
    chunks = splitter.create_documents(docs)

    embeddings = HuggingFaceEmbeddings(
        model_name="sentence-transformers/all-MiniLM-L6-v2"
    )
    db = FAISS.from_documents(chunks, embeddings)
    return db

def rag_search(query: str, k: int = 5):
    db = build_rag_db()
    return db.similarity_search(query, k=k)
```

### 3. `pbpe_engine.py`（PBPE金融モデル）

python

```
import json
from math import pow

def load_canonical(path: str = "canonical_data.json"):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def calc_npv(cashflow_per_year: float, years: int, discount_rate: float) -> float:
    return sum(
        cashflow_per_year / pow(1 + discount_rate, t)
        for t in range(1, years + 1)
    )

def pbpe_summary():
    data = load_canonical()
    f = data["finance"]
    p = data["pbpe"]
    m = data["mbt55"]

    # 単純化した例：トークン収入＋グリーンプレミアム
    token_income_per_year = (
        m["units"] * p["token_per_unit_per_year"] * p["token_price_usd"]
    )

    cashflow_per_year = token_income_per_year - f["opex_usd_per_year"]
    npv = calc_npv(cashflow_per_year, m["operating_years"], f["discount_rate"])

    return {
        "token_income_per_year": token_income_per_year,
        "cashflow_per_year": cashflow_per_year,
        "npv_usd_20y_model": npv
    }

if __name__ == "__main__":
    print(pbpe_summary())
```

### 4. `mbt_engine.py`（MBT55処理モデル）

python

```
import json

def load_canonical(path: str = "canonical_data.json"):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def mbt_summary():
    data = load_canonical()
    m = data["mbt55"]

    total_waste_ton_per_year = (
        m["units"] * m["waste_ton_per_unit_per_year"]
    )
    total_waste_ton_total = (
        total_waste_ton_per_year * m["operating_years"]
    )

    return {
        "waste_ton_per_year": total_waste_ton_per_year,
        "waste_ton_total": total_waste_ton_total
    }

if __name__ == "__main__":
    print(mbt_summary())
```
