# 📁 **interfaces/marketplace/healthbook_registry.py**

PBPE‑Marketplace の registry に **HealthBook MI の Snapshot / Engine Results / Disease Risk / Probiotics Recommendation** を登録するためのモジュール。

python

```
# interfaces/marketplace/healthbook_registry.py

import json
from datetime import datetime

class HealthBookRegistry:
    """
    PBPE-Marketplace Integration
    HealthBook-MI Registry Module
    -----------------------------------------
    - Snapshot v2 を PBPE registry に登録
    - Engine Results を保存
    - Disease Risk / Probiotics Recommendation を保存
    """

    def __init__(self, storage_path="pbpe_registry.json"):
        self.storage_path = storage_path
        self._init_storage()

    def _init_storage(self):
        try:
            json.load(open(self.storage_path))
        except:
            json.dump({"records": []}, open(self.storage_path, "w"))

    def register_snapshot(self, user_id, snapshot):
        entry = {
            "type": "snapshot",
            "user_id": user_id,
            "timestamp": datetime.utcnow().isoformat(),
            "data": snapshot
        }
        self._save(entry)

    def register_engine_results(self, user_id, engine_results):
        entry = {
            "type": "engine_results",
            "user_id": user_id,
            "timestamp": datetime.utcnow().isoformat(),
            "data": engine_results
        }
        self._save(entry)

    def register_disease_risk(self, user_id, disease_risk):
        entry = {
            "type": "disease_risk",
            "user_id": user_id,
            "timestamp": datetime.utcnow().isoformat(),
            "data": disease_risk
        }
        self._save(entry)

    def register_probiotics(self, user_id, probiotics):
        entry = {
            "type": "probiotics_recommendation",
            "user_id": user_id,
            "timestamp": datetime.utcnow().isoformat(),
            "data": probiotics
        }
        self._save(entry)

    def _save(self, entry):
        db = json.load(open(self.storage_path))
        db["records"].append(entry)
        json.dump(db, open(self.storage_path, "w"), indent=2)
```

# 📁 **interfaces/marketplace/healthbook_kpis.py**

PBPE‑Marketplace の KPI システムに **代謝OSの指標（Phenotype / Estrobolome / ATP / Inflammation / Disease Risk）** を追加するモジュール。

```
# interfaces/marketplace/healthbook_kpis.py

import json

class HealthBookKPIs:
    """
    PBPE-Marketplace Integration
    HealthBook-MI KPI Module
    -----------------------------------------
    - Phenotype Score
    - Estrobolome Score
    - AcetylCoA / ATP Index
    - Inflammation Load
    - Disease Risk (Causal)
    """

    def __init__(self, storage_path="pbpe_kpis.json"):
        self.storage_path = storage_path
        self._init_storage()

    def _init_storage(self):
        try:
            json.load(open(self.storage_path))
        except:
            json.dump({"kpis": {}}, open(self.storage_path, "w"))

    def update_kpis(self, user_id, causal_map_results):
        kpis = {
            "phenotype_score": causal_map_results.get("phenotype_score", None),
            "estrobolome_score": causal_map_results["estrobolome_score_causal"],
            "acetylcoa_index": causal_map_results["acetylcoa_index_causal"],
            "atp": causal_map_results["atp_causal"],
            "inflammation": causal_map_results["inflammation_causal"],
            "disease_risk": causal_map_results["disease_risk_causal"]
        }

        db = json.load(open(self.storage_path))
        db["kpis"][user_id] = kpis
        json.dump(db, open(self.storage_path, "w"), indent=2)

        return kpis
```
