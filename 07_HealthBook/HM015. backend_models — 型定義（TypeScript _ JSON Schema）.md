```
export class EstrobolomeService {
  computeEstrobolomeScores(input: Hb003InputLayer["gut_microbiome"]) {
    const activity =
      (input.scfa_production_score * 0.4) +
      (input.microbiome_diversity_score * 0.3) +
      (input.beta_glucuronidase_activity === "high" ? 30 : 10);

    const recycling = activity * 0.85;
    const mitoResponse = activity * 0.62;

    return {
      estrobolome_activity_score: Math.round(activity),
      estrogen_recycling_efficiency: Math.round(recycling),
      mitochondrial_estrogen_response: Math.round(mitoResponse),
      pain_sensitivity_correlation: Math.round(activity * 0.45)
    };
  }
}
```

```
export class AcetylCoAService {
  computeAcetylCoA(input: Hb003InputLayer) {
    const scfa = input.gut_microbiome.scfa_production_score * 0.5;
    const minerals =
      input.minerals_vitamins.magnesium_status_score * 0.3 +
      input.minerals_vitamins.b_complex_status_score * 0.2;

    const inhibition =
      input.medications.antihypertensive_use
        ? input.medications.antihypertensive_mg_loss_risk_score * 0.5
        : 0;

    const index = scfa + minerals - inhibition;

    return {
      acetylcoa_index: Math.round(index),
      tca_support_score: Math.round(minerals),
      electron_transport_efficiency: Math.round(index * 0.7)
    };
  }
}
```

```
export class MitochondriaService {
  computeMitochondria(input: Hb003InputLayer) {
    const estrogen = input.hormones.estrogen_status_score * 0.6;
    const magnesium = input.minerals_vitamins.magnesium_status_score * 0.4;
    const coldPenalty = input.environmental_factors.cold_exposure_score * 0.5;

    const atp = estrogen + magnesium - coldPenalty;

    return {
      mitochondrial_atp_generation_score: Math.round(atp),
      oxidative_stress_score: Math.round(100 - atp)
    };
  }
}
```

# 🟦 **backend/controllers — API 実装**

## 📁 `backend/controllers/Hb003Controller.ts`

ts

```
import { EstrobolomeService } from "../services/EstrobolomeService";
import { AcetylCoAService } from "../services/AcetylCoAService";
import { MitochondriaService } from "../services/MitochondriaService";

export class Hb003Controller {
  static createSnapshot(req, res) {
    const input = req.body;

    const estro = new EstrobolomeService().computeEstrobolomeScores(
      input.input_layer.gut_microbiome
    );

    const acetyl = new AcetylCoAService().computeAcetylCoA(input.input_layer);

    const mito = new MitochondriaService().computeMitochondria(
      input.input_layer
    );

    const output = {
      metabolic_age_score: Math.round(
        (acetyl.acetylcoa_index + mito.mitochondrial_atp_generation_score) / 2
      ),
      hormone_balance_score: estro.estrogen_recycling_efficiency,
      immune_resilience_score: estro.estrobolome_activity_score,
      recommended_focus_areas: ["fiber", "magnesium", "mbt55", "estrogen_support"]
    };

    return res.json({
      hb003: {
        meta: input.meta,
        input_layer: input.input_layer,
        estrobolome_module: estro,
        acetylcoa_module: acetyl,
        mitochondria_module: mito,
        inflammation_module: input.inflammation_module,
        pain_module: input.pain_module,
        mbt55_functional_mapping: input.mbt55_functional_mapping,
        output_layer: output
      }
    });
  }
}
```
